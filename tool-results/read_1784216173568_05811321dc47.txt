   291→  saveBlend(id) {

   294→    const rows = document.querySelectorAll('#f-bl-ingredientes .ing-row');
   295→    const ingredientes = [];
   296→    rows.forEach(row => {
   297→      const espId = row.querySelector('[data-field="especiaId"]').value;
   298→      const cant = Number(row.querySelector('[data-field="cantidad"]').value) || 0;
   299→      if (espId && cant > 0) ingredientes.push({ especiaId: espId, cantidad: cant });
   300→    });
   301→    if (ingredientes.length === 0) { alert('Agrega al menos un ingrediente'); return; }
   302→
   303→    const data = {
   304→      nombre,
   305→      categoria: document.getElementById('f-bl-cat').value,
   306→      precioChico: Number(document.getElementById('f-bl-pchico').value) || 0,
   307→      precioGrande: Number(document.getElementById('f-bl-pgrande').value) || 0,
   308→      ingredientes
   309→    };
   310→    if (id) data.id = id;
   311→    try {
   312→      ArcanoDB.saveBlend(data);
   313→      document.querySelector('.modal-overlay').remove();
