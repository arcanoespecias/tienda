   321→   600→function producirEspecia(especiaId, talla, cantidad) {
   322→   601→  _ensureStructure();
   323→   602→  var esp = _db.especias[especiaId];
   324→   603→  if (!esp) throw new Error('Especia no encontrada');
   325→   604→  talla = (talla === 'grande') ? 'grande' : 'chico';
   326→   605→  cantidad = Number(cantidad) || 0;
   327→   606→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   328→   607→
   329→   608→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   330→   609→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   331→   610→
   332→   611→  var grsTotal = gramosPorFrasco * cantidad;
   333→   612→
   334→   613→  // Check & consume pala (raw material)
   335→   614→  if ((esp.stockBolsa || 0) < grsTotal) {
   336→   615→    throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   337→   616→  }
   338→   617→
   339→   618→  // Check & consume envases
   340→   619→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   341→   620→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   342→   621→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   343→   622→  }
   344→   623→
   345→   624→  // Check & consume stickers
   346→   625→  var stk = _findStickerByNombre(esp.nombre);
   347→   626→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   348→   627→  if (stkStock < cantidad) {
   349→   628→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   350→   629→  }
   351→   630→
   352→   631→  // Check & consume bolsas (packaging)
   353→   632→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   354→   633→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   355→   634→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   356→   635→  }
   357→   636→
   358→   637→  // All checks passed — consume
   359→   638→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   360→   639→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   361→   640→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   362→   641→  if (stk) {
   363→   642→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   364→   643→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   365→   644→  }
   366→   645→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   367→   646→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   368→   647→
   369→   648→  // Record
   370→   649→  var prodId = nextId('producciones');
   371→   650→  var prod = {
   372→   651→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   373→   652→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   374→   653→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   375→   654→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   376→   655→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   377→   656→  };
   378→   657→  _db.producciones[prodId] = prod;
   379→   658→  _saveToFirebase(); _cacheLocal();
   380→   659→  _notify('create', 'producciones', prodId);
   381→   660→  _notify('update', 'especias', especiaId);
   382→   661→  return { producto: esp, produccion: prod };
   383→   662→}

   385→   664→function producirBlend(blendId, talla, cantidad) {
   386→   665→  _ensureStructure();
   387→   666→  var blend = _db.blends[blendId];
   388→   667→  if (!blend) throw new Error('Blend no encontrado');
   389→   668→  talla = (talla === 'grande') ? 'grande' : 'chico';
   390→   669→  cantidad = Number(cantidad) || 0;
   391→   670→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   392→   671→
   393→   672→  var ingredientes = blend.ingredientes || [];
   394→   673→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   395→   674→
   396→   675→  // Check ingredient stock
   397→   676→  var detalleIngredientes = [];
   398→   677→  for (var i = 0; i < ingredientes.length; i++) {
   399→   678→    var ing = ingredientes[i];
   400→   679→    var esp = _db.especias[ing.especiaId];
   401→   680→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   402→   681→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   403→   682→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   404→   683→    var grsNeeded = grsPorFrasco * cantidad;
   405→   684→    if ((esp.stockBolsa || 0) < grsNeeded) {
   406→   685→      throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   407→   686→    }
   408→   687→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   409→   688→  }
   410→   689→
   411→   690→  // Check envases
   412→   691→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   413→   692→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   414→   693→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   415→   694→  }
   416→   695→
   417→   696→  // Check stickers
   418→   697→  var stk = _findStickerByNombre(blend.nombre);
   419→   698→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   420→   699→  if (stkStock < cantidad) {
   421→   700→    throw new Error('Stickers ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   422→   701→  }
   423→   702→
   424→   703→  // Check & consume bolsas (packaging)
   425→   704→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   426→   705→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   427→   706→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   428→   707→  }
   429→   708→
   430→   709→  // All checks passed — consume
   431→   710→  var grsTotalGeneral = 0;
   432→   711→  for (var i = 0; i < detalleIngredientes.length; i++) {
   433→   712→    var d = detalleIngredientes[i];
   434→   713→    var esp = _db.especias[d.especiaId];
   435→   714→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   436→   715→    grsTotalGeneral += d.gramosTotal;
   437→   716→  }
   438→   717→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   439→   718→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   440→   719→  if (stk) {
   441→   720→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   442→   721→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   443→   722→  }
   444→   723→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   445→   724→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   446→   725→
   447→   726→  var prodId = nextId('producciones');
   448→   727→  var prod = {
   449→   728→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   450→   729→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   451→   730→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   452→   731→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   453→   732→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   454→   733→  };
   455→   734→  _db.producciones[prodId] = prod;
   456→   735→  _saveToFirebase(); _cacheLocal();
   457→   736→  _notify('create', 'producciones', prodId);
   458→   737→  _notify('update', 'blends', blendId);
   459→   738→  return { producto: blend, produccion: prod };
   460→   739→}

   668→   947→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   669→   948→function findEspeciaByName(nombre) {
   670→   949→  if (!nombre) return null;
   671→   950→  var target = nombre.trim().toLowerCase();
   672→   951→  var keys = Object.keys(_db.especias || {});
   673→   952→  // 1. Exact match
   674→   953→  for (var i = 0; i < keys.length; i++) {
   675→   954→    var e = _db.especias[keys[i]];
   676→   955→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   677→   956→  }
   678→   957→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   679→   958→  for (var i = 0; i < keys.length; i++) {
   680→   959→    var e = _db.especias[keys[i]];
   681→   960→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   682→   961→  }
   683→   962→  // 3. Target is contained in especia name
   684→   963→  for (var i = 0; i < keys.length; i++) {
   685→   964→    var e = _db.especias[keys[i]];
   686→   965→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   687→   966→  }
   688→   967→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   689→   968→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   690→   969→  for (var w = 0; w < targetWords.length; w++) {
   691→   970→    for (var i = 0; i < keys.length; i++) {
   692→   971→      var e = _db.especias[keys[i]];
   693→   972→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   694→   973→    }
   695→   974→  }
   696→   975→  return null;
   697→   976→}

   707→   986→  return 'Comidas';

   710→   989→/**
   711→   990→ * Import especias and blends from parsed Excel data.

   713→   992→ *
   714→   993→ * especiasList: [{ nombre, categoria? }]

   719→   998→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   720→   999→  _ensureStructure();
   721→  1000→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   722→  1001→
   723→  1002→  // 1. Create especias (skip if name already exists)
   724→  1003→  for (var i = 0; i < especiasList.length; i++) {
   725→  1004→    var esp = especiasList[i];
   726→  1005→    var nombre = (esp.nombre || '').trim();

   729→  1008→    if (existing) {
   730→  1009→      resultado.especiasExistentes++;
   731→  1010→      continue;
   732→  1011→    }
   733→  1012→    saveEspecia({
   734→  1013→      nombre: nombre,
   735→  1014→      categoria: esp.categoria || 'Comidas',
   736→  1015→      precioChico: 0,
   737→  1016→      precioGrande: 0,
   738→  1017→      gramosChico: Number(gramosChico) || 30,
   739→  1018→      gramosGrande: Number(gramosGrande) || 80,
   740→  1019→      stockBolsa: 0,
   741→  1020→      stockChico: 0,
   742→  1021→      stockGrande: 0
   743→  1022→    });
   744→  1023→    resultado.especiasCreadas++;
   745→  1024→  }
   746→  1025→
   747→  1026→  // 2. Create blends (skip if name already exists)
   748→  1027→  for (var j = 0; j < blendsList.length; j++) {
   749→  1028→    var bl = blendsList[j];
   750→  1029→    var nombre = (bl.nombre || '').trim();
   751→  1030→    if (!nombre) continue;
   752→  1031→    var existingBlend = null;
   753→  1032→    var blKeys = Object.keys(_db.blends || {});
   754→  1033→    for (var k = 0; k < blKeys.length; k++) {
   755→  1034→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
   756→  1035→        existingBlend = _db.blends[blKeys[k]];
   757→  1036→        break;
   758→  1037→      }
   759→  1038→    }
   760→  1039→    if (existingBlend) {
   761→  1040→      resultado.blendsExistentes++;
   762→  1041→      continue;
   763→  1042→    }
   764→  1043→
   765→  1044→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
   766→  1045→    var ings = bl.ingredientes || [];
   767→  1046→    var recipeTotal = 0;
   768→  1047→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
   769→  1048→    if (recipeTotal <= 0) recipeTotal = 500;
   770→  1049→
   771→  1050→    var resolvedIngs = [];
   772→  1051→    for (var ii = 0; ii < ings.length; ii++) {
   773→  1052→      var ing = ings[ii];
   774→  1053→      var espObj = findEspeciaByName(ing.especia);
   775→  1054→      if (!espObj) {
   776→  1055→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
   777→  1056→        continue;
   778→  1057→      }
   779→  1058→      var ingG = Number(ing.g) || 0;
   780→  1059→      resolvedIngs.push({
   781→  1060→        especiaId: espObj.id,
   782→  1061→        especiaNombre: espObj.nombre,
   783→  1062→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
   784→  1063→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
   785→  1064→        gramosReceta: ingG
   786→  1065→      });
   787→  1066→    }
   788→  1067→
   789→  1068→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
   790→  1069→    saveBlend({
   791→  1070→      nombre: nombre,
   792→  1071→      categoria: cat,
   793→  1072→      region: bl.region || '',
   794→  1073→      uso: bl.uso || '',
   795→  1074→      precioChico: 0,
   796→  1075→      precioGrande: 0,
   797→  1076→      ingredientes: resolvedIngs,
   798→  1077→      stockChico: 0,
   799→  1078→      stockGrande: 0
   800→  1079→    });
   801→  1080→    resultado.blendsCreados++;
   802→  1081→  }
   803→  1082→
   804→  1083→  return resultado;
   805→  1084→}

   820→  1099→  _ensureStructure();
   821→  1100→  tagName = (tagName || '').trim();
   822→  1101→  if (!tagName) return false;
   823→  1102→  if (!_db.productTags) _db.productTags = {};
   824→  1103→  if (!_db.productTags[cat]) _db.productTags[cat] = [];
   825→  1104→  // Check duplicate (case-insensitive)
   826→  1105→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   827→  1106→    if (_db.productTags[cat][i].toLowerCase() === tagName.toLowerCase()) return false;
   828→  1107→  }
   829→  1108→  _db.productTags[cat].push(tagName);
   830→  1109→  _saveToFirebase(); _cacheLocal();
   831→  1110→  return true;
   832→  1111→}

   834→  1113→function removeProductTag(cat, tagName) {
   835→  1114→  _ensureStructure();
   836→  1115→  if (!_db.productTags || !_db.productTags[cat]) return false;
   837→  1116→  var idx = -1;
   838→  1117→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   839→  1118→    if (_db.productTags[cat][i] === tagName) { idx = i; break; }
   840→  1119→  }
   841→  1120→  if (idx < 0) return false;
   842→  1121→  _db.productTags[cat].splice(idx, 1);
   843→  1122→  // Also remove from all products that have this tag
   844→  1123→  var allEsp = getEspecias();
   845→  1124→  for (var e = 0; e < allEsp.length; e++) {
   846→  1125→    if (allEsp[e].tags) {
   847→  1126→      var ti = allEsp[e].tags.indexOf(tagName);
   848→  1127→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
   849→  1128→    }
   850→  1129→  }
   851→  1130→  var allBl = getBlends();
   852→  1131→  for (var b = 0; b < allBl.length; b++) {
   853→  1132→    if (allBl[b].tags) {
   854→  1133→      var bi = allBl[b].tags.indexOf(tagName);
   855→  1134→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
   856→  1135→    }
   857→  1136→  }
   858→  1137→  _saveToFirebase(); _cacheLocal();
   859→  1138→  return true;
   860→  1139→}

   864→  1143→function compressImage(file, maxW, quality, cb) {
   865→  1144→  var reader = new FileReader();
   866→  1145→  reader.onload = function(ev) {
   867→  1146→    var img = new Image();
   868→  1147→    img.onload = function() {
   869→  1148→      var w = img.width, h = img.height;
   870→  1149→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
   871→  1150→      var canvas = document.createElement('canvas');
   872→  1151→      canvas.width = w; canvas.height = h;
   873→  1152→      var ctx = canvas.getContext('2d');
   874→  1153→      ctx.drawImage(img, 0, 0, w, h);
   875→  1154→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
   876→  1155→      cb(null, dataUrl);
   877→  1156→    };
   878→  1157→    img.onerror = function() { cb('Error al cargar imagen'); };
   879→  1158→    img.src = ev.target.result;
   880→  1159→  };
   881→  1160→  reader.onerror = function() { cb('Error al leer archivo'); };
   882→  1161→  reader.readAsDataURL(file);
   883→  1162→}
