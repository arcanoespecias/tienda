   150→function initDB() {
   151→  return new Promise(function(resolve) {
   152→    _initFirebase();
   153→
   154→    // Try localStorage cache
   155→    var cached = null;
   156→    try { cached = JSON.parse(localStorage.getItem(DB_KEY)); } catch (e) {}
   157→
   158→    if (cached && cached.meta && cached.meta.version === DB_VERSION && _ensureStructureOn(cached)) {
   159→      _db = cached;
   160→      _ready = true;
   161→      _startFirebaseListener();
   162→      resolve();
   163→      return;
   164→    }
   165→
   166→    // Load from Firebase
   167→    if (_firebaseRef) {
   168→      _firebaseRef.once('value').then(function(snap) {
   169→        var fbData = snap.val();
   170→        if (fbData && fbData.meta && fbData.meta.version === DB_VERSION && _ensureStructureOn(fbData)) {
   171→          _db = fbData;
   172→        } else {
   173→          // Fresh DB (old version or empty)
   174→          _db = _emptyDB();
   175→          _ensureStructure();
   176→          _saveToFirebase();
   177→        }
   178→        _ready = true;
   179→        _cacheLocal();
   180→        _startFirebaseListener();
   181→        resolve();
   182→      }).catch(function() {
   183→        _db = _emptyDB();
   184→        _ensureStructure();
   185→        _ready = true;
   186→        resolve();
   187→      });
   188→    } else {
   189→      _db = _emptyDB();
   190→      _ensureStructure();
   191→      _ready = true;
   192→      resolve();
   193→    }
   194→  });
   195→}

   404→        if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   405→        _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) + (Number(item.cantidad) || 0);

   412→          stk.stockChico = (stk.stockChico || 0) + (Number(item.cantidad) || 0);

   437→function producirEspecia(especiaId, talla, cantidad) {
   438→  _ensureStructure();
   439→  var esp = _db.especias[especiaId];
   440→  if (!esp) throw new Error('Especia no encontrada');
   441→  talla = (talla === 'grande') ? 'grande' : 'chico';
   442→  cantidad = Number(cantidad) || 0;
   443→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   444→
   445→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   446→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   447→
   448→  var grsTotal = gramosPorFrasco * cantidad;
   449→
   450→  // Check & consume pala (raw material)
   451→  if ((esp.stockBolsa || 0) < grsTotal) {
   452→    throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   453→  }
   454→
   455→  // Check & consume envases
   456→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   457→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   458→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   459→  }
   460→
   461→  // Check & consume stickers
   462→  var stk = _findStickerByNombre(esp.nombre);
   463→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   464→  if (stkStock < cantidad) {
   465→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   466→  }
   467→
   468→  // Check & consume bolsas (packaging)
   469→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   470→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   471→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   472→  }
   473→
   474→  // All checks passed — consume
   475→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   476→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   477→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   478→  if (stk) {
   479→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   480→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   481→  }
   482→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   483→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   484→
   485→  // Record
   486→  var prodId = nextId('producciones');

   501→function producirBlend(blendId, talla, cantidad) {
   502→  _ensureStructure();
   503→  var blend = _db.blends[blendId];
   504→  if (!blend) throw new Error('Blend no encontrado');
   505→  talla = (talla === 'grande') ? 'grande' : 'chico';
   506→  cantidad = Number(cantidad) || 0;
   507→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   508→
   509→  var ingredientes = blend.ingredientes || [];
   510→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   511→
   512→  // Check ingredient stock
   513→  var detalleIngredientes = [];
   514→  for (var i = 0; i < ingredientes.length; i++) {
   515→    var ing = ingredientes[i];
   516→    var esp = _db.especias[ing.especiaId];
   517→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   518→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   519→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   520→    var grsNeeded = grsPorFrasco * cantidad;
   521→    if ((esp.stockBolsa || 0) < grsNeeded) {
   522→      throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   523→    }
   524→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   525→  }
   526→
   527→  // Check envases
   528→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   529→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   530→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   531→  }
   532→
   533→  // Check stickers
   534→  var stk = _findStickerByNombre(blend.nombre);
   535→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   536→  if (stkStock < cantidad) {
   537→    throw new Error('Stickers ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   538→  }
   539→
   540→  // Check & consume bolsas (packaging)
   541→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   542→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   543→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   544→  }
   545→
   546→  // All checks passed — consume
   547→  var grsTotalGeneral = 0;
   548→  for (var i = 0; i < detalleIngredientes.length; i++) {
   549→    var d = detalleIngredientes[i];
   550→    var esp = _db.especias[d.especiaId];
   551→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   552→    grsTotalGeneral += d.gramosTotal;
   553→  }
   554→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   555→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   556→  if (stk) {
   557→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   558→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   559→  }
   560→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   561→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   562→
   563→  var prodId = nextId('producciones');
   564→  var prod = {
   565→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   566→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   567→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   568→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   569→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   570→  };
   571→  _db.producciones[prodId] = prod;
   572→  _saveToFirebase(); _cacheLocal();
   573→  _notify('create', 'producciones', prodId);
   574→  _notify('update', 'blends', blendId);
   575→  return { producto: blend, produccion: prod };
   576→}

   588→function saveVenta(data) {
   589→  _ensureStructure();
   590→  var isNew = !data.id;
   591→  if (isNew) {
   592→    data.id = nextId('ventas');
   593→    data.creado = new Date().toISOString();
   594→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   595→    data.items = data.items || [];
   596→    data.total = Number(data.total) || 0;
   597→  }
   598→  if (isNew) {
   599→    for (var i = 0; i < data.items.length; i++) {
   600→      var item = data.items[i];
   601→      var producto;
   602→      if (item.tipo === 'especia') {
   603→        producto = _db.especias[item.productoId];
   604→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   605→      } else {
   606→        producto = _db.blends[item.productoId];
   607→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   608→      }
   609→      item.productoNombre = producto.nombre;
   610→      var cant = Number(item.cantidad) || 0;
   611→      var talla = item.talla || 'chico';
   612→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   613→      if ((producto[stockKey] || 0) < cant) {
   614→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   615→      }
   616→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   617→      item.precioUnitario = Number(item.precioUnitario) || 0;
   618→      item.subtotal = item.precioUnitario * cant;
   619→    }
   620→    // Recalculate total
   621→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   622→  }
   623→  _db.ventas[data.id] = data;
   624→  _saveToFirebase(); _cacheLocal();
   625→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   626→  return data;
   627→}

   784→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   785→function findEspeciaByName(nombre) {
   786→  if (!nombre) return null;
   787→  var target = nombre.trim().toLowerCase();
   788→  var keys = Object.keys(_db.especias || {});
   789→  // 1. Exact match
   790→  for (var i = 0; i < keys.length; i++) {
   791→    var e = _db.especias[keys[i]];
   792→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   793→  }
   794→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   795→  for (var i = 0; i < keys.length; i++) {
   796→    var e = _db.especias[keys[i]];
   797→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   798→  }
   799→  // 3. Target is contained in especia name
   800→  for (var i = 0; i < keys.length; i++) {
   801→    var e = _db.especias[keys[i]];
   802→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   803→  }
   804→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   805→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   806→  for (var w = 0; w < targetWords.length; w++) {
   807→    for (var i = 0; i < keys.length; i++) {
   808→      var e = _db.especias[keys[i]];
   809→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   810→    }
   811→  }
   812→  return null;
   813→}

   815→/** Auto-detect categoria from blend USO field */
   816→function _categoriaFromUso(uso) {
   817→  if (!uso) return 'Comidas';
   818→  var u = uso.toLowerCase();
   819→  // Cocteleria keywords
   820→  if (/\b(gin|ron|vodka|whisky|mojito|mule|vermouth|aperitif|coctel)\b/.test(u)) return 'Cocteleria';
   821→  // Infusiones keywords
   822→  if (/\b(relajant|sueño|digestiv|energiz|té|calidez|respiratorio|meditaci|antioxidant|bienestar|infusi)\b/.test(u)) return 'Infusiones';
   823→  return 'Comidas';
   824→}

   826→/**
   827→ * Import especias and blends from parsed Excel data.
   828→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   829→ *
   830→ * especiasList: [{ nombre, categoria? }]
   831→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   832→ * gramosChico:  default grams for frasco chico (e.g. 30)
   833→ * gramosGrande: default grams for frasco grande (e.g. 80)
   834→ */
   835→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   836→  _ensureStructure();
   837→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   838→
   839→  // 1. Create especias (skip if name already exists)
   840→  for (var i = 0; i < especiasList.length; i++) {
   841→    var esp = especiasList[i];
   842→    var nombre = (esp.nombre || '').trim();
   843→    if (!nombre) continue;
   844→    var existing = findEspeciaByName(nombre);
   845→    if (existing) {
   846→      resultado.especiasExistentes++;
   847→      continue;
   848→    }
   849→    saveEspecia({
   850→      nombre: nombre,
   851→      categoria: esp.categoria || 'Comidas',
   852→      precioChico: 0,
   853→      precioGrande: 0,
   854→      gramosChico: Number(gramosChico) || 30,
   855→      gramosGrande: Number(gramosGrande) || 80,
   856→      stockBolsa: 0,
   857→      stockChico: 0,
   858→      stockGrande: 0
   859→    });
   860→    resultado.especiasCreadas++;
   861→  }
   862→
   863→  // 2. Create blends (skip if name already exists)
   864→  for (var j = 0; j < blendsList.length; j++) {
   865→    var bl = blendsList[j];
   866→    var nombre = (bl.nombre || '').trim();
   867→    if (!nombre) continue;
   868→    var existingBlend = null;
   869→    var blKeys = Object.keys(_db.blends || {});
   870→    for (var k = 0; k < blKeys.length; k++) {
   871→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
   872→        existingBlend = _db.blends[blKeys[k]];
   873→        break;
   874→      }
   875→    }
   876→    if (existingBlend) {
   877→      resultado.blendsExistentes++;
   878→      continue;
   879→    }
   880→
   881→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
   882→    var ings = bl.ingredientes || [];
   883→    var recipeTotal = 0;
   884→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
   885→    if (recipeTotal <= 0) recipeTotal = 500;
   886→
   887→    var resolvedIngs = [];
   888→    for (var ii = 0; ii < ings.length; ii++) {
   889→      var ing = ings[ii];
   890→      var espObj = findEspeciaByName(ing.especia);
   891→      if (!espObj) {
   892→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
   893→        continue;
   894→      }
   895→      var ingG = Number(ing.g) || 0;
   896→      resolvedIngs.push({
   897→        especiaId: espObj.id,
   898→        especiaNombre: espObj.nombre,
   899→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
   900→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
   901→        gramosReceta: ingG
   902→      });
   903→    }
   904→
   905→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
   906→    saveBlend({
   907→      nombre: nombre,
   908→      categoria: cat,
   909→      region: bl.region || '',

   935→function addProductTag(cat, tagName) {
   936→  _ensureStructure();
   937→  tagName = (tagName || '').trim();
   938→  if (!tagName) return false;
   939→  if (!_db.productTags) _db.productTags = {};
   940→  if (!_db.productTags[cat]) _db.productTags[cat] = [];
   941→  // Check duplicate (case-insensitive)
   942→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   943→    if (_db.productTags[cat][i].toLowerCase() === tagName.toLowerCase()) return false;
   944→  }
   945→  _db.productTags[cat].push(tagName);
   946→  _saveToFirebase(); _cacheLocal();
   947→  return true;
   948→}
   949→
   950→function removeProductTag(cat, tagName) {
   951→  _ensureStructure();
   952→  if (!_db.productTags || !_db.productTags[cat]) return false;
   953→  var idx = -1;
   954→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   955→    if (_db.productTags[cat][i] === tagName) { idx = i; break; }
   956→  }
   957→  if (idx < 0) return false;
   958→  _db.productTags[cat].splice(idx, 1);
   959→  // Also remove from all products that have this tag
   960→  var allEsp = getEspecias();
   961→  for (var e = 0; e < allEsp.length; e++) {
   962→    if (allEsp[e].tags) {
   963→      var ti = allEsp[e].tags.indexOf(tagName);
   964→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
   965→    }
   966→  }
   967→  var allBl = getBlends();
   968→  for (var b = 0; b < allBl.length; b++) {
   969→    if (allBl[b].tags) {
   970→      var bi = allBl[b].tags.indexOf(tagName);
   971→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
   972→    }
   973→  }
   974→  _saveToFirebase(); _cacheLocal();
   975→  return true;
   976→}

   980→function compressImage(file, maxW, quality, cb) {
   981→  var reader = new FileReader();
   982→  reader.onload = function(ev) {
   983→    var img = new Image();
   984→    img.onload = function() {
   985→      var w = img.width, h = img.height;
   986→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
   987→      var canvas = document.createElement('canvas');
   988→      canvas.width = w; canvas.height = h;
   989→      var ctx = canvas.getContext('2d');
   990→      ctx.drawImage(img, 0, 0, w, h);
   991→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
   992→      cb(null, dataUrl);
   993→    };
   994→    img.onerror = function() { cb('Error al cargar imagen'); };
   995→    img.src = ev.target.result;
   996→  };
   997→  reader.onerror = function() { cb('Error al leer archivo'); };
   998→  reader.readAsDataURL(file);
   999→}
