    26→function renderStoreProducts() {
    27→  var db = getDB();
    28→  var products = [];
    29→  var container = document.getElementById('store-products');
    30→  var empty = document.getElementById('store-empty');
    31→
    32→  // Get sellable especias (with precioVenta > 0 and stock > 0)
    33→  var especias = (db.productos || []).filter(function(p) {
    34→    return p.tipo === 'especia' && p.precioVenta > 0 && p.stock > 0;
    35→  });
    36→  especias.forEach(function(p) {
    37→    // Filter by storeCategory
    38→    if (storeCategory !== 'todos' && storeCategory !== 'blends') {
    39→      if ((p.categoria || '') !== storeCategory) return;
    40→    }
    41→    products.push({
    42→      id: p.id, nombre: p.nombre, tipo: p.categoria || 'Especia',
    43→      desc: (p.usoPrincipal || p.notas || ''), precio: p.precioVenta,
    44→      stock: p.stock, unidad: p.unidad, blend: false,
    45→      origen: p.origen || '', perfil: p.perfil || ''
    46→    });
    47→  });
    48→
    49→  // Get blends with stock > 0 (only when "todos" or "blends" selected)
    50→  if (storeCategory === 'todos' || storeCategory === 'blends') {
    51→    var blends = (db.blends || []).filter(function(b) {
    52→      return b.stock > 0 && b.precioVenta > 0;
    53→    });
    54→    blends.forEach(function(b) {
    55→      products.push({
    56→        id: b.id, nombre: b.nombre, tipo: 'Blend',
    57→        desc: b.descripcion || '', precio: b.precioVenta,
    58→        stock: b.stock, unidad: 'unidad', blend: true,
    59→        origen: '', perfil: ''
    60→      });
    61→    });
    62→  }
    63→
    64→  if (products.length === 0) {
    65→    container.innerHTML = '';
    66→    empty.style.display = 'block';
    67→    return;
    68→  }
    69→  empty.style.display = 'none';
    70→
    71→  var html = '';
    72→  products.forEach(function(p) {
    73→    html += '<div class="product-card">' +
    74→      '<div class="product-card-type">' + esc(p.tipo) + (p.origen ? ' &mdash; ' + esc(p.origen) : '') + '</div>' +
    75→      '<div class="product-card-name">' + esc(p.nombre) + '</div>' +
    76→      (p.perfil ? '<div class="product-card-meta">' + esc(p.perfil) + '</div>' : '') +
    77→      (p.desc ? '<div class="product-card-desc">' + esc(p.desc) + '</div>' : '<div class="product-card-desc" style="opacity:.3">Sin descripción</div>') +
    78→      '<div class="product-card-footer">' +
    79→        '<div><div class="product-card-price">' + fmt(p.precio) + '</div>' +
    80→        '<div class="product-card-stock">' + p.stock + ' ' + esc(p.unidad) + ' disponibles</div></div>' +
    81→        '<button class="btn-add-cart" onclick="addToCart(' + (p.blend ? 'true' : 'false') + ',' + p.id + ',\'' + esc(p.nombre).replace(/'/g,"\\'") + '\',' + p.precio + ',\'' + esc(p.unidad) + '\',' + p.stock + ')">+ Agregar</button>' +
    82→      '</div></div>';
    83→  });
    84→  container.innerHTML = html;
    85→}

    87→function addToCart(isBlend, id, nombre, precio, unidad, stock) {
    88→  var existing = cart.find(function(c) { return c.isBlend === isBlend && c.id === id; });
    89→  if (existing) {
    90→    if (existing.qty >= stock) { toast('No hay más stock disponible', 'err'); return; }
    91→    existing.qty++;
    92→  } else {
    93→    cart.push({ isBlend: isBlend, id: id, nombre: nombre, precio: precio, unidad: unidad, qty: 1, maxStock: stock });
    94→  }
    95→  updateCartUI();
    96→  toast(nombre + ' agregado al carrito');
    97→}

    99→function updateCartUI() {
   100→  var countEl = document.getElementById('cart-count');
   101→  var total = 0;
   102→  var count = 0;
   103→  cart.forEach(function(c) { total += c.precio * c.qty; count += c.qty; });
   104→
   105→  if (count > 0) {
   106→    countEl.style.display = 'flex';
   107→    countEl.textContent = count;
   108→  } else {
   109→    countEl.style.display = 'none';
   110→  }
   111→
   112→  var itemsEl = document.getElementById('cart-items');
   113→  var footerEl = document.getElementById('cart-footer');
   114→
   115→  if (cart.length === 0) {
   116→    itemsEl.innerHTML = '<div class="cart-empty">Tu carrito está vacío</div>';
   117→    footerEl.style.display = 'none';
   118→    return;
   119→  }
   120→
   121→  footerEl.style.display = 'block';
   122→  document.getElementById('cart-total-val').textContent = fmt(total);
   123→
   124→  var html = '';
   125→  cart.forEach(function(c, i) {
   126→    html += '<div class="cart-item">' +
   127→      '<div class="cart-item-info"><div class="cart-item-name">' + esc(c.nombre) + '</div>' +
   128→      '<div class="cart-item-price">' + fmt(c.precio) + ' / ' + esc(c.unidad) + '</div></div>' +
   129→      '<div class="cart-item-qty">' +
   130→        '<button onclick="cartQty(' + i + ',-1)">-</button>' +
   131→        '<span>' + c.qty + '</span>' +
   132→        '<button onclick="cartQty(' + i + ',1)">+</button>' +
   133→      '</div></div>';
   134→  });
   135→  itemsEl.innerHTML = html;
   136→}
   137→
   138→function cartQty(idx, delta) {
   139→  if (!cart[idx]) return;
   140→  cart[idx].qty += delta;
   141→  if (cart[idx].qty <= 0) cart.splice(idx, 1);
   142→  else if (cart[idx].qty > cart[idx].maxStock) { cart[idx].qty = cart[idx].maxStock; toast('Stock máximo alcanzado', 'err'); }
   143→  updateCartUI();
   144→}

   181→function submitOrder() {
   182→  var nombre = document.getElementById('chk-nombre').value.trim();
   183→  var tel = document.getElementById('chk-tel').value.trim();
   184→  var dir = document.getElementById('chk-dir').value.trim();
   185→  var notas = document.getElementById('chk-notas').value.trim();
   186→  var alertEl = document.getElementById('checkout-alert');
   187→
   188→  if (!nombre) {
   189→    alertEl.textContent = 'Ingresa tu nombre';
   190→    alertEl.className = 'alert alert-err show';
   191→    return;
   192→  }
   193→
   194→  var db = getDB();
   195→  var items = [];
   196→  var total = 0;
   197→
   198→  cart.forEach(function(c) {
   199→    var sub = c.precio * c.qty;
   200→    total += sub;
   201→    items.push({
   202→      tipo: c.isBlend ? 'blend' : 'especia',
   203→      id: c.id,
   204→      nombre: c.nombre,
   205→      cantidad: c.qty,
   206→      unidad: c.unidad,
   207→      precioUnitario: c.precio,
   208→      subtotal: sub
   209→    });
   210→  });
   211→
   212→  var venta = {
   213→    id: nextId(),
   214→    fecha: new Date().toISOString(),
   215→    clienteNombre: nombre,
   216→    clienteTelefono: tel,
   217→    clienteDireccion: dir,
   218→    items: items,
   219→    total: total,
   220→    estado: 'pendiente',
   221→    tipo: 'tienda', // marks as storefront order
   222→    creadoPor: null,
   223→    creadoEn: new Date().toISOString()
   224→  };
   225→
   226→  db.ventas.push(venta);
   227→
   228→  // Update stock
   229→  items.forEach(function(it) {
   230→    if (it.tipo === 'especia') {
   231→      var prod = db.productos.find(function(p) { return p.id === it.id; });
   232→      if (prod) {
   233→        prod.stock = Math.max(0, prod.stock - it.cantidad);
   234→        prod.actualizadoEn = new Date().toISOString();
   235→        addMovimiento('venta', 'ventas', venta.id, it.nombre, -it.cantidad, 'Pedido de tienda');
   236→      }
   237→    } else if (it.tipo === 'blend') {
   238→      var blend = db.blends.find(function(b) { return b.id === it.id; });
   239→      if (blend) {
   240→        blend.stock = Math.max(0, blend.stock - it.cantidad);
   241→        blend.actualizadoEn = new Date().toISOString();
   242→        addMovimiento('venta', 'ventas', venta.id, it.nombre, -it.cantidad, 'Pedido de tienda');
   243→      }
   244→    }
   245→  });
   246→
   247→  saveDB();
   248→  cart = [];
   249→  closeModal();
   250→  renderStoreProducts();
   251→  updateCartUI();
   252→  toast('Pedido realizado con éxito');
   253→}

   258→function renderDashboard() {
   259→  if (!currentUser || currentUser.rol !== 'admin') return;
   260→  var db = getDB();
   261→  var el = document.getElementById('page-dashboard');
   262→  var productos = db.productos || [];
   263→  var compras = db.compras || [];
   264→  var blends = db.blends || [];
   265→  var ventas = db.ventas || [];
   266→  var movimientos = db.movimientos || [];
   267→
   268→  // Stats
   269→  var especias = productos.filter(function(p) { return p.tipo === 'especia'; });
   270→  var totalVentas = ventas.filter(function(v) { return v.estado !== 'cancelada'; }).reduce(function(s, v) { return s + (v.total || 0); }, 0);
   271→  var totalCompras = compras.filter(function(c) { return c.estado === 'recibida'; }).reduce(function(s, c) { return s + (c.total || 0); }, 0);
   272→  var ventasPendientes = ventas.filter(function(v) { return v.estado === 'pendiente'; }).length;
   273→  var lowStock = especias.filter(function(p) { return p.stock <= p.stockMin; });
   274→
   275→  var html = '<h1 class="page-title">Dashboard</h1>';
   276→
   277→  // KPI cards
   278→  html += '<div class="g4 mb-20">';
   279→  html += statCard('Ventas Totales', fmt(totalVentas), ventas.length + ' transacciones');
   280→  html += statCard('Compras', fmt(totalCompras), compras.length + ' órdenes');
   281→  html += statCard('Productos', especias.length + ' especias', productos.length + ' total items');
   282→  html += statCard('Blends', blends.length + ' mezclas', 'Activos');
   283→  html += '</div>';
   284→
   285→  // Alerts
   286→  if (lowStock.length > 0 || ventasPendientes > 0) {
   287→    html += '<div class="section-title mb-12">Alertas</div>';
   288→    if (ventasPendientes > 0) {
   289→      html += '<div class="card mb-12" style="border-color:var(--yellow)">' +
   290→        '<div class="flex items-center gap-8"><span style="font-size:1.2rem">\u26A0</span>' +
   291→        '<div><div class="fw7 text-gold">' + ventasPendientes + ' ventas pendientes</div>' +
   292→        '<div class="text-xs text-muted">Requieren atención</div></div>' +
   293→        '<button class="btn btn-gold btn-sm" style="margin-left:auto" onclick="navigateTo(\'ventas\')">Ver Ventas</button></div></div>';
   294→    }
   295→    if (lowStock.length > 0) {
   296→      html += '<div class="card mb-20" style="border-color:var(--red)">' +
   297→        '<div class="flex items-center gap-8"><span style="font-size:1.2rem">\u26A0</span>' +
   298→        '<div><div class="fw7 text-red">' + lowStock.length + ' productos con stock bajo</div>' +
   299→        '<div class="text-xs text-muted">' + lowStock.slice(0, 3).map(function(p) { return p.nombre; }).join(', ') + (lowStock.length > 3 ? '...' : '') + '</div></div>' +
   300→        '<button class="btn btn-ghost btn-sm" style="margin-left:auto" onclick="navigateTo(\'productos\')">Ver Productos</button></div></div>';
   301→    }
   302→  }
   303→
   304→  // Recent sales
   305→  html += '<div class="section-title mb-12">Ventas Recientes</div>';
   306→  var recentVentas = ventas.slice().sort(function(a, b) { return new Date(b.creadoEn) - new Date(a.creadoEn); }).slice(0, 5);
   307→  if (recentVentas.length === 0) {
   308→    html += '<div class="card"><div class="empty"><div class="empty-icon">\u2193</div><p>No hay ventas registradas</p></div></div>';
   309→  } else {
   310→    html += '<div class="card"><div class="tw"><table><tr><th>Fecha</th><th>Cliente</th><th>Total</th><th>Estado</th></tr>';
   311→    recentVentas.forEach(function(v) {
   312→      var estadoBadge = v.estado === 'completada' ? 'bg' : (v.estado === 'pendiente' ? 'by' : 'br');

   314→        '<td>' + esc(v.clienteNombre || (v.tipo === 'tienda' ? 'Tienda' : 'Mostrador')) + '</td>' +
   315→        '<td class="text-gold fw7">' + fmt(v.total) + '</td>' +
   316→        '<td><span class="badge ' + estadoBadge + '">' + esc(v.estado) + '</span></td></tr>';
   317→    });
   318→    html += '</table></div></div>';
   319→  }
   320→
   321→  // Recent movements
   322→  html += '<div class="section-title mt-16 mb-12">Movimientos Recientes</div>';
   323→  var recentMovs = movimientos.slice().sort(function(a, b) { return new Date(b.fecha) - new Date(a.fecha); }).slice(0, 8);
   324→  if (recentMovs.length === 0) {
   325→    html += '<div class="card"><div class="empty"><div class="empty-icon">\u21C4</div><p>No hay movimientos</p></div></div>';
   326→  } else {

   333→        '<span class="mov-date">' + fmtDateTime(m.fecha) + '</span></div>';
   334→    });
   335→    html += '</div>';
   336→  }
   337→
   338→  el.innerHTML = html;
   339→}

   426→function cmpItemChanged(el) {
   427→  var row = el.closest('.ing-row');
   428→  var select = row.querySelector('select');
   429→  var qtyInput = row.querySelectorAll('input')[0];
   430→  var priceInput = row.querySelectorAll('input')[1];
   431→  var costEl = row.querySelector('.ing-cost');

   747→function calcBlendCost(blend) {
   748→  if (!blend || !blend.receta) return 0;
   749→  var db = getDB();
   750→  var costo = 0;
   751→  blend.receta.forEach(function(r) {
   752→    var prod = (db.productos || []).find(function(p) { return p.id === r.productoId; });
   753→    if (prod) {
   754→      // Convert to same unit for cost calc (simplified: assumes recipe unit matches product unit)
   755→      var unitCost = prod.precioCosto / 1; // cost per base unit
   756→      // If product is in kg and recipe in gr, convert
   757→      var factor = 1;
   758→      if (prod.unidad === 'kg' && r.unidad === 'gr') factor = 0.001;
   759→      else if (prod.unidad === 'L' && r.unidad === 'ml') factor = 0.001;
   760→      costo += r.cantidad * unitCost * factor;
   761→    }
   762→  });
   763→  return Math.round(costo);
   764→}

   927→function ejecutarProduccion(blendId) {

   929→  if (cant < 1) return;

   931→  var db = getDB();

   940→      toast('Stock insuficiente de ' + r.nombre, 'err');

   944→
   945→  // Deduct stock from ingredients
   946→  (b.receta || []).forEach(function(r) {
   947→    var needed = r.cantidad * cant;
   948→    var prod = db.productos.find(function(p) { return p.id === r.productoId; });

   950→      prod.stock -= needed;
   951→      prod.actualizadoEn = new Date().toISOString();

   953→    }
   954→  });
   955→
   956→  b.stock += cant;
   957→  b.actualizadoEn = new Date().toISOString();
   958→  saveDB();
   959→  closeModal();
   960→  renderBlends();
   961→  toast(cant + ' unidad(es) de ' + b.nombre + ' producida(s)');

  1429→function handleExcelImport(input) {
  1430→  var file = input.files[0];
  1431→  if (!file) return;
  1432→
  1433→  var reader = new FileReader();
  1434→  reader.onload = function(e) {
  1435→    try {
  1436→      var data = new Uint8Array(e.target.result);
  1437→      var workbook = XLSX.read(data, { type: 'array' });
  1438→      var firstSheet = workbook.Sheets[workbook.SheetNames[0]];
  1439→      var rows = XLSX.utils.sheet_to_json(firstSheet, { defval: '' });
  1440→
  1441→      if (rows.length === 0) {
  1442→        toast('El archivo está vacío', 'err');
  1443→        return;
  1444→      }
  1445→
  1446→      // Detect column mapping
  1447→      var headers = Object.keys(rows[0]);
  1448→      var colMap = detectColumnMap(headers);
  1449→
  1450→      if (!colMap.nombre) {
  1451→        toast('No se encontró columna de nombre (Blend/Nombre/Producto)', 'err');
  1452→        return;
  1453→      }
  1454→
  1455→      // Show preview before importing
  1456→      showImportPreview(rows, colMap);
  1457→
  1458→    } catch(err) {
  1459→      toast('Error al leer el archivo: ' + err.message, 'err');
  1460→      console.error('[Import]', err);
  1461→    }
  1462→  };
  1463→  reader.readAsArrayBuffer(file);
  1464→  input.value = ''; // Reset so same file can be re-selected
  1465→}

  1467→function detectColumnMap(headers) {
  1468→  var map = { nombre: null, categoria: null, origen: null, uso: null, perfil: null, precioCosto: null, precioVenta: null, stock: null, unidad: null };
  1469→
  1470→  headers.forEach(function(h) {
  1471→    var hl = h.toLowerCase().trim();
  1472→    if (['blend', 'nombre', 'producto', 'name', 'especia', 'item'].indexOf(hl) !== -1) map.nombre = h;
  1473→    else if (['categoría', 'categoria', 'category', 'tipo', 'type'].indexOf(hl) !== -1) map.categoria = h;
  1474→    else if (['origen', 'origin', 'país', 'pais', 'procedencia'].indexOf(hl) !== -1) map.origen = h;
  1475→    else if (['uso principal', 'uso', 'use', 'aplicación', 'aplicacion'].indexOf(hl) !== -1) map.uso = h;
  1476→    else if (['perfil', 'profile', 'sabor', 'flavor'].indexOf(hl) !== -1) map.perfil = h;
  1477→    else if (['precio costo', 'costo', 'precio_compra', 'cost', 'precio de costo'].indexOf(hl) !== -1) map.precioCosto = h;
  1478→    else if (['precio venta', 'venta', 'precio', 'price', 'precio de venta'].indexOf(hl) !== -1) map.precioVenta = h;
  1479→    else if (['stock', 'cantidad', 'inventario', 'inventory', 'existencia'].indexOf(hl) !== -1) map.stock = h;
  1480→    else if (['unidad', 'unit', 'medida', 'um'].indexOf(hl) !== -1) map.unidad = h;
  1481→  });
  1482→
  1483→  return map;

  1529→function executeImport() {
  1530→  var rows = window._importRows;
  1531→  var colMap = window._importColMap;
  1532→  var tipo = document.getElementById('import-tipo').value;
  1533→  var dupeAction = document.getElementById('import-dupes').value;
  1534→  var alertEl = document.getElementById('import-alert');
  1535→
  1536→  if (!rows || rows.length === 0) return;
  1537→
  1538→  var db = getDB();
  1539→  var now = new Date().toISOString();
  1540→  var imported = 0, updated = 0, skipped = 0;
  1541→
  1542→  // If overwrite, clear existing products of this type
  1543→  if (dupeAction === 'overwrite') {
  1544→    var before = db.productos.length;
  1545→    db.productos = db.productos.filter(function(p) { return p.tipo !== tipo; });
  1546→    toast('Eliminados ' + (before - db.productos.length) + ' productos de tipo ' + tipo);
  1547→  }
  1548→
  1549→  rows.forEach(function(r) {
  1550→    var nombre = String(r[colMap.nombre] || '').trim();
  1551→    if (!nombre) return;
  1552→
  1553→    var existing = db.productos.find(function(p) {
  1554→      return p.nombre.toLowerCase() === nombre.toLowerCase() && p.tipo === tipo;
  1555→    });
  1556→
  1557→    if (existing) {
  1558→      if (dupeAction === 'skip') { skipped++; return; }
  1559→      // Update existing
  1560→      if (colMap.categoria) existing.categoria = String(r[colMap.categoria] || '').trim();
  1561→      if (colMap.origen) existing.origen = String(r[colMap.origen] || '').trim();
  1562→      if (colMap.uso) existing.usoPrincipal = String(r[colMap.uso] || '').trim();
  1563→      if (colMap.perfil) existing.perfil = String(r[colMap.perfil] || '').trim();
  1564→      if (colMap.precioCosto) existing.precioCosto = parseFloat(r[colMap.precioCosto]) || existing.precioCosto;
  1565→      if (colMap.precioVenta) existing.precioVenta = parseFloat(r[colMap.precioVenta]) || existing.precioVenta;
  1566→      if (colMap.stock) existing.stock = parseFloat(r[colMap.stock]) || existing.stock;
  1567→      if (colMap.unidad) existing.unidad = String(r[colMap.unidad] || '').trim();
  1568→      existing.actualizadoEn = now;
  1569→      updated++;
  1570→    } else {
  1571→      var notas = [];
  1572→      if (colMap.uso) notas.push('Uso: ' + String(r[colMap.uso] || '').trim());
  1573→      if (colMap.perfil) notas.push('Perfil: ' + String(r[colMap.perfil] || '').trim());
  1574→
  1575→      var prod = {
  1576→        id: nextId(),
  1577→        nombre: nombre,
  1578→        tipo: tipo,
  1579→        unidad: colMap.unidad ? String(r[colMap.unidad] || 'gr').trim() : 'gr',
  1580→        precioCosto: colMap.precioCosto ? (parseFloat(r[colMap.precioCosto]) || 0) : 0,
  1581→        precioVenta: colMap.precioVenta ? (parseFloat(r[colMap.precioVenta]) || 0) : 0,
  1582→        stock: colMap.stock ? (parseFloat(r[colMap.stock]) || 0) : 0,
  1583→        stockMin: 0,
  1584→        proveedor: '',
  1585→        notas: notas.join(' | '),
  1586→        categoria: colMap.categoria ? String(r[colMap.categoria] || '').trim() : '',
  1587→        origen: colMap.origen ? String(r[colMap.origen] || '').trim() : '',
  1588→        usoPrincipal: colMap.uso ? String(r[colMap.uso] || '').trim() : '',
  1589→        perfil: colMap.perfil ? String(r[colMap.perfil] || '').trim() : '',
  1590→        creadoEn: now,
  1591→        actualizadoEn: now
  1592→      };
  1593→      db.productos.push(prod);
  1594→      imported++;
  1595→    }
  1596→  });
  1597→
  1598→  saveDB();
  1599→  closeModal();
  1600→  renderAjustes();
  1601→  toast('Importados: ' + imported + ' | Actualizados: ' + updated + ' | Ignorados: ' + skipped);
  1602→}

  1640→        'ID': b.id,
  1641→        'Nombre': b.nombre,
  1642→        'Descripción': b.descripcion || '',
  1643→        'Receta': recetaStr,
  1644→        'Costo Unitario': b.costoUnitario || calcBlendCost(b),
  1645→        'Precio Venta': b.precioVenta || 0,
  1646→        'Stock': b.stock || 0
  1647→      };
  1648→    });
  1649→    var wsBlend = XLSX.utils.json_to_sheet(blends);
  1650→    wsBlend['!cols'] = [{ wch: 6 }, { wch: 30 }, { wch: 30 }, { wch: 60 }, { wch: 16 }, { wch: 14 }, { wch: 10 }];
  1651→    XLSX.utils.book_append_sheet(wb, wsBlend, 'Blends');
  1652→  }
  1653→
  1654→  if (tipo === 'ventas' || tipo === 'todo') {
  1655→    var ventas = (db.ventas || []).map(function(v) {
  1656→      var itemsStr = (v.items || []).map(function(i) { return i.nombre + ' x' + i.cantidad + ' (' + fmt(i.subtotal) + ')'; }).join(', ');
  1657→      return {
  1658→        'ID': v.id,
  1659→        'Fecha': fmtDateTime(v.fecha || v.creadoEn),
  1660→        'Tipo': v.tipo === 'tienda' ? 'Tienda' : 'Mostrador',
  1661→        'Cliente': v.clienteNombre || '',
  1662→        'Teléfono': v.clienteTelefono || '',
  1663→        'Dirección': v.clienteDireccion || '',

  1665→        'Total': v.total || 0,
  1666→        'Estado': v.estado || '',

  1671→    wsVentas['!cols'] = [{ wch: 6 }, { wch: 18 }, { wch: 12 }, { wch: 20 }, { wch: 16 }, { wch: 24 }, { wch: 60 }, { wch: 14 }, { wch: 12 }, { wch: 12 }];

  1674→

  1678→      var estado = p.stock <= (p.stockMin || 0) ? 'BAJO' : 'OK';

  1681→        'Tipo': p.tipo,

  1688→        'Valor en Stock': (p.stock || 0) * (p.precioCosto || 0)
  1689→      });
  1690→    });
  1691→    (db.blends || []).forEach(function(b) {
  1692→      stockRows.push({
  1693→        'Nombre': b.nombre,
  1694→        'Tipo': 'Blend',
  1695→        'Categoría': '',
  1696→        'Stock Actual': b.stock || 0,
  1697→        'Unidad': 'unidad',
  1698→        'Stock Mínimo': 0,
  1699→        'Estado': b.stock > 0 ? 'OK' : 'SIN STOCK',
  1700→        'Precio Venta': b.precioVenta || 0,
  1701→        'Valor en Stock': (b.stock || 0) * (b.costoUnitario || 0)

  1703→    });

  1705→    wsStock['!cols'] = [{ wch: 30 }, { wch: 10 }, { wch: 14 }, { wch: 12 }, { wch: 8 }, { wch: 12 }, { wch: 12 }, { wch: 14 }, { wch: 16 }];

  1708→
  1709→  if (tipo === 'todo') {
  1710→    // Also add a summary sheet
  1711→    var totalVentas = (db.ventas || []).filter(function(v) { return v.estado !== 'cancelada'; }).reduce(function(s, v) { return s + (v.total || 0); }, 0);

  1714→      { 'Métrica': 'Total Productos', 'Valor': (db.productos || []).length },
  1715→      { 'Métrica': 'Total Blends', 'Valor': (db.blends || []).length },
  1716→      { 'Métrica': 'Total Ventas', 'Valor': (db.ventas || []).length },
  1717→      { 'Métrica': 'Total Compras', 'Valor': (db.compras || []).length },
  1718→      { 'Métrica': 'Ventas ($)', 'Valor': totalVentas },
  1719→      { 'Métrica': 'Compras ($)', 'Valor': totalCompras },
  1720→      { 'Métrica': 'Fecha Exportación', 'Valor': fecha }
  1721→    ];
  1722→    var wsSum = XLSX.utils.json_to_sheet(summary);
  1723→    wsSum['!cols'] = [{ wch: 24 }, { wch: 18 }];
  1724→    XLSX.utils.book_append_sheet(wb, wsSum, 'Resumen');
  1725→  }
  1726→
  1727→  var fileName = 'Arcano_' + tipo + '_' + fecha + '.xlsx';
  1728→  XLSX.writeFile(wb, fileName);
  1729→  toast('Excel descargado: ' + fileName);
  1730→}
