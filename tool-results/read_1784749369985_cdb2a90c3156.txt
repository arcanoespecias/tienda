    17→  toast._t = setTimeout(function() { el.className = ''; }, 3000);

    62→function pinDigit(d) {
    63→  if (!window._pinVal) window._pinVal = '';
    64→  if (window._pinVal.length >= 4) return;
    65→  window._pinVal += d;
    66→  updatePinDots();
    67→  if (window._pinVal.length === 4) verifyPin();
    68→}
    69→
    70→function pinBack() { if (window._pinVal) { window._pinVal = window._pinVal.slice(0,-1); updatePinDots(); } }
    71→
    72→function updatePinDots() {
    73→  var dots = document.querySelectorAll('#pin-dots .dot');
    74→  var len = (window._pinVal||'').length;
    75→  for (var i = 0; i < dots.length; i++) dots[i].classList.toggle('filled', i < len);
    76→}

    78→function verifyPin() {
    79→  var user = DB.find('usuarios', window._pinTarget);
    80→  if (user && user.pin === window._pinVal) {
    81→    currentUser = user;
    82→    enterApp();
    83→  } else {
    84→    var err = document.getElementById('pin-error');
    85→    if (err) err.textContent = 'PIN incorrecto';
    86→    window._pinVal = '';
    87→    updatePinDots();
    88→  }
    89→}

   149→function renderDashboard() {
   150→  var el = document.getElementById('page-dashboard');
   151→  if (!el || !DB.data) return;
   152→  var d = DB.data;
   153→  var totalVentasMes = 0, totalComprasMes = 0, ventasMes = 0;
   154→  var now = new Date(), mes = now.getMonth(), anio = now.getFullYear();
   155→  (d.ventas||[]).forEach(function(v) { var vd=new Date(v.fecha); if(vd.getMonth()===mes&&vd.getFullYear()===anio){totalVentasMes+=v.total||0;ventasMes++;} });
   156→  (d.compras||[]).forEach(function(c) { var cd=new Date(c.fecha); if(cd.getMonth()===mes&&cd.getFullYear()===anio) totalComprasMes+=c.total||0; });
   157→
   158→  var lowStock = 0;
   159→  (d.especias||[]).forEach(function(e) { if((e.stock||0)<=(e.stockMin||0)) lowStock++; });
   160→  (d.blends||[]).forEach(function(b) { if((b.stock||0)<=5) lowStock++; });
   161→
   162→  var html = '<div class="page-header"><h2>Dashboard</h2></div>';
   163→  html += '<div class="g4 mb-16">' +
   164→    '<div class="stat-card"><div class="stat-value">'+(d.especias||[]).length+'</div><div class="stat-label">Especias</div></div>' +
   165→    '<div class="stat-card"><div class="stat-value">'+(d.blends||[]).length+'</div><div class="stat-label">Blends</div></div>' +
   166→    '<div class="stat-card"><div class="stat-value">'+ventasMes+'</div><div class="stat-label">Ventas del mes</div></div>' +
   167→    '<div class="stat-card"><div class="stat-value" style="color:'+(lowStock>0?'var(--red)':'var(--green)')+'">'+lowStock+'</div><div class="stat-label">Stock bajo</div></div></div>';
   168→  html += '<div class="g2 mb-16">' +
   169→    '<div class="stat-card"><div class="stat-value text-sm">'+fmt(totalVentasMes)+'</div><div class="stat-label">Ventas mes</div></div>' +
   170→    '<div class="stat-card"><div class="stat-value text-sm">'+fmt(totalComprasMes)+'</div><div class="stat-label">Compras mes</div></div></div>';
   171→
   172→  // Recent activity
   173→  var recent = [];
   174→  (d.compras||[]).slice(-5).reverse().forEach(function(c) { recent.push({txt:'Compra #'+c.id+' — '+esc(c.proveedor||''),date:c.fecha,cls:'badge-yellow'}); });
   175→  (d.ventas||[]).slice(-5).reverse().forEach(function(v) { recent.push({txt:'Venta #'+v.id+' — '+esc(v.vendedorNombre||''),date:v.fecha,cls:'badge-green'}); });
   176→  (d.produccion||[]).slice(-5).reverse().forEach(function(p) { recent.push({txt:'Produccion: '+esc(p.blendNombre||''),date:p.fecha,cls:'badge-blue'}); });
   177→  recent.sort(function(a,b) { return new Date(b.date)-new Date(a.date); });
   178→  recent = recent.slice(0,8);
   179→
   180→  if (recent.length) {
   181→    html += '<h3 class="text-sm text-muted mb-8">Actividad reciente</h3>';
   182→    recent.forEach(function(r) {
   183→      html += '<div class="card" style="padding:10px 14px;display:flex;align-items:center;justify-content:space-between">' +
   184→        '<div><span class="badge '+r.cls+' mb-8"></span> '+r.txt+'</div><div class="text-xs text-muted">'+fmtDateTime(r.date)+'</div></div>';
   185→    });
   186→  }
   187→  el.innerHTML = html;
   188→}

   208→    html += '<p class="empty-msg">No hay compras</p>';

   216→    });

   218→  }
   219→  el.innerHTML = html;
   220→}

   284→function calcCmpTotal() {
   285→  var rows = document.querySelectorAll('#cmp-items .compra-row');
   286→  var total = 0;
   287→  rows.forEach(function(row) {
   288→    var inputs = row.querySelectorAll('input[type="number"]');
   289→    var q = inputs[0] ? Number(inputs[0].value)||0 : 0;
   290→    var p = inputs[1] ? Number(inputs[1].value)||0 : 0;
   291→    var sub = q * p;
   292→    total += sub;
   293→    var sp = row.querySelector('.subtotal-disp');
   294→    if (sp) sp.textContent = fmt(sub);
   295→  });
   296→  var te = document.getElementById('cmp-total');
   297→  if (te) te.textContent = fmt(total);
   298→}

   300→function guardarCompra() {
   301→  var fecha = document.getElementById('cmp-fecha').value;
   302→  var proveedor = (document.getElementById('cmp-proveedor').value||'').trim();
   303→  var recibida = document.getElementById('cmp-recibida').checked;
   304→  fecha = fecha ? new Date(fecha+'T12:00:00').toISOString() : new Date().toISOString();
   305→
   306→  var rows = document.querySelectorAll('#cmp-items .compra-row');
   307→  var items = [], total = 0;
   308→
   309→  rows.forEach(function(row) {
   310→    var tipo = row.getAttribute('data-tipo');
   311→    var numInputs = row.querySelectorAll('input[type="number"]');
   312→
   313→    if (tipo === 'especia') {
   314→      var sel = row.querySelector('select');
   315→      var existId = sel ? Number(sel.value)||0 : 0;
   316→      var nombreInput = row.querySelector('input[name="nombre"]');
   317→      var nombre = nombreInput ? (nombreInput.value||'').trim() : '';
   318→      var cantGr = numInputs[0] ? Number(numInputs[0].value)||0 : 0;
   319→      var precioTotal = numInputs[1] ? Number(numInputs[1].value)||0 : 0;
   320→      if (cantGr <= 0 || precioTotal <= 0) return;
   321→
   322→      var prodId, prodNombre;
   323→      if (existId) {
   324→        prodId = existId;
   325→        var f = DB.find('especias', existId);
   326→        if (!f) return;
   327→        prodNombre = f.nombre;
   328→      } else {
   329→        if (!nombre) return;
   330→        var existing = null;
   331→        DB.data.especias.forEach(function(e) { if (e.nombre.toLowerCase()===nombre.toLowerCase()) existing=e; });
   332→        if (existing) { prodId=existing.id; prodNombre=existing.nombre; }
   333→        else {
   334→          prodId = DB.nextId('especias');
   335→          prodNombre = nombre;
   336→          DB.data.especias.push({id:prodId,nombre:nombre,stock:0,stockMin:0,costoPor1000gr:Math.round(precioTotal/cantGr*1000),precioVentaGr:0,categoria:'',proveedor:proveedor,creadoEn:new Date().toISOString()});
   337→        }
   338→      }
   339→      if (recibida) {
   340→        var e = DB.find('especias', prodId);
   341→        if (e) e.stock = Math.round(((e.stock||0)+cantGr)*100)/100;
   342→      }
   343→      total += precioTotal;
   344→      items.push({productoId:prodId,productoNombre:prodNombre,tipo:'especia',cantidad:cantGr,precioUnitario:Math.round(precioTotal/cantGr*100)/100,subtotal:precioTotal});
   345→
   346→    } else if (tipo === 'frasco') {
   347→      var frascoTipo = row.querySelector('select').value;
   348→      var cant = numInputs[0]?Number(numInputs[0].value)||0:0;
   349→      var pUnit = numInputs[1]?Number(numInputs[1].value)||0:0;
   350→      if (cant<=0||pUnit<=0) return;
   351→      var fname = frascoTipo==='grande'?'Frasco Grande':'Frasco Pequeno';
   352→      var frasco = DB.data.frascos.find(function(f){return f.tipo===frascoTipo;});
   353→      if (frasco && recibida) frasco.stock = (frasco.stock||0)+cant;
   354→      total += cant*pUnit;
   355→      items.push({productoId:frasco?frasco.id:0,productoNombre:fname,tipo:'frasco',cantidad:cant,precioUnitario:pUnit,subtotal:cant*pUnit});
   356→
   357→    } else if (tipo === 'etiqueta') {
   358→      var sel2 = row.querySelector('select');
   359→      var existId2 = sel2?Number(sel2.value)||0:0;
   360→      var nombreInput2 = row.querySelector('input[name="nombre"]');
   361→      var nombre2 = nombreInput2?(nombreInput2.value||'').trim():'';
   362→      var cant2 = numInputs[0]?Number(numInputs[0].value)||0:0;
   363→      var pUnit2 = numInputs[1]?Number(numInputs[1].value)||0:0;
   364→      if (cant2<=0||pUnit2<=0) return;
   365→      var prodId2, prodNombre2;
   366→      if (existId2) {
   367→        prodId2=existId2; var f2=DB.find('etiquetas',existId2); if(!f2)return; prodNombre2=f2.nombre;
   368→      } else {
   369→        if(!nombre2)return;
   370→        var ex2=null; DB.data.etiquetas.forEach(function(e){if(e.nombre.toLowerCase()===nombre2.toLowerCase())ex2=e;});
   371→        if(ex2){prodId2=ex2.id;prodNombre2=ex2.nombre;}
   372→        else{prodId2=DB.nextId('etiquetas');prodNombre2=nombre2;DB.data.etiquetas.push({id:prodId2,nombre:nombre2,stock:0,costoUnit:pUnit2,creadoEn:new Date().toISOString()});}
   373→      }
   374→      if(recibida){var et=DB.find('etiquetas',prodId2);if(et)et.stock=(et.stock||0)+cant2;}
   375→      total+=cant2*pUnit2;

   750→function guardarVenta() {
   751→  var blendId = Number(document.getElementById('vta-blend').value)||0;
   752→  var cant = Number(document.getElementById('vta-cant').value)||0;
   753→  var items = window._ventaItems || [];
   754→
   755→  // Add current item if selected
   756→  if (blendId && cant > 0) {
   757→    var b = DB.find('blends', blendId);
   758→    if (b) {
   759→      var frasco = document.getElementById('vta-frasco').value;
   760→      items.push({blendId:blendId,blendNombre:b.nombre,frascoTipo:frasco,cantidad:cant,precioUnit:b.precioVenta,subtotal:cant*b.precioVenta});
   761→    }
   762→  }
   763→  if (items.length===0) { toast('Agregue al menos un item','err'); return; }
   764→
   765→  // Check stock
   766→  items.forEach(function(item) {
   767→    var b = DB.find('blends', item.blendId);
   768→    if (!b || (b.stock||0) < item.cantidad) { toast('Stock insuficiente: '+item.blendNombre,'err'); }
   769→  });
   770→
   771→  var total = 0;
   772→  items.forEach(function(item) {
   773→    total += item.subtotal;
   774→    // Deduct blend stock
   775→    var b = DB.find('blends', item.blendId);
   776→    if (b) b.stock = Math.max(0, (b.stock||0) - item.cantidad);
   777→    // Deduct frasco stock
   778→    var frasco = DB.data.frascos.find(function(f) { return f.tipo === item.frascoTipo; });
   779→    if (frasco) frasco.stock = Math.max(0, (frasco.stock||0) - item.cantidad);
   780→  });
   781→
   782→  var venta = {id:DB.nextId('ventas'),fecha:new Date().toISOString(),items:items,total:total,vendedorId:currentUser?currentUser.id:null,vendedorNombre:currentUser?currentUser.nombre:'Tienda',creadoEn:new Date().toISOString()};
   783→  DB.data.ventas.push(venta);
   784→  DB.save();
   785→  closeModal();
   786→  toast('Venta #'+venta.id+' registrada');
   787→  renderVentas();
   788→}

   968→function submitOrder() {
   969→  if (cart.length === 0) return;
   970→  var items = [];
   971→  var total = 0;
   972→  cart.forEach(function(c) {
   973→    var subtotal = c.precio * c.cantidad;
   974→    total += subtotal;
   975→    if (c.tipo === 'blend') {
   976→      // Deduct blend stock
   977→      var b = DB.find('blends', c.blendId);
   978→      if (b) b.stock = Math.max(0, (b.stock||0)-c.cantidad);
   979→      // Deduct frasco (default pequeno for storefront)
   980→      var frasco = DB.data.frascos.find(function(f) { return f.tipo === 'pequeno'; });
   981→      if (frasco) frasco.stock = Math.max(0, (frasco.stock||0)-c.cantidad);
   982→      items.push({blendId:c.blendId,blendNombre:c.nombre,frascoTipo:'pequeno',cantidad:c.cantidad,precioUnit:c.precio,subtotal:subtotal});
   983→    } else {
   984→      // Especia directa sale
   985→      var e = DB.find('especias', c.especiaId);
   986→      if (e) e.stock = Math.max(0, (e.stock||0)-c.cantidad);
   987→      items.push({blendId:0,blendNombre:c.nombre,frascoTipo:'granel',cantidad:c.cantidad,precioUnit:c.precio,subtotal:subtotal});
   988→    }
   989→  });
   990→  var venta = {id:DB.nextId('ventas'),fecha:new Date().toISOString(),items:items,total:total,vendedorId:null,vendedorNombre:'Tienda Online',creadoEn:new Date().toISOString()};
   991→  DB.data.ventas.push(venta);
   992→  DB.save();
   993→  cart = [];
   994→  toggleCart();
   995→  updateCartUI();
   996→  toast('Pedido registrado! Venta #'+venta.id);
   997→  renderStorefront();
   998→}
