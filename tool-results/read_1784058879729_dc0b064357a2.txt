   736→function calcBlendCost(blend) {
   737→  if (!blend || !blend.receta) return 0;
   738→  var db = getDB();
   739→  var costo = 0;
   740→  blend.receta.forEach(function(r) {
   741→    var prod = (db.productos || []).find(function(p) { return p.id === r.productoId; });
   742→    if (prod) {
   743→      // Convert to same unit for cost calc (simplified: assumes recipe unit matches product unit)
   744→      var unitCost = prod.precioCosto / 1; // cost per base unit
   745→      // If product is in kg and recipe in gr, convert
   746→      var factor = 1;
   747→      if (prod.unidad === 'kg' && r.unidad === 'gr') factor = 0.001;
   748→      else if (prod.unidad === 'L' && r.unidad === 'ml') factor = 0.001;
   749→      costo += r.cantidad * unitCost * factor;
   750→    }
   751→  });
   752→  return Math.round(costo);
   753→}

   829→function guardarBlend(existingId) {
   830→  var nombre = document.getElementById('bl-nombre').value.trim();

   832→  var precio = parseFloat(document.getElementById('bl-precio').value) || 0;
   833→  var alertEl = document.getElementById('bl-alert');
   834→
   835→  if (!nombre) {
   836→    alertEl.textContent = 'El nombre es obligatorio';
   837→    alertEl.className = 'alert alert-err show';
   838→    return;
   839→  }
   840→
   841→  var db = getDB();
   842→  var rows = document.querySelectorAll('#bl-receta-container .ing-row');
   843→  var receta = [];
   844→  rows.forEach(function(row) {
   845→    var select = row.querySelector('select');

   847→    var unitSelect = row.querySelectorAll('select')[1];
   848→    var productId = parseInt(select.value);
   849→    var qty = parseFloat(qtyInput.value) || 0;
   850→    var unit = unitSelect ? unitSelect.value : 'gr';
   851→    if (productId && qty > 0) {
   852→      var prod = db.productos.find(function(p) { return p.id === productId; });
   853→      receta.push({ productoId: productId, nombre: prod ? prod.nombre : '?', cantidad: qty, unidad: unit });

   855→  });

   857→  var now = new Date().toISOString();
   858→
   859→  if (existingId) {
   860→    var blend = db.blends.find(function(b) { return b.id === existingId; });
   861→    if (blend) {
   862→      blend.nombre = nombre; blend.descripcion = desc; blend.receta = receta;
   863→      blend.precioVenta = precio; blend.costoUnitario = calcBlendCost({ receta: receta });
   864→      blend.actualizadoEn = now;
   865→    }
   866→  } else {
   867→    db.blends.push({
   868→      id: nextId(), nombre: nombre, descripcion: desc, receta: receta,
   869→      costoUnitario: calcBlendCost({ receta: receta }), precioVenta: precio,
   870→      stock: 0, creadoEn: now, actualizadoEn: now
   871→    });
   872→  }
   873→
   874→  saveDB();
   875→  closeModal();
   876→  renderBlends();
   877→  toast(existingId ? 'Blend actualizado' : 'Blend creado');
   878→}

   916→function ejecutarProduccion(blendId) {
   917→  var cant = parseInt(document.getElementById('bl-prod-cant').value) || 1;
   918→  if (cant < 1) return;
   919→
   920→  var db = getDB();
   921→  var b = db.blends.find(function(x) { return x.id === blendId; });
   922→  if (!b) return;
   923→
   924→  // Check stock for multiplied recipe
   925→  (b.receta || []).forEach(function(r) {
   926→    var needed = r.cantidad * cant;
   927→    var prod = db.productos.find(function(p) { return p.id === r.productoId; });
   928→    if (!prod || prod.stock < needed) {
   929→      toast('Stock insuficiente de ' + r.nombre, 'err');
   930→      return;
   931→    }
   932→  });
   933→
   934→  // Deduct stock from ingredients
   935→  (b.receta || []).forEach(function(r) {
   936→    var needed = r.cantidad * cant;
   937→    var prod = db.productos.find(function(p) { return p.id === r.productoId; });
   938→    if (prod) {
   939→      prod.stock -= needed;
   940→      prod.actualizadoEn = new Date().toISOString();
   941→      addMovimiento('blend', 'blends', b.id, r.nombre, -needed, 'Producción: ' + b.nombre + ' x' + cant);
   942→    }
   943→  });
   944→
   945→  b.stock += cant;
   946→  b.actualizadoEn = new Date().toISOString();
   947→  saveDB();
   948→  closeModal();
   949→  renderBlends();
   950→  toast(cant + ' unidad(es) de ' + b.nombre + ' producida(s)');
   951→}

  1164→        prod.stock = Math.max(0, prod.stock - it.cantidad);

  1166→        addMovimiento('venta', 'ventas', venta.id, it.nombre, -it.cantidad, 'Venta mostrador');

  1171→        blend.stock = Math.max(0, blend.stock - it.cantidad);
  1172→        blend.actualizadoEn = new Date().toISOString();
  1173→        addMovimiento('venta', 'ventas', venta.id, it.nombre, -it.cantidad, 'Venta mostrador');
  1174→      }
  1175→    }
  1176→  });
  1177→
  1178→  saveDB();
  1179→  ventaActual = null;
  1180→  renderVentas();
