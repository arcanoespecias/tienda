   291→  saveBlend(id) {
   292→    const nombre = document.getElementById('f-bl-nombre').value.trim();
   293→    if (!nombre) { alert('Nombre requerido'); return; }
   294→    const rows = document.querySelectorAll('#f-bl-ingredientes .ing-row');
   295→    const ingredientes = [];
   296→    rows.forEach(row => {
   297→      const espId = row.querySelector('[data-field="especiaId"]').value;
   298→      const cant = Number(row.querySelector('[data-field="cantidad"]').value) || 0;
   299→      if (espId && cant > 0) ingredientes.push({ especiaId: espId, cantidad: cant });
   300→    });
   301→    if (ingredientes.length === 0) { alert('Agrega al menos un ingrediente'); return; }
   302→
   303→    const data = {
   304→      nombre,
   305→      categoria: document.getElementById('f-bl-cat').value,
   306→      precioChico: Number(document.getElementById('f-bl-pchico').value) || 0,
   307→      precioGrande: Number(document.getElementById('f-bl-pgrande').value) || 0,
   308→      ingredientes
   309→    };
   310→    if (id) data.id = id;
   311→    try {
   312→      ArcanoDB.saveBlend(data);
   313→      document.querySelector('.modal-overlay').remove();
   314→      App.renderPage('blends');
   315→    } catch (e) { alert('Error: ' + e.message); }

   409→    const allItems = compra ? compra.items || [] : [];

   412→    if (espItems.length === 0 && !readonly) espItems.push({tipo:'especia', especiaId:'', cantidad:'', costoUnitario:''});
   413→    if (etqItems.length === 0 && !readonly) etqItems.push({tipo:'etiqueta', etiquetaNombre:'', cantidad:'', costoUnitario:''});

   415→    const modal = document.createElement('div');
   416→    modal.className = 'modal-overlay';
   417→    modal.innerHTML = `
   418→      <div class="modal modal-lg">
   419→        <div class="modal-header"><h3>${id ? 'Detalle Compra' : 'Nueva Compra'}</h3><button class="btn btn-ghost" onclick="this.closest('.modal-overlay').remove()">X</button></div>

  1039→  /** Actualiza el calculo de especias necesarias en tiempo real */
  1040→  _updateProduccionCalculo() {
  1041→    const blendId = document.getElementById('f-etq-blend').value;
  1042→    const cant = Number(document.getElementById('f-etq-cant').value) || 0;
  1043→    const calcDiv = document.getElementById('f-etq-calculo');
  1044→    const calcBody = document.getElementById('f-etq-calculo-body');
  1045→
  1046→    if (!blendId || cant <= 0) {
  1047→      calcDiv.style.display = 'none';
  1048→      return;
  1049→    }
  1050→
  1051→    const blend = ArcanoDB.getBlend(blendId);
  1052→    if (!blend) { calcDiv.style.display = 'none'; return; }
  1053→
  1054→    const ingredientes = blend.ingredientes || [];
  1055→    let html = '<p class="mb-8">Para producir <b>' + cant + '</b> etiquetas de <b class="text-gold">"' + blend.nombre + '"</b>:</p>';
  1056→    html += '<table class="table"><thead><tr><th>Especia</th><th>Por unidad</th><th>Total necesario</th><th>Stock actual</th><th>Estado</th></tr></thead><tbody>';
  1057→
  1058→    let todoOk = true;
  1059→    for (const ing of ingredientes) {
  1060→      const esp = ArcanoDB.getEspecia(ing.especiaId);
  1061→      if (!esp) continue;
  1062→      const porUnidad = ing.cantidad || 0;
  1063→      const total = porUnidad * cant;
  1064→      const ok = esp.stock >= total;
  1065→      if (!ok) todoOk = false;
  1066→      html += '<tr>' +
  1067→        '<td class="fw7">' + esp.nombre + '</td>' +
  1068→        '<td>' + porUnidad + '</td>' +
  1069→        '<td class="fw7">' + total + '</td>' +
  1070→        '<td>' + esp.stock + '</td>' +
  1071→        '<td>' + (ok
  1072→          ? '<span class="badge badge-green">OK</span>'
  1073→          : '<span class="badge badge-red">Faltan ' + (total - esp.stock) + '</span>') +
  1074→        '</td></tr>';
  1075→    }
  1076→    html += '</tbody></table>';
  1077→
  1078→    // Show etiqueta info
  1079→    var etqStockList = ArcanoDB.getEtiquetasStock();
  1080→    var etq = null;
  1081→    for (var ei = 0; ei < etqStockList.length; ei++) {
  1082→      if (etqStockList[ei].nombre === blend.nombre) { etq = etqStockList[ei]; break; }
  1083→    }
  1084→    html += '<div class="mt-12 card"><div class="card-header"><h3>Etiqueta fisica requerida</h3></div><div class="card-body">';
  1085→    if (etq) {
  1086→      var etqOk = etq.stock >= cant;
  1087→      html += '<div class="list-row"><span class="fw7">Etiqueta "' + etq.nombre + '"</span><span class="' + (etqOk ? 'text-green' : 'text-red') + '">Stock: ' + etq.stock + ' / Necesario: ' + cant + '</span></div>';
  1088→      if (!etqOk) {
  1089→        html += '<p class="text-red mt-8 fw7">Faltan ' + (cant - etq.stock) + ' etiquetas de "' + blend.nombre + '". Compralas en Compras.</p>';
  1090→        todoOk = false;
  1091→      }
  1092→    } else {
  1093→      html += '<p class="text-red fw7">No hay etiquetas "' + blend.nombre + '" en stock. Compralas en Compras primero.</p>';
  1094→      todoOk = false;
  1095→    }
  1096→    html += '</div></div>';
  1097→
  1098→    if (!todoOk) {
  1099→      html += '<p class="text-red mt-8 fw7">No se puede producir. Revisa los faltantes.</p>';
  1100→    } else {
  1101→      html += '<p class="text-green mt-8 fw7">Todo disponible. Se produciran ' + cant + ' etiquetas de "' + blend.nombre + '".</p>';
  1102→    }
  1103→
  1104→    calcBody.innerHTML = html;
  1105→    calcDiv.style.display = 'block';
  1106→  },
