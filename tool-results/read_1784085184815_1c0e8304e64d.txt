    11→var storeCategory = 'todos';

    13→function renderStorefront() {
    14→  // Build category list from active especias
    15→  var db = getDB();
    16→  var categories = ['todos', 'blends'];
    17→  db.productos.forEach(function(p) {
    18→    if (p.tipo === 'especia' && p.precioVenta > 0 && p.stock > 0 && p.categoria) {
    19→      var found = false;
    20→      categories.forEach(function(c) { if (c === p.categoria) found = true; });
    21→      if (!found) categories.push(p.categoria);
    22→    }
    23→  });
    24→
    25→  // Update category nav in the static HTML
    26→  var storeNav = document.getElementById('store-nav');
    27→  if (storeNav) {
    28→    var navHtml = '';
    29→    categories.forEach(function(cat) {
    30→      var label = cat === 'todos' ? 'Todos' : cat === 'blends' ? 'Blends' : cat;
    31→      var cls = (cat === storeCategory) ? ' store-nav-btn active' : ' store-nav-btn';
    32→      navHtml += '<button class="' + cls + '" data-cat="' + esc(cat) + '">' + esc(label) + '</button>';
    33→    });
    34→    storeNav.innerHTML = navHtml;
    35→    setupStoreNav();
    36→  }
    37→
    38→  renderStoreProducts();
    39→  updateCartUI();
    40→}

    53→function renderStoreProducts() {
    54→  var container = document.getElementById('store-products');
    55→  var emptyEl = document.getElementById('store-empty');
    56→  if (!container) return;
    57→
    58→  var db = getDB();
    59→  var html = '';
    60→  var hasProducts = false;
    61→
    62→  // --- Especias ---
    63→  db.productos.forEach(function(p) {
    64→    if (p.tipo !== 'especia') return;
    65→    if (p.precioVenta <= 0 || p.stock <= 0) return;
    66→    if (storeCategory !== 'todos' && storeCategory !== p.categoria) return;
    67→
    68→    hasProducts = true;
    69→    html += '<div class="product-card">' +
    70→      '<div class="product-card-type">Especia</div>' +
    71→      '<div class="product-card-name">' + esc(p.nombre) + '</div>' +
    72→      '<div class="product-card-desc">' + esc(p.categoria || '') + '</div>' +
    73→      '<div class="product-card-footer">' +
    74→        '<div>' +
    75→          '<div class="product-card-price">' + fmt(p.precioVenta) + '/gr</div>' +
    76→          '<div class="product-card-stock">' + p.stock + ' gr disponibles</div>' +
    77→        '</div>' +
    78→        '<button class="btn-add-cart" onclick="addToCart(false,' + p.id + ',\'' + jsEsc(p.nombre) + '\',' + p.precioVenta + ',\'' + esc(p.unidad) + '\',' + p.stock + ')">Agregar</button>' +
    79→      '</div>' +
    80→      '</div>';
    81→  });
    82→
    83→  // --- Blends ---
    84→  if (storeCategory === 'todos' || storeCategory === 'blends') {
    85→    db.blends.forEach(function(b) {
    86→      if (b.precioVenta <= 0 || (b.stock || 0) <= 0) return;
    87→
    88→      var maxFrascoStock = 0;
    89→      db.productos.forEach(function(p) {
    90→        if (p.tipo === 'frasco' && p.stock > 0 && p.stock > maxFrascoStock) {
    91→          maxFrascoStock = p.stock;
    92→        }
    93→      });
    94→
    95→      var etiquetaStock = 999999;
    96→      if (b.etiquetaId) {
    97→        db.productos.forEach(function(p) {
    98→          if (p.id === b.etiquetaId) etiquetaStock = p.stock;
    99→        });
   100→      }
   101→
   102→      var efectivo = Math.min(b.stock || 0, maxFrascoStock, etiquetaStock);
   103→      if (efectivo <= 0) return;
   104→
   105→      hasProducts = true;
   106→      html += '<div class="product-card">' +
   107→        '<div class="product-card-type" style="color:#e55">Blend</div>' +
   108→        '<div class="product-card-name">' + esc(b.nombre) + '</div>' +
   109→        '<div class="product-card-desc">' + esc(b.formato || 'polvo') + ' - ' + (b.gramosPorUnidad || 100) + 'gr</div>' +
   110→        '<div class="product-card-footer">' +
   111→          '<div>' +
   112→            '<div class="product-card-price">' + fmt(b.precioVenta) + '/und</div>' +
   113→            '<div class="product-card-stock">' + efectivo + ' unidades</div>' +
   114→          '</div>' +
   115→          '<button class="btn-add-cart" onclick="mostrarSelectorFrasco(' + b.id + ',\'carrito\',\'' + jsEsc(b.nombre) + '\',' + b.precioVenta + ',' + efectivo + ')">Agregar</button>' +
   116→        '</div>' +
   117→        '</div>';
   118→    });
   119→  }
   120→
   121→  container.innerHTML = html;
   122→  if (emptyEl) emptyEl.style.display = hasProducts ? 'none' : 'block';
   123→}

   125→function addToCart(isBlend, id, nombre, precio, unidad, stock) {
   126→  var existingIdx = -1;
   127→  cart.forEach(function(item, idx) {
   128→    if (!item.isBlend && !isBlend && item.id === id) {
   129→      existingIdx = idx;
   130→    }
   131→  });
   132→
   133→  if (existingIdx >= 0) {
   134→    if (cart[existingIdx].qty < cart[existingIdx].maxStock) {
   135→      cart[existingIdx].qty++;
   136→    } else {
   137→      toast('Stock maximo alcanzado', 'err');
   138→      return;
   139→    }
   140→  } else {
   141→    cart.push({
   142→      isBlend: false,
   143→      id: id,
   144→      nombre: nombre,
   145→      precio: precio,
   146→      unidad: unidad,
   147→      qty: 1,
   148→      maxStock: stock
   149→    });
   150→  }
   151→
   152→  updateCartUI();
   153→  toast(nombre + ' agregado al carrito');
   154→}

   156→function mostrarSelectorFrasco(blendId, contexto, nombre, precio, stock) {
   157→  var db = getDB();
   158→  var frascos = [];
   159→  db.productos.forEach(function(p) {
   160→    if (p.tipo === 'frasco' && p.stock > 0) frascos.push(p);
   161→  });
   162→
   163→  if (frascos.length === 0) {
   164→    toast('No hay frascos disponibles', 'err');
   165→    return;
   166→  }
   167→
   168→  var html = '<p class="mb-2">Seleccione un frasco para <strong>' + esc(nombre) + '</strong></p>' +
   169→    '<p>Precio: ' + fmt(precio) + ' / unidad</p><div class="frasco-list">';
   170→
   171→  frascos.forEach(function(f) {
   172→    var disponible = Math.min(stock, f.stock);
   173→    if (disponible <= 0) return;
   174→    html += '<div class="frasco-option">' +
   175→      '<div><strong>' + esc(f.nombre) + '</strong><br>Disponible: ' + disponible + ' unidades</div>' +
   176→      '<button class="btn btn-primary btn-sm" onclick="confirmarFrasco(' + blendId + ',' + f.id + ',\'' + jsEsc(contexto) + '\',' + precio + ',' + stock + ')">Seleccionar</button>' +
   177→      '</div>';
   178→  });
   179→
   180→  html += '</div>';
   181→  openModal('Seleccionar Frasco', html);
   182→}

   184→function confirmarFrasco(blendId, frascoId, contexto, precio, stock) {
   185→  var db = getDB();
   186→  var blend = null;
   187→  db.blends.forEach(function(b) { if (b.id === blendId) blend = b; });
   188→  if (!blend) return;
   189→
   190→  var frasco = null;
   191→  db.productos.forEach(function(p) { if (p.id === frascoId) frasco = p; });
   192→  if (!frasco) return;
   193→
   194→  var etiquetaStock = 999999;
   195→  if (blend.etiquetaId) {
   196→    db.productos.forEach(function(p) {
   197→      if (p.id === blend.etiquetaId) etiquetaStock = p.stock;
   198→    });
   199→  }
   200→
   201→  var disponible = Math.min(blend.stock, frasco.stock, etiquetaStock);
   202→  if (disponible <= 0) {
   203→    toast('No hay stock suficiente', 'err');
   204→    closeModal();
   205→    return;
   206→  }
   207→
   208→  if (contexto === 'carrito') {
   209→    var existingIdx = -1;
   210→    cart.forEach(function(item, idx) {
   211→      if (item.isBlend && item.id === blendId && item.frascoId === frascoId) {
   212→        existingIdx = idx;
   213→      }
   214→    });
   215→
   216→    if (existingIdx >= 0) {
   217→      if (cart[existingIdx].qty < cart[existingIdx].maxStock) {
   218→        cart[existingIdx].qty++;
   219→      } else {
   220→        toast('Stock maximo alcanzado', 'err');
   221→        closeModal();
   222→        return;
   223→      }
   224→    } else {
   225→      cart.push({
   226→        isBlend: true,
   227→        id: blendId,
   228→        nombre: blend.nombre,
   229→        precio: precio,
   230→        unidad: 'unidad',
   231→        qty: 1,
   232→        maxStock: disponible,
   233→        frascoId: frascoId,
   234→        frascoNombre: frasco.nombre,
   235→        etiquetaId: blend.etiquetaId || null
   236→      });
   237→    }
   238→
   239→    updateCartUI();
   240→    toast(blend.nombre + ' agregado al carrito');
   241→  } else if (contexto === 'venta') {
   242→    if (typeof agregarBlendAVenta === 'function') {
   243→      agregarBlendAVenta(blendId, frascoId, blend.nombre, precio, disponible);
   244→    }
   245→    closeModal();
   246→    return;
   247→  }
   248→
   249→  closeModal();
   250→}

   252→function updateCartUI() {
   253→  var countEl = document.getElementById('cart-count');
   254→  var total = 0;
   255→  cart.forEach(function(item) { total += item.qty; });
   256→  if (countEl) {

   258→    countEl.style.display = total > 0 ? 'inline' : 'none';

   262→function cartQty(idx, delta) {
   263→  if (!cart[idx]) return;
   264→  cart[idx].qty += delta;
   265→  if (cart[idx].qty <= 0) {
   266→    cart.splice(idx, 1);
   267→  } else if (cart[idx].qty > cart[idx].maxStock) {
   268→    cart[idx].qty = cart[idx].maxStock;
   269→    toast('Stock maximo alcanzado', 'err');
   270→  }
   271→  updateCartUI();
   272→  renderCartDrawer();
   273→}

   365→function submitOrder() {
   366→  if (!cart || cart.length === 0) {
   367→    toast('El carrito esta vacio', 'err');
   368→    return;
   369→  }
   370→
   371→  var db = getDB();
   372→  var total = 0;
   373→  var items = [];
   374→
   375→  cart.forEach(function(ci) {
   376→    var subtotal = ci.precio * ci.qty;
   377→    total += subtotal;
   378→    items.push({
   379→      isBlend: ci.isBlend,
   380→      id: ci.id,
   381→      nombre: ci.nombre,
   382→      precio: ci.precio,
   383→      unidad: ci.unidad,
   384→      qty: ci.qty,
   385→      frascoId: ci.frascoId || null,
   386→      frascoNombre: ci.frascoNombre || '',
   387→      etiquetaId: ci.etiquetaId || null,
   388→      subtotal: subtotal
   389→    });
   390→  });
   391→
   392→  var venta = {
   393→    id: nextId(),
   394→    estado: 'pendiente',
   395→    tipo: 'tienda',
   396→    creadoPor: null,
   397→    fecha: new Date().toISOString(),
   398→    cliente: 'Tienda Online',
   399→    items: items,
   400→    total: total,
   401→    notas: ''
   402→  };
   403→  db.ventas.push(venta);
   404→
   405→  // Process stock deductions
   406→  cart.forEach(function(ci) {
   407→    if (!ci.isBlend) {
   408→      // Especia: deduct stock from product
   409→      db.productos.forEach(function(p) {
   410→        if (p.id === ci.id) {
   411→          p.stock -= ci.qty;
   412→          addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Venta tienda #' + venta.id);
   413→        }
   414→      });
   415→    } else {
   416→      // Blend: deduct blend stock
   417→      db.blends.forEach(function(b) {
   418→        if (b.id === ci.id) {
   419→          b.stock -= ci.qty;
   420→          addMovimiento('venta', b.id, b.nombre, -ci.qty, 'Venta tienda blend #' + venta.id);
   421→        }
   422→      });
   423→      // Deduct frasco stock
   424→      if (ci.frascoId) {
   425→        db.productos.forEach(function(p) {
   426→          if (p.id === ci.frascoId) {
   427→            p.stock -= ci.qty;
   428→            addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Frasco para venta #' + venta.id);
   429→          }
   430→        });
   431→      }
   432→      // Deduct etiqueta stock
   433→      if (ci.etiquetaId) {
   434→        db.productos.forEach(function(p) {
   435→          if (p.id === ci.etiquetaId) {
   436→            p.stock -= ci.qty;
   437→            addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Etiqueta para venta #' + venta.id);
   438→          }
   439→        });
   440→      }
   441→    }
   442→  });
   443→
   444→  cart = [];
   445→  saveDB();
   446→  // Close drawer if open
   447→  var drawer = document.getElementById('cart-drawer');
   448→  var overlay = document.getElementById('cart-overlay');
   449→  if (drawer) drawer.classList.remove('open');
   450→  if (overlay) overlay.classList.remove('open');
   451→  toast('Pedido #' + venta.id + ' creado exitosamente');
   452→  renderStorefront();
   453→}

   465→function renderDashboard() {
   466→  var el = document.getElementById('page-dashboard');
   467→  if (!el) return;
   468→
   469→  var db = getDB();
   470→
   471→  // KPI 1: Ventas Totales (non-cancelled)
   472→  var ventasTotal = 0;
   473→  db.ventas.forEach(function(v) {
   474→    if (v.estado !== 'cancelada') ventasTotal += (v.total || 0);
   475→  });
   476→
   477→  // KPI 2: Compras (recibidas)
   478→  var comprasTotal = 0;
   479→  db.compras.forEach(function(c) {
   480→    if (c.estado === 'recibida') comprasTotal += (c.total || 0);
   481→  });
   482→
   483→  // KPI 3: Especias count + total items
   484→  var especiaCount = 0;
   485→  var totalItems = 0;
   486→  db.productos.forEach(function(p) {
   487→    if (p.tipo === 'especia') especiaCount++;
   488→    totalItems++;
   489→  });
   490→
   491→  // KPI 4: Blends count
   492→  var blendCount = db.blends.length;
   493→
   494→  // Alerts
   495→  var pendientes = 0;
   496→  db.ventas.forEach(function(v) { if (v.estado === 'pendiente') pendientes++; });
   497→
   498→  var lowStock = [];
   499→  db.productos.forEach(function(p) {
   500→    if (p.tipo === 'especia' && p.stock <= (p.stockMin || 0)) {
   501→      lowStock.push(p);
   502→    }
   503→  });
   504→
   505→  // Build HTML
   506→  var html = '<div class="g4">' +
   507→    statCard('Ventas Totales', fmt(ventasTotal), 'Ingreso total') +
   508→    statCard('Compras', fmt(comprasTotal), 'Recibidas') +
   509→    statCard('Especias', String(especiaCount), totalItems + ' productos total') +
   510→    statCard('Blends', String(blendCount), 'Mezclas activas') +
   511→    '</div>';
   512→
   513→  // Alerts section
   514→  html += '<div class="alerts-section">';
   515→  if (pendientes > 0) {
   516→    html += '<div class="alert-card"><div><strong>' + pendientes + ' ventas pendientes</strong></div>' +
   517→      '<button class="btn btn-primary btn-sm" onclick="navigateTo(\'ventas\')">Ver Ventas</button></div>';
   518→  }
   519→  if (lowStock.length > 0) {
   520→    html += '<div class="alert-card alert-warn"><div><strong>' + lowStock.length + ' productos con stock bajo</strong></div>' +
   521→      '<button class="btn btn-warn btn-sm" onclick="navigateTo(\'productos\')">Ver Stock</button></div>';
   522→  }
   523→  if (pendientes === 0 && lowStock.length === 0) {
   524→    html += '<div class="alert-card"><div>Todo al dia</div></div>';
   525→  }
   526→  html += '</div>';

   719→function guardarCompra() {
   720→  var fecha = document.getElementById('cmp-fecha').value;
   721→  var proveedor = (document.getElementById('cmp-proveedor').value || '').trim();
   722→  var notas = (document.getElementById('cmp-notas').value || '').trim();
   723→  var esRecibida = document.getElementById('cmp-recibida').checked;
   724→
   725→  var container = document.getElementById('compra-items');
   726→  var rows = container.querySelectorAll('.compra-item-row');
   727→  var items = [];
   728→  var total = 0;
   729→
   730→  var db = getDB();
   731→
   732→  rows.forEach(function(row) {
   733→    var select = row.querySelector('select');
   734→    var inputs = row.querySelectorAll('input');
   735→    var productId = Number(select.value);
   736→    var cantidad = Number(inputs[0].value) || 0;
   737→    var precioUnit = Number(inputs[1].value) || 0;
   738→
   739→    if (productId && cantidad > 0) {
   740→      var producto = null;
   741→      db.productos.forEach(function(p) { if (p.id === productId) producto = p; });
   742→      if (!producto) return;
   743→
   744→      var subtotal = cantidad * precioUnit;
   745→      total += subtotal;
   746→      items.push({
   747→        productoId: productId,
   748→        productoNombre: producto.nombre,
   749→        cantidad: cantidad,
   750→        precioUnitario: precioUnit,
   751→        subtotal: subtotal
   752→      });
   753→    }
   754→  });
   755→
   756→  if (items.length === 0) {
   757→    toast('Agregue al menos un producto valido', 'err');
   758→    return;
   759→  }
   760→
   761→  var compra = {
   762→    id: nextId(),
   763→    fecha: fecha || new Date().toISOString(),
   764→    proveedor: proveedor,
   765→    notas: notas,
   766→    estado: esRecibida ? 'recibida' : 'pendiente',
   767→    items: items,
   768→    total: total,
   769→    creadoPor: currentUser ? currentUser.id : null,
   770→    creadoEn: new Date().toISOString()
   771→  };
   772→
   773→  db.compras.push(compra);
   774→
   775→  // If received, add stock to each product
   776→  if (esRecibida) {
   777→    items.forEach(function(item) {
   778→      db.productos.forEach(function(p) {
   779→        if (p.id === item.productoId) {
   780→          p.stock += item.cantidad;
   781→          addMovimiento('compra', p.id, p.nombre, item.cantidad, 'Compra #' + compra.id + ' recibida');
   782→        }
   783→      });
   784→    });
   785→  }
   786→
   787→  saveDB();
   788→  closeModal();
   789→  toast('Compra #' + compra.id + ' registrada');
   790→  renderCompras();
   791→}

   839→function cambiarEstadoCompra(id, nuevoEstado) {
   840→  var db = getDB();
   841→  var compra = null;
   842→  db.compras.forEach(function(c) { if (c.id === id) compra = c; });
   843→  if (!compra) return;
   844→
   845→  var oldEstado = compra.estado;
   846→  if (oldEstado === nuevoEstado) return;
   847→
   848→  // Changing TO recibida: add stock to products
   849→  if (nuevoEstado === 'recibida' && oldEstado !== 'recibida') {
   850→    (compra.items || []).forEach(function(item) {
   851→      db.productos.forEach(function(p) {
   852→        if (p.id === item.productoId) {
   853→          p.stock += item.cantidad;
   854→          addMovimiento('compra', p.id, p.nombre, item.cantidad, 'Compra #' + compra.id + ' recibida');
   855→        }
   856→      });
   857→    });
   858→  }
   859→
   860→  // Changing FROM recibida to something else: reverse stock addition
   861→  if (oldEstado === 'recibida' && nuevoEstado !== 'recibida') {
   862→    (compra.items || []).forEach(function(item) {
   863→      db.productos.forEach(function(p) {
   864→        if (p.id === item.productoId) {
   865→          p.stock -= item.cantidad;
   866→          addMovimiento('ajuste', p.id, p.nombre, -item.cantidad, 'Reversion compra #' + compra.id);
   867→        }
   868→      });
   869→    });
   870→  }
   871→
   872→  compra.estado = nuevoEstado;
   873→  saveDB();
   874→  closeModal();
   875→  toast('Estado actualizado a ' + nuevoEstado);
   876→  renderCompras();
   877→}

  1596→function recalcProduccion(id) {
  1597→  var db = getDB();

  1599→  if (!blend) return;

  1606→  var detailHtml = '';

  1620→      '<span>' + fmt(cost) + '</span>' +
  1621→    '</div>';
  1622→  });
  1623→
  1624→  var costoEl = document.getElementById('prod-costo-total');
  1625→  if (costoEl) costoEl.textContent = fmt(totalCost);
  1626→
  1627→  var detailEl = document.getElementById('prod-detail');

  1629→    detailEl.innerHTML = cantidad > 0

  1639→function ejecutarProduccion(id) {
  1640→  var db = getDB();
  1641→  var blend = db.blends.find(function(b) { return b.id === id; });
  1642→  if (!blend) return;
  1643→
  1644→  var cantidadInput = document.getElementById('prod-cantidad');
  1645→  var cantidad = Number(cantidadInput.value) || 0;
  1646→  if (cantidad < 1) {
  1647→    toast('Cantidad invalida', 'err');
  1648→    return;
  1649→  }
  1650→
  1651→  var gramosPorUnidad = blend.gramosPorUnidad || 100;
  1652→  var receta = blend.receta || [];
  1653→  var allOk = true;
  1654→
  1655→  // Validate stock
  1656→  receta.forEach(function(ing) {
  1657→    var spice = db.productos.find(function(p) { return p.id === ing.productoId; });
  1658→    if (!spice) { allOk = false; return; }
  1659→    var needed = (ing.porcentaje / 100) * gramosPorUnidad * cantidad;
  1660→    if ((spice.stock || 0) < needed) allOk = false;
  1661→  });
  1662→
  1663→  if (!allOk) {
  1664→    var alertEl = document.getElementById('prod-alert');
  1665→    if (alertEl) { alertEl.textContent = 'Stock insuficiente para producir ' + cantidad + ' unidades'; alertEl.classList.add('show'); }
  1666→    return;
  1667→  }
  1668→
  1669→  // Deduct stock and record movimientos
  1670→  receta.forEach(function(ing) {
  1671→    var spice = db.productos.find(function(p) { return p.id === ing.productoId; });
  1672→    if (!spice) return;
  1673→    var needed = (ing.porcentaje / 100) * gramosPorUnidad * cantidad;
  1674→    spice.stock = (spice.stock || 0) - needed;
  1675→    spice.stock = Math.round(spice.stock * 100) / 100;
  1676→    addMovimiento('produccion', spice.id, spice.nombre, -needed,
  1677→      'Produccion: ' + blend.nombre + ' x' + cantidad + ' (' + needed.toFixed(1) + 'gr)');
  1678→  });
  1679→
  1680→  // Add blend stock
  1681→  blend.stock = (blend.stock || 0) + cantidad;
  1682→
  1683→  saveDB();
  1684→  closeModal();
  1685→  refreshPage();
  1686→  toast('Producidas ' + cantidad + ' unidades de ' + blend.nombre);
  1687→}
