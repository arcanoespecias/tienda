  1130→  handleOCRResult(text) {
  1131→    var statusEl = document.getElementById('cam-status');
  1132→    var confirmArea = document.getElementById('cam-confirm-area');
  1133→    var detectedTextEl = document.getElementById('cam-detected-text');
  1134→    var prodSelect = document.getElementById('cam-prod-select');
  1135→    if (!text || text.length < 2) {
  1136→      if (statusEl) statusEl.innerHTML = '<span style="color:var(--red)">No se detecto texto. Intenta de nuevo.</span>';
  1137→      setTimeout(function() { if (statusEl) statusEl.textContent = 'Apunta la camara a la etiqueta del producto'; }, 2000);
  1138→      return;
  1139→    }
  1140→    var especias = ArcanoDB.getEspecias();
  1141→    var blends = ArcanoDB.getBlends();
  1142→    var allProducts = [];
  1143→    for (var i = 0; i < especias.length; i++) { allProducts.push({ tipo: 'especia', producto: especias[i] }); }
  1144→    for (var i = 0; i < blends.length; i++) { allProducts.push({ tipo: 'blend', producto: blends[i] }); }
  1145→    var ocrLower = text.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  1146→    var scored = [];
  1147→    for (var i = 0; i < allProducts.length; i++) {
  1148→      var p = allProducts[i];
  1149→      var name = (p.producto.nombre || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  1150→      var nameWords = name.split(/\s+/);
  1151→      var matchCount = 0;
  1152→      for (var w = 0; w < nameWords.length; w++) {
  1153→        if (nameWords[w].length < 2) continue;
  1154→        if (ocrLower.indexOf(nameWords[w]) !== -1) matchCount++;
  1155→      }
  1156→      var score = nameWords.length > 0 ? matchCount / nameWords.length : 0;
  1157→      if (ocrLower.indexOf(name) !== -1) score = Math.max(score, 1.0);
  1158→      if (name.length >= 3 && ocrLower.indexOf(name.substring(0, Math.min(name.length, 6))) !== -1) score = Math.max(score, 0.7);
  1159→      if (score >= 0.5) scored.push({ tipo: p.tipo, producto: p.producto, score: score });
  1160→    }
  1161→    scored.sort(function(a, b) { return b.score - a.score; });
  1162→    if (confirmArea) confirmArea.style.display = 'block';

  1168→          var s = scored[i];
  1169→          var pct = Math.round(s.score * 100);
  1170→          prodSelect.innerHTML += '<option value="' + s.tipo + '|' + s.producto.id + '">' + s.producto.nombre + ' (' + pct + '%)</option>';
  1171→        }
  1172→        if (scored[0].score >= 0.7) {
  1173→          prodSelect.value = scored[0].tipo + '|' + scored[0].producto.id;
  1174→        }
  1175→        if (statusEl) statusEl.innerHTML = '<span style="color:var(--green)">Producto detectado - confirma abajo</span>';
  1176→      } else {
  1177→        if (statusEl) statusEl.innerHTML = '<span style="color:var(--red)">No se encontro producto. Selecciona manualmente.</span>';
