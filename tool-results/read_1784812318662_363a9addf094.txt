  1353→    if (!file) return;

  1357→      var placeholder = area.querySelector('.img-upload-placeholder');

  1359→      var existing = document.getElementById('img-preview-' + areaId.split('-').pop());
  1360→      if (existing) existing.remove();
  1361→      var removeBtn = area.querySelector('.btn-red');
  1362→      if (removeBtn) removeBtn.remove();

  1364→      img.src = dataUrl;
  1365→      img.className = 'img-preview';
  1366→      img.id = 'img-preview-' + areaId.split('-').pop();
  1367→      area.insertBefore(img, placeholder);
  1368→      var rmBtn = document.createElement('button');
  1369→      rmBtn.className = 'btn btn-sm btn-red';
  1370→      rmBtn.style.marginTop = '6px';
  1371→      rmBtn.textContent = 'Quitar imagen';
  1372→      rmBtn.onclick = function() { Pages.removeImage(areaId, input.id); };
  1373→      area.insertBefore(rmBtn, placeholder);
  1374→    });
  1375→  },
