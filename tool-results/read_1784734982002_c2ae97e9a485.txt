  1268→  handleImageUpload(input, areaId) {
  1269→    var file = input.files && input.files[0];
  1270→    if (!file) return;
  1271→    ArcanoDB.compressImage(file, 400, 0.75, function(err, dataUrl) {
  1272→      if (err) { alert(err); return; }
  1273→      var area = document.getElementById(areaId);
  1274→      var placeholder = area.querySelector('.img-upload-placeholder');
  1275→      if (placeholder) placeholder.style.display = 'none';
  1276→      var existing = document.getElementById('img-preview-' + areaId.split('-').pop());
  1277→      if (existing) existing.remove();
  1278→      var removeBtn = area.querySelector('.btn-red');
  1279→      if (removeBtn) removeBtn.remove();
  1280→      var img = document.createElement('img');
  1281→      img.src = dataUrl;
  1282→      img.className = 'img-preview';
  1283→      img.id = 'img-preview-' + areaId.split('-').pop();
  1284→      area.insertBefore(img, placeholder);
  1285→      var rmBtn = document.createElement('button');
  1286→      rmBtn.className = 'btn btn-sm btn-red';
  1287→      rmBtn.style.marginTop = '6px';
  1288→      rmBtn.textContent = 'Quitar imagen';
  1289→      rmBtn.onclick = function() { Pages.removeImage(areaId, input.id); };
  1290→      area.insertBefore(rmBtn, placeholder);
  1291→    });
  1292→  },
