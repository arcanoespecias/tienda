   331→   331→/** Producir frascos de ESPECIA pura: consume grs de bolsa + etiquetas (por talla), agrega frascos */
   332→   332→function producirEspeciaFrascos(especiaId, grsPorFrasco, cantidad, talla) {
   333→   333→  _ensureStructure();
   334→   334→  var esp = _db.especias[especiaId];
   335→   335→  if (!esp) throw new Error('Especia no encontrada');
   336→   336→  grsPorFrasco = Number(grsPorFrasco) || 0;
   337→   337→  cantidad = Number(cantidad) || 0;
   338→   338→  talla = (talla === 'grande') ? 'grande' : 'chico';
   339→   339→  if (grsPorFrasco <= 0) throw new Error('Los gramos por frasco deben ser mayor a 0');
   340→   340→  if (cantidad <= 0) throw new Error('La cantidad de frascos debe ser mayor a 0');
   341→   341→
   342→   342→  var grsTotal = grsPorFrasco * cantidad;
   343→   343→  if ((esp.stockBolsa || 0) < grsTotal) {
   344→   344→    throw new Error('Stock insuficiente en BOLSA de "' + esp.nombre + '". Necesario: ' + grsTotal + 'grs, Disponible: ' + (esp.stockBolsa || 0) + 'grs');
   345→   345→  }
   346→   346→
   347→   347→  // Check etiqueta por talla
   348→   348→  var etiqueta = _findEtiquetaByNombre(esp.nombre);
   349→   349→  var detalleEtiqueta = null;
   350→   350→  var stockKey = (talla === 'grande') ? 'stockGrande' : 'stockChico';
   351→   351→  if (etiqueta) {
   352→   352→    var etqDisp = Number(etiqueta[stockKey]) || 0;
   353→   353→    if (etqDisp < cantidad) {
   354→   354→      throw new Error('Stock insuficiente de ETIQUETAS ' + talla.toUpperCase() + ' de "' + esp.nombre + '". Necesario: ' + cantidad + ', Disponible: ' + etqDisp);
   355→   355→    }
   356→   356→    etiqueta[stockKey] = etqDisp - cantidad;
   357→   357→    detalleEtiqueta = { etiquetaNombre: esp.nombre, talla: talla, cantidadConsumida: cantidad, stockRestante: etiqueta[stockKey] };
   358→   358→  } else {
   359→   359→    throw new Error('No hay etiquetas fisicas registradas para "' + esp.nombre + '". Compralas en Compras primero.');
   360→   360→  }
   361→   361→
   362→   362→  // Consume bolsa, add frascos
   363→   363→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   364→   364→  esp.stockFrascos = (esp.stockFrascos || 0) + cantidad;
   365→   365→
   366→   366→  // Record
   367→   367→  var prodId = nextId('producciones');
   368→   368→  var produccion = {
   369→   369→    id: prodId, tipo: 'especia', especiaId: especiaId, especiaNombre: esp.nombre,
   370→   370→    categoria: esp.categoria || '', talla: talla, grsPorFrasco: grsPorFrasco, cantidad: cantidad,
   371→   371→    grsConsumidos: grsTotal, etiquetaUsada: detalleEtiqueta,
   372→   372→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   373→   373→  };
   374→   374→  _db.producciones[prodId] = produccion;
   375→   375→  _saveToFirebase();
   376→   376→  _notify('create', 'producciones', prodId);
   377→   377→  _notify('update', 'especias', especiaId);
   378→   378→  return { especia: esp, produccion: produccion };
   379→   379→}

   381→   381→/** Producir frascos de BLEND: consume grs de especias (bolsa) + etiquetas (por talla), agrega frascos al blend */
   382→   382→function producirBlend(blendId, cantidad, talla) {
   383→   383→  _ensureStructure();
   384→   384→  var blend = _db.blends[blendId];
   385→   385→  if (!blend) throw new Error('Blend no encontrado');
   386→   386→  cantidad = Number(cantidad) || 0;
   387→   387→  talla = (talla === 'grande') ? 'grande' : 'chico';
   388→   388→  if (cantidad <= 0) throw new Error('Cantidad debe ser mayor a 0');
   389→   389→
   390→   390→  var ingredientes = blend.ingredientes || [];
   391→   391→  var detalleIngredientes = [];
   392→   392→  var detalleEtiqueta = null;
   393→   393→
   394→   394→  // 1. Check especias stockBolsa (grams)
   395→   395→  for (var i = 0; i < ingredientes.length; i++) {
   396→   396→    var ing = ingredientes[i];
   397→   397→    var esp = _db.especias[ing.especiaId];
   398→   398→    if (!esp) throw new Error('Especia no encontrada: ' + ing.especiaId);
   399→   399→    var grsNeeded = (ing.cantidad || 0) * cantidad;
   400→   400→    if ((esp.stockBolsa || 0) < grsNeeded) {
   401→   401→      throw new Error('Stock insuficiente en BOLSA de "' + esp.nombre + '". Necesario: ' + grsNeeded + 'grs, Disponible: ' + (esp.stockBolsa || 0) + 'grs');
   402→   402→    }
   403→   403→  }
   404→   404→  // 2. Consume especias from bolsa
   405→   405→  for (var i = 0; i < ingredientes.length; i++) {
   406→   406→    var ing = ingredientes[i];
   407→   407→    var esp = _db.especias[ing.especiaId];
   408→   408→    var grsNeeded = (ing.cantidad || 0) * cantidad;
   409→   409→    esp.stockBolsa = (esp.stockBolsa || 0) - grsNeeded;
   410→   410→    detalleIngredientes.push({
   411→   411→      especiaId: ing.especiaId, especiaNombre: esp.nombre,
   412→   412→      grsPorFrasco: ing.cantidad || 0, grsTotal: grsNeeded
   413→   413→    });
   414→   414→  }
   415→   415→
   416→   416→  // 3. Check & consume etiqueta por talla
   417→   417→  var stockKey = (talla === 'grande') ? 'stockGrande' : 'stockChico';
   418→   418→  var etiqueta = _findEtiquetaByNombre(blend.nombre);
   419→   419→  if (etiqueta) {
   420→   420→    var etqDisp = Number(etiqueta[stockKey]) || 0;
   421→   421→    if (etqDisp < cantidad) {
   422→   422→      throw new Error('Stock insuficiente de ETIQUETAS ' + talla.toUpperCase() + ' de "' + blend.nombre + '". Necesario: ' + cantidad + ', Disponible: ' + etqDisp);
   423→   423→    }
   424→   424→    etiqueta[stockKey] = etqDisp - cantidad;
   425→   425→    detalleEtiqueta = { etiquetaNombre: etiqueta.nombre, talla: talla, cantidadConsumida: cantidad, stockRestante: etiqueta[stockKey] };
   426→   426→  } else {
   427→   427→    throw new Error('No hay etiquetas fisicas registradas para "' + blend.nombre + '". Compralas en Compras primero.');
   428→   428→  }
   429→   429→
   430→   430→  // 4. Add blend frascos
   431→   431→  blend.stockFrascos = (blend.stockFrascos || 0) + cantidad;
   432→   432→
   433→   433→  // 5. Record
   434→   434→  var prodId = nextId('producciones');
   435→   435→  var produccion = {
   436→   436→    id: prodId, tipo: 'blend', blendId: blendId, blendNombre: blend.nombre,
   437→   437→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   438→   438→    ingredientesUsados: detalleIngredientes, etiquetaUsada: detalleEtiqueta,
   439→   439→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   440→   440→  };
   441→   441→  _db.producciones[prodId] = produccion;
   442→   442→  _saveToFirebase();
   443→   443→  _notify('create', 'producciones', prodId);
   444→   444→  _notify('update', 'blends', blendId);
   445→   445→  return { blend: blend, produccion: produccion };
   446→   446→}

   520→   520→function saveCompra(data) {
   521→   521→  _ensureStructure();
   522→   522→  var isNew = !data.id;
   523→   523→  if (isNew) {
   524→   524→    data.id = nextId('compras');

   527→   527→    data.items = data.items || [];
   528→   528→    data.total = Number(data.total) || 0;
   529→   529→  }
   530→   530→  if (isNew) {
   531→   531→    for (var i = 0; i < data.items.length; i++) {
   532→   532→      var item = data.items[i];
   533→   533→      if (item.tipo === 'etiqueta') {
   534→   534→        var etq = _getOrCreateEtiqueta(item.etiquetaNombre);
   535→   535→        var talla = item.talla || 'chico';

   537→   537→          etq.stockGrande = (etq.stockGrande || 0) + (Number(item.cantidad) || 0);
   538→   538→        } else {
   539→   539→          etq.stockChico = (etq.stockChico || 0) + (Number(item.cantidad) || 0);
   540→   540→        }
   541→   541→      } else if (item.tipo === 'especia' && item.especiaId && _db.especias[item.especiaId]) {
   542→   542→        // Add to stockBolsa (grams of raw spice)
   543→   543→        _db.especias[item.especiaId].stockBolsa = (_db.especias[item.especiaId].stockBolsa || 0) + (Number(item.cantidad) || 0);
   544→   544→      }
   545→   545→    }
   546→   546→  }
   547→   547→  _db.compras[data.id] = data;
   548→   548→  _saveToFirebase();
   549→   549→  _notify(isNew ? 'create' : 'update', 'compras', data.id);

   551→   551→}

   571→   571→    data.id = nextId('ventas');

   574→   574→    data.items = data.items || [];
   575→   575→    data.total = Number(data.total) || 0;
   576→   576→  }
   577→   577→  if (isNew) {
   578→   578→    for (var i = 0; i < data.items.length; i++) {
   579→   579→      var item = data.items[i];
   580→   580→      var producto;
   581→   581→      if (item.tipo === 'especia') {

   586→   586→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   587→   587→      }
   588→   588→      var cant = Number(item.cantidad) || 0;
   589→   589→      var stockDisponible = (producto.stockFrascos || 0);

   592→   592→      }
   593→   593→      item.productoNombre = producto.nombre;
   594→   594→      producto.stockFrascos = (producto.stockFrascos || 0) - cant;
   595→   595→    }
   596→   596→  }
   597→   597→  _db.ventas[data.id] = data;
   598→   598→  _saveToFirebase();
   599→   599→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);

   657→   657→function getStats() {
   658→   658→  var db = getDB();
   659→   659→  var especias = _filterValid(Object.values(db.especias || {}));
   660→   660→  var blends = _filterValid(Object.values(db.blends || {}));
   661→   661→  var ventas = _filterValid(Object.values(db.ventas || {}));
   662→   662→  var compras = _filterValid(Object.values(db.compras || {}));
   663→   663→  var etiquetas = _filterValid(Object.values(db.etiquetas || {}));
   664→   664→
   665→   665→  var today = new Date().toISOString().slice(0, 10);
   666→   666→  var mes = new Date().toISOString().slice(0, 7);
   667→   667→  var ventasHoy = ventas.filter(function(v) { return v.fecha === today; });
   668→   668→  var ventasMes = ventas.filter(function(v) { return v.fecha && v.fecha.startsWith(mes); });
   669→   669→  var comprasMes = compras.filter(function(c) { return c.fecha && c.fecha.startsWith(mes); });
   670→   670→  var etqBajoStock = etiquetas.filter(function(e) {
   671→   671→    var chico = Number(e.stockChico) || 0;
   672→   672→    var grande = Number(e.stockGrande) || 0;
   673→   673→    return chico + grande <= 5;
   674→   674→  });
   675→   675→  var totalFrascos = especias.reduce(function(s, e) { return s + (e.stockFrascos || 0); }, 0) +
   676→   676→                      blends.reduce(function(s, b) { return s + (b.stockFrascos || 0); }, 0);
   677→   677→
   678→   678→  return {
   679→   679→    totalEspecias: especias.length,
   680→   680→    totalBlends: blends.length,
   681→   681→    totalEtiquetas: etiquetas.length,
   682→   682→    totalFrascos: totalFrascos,
   683→   683→    totalVentas: ventas.length,
   684→   684→    totalCompras: compras.length,
   685→   685→    ventasHoy: ventasHoy.length,
   686→   686→    totalVentasHoy: ventasHoy.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   687→   687→    ventasMes: ventasMes.length,
   688→   688→    totalVentasMes: ventasMes.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   689→   689→    totalComprasMes: comprasMes.reduce(function(s, c) { return s + (Number(c.total) || 0); }, 0),
   690→   690→    especiasBajoStockBolsa: especias.filter(function(e) { return (e.stockBolsa || 0) <= 50; }),
   691→   691→    especiasBajoStockFrascos: especias.filter(function(e) { return (e.stockFrascos || 0) <= 3; }),
   692→   692→    blendsBajoStockFrascos: blends.filter(function(b) { return (b.stockFrascos || 0) <= 3; }),
   693→   693→    etiquetasBajoStock: etqBajoStock
   694→   694→  };
   695→   695→}
