  1131→  handleOCRResult(text) {
  1132→    var statusEl = document.getElementById('cam-status');
  1133→    var confirmArea = document.getElementById('cam-confirm-area');
  1134→    var detectedTextEl = document.getElementById('cam-detected-text');
  1135→    var prodSelect = document.getElementById('cam-prod-select');
  1136→    if (!text || text.length < 2) {
  1137→      if (statusEl) statusEl.innerHTML = '<span style="color:var(--red)">No se detecto texto. Intenta de nuevo.</span>';
  1138→      setTimeout(function() { if (statusEl) statusEl.textContent = 'Apunta la camara a la etiqueta del producto'; }, 2000);
  1139→      return;
  1140→    }
  1141→    var especias = ArcanoDB.getEspecias();
  1142→    var blends = ArcanoDB.getBlends();
  1143→    var allProducts = [];
  1144→    for (var i = 0; i < especias.length; i++) { allProducts.push({ tipo: 'especia', producto: especias[i] }); }
  1145→    for (var i = 0; i < blends.length; i++) { allProducts.push({ tipo: 'blend', producto: blends[i] }); }
  1146→    var ocrLower = text.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  1147→    var scored = [];
  1148→    for (var i = 0; i < allProducts.length; i++) {
  1149→      var p = allProducts[i];
  1150→      var name = (p.producto.nombre || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  1151→      var nameWords = name.split(/\s+/);
  1152→      var matchCount = 0;
  1153→      for (var w = 0; w < nameWords.length; w++) {
  1154→        if (nameWords[w].length < 2) continue;
  1155→        if (ocrLower.indexOf(nameWords[w]) !== -1) matchCount++;
  1156→      }
  1157→      var score = nameWords.length > 0 ? matchCount / nameWords.length : 0;
  1158→      if (ocrLower.indexOf(name) !== -1) score = Math.max(score, 1.0);
  1159→      if (name.length >= 3 && ocrLower.indexOf(name.substring(0, Math.min(name.length, 6))) !== -1) score = Math.max(score, 0.7);
  1160→      if (score >= 0.5) scored.push({ tipo: p.tipo, producto: p.producto, score: score });
  1161→    }
  1162→    scored.sort(function(a, b) { return b.score - a.score; });
  1163→    if (confirmArea) confirmArea.style.display = 'block';
  1164→    if (detectedTextEl) detectedTextEl.textContent = 'Texto leido: "' + text.substring(0, 80) + (text.length > 80 ? '...' : '') + '"';

  1166→      prodSelect.innerHTML = '<option value="">Seleccionar producto</option>';
  1167→      if (scored.length > 0) {
  1168→        for (var i = 0; i < Math.min(scored.length, 5); i++) {
  1169→          var s = scored[i];
  1170→          var pct = Math.round(s.score * 100);
  1171→          prodSelect.innerHTML += '<option value="' + s.tipo + '|' + s.producto.id + '">' + s.producto.nombre + ' (' + pct + '%)</option>';
  1172→        }
  1173→        if (scored[0].score >= 0.7) {
  1174→          prodSelect.value = scored[0].tipo + '|' + scored[0].producto.id;
  1175→        }
  1176→        if (statusEl) statusEl.innerHTML = '<span style="color:var(--green)">Producto detectado - confirma abajo</span>';
  1177→      } else {
  1178→        if (statusEl) statusEl.innerHTML = '<span style="color:var(--red)">No se encontro producto. Selecciona manualmente.</span>';
  1179→        for (var i = 0; i < allProducts.length; i++) {
  1180→          var ap = allProducts[i];
  1181→          prodSelect.innerHTML += '<option value="' + ap.tipo + '|' + ap.producto.id + '">' + ap.producto.nombre + '</option>';
