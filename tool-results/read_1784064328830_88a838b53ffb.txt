  1723→  1723→
  1724→  1724→  // If overwrite, clear existing products of this type

  1735→  1735→    var existing = db.productos.find(function(p) {
  1736→  1736→      return p.nombre.toLowerCase() === nombre.toLowerCase() && p.tipo === tipo;
  1737→  1737→    });
  1738→  1738→
  1739→  1739→    if (existing) {

  1759→  1759→        nombre: nombre,
  1760→  1760→        tipo: tipo,
  1761→  1761→        unidad: colMap.unidad ? String(r[colMap.unidad] || 'gr').trim() : 'gr',
  1762→  1762→        precioCosto: colMap.precioCosto ? (parseFloat(r[colMap.precioCosto]) || 0) : 0,
  1763→  1763→        precioVenta: colMap.precioVenta ? (parseFloat(r[colMap.precioVenta]) || 0) : 0,
  1764→  1764→        stock: colMap.stock ? (parseFloat(r[colMap.stock]) || 0) : 0,

  1767→  1767→        notas: notas.join(' | '),

  1769→  1769→        origen: colMap.origen ? String(r[colMap.origen] || '').trim() : '',
  1770→  1770→        usoPrincipal: colMap.uso ? String(r[colMap.uso] || '').trim() : '',
  1771→  1771→        perfil: colMap.perfil ? String(r[colMap.perfil] || '').trim() : '',
  1772→  1772→        creadoEn: now,
  1773→  1773→        actualizadoEn: now
  1774→  1774→      };
  1775→  1775→      db.productos.push(prod);
  1776→  1776→      imported++;
  1777→  1777→    }
  1778→  1778→  });
  1779→  1779→
  1780→  1780→  saveDB();
  1781→  1781→  closeModal();
  1782→  1782→  renderAjustes();
  1783→  1783→  toast('Importados: ' + imported + ' | Actualizados: ' + updated + ' | Ignorados: ' + skipped);
  1784→  1784→}
