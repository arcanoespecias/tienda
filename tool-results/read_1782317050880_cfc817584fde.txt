   343→   343→    const i = db.ventas.findIndex(x => x.id === editingVentaId);
   344→   344→    if (i >= 0) db.ventas[i] = v;
   345→   345→  } else {

   396→   396→function renderReportes() {
   397→   397→  const db   = getDB(); const c = db.costos;
   398→   398→  const comp = db.ventas.filter(v => v.estado !== 'cancelada');
   399→   399→  const total = comp.reduce((s, v) => s + v.total, 0);
   400→   400→  const mesKey = new Date().toLocaleDateString('es-AR', { year:'numeric', month:'long' });
   401→   401→
   402→   402→  // Por mes
   403→   403→  const porMes = {};
   404→   404→  comp.forEach(v => {
   405→   405→    const k = new Date(v.fecha).toLocaleDateString('es-AR', { year:'numeric', month:'long' });
   406→   406→    if (!porMes[k]) porMes[k] = { count:0, total:0 };
   407→   407→    porMes[k].count++; porMes[k].total += v.total;
   408→   408→  });
   409→   409→  const mesActual = porMes[mesKey] || { count:0, total:0 };
   410→   410→
   411→   411→  document.getElementById('rep-stats').innerHTML = `
   412→   412→    <div class="stat"><div class="stat-label">Total histórico</div><div class="stat-value">${fmt(total)}</div></div>
   413→   413→    <div class="stat"><div class="stat-label">Ventas totales</div><div class="stat-value">${comp.length}</div></div>
   414→   414→    <div class="stat"><div class="stat-label">Este mes</div><div class="stat-value">${fmt(mesActual.total)}</div><div class="stat-sub">${mesActual.count} ventas</div></div>
   415→   415→  `;
   416→   416→
   417→   417→  document.getElementById('rep-mes-tabla').innerHTML = Object.keys(porMes).reverse().slice(0,12).map(k => `<tr>
   418→   418→    <td>${k}</td><td>${porMes[k].count}</td>
   419→   419→    <td class="fw7 text-gold">${fmt(porMes[k].total)}</td>
   420→   420→    <td>${fmt(porMes[k].total / porMes[k].count)}</td>
   421→   421→  </tr>`).join('') || '<tr><td colspan="4" class="text-muted">Sin datos</td></tr>';
   422→   422→
   423→   423→  // Por producto
   424→   424→  const porProd = {};
   425→   425→  comp.forEach(v => v.items.forEach(i => {
   426→   426→    if (!porProd[i.nombre]) porProd[i.nombre] = { qty:0, total:0 };
   427→   427→    porProd[i.nombre].qty   += i.qty;
   428→   428→    porProd[i.nombre].total += i.precio * i.qty;
   429→   429→  }));
   430→   430→  const prodList = Object.entries(porProd).sort((a,b) => b[1].total - a[1].total);
   431→   431→  document.getElementById('rep-prod-tabla').innerHTML = prodList.map(([n,{qty,t2}],_,arr) => {
   432→   432→    const tot = porProd[n].total;
   433→   433→    return `<tr>
   434→   434→      <td>${n}</td><td>${porProd[n].qty}</td>
   435→   435→      <td class="fw7 text-gold">${fmt(tot)}</td>
   436→   436→      <td>${total ? Math.round(tot/total*100)+'%' : '—'}</td>
   437→   437→    </tr>`;
   438→   438→  }).join('') || '<tr><td colspan="4" class="text-muted">Sin datos</td></tr>';
   439→   439→
   440→   440→  // Rentabilidad de blends
   441→   441→  document.getElementById('rep-rent-tabla').innerHTML = db.blends.map(bl => {
   442→   442→    const cd = calcBlendCostData(bl, c);
   443→   443→    return `<tr>
   444→   444→      <td><strong>${bl.nombre}</strong></td>
   445→   445→      <td>${fmt(cd.costoKg)}</td>
   446→   446→      <td>${fmt(cd.totalC)} ${bl.pVentaChico ? `<span class="badge ${cd.margenC>0?'bg':'br'} ml-4">${Math.round(cd.margenC)}%</span>` : '—'}</td>
   447→   447→      <td>${fmt(cd.totalG)} ${bl.pVentaGrande ? `<span class="badge ${cd.margenG>0?'bg':'br'}">${Math.round(cd.margenG)}%</span>` : '—'}</td>
   448→   448→    </tr>`;
   449→   449→  }).join('') || '<tr><td colspan="4" class="text-muted">Sin blends</td></tr>';
   450→   450→}
