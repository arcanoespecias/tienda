     6→function openModalBlend(id = null) {
     7→  editingBlendId = id;
     8→  blendIngRows = [];
     9→  document.getElementById('modal-blend-titulo').textContent = id ? 'Editar blend' : 'Nuevo Blend';
    10→  const db = getDB();
    11→  if (id) {
    12→    const bl = db.blends.find(b => b.id === id);
    13→    if (bl) {
    14→      document.getElementById('bl-nombre').value       = bl.nombre;
    15→      document.getElementById('bl-peso-c').value       = bl.pesoChico || '';
    16→      document.getElementById('bl-peso-g').value       = bl.pesoGrande || '';
    17→      document.getElementById('bl-pventa-c').value     = bl.pVentaChico || '';
    18→      document.getElementById('bl-pventa-g').value     = bl.pVentaGrande || '';
    19→      document.getElementById('bl-notas').value        = bl.notas || '';
    20→      blendIngRows = bl.ingredientes.map(i => ({ ...i, _rid: nid() }));
    21→    }
    22→  } else {
    23→    document.getElementById('bl-nombre').value    = '';
    24→    document.getElementById('bl-peso-c').value    = '';
    25→    document.getElementById('bl-peso-g').value    = '';
    26→    document.getElementById('bl-pventa-c').value  = '';
    27→    document.getElementById('bl-pventa-g').value  = '';
    28→    document.getElementById('bl-notas').value     = '';
    29→    blendIngRows = [{ _rid: nid(), espId: null, gramos: 0 }];
    30→  }
    31→  renderIngRows();
    32→  calcBlendModal();
    33→  openModal('modal-blend');
    34→}

    47→function renderIngRows() {
    48→  const esp = getDB().especias;
    49→  const opts = esp.map(e => `<option value="${e.id}">${e.nombre} (${fmt(e.precioKg)}/kg)</option>`).join('');
    50→  document.getElementById('blend-ings-rows').innerHTML = blendIngRows.map(r => `
    51→    <div class="ing-row" id="row-${r._rid}">
    52→      <select onchange="ingChange(${r._rid},'espId',this.value)">
    53→        <option value="">-- Especia --</option>${opts}
    54→      </select>
    55→      <input type="number" min="0" max="1000" placeholder="gramos" value="${r.gramos || ''}"
    56→        oninput="ingChange(${r._rid},'gramos',+this.value)">
    57→      <div class="ing-cost" id="ic-${r._rid}">$0</div>
    58→      <button class="btn btn-red btn-sm" onclick="removeIngRow(${r._rid})">×</button>
    59→    </div>
    60→  `).join('');

    62→    const sel = document.querySelector(`#row-${r._rid} select`);
    63→    if (sel && r.espId) sel.value = r.espId;
    64→    updateIngCost(r);
    65→  });

    68→function ingChange(rid, field, val) {
    69→  const r = blendIngRows.find(x => x._rid === rid);
    70→  if (r) { r[field] = val; updateIngCost(r); calcBlendModal(); }

    73→function updateIngCost(r) {
    74→  const el = document.getElementById('ic-' + r._rid);
    75→  if (!el) return;
    76→  if (!r.espId || !r.gramos) { el.textContent = '$0'; return; }
    77→  const esp = getDB().especias.find(e => e.id == r.espId);
    78→  el.textContent = esp ? fmt((esp.precioKg / 1000) * r.gramos) : '$0';
    79→}

    81→function calcBlendModal() {
    82→  const db = getDB(); const c = db.costos;
    83→  let costoKg = 0, totalG = 0;
    84→  blendIngRows.forEach(r => {
    85→    if (!r.espId || !r.gramos) return;
    86→    const esp = db.especias.find(e => e.id == r.espId);
    87→    if (!esp) return;
    88→    costoKg += (esp.precioKg / 1000) * r.gramos;
    89→    totalG  += +r.gramos;
    90→  });
    91→  const costo100 = costoKg / 10;
    92→  const pC = +document.getElementById('bl-peso-c')?.value || 0;
    93→  const pG = +document.getElementById('bl-peso-g')?.value || 0;
    94→  const mpC = (costoKg / 1000) * pC;
    95→  const mpG = (costoKg / 1000) * pG;
    96→  const fC = c.envChico  + c.pkgChico  + c.etiqueta + c.mo + c.otros;
    97→  const fG = c.envGrande + c.pkgGrande + c.etiqueta + c.mo + c.otros;
    98→
    99→  const set = (id, val) => { const el = document.getElementById(id); if (el) el.textContent = val; };
   100→  set('bc-kg',    fmt(costoKg));
   101→  set('bc-100',   fmt(costo100));
   102→  set('bc-grams', `${totalG}g de 1000g`);
   103→  set('bc-mp-c',  fmt(mpC));
   104→  set('bc-fij-c', fmt(fC));
   105→  set('bc-tot-c', fmt(mpC + fC));
   106→  set('bc-mp-g',  fmt(mpG));
   107→  set('bc-fij-g', fmt(fG));
   108→  set('bc-tot-g', fmt(mpG + fG));
   109→
   110→  const gramEl = document.getElementById('bc-grams');
   111→  if (gramEl) gramEl.style.color = Math.abs(totalG - 1000) < 1 ? 'var(--green)' : 'var(--yellow)';
   112→}

   114→function guardarBlend() {
   115→  const nombre = document.getElementById('bl-nombre').value.trim();
   116→  if (!nombre) { showAlert('alert-blend','El nombre es obligatorio.','err'); return; }
   117→  const db = getDB();
   118→  const ings = blendIngRows.filter(r => r.espId && r.gramos > 0).map(r => {
   119→    const esp = db.especias.find(e => e.id == r.espId);
   120→    return { espId: +r.espId, nombre: esp?.nombre || '', gramos: +r.gramos, precioKg: esp?.precioKg || 0 };
   121→  });
   122→  if (!ings.length) { showAlert('alert-blend','Agregá al menos una especia.','err'); return; }
   123→  const bl = {
   124→    nombre, ingredientes: ings,
   125→    pesoChico:   +document.getElementById('bl-peso-c').value   || 0,
   126→    pesoGrande:  +document.getElementById('bl-peso-g').value   || 0,
   127→    pVentaChico: +document.getElementById('bl-pventa-c').value || 0,
   128→    pVentaGrande:+document.getElementById('bl-pventa-g').value || 0,
   129→    notas:        document.getElementById('bl-notas').value.trim(),
   130→  };
   131→  if (editingBlendId) {
   132→    bl.id = editingBlendId;
   133→    const i = db.blends.findIndex(b => b.id === editingBlendId);
   134→    if (i >= 0) db.blends[i] = bl;
   135→  } else { bl.id = nid(); db.blends.push(bl); }
   136→  saveDB(db);
   137→  toast(editingBlendId ? 'Blend actualizado ✓' : 'Blend creado ✓');
   138→  setTimeout(() => closeModal('modal-blend'), 700);
   139→  renderBlends();
   140→}

   255→function _doExportBlendsExcel() {
   256→  const db = getDB();
   257→  const c = db.costos;
   258→
   259→  // ---- HOJA 1: Resumen de Blends ----
   260→  const resumenRows = db.blends.map(bl => {
   261→    const cd = calcBlendCostData(bl, c);
   262→    const prod = calcProduccion(bl, db.especias);
   263→    const totalG = bl.ingredientes.reduce((s, i) => s + i.gramos, 0);
   264→    const ingNombres = bl.ingredientes.map(i => i.nombre + ' (' + i.gramos + 'g)').join(', ');
   265→    return {
   266→      'Nombre': bl.nombre,
   267→      'Ingredientes': ingNombres,
   268→      'Cantidad ingredientes': bl.ingredientes.length,
   269→      'Peso total formula': totalG + 'g',
   270→      'Costo/kg': Math.round(cd.costoKg),
   271→      'Costo/100g': Math.round(cd.costo100g),
   272→      'Peso envase chico (g)': bl.pesoChico || '',
   273→      'Costo total chico': Math.round(cd.totalC),
   274→      'Precio venta chico': bl.pVentaChico || '',
   275→      'Margen chico %': cd.margenC !== null ? Math.round(cd.margenC) + '%' : '',
   276→      'Produccion posible chico (u.)': prod.unidadesC,
   277→      'Peso envase grande (g)': bl.pesoGrande || '',

   279→      'Precio venta grande': bl.pVentaGrande || '',
   280→      'Margen grande %': cd.margenG !== null ? Math.round(cd.margenG) + '%' : '',
   281→      'Produccion posible grande (u.)': prod.unidadesG,
   282→      'Notas': bl.notas || ''
   283→    };
   284→  });

   295→        '#': idx + 1,

   301→        'Costo/kg del blend': Math.round(cd.costoKg),
   302→        'Peso envase chico (g)': bl.pesoChico || '',
   303→        'Peso envase grande (g)': bl.pesoGrande || ''
   304→      });
   305→    });
   306→    // Fila subtotal del blend
   307→    ingRows.push({
   308→      'Blend': 'SUBTOTAL: ' + bl.nombre,
   309→      '#': '',
   310→      'Especia': bl.ingredientes.length + ' ingredientes',
   311→      'Gramos en formula': totalG,
   312→      '% sobre 1kg': Math.round(totalG / 10) + '%',
   313→      'Precio/kg especia': '',
   314→      'Costo en formula': Math.round(cd.costoKg),
   315→      'Costo/kg del blend': '',
   316→      'Peso envase chico (g)': '',

   433→function agregarItemVenta() {

   436→  const qty = parseInt(document.getElementById('v-qty').value) || 1;

   453→  if (ex) ex.qty += qty;

   478→function recalcVenta() {
   479→  const sub  = ventaItems.reduce((s, i) => s + i.precio * i.qty, 0);
   480→  const desc = parseFloat(document.getElementById('v-desc')?.value || 0) || 0;

   483→  if (el) el.textContent = fmt(total);

   486→function guardarVenta() {
   487→  if (!ventaItems.length) { showAlert('alert-venta','Agregá al menos un producto.','err'); return; }
   488→  const db   = getDB();
   489→  const desc = parseFloat(document.getElementById('v-desc').value) || 0;
   490→  const sub  = ventaItems.reduce((s, i) => s + i.precio * i.qty, 0);
   491→  const v = {
   492→    items:     [...ventaItems],
   493→    subtotal:  sub,
   494→    descuento: desc,
   495→    total:     Math.max(0, sub - desc),
   496→    estado:    document.getElementById('v-estado').value,
   497→    nota:      document.getElementById('v-nota').value.trim(),
   498→    usuario:   currentUser?.nombre || 'Sistema',
   499→  };
   500→  if (editingVentaId) {
   501→    v.id = editingVentaId;
   502→    const old = db.ventas.find(x => x.id === editingVentaId);
   503→    v.fecha = old?.fecha || new Date().toISOString();
   504→    const i = db.ventas.findIndex(x => x.id === editingVentaId);
   505→    if (i >= 0) db.ventas[i] = v;
   506→  } else {
   507→    v.id = nid(); v.fecha = new Date().toISOString();
   508→    db.ventas.unshift(v);
   509→  }
   510→  saveDB(db);
   511→  toast('Venta guardada ✓');
   512→  setTimeout(() => closeModal('modal-venta'), 700);
   513→  renderVentas(); renderDashboard();
   514→}

   557→function renderReportes() {
   558→  const db   = getDB(); const c = db.costos;
   559→  const comp = db.ventas.filter(v => v.estado !== 'cancelada');
   560→  const total = comp.reduce((s, v) => s + v.total, 0);
   561→  const mesKey = new Date().toLocaleDateString('es-AR', { year:'numeric', month:'long' });
   562→
   563→  // Por mes
   564→  const porMes = {};
   565→  comp.forEach(v => {
   566→    const k = new Date(v.fecha).toLocaleDateString('es-AR', { year:'numeric', month:'long' });
   567→    if (!porMes[k]) porMes[k] = { count:0, total:0 };
   568→    porMes[k].count++; porMes[k].total += v.total;
   569→  });
   570→  const mesActual = porMes[mesKey] || { count:0, total:0 };
   571→
   572→  document.getElementById('rep-stats').innerHTML = `
   573→    <div class="stat"><div class="stat-label">Total histórico</div><div class="stat-value">${fmt(total)}</div></div>
   574→    <div class="stat"><div class="stat-label">Ventas totales</div><div class="stat-value">${comp.length}</div></div>
   575→    <div class="stat"><div class="stat-label">Este mes</div><div class="stat-value">${fmt(mesActual.total)}</div><div class="stat-sub">${mesActual.count} ventas</div></div>
   576→  `;
   577→
   578→  document.getElementById('rep-mes-tabla').innerHTML = Object.keys(porMes).reverse().slice(0,12).map(k => `<tr>
   579→    <td>${k}</td><td>${porMes[k].count}</td>
   580→    <td class="fw7 text-gold">${fmt(porMes[k].total)}</td>
   581→    <td>${fmt(porMes[k].total / porMes[k].count)}</td>
   582→  </tr>`).join('') || '<tr><td colspan="4" class="text-muted">Sin datos</td></tr>';
   583→
   584→  // Por producto
   585→  const porProd = {};
   586→  comp.forEach(v => v.items.forEach(i => {
   587→    if (!porProd[i.nombre]) porProd[i.nombre] = { qty:0, total:0 };
   588→    porProd[i.nombre].qty   += i.qty;
   589→    porProd[i.nombre].total += i.precio * i.qty;
   590→  }));
   591→  const prodList = Object.entries(porProd).sort((a,b) => b[1].total - a[1].total);
   592→  document.getElementById('rep-prod-tabla').innerHTML = prodList.map(([n,{qty,t2}],_,arr) => {
   593→    const tot = porProd[n].total;
   594→    return `<tr>
   595→      <td>${n}</td><td>${porProd[n].qty}</td>
   596→      <td class="fw7 text-gold">${fmt(tot)}</td>
   597→      <td>${total ? Math.round(tot/total*100)+'%' : '—'}</td>
   598→    </tr>`;
   599→  }).join('') || '<tr><td colspan="4" class="text-muted">Sin datos</td></tr>';
   600→
   601→  // Rentabilidad de blends
   602→  document.getElementById('rep-rent-tabla').innerHTML = db.blends.map(bl => {
   603→    const cd = calcBlendCostData(bl, c);
   604→    return `<tr>
   605→      <td><strong>${bl.nombre}</strong></td>
   606→      <td>${fmt(cd.costoKg)}</td>
   607→      <td>${fmt(cd.totalC)} ${bl.pVentaChico ? `<span class="badge ${cd.margenC>0?'bg':'br'} ml-4">${Math.round(cd.margenC)}%</span>` : '—'}</td>
   608→      <td>${fmt(cd.totalG)} ${bl.pVentaGrande ? `<span class="badge ${cd.margenG>0?'bg':'br'}">${Math.round(cd.margenG)}%</span>` : '—'}</td>
   609→    </tr>`;
   610→  }).join('') || '<tr><td colspan="4" class="text-muted">Sin blends</td></tr>';
   611→}
