    13→    13→function renderStorefront() {
    14→    14→  // Build category list from active especias
    15→    15→  var db = getDB();
    16→    16→  var categories = ['todos', 'blends'];
    17→    17→  db.productos.forEach(function(p) {
    18→    18→    if (p.tipo === 'especia' && p.precioVenta > 0 && p.stock > 0 && p.categoria) {
    19→    19→      var found = false;
    20→    20→      categories.forEach(function(c) { if (c === p.categoria) found = true; });
    21→    21→      if (!found) categories.push(p.categoria);
    22→    22→    }
    23→    23→  });
    24→    24→
    25→    25→  // Update category nav in the static HTML
    26→    26→  var storeNav = document.getElementById('store-nav');
    27→    27→  if (storeNav) {
    28→    28→    var navHtml = '';
    29→    29→    categories.forEach(function(cat) {
    30→    30→      var label = cat === 'todos' ? 'Todos' : cat === 'blends' ? 'Blends' : cat;
    31→    31→      var cls = (cat === storeCategory) ? ' store-nav-btn active' : ' store-nav-btn';
    32→    32→      navHtml += '<button class="' + cls + '" data-cat="' + esc(cat) + '">' + esc(label) + '</button>';
    33→    33→    });
    34→    34→    storeNav.innerHTML = navHtml;
    35→    35→    setupStoreNav();
    36→    36→  }
    37→    37→
    38→    38→  renderStoreProducts();
    39→    39→  updateCartUI();
    40→    40→}

   125→   125→function addToCart(isBlend, id, nombre, precio, unidad, stock) {
   126→   126→  var existingIdx = -1;
   127→   127→  cart.forEach(function(item, idx) {
   128→   128→    if (!item.isBlend && !isBlend && item.id === id) {
   129→   129→      existingIdx = idx;
   130→   130→    }
   131→   131→  });
   132→   132→
   133→   133→  if (existingIdx >= 0) {
   134→   134→    if (cart[existingIdx].qty < cart[existingIdx].maxStock) {
   135→   135→      cart[existingIdx].qty++;
   136→   136→    } else {
   137→   137→      toast('Stock maximo alcanzado', 'err');
   138→   138→      return;
   139→   139→    }
   140→   140→  } else {
   141→   141→    cart.push({
   142→   142→      isBlend: false,
   143→   143→      id: id,
   144→   144→      nombre: nombre,
   145→   145→      precio: precio,
   146→   146→      unidad: unidad,
   147→   147→      qty: 1,
   148→   148→      maxStock: stock
   149→   149→    });
   150→   150→  }
   151→   151→
   152→   152→  updateCartUI();
   153→   153→  toast(nombre + ' agregado al carrito');
   154→   154→}

   184→   184→function confirmarFrasco(blendId, frascoId, contexto, precio, stock) {
   185→   185→  var db = getDB();
   186→   186→  var blend = null;
   187→   187→  db.blends.forEach(function(b) { if (b.id === blendId) blend = b; });
   188→   188→  if (!blend) return;
   189→   189→
   190→   190→  var frasco = null;
   191→   191→  db.productos.forEach(function(p) { if (p.id === frascoId) frasco = p; });
   192→   192→  if (!frasco) return;
   193→   193→
   194→   194→  var etiquetaStock = 999999;
   195→   195→  if (blend.etiquetaId) {
   196→   196→    db.productos.forEach(function(p) {
   197→   197→      if (p.id === blend.etiquetaId) etiquetaStock = p.stock;
   198→   198→    });
   199→   199→  }
   200→   200→
   201→   201→  var disponible = Math.min(blend.stock, frasco.stock, etiquetaStock);
   202→   202→  if (disponible <= 0) {
   203→   203→    toast('No hay stock suficiente', 'err');
   204→   204→    closeModal();
   205→   205→    return;
   206→   206→  }
   207→   207→
   208→   208→  if (contexto === 'carrito') {
   209→   209→    var existingIdx = -1;
   210→   210→    cart.forEach(function(item, idx) {
   211→   211→      if (item.isBlend && item.id === blendId && item.frascoId === frascoId) {
   212→   212→        existingIdx = idx;
   213→   213→      }
   214→   214→    });
   215→   215→
   216→   216→    if (existingIdx >= 0) {
   217→   217→      if (cart[existingIdx].qty < cart[existingIdx].maxStock) {
   218→   218→        cart[existingIdx].qty++;
   219→   219→      } else {
   220→   220→        toast('Stock maximo alcanzado', 'err');
   221→   221→        closeModal();
   222→   222→        return;
   223→   223→      }
   224→   224→    } else {
   225→   225→      cart.push({
   226→   226→        isBlend: true,
   227→   227→        id: blendId,
   228→   228→        nombre: blend.nombre,
   229→   229→        precio: precio,
   230→   230→        unidad: 'unidad',
   231→   231→        qty: 1,
   232→   232→        maxStock: disponible,
   233→   233→        frascoId: frascoId,
   234→   234→        frascoNombre: frasco.nombre,
   235→   235→        etiquetaId: blend.etiquetaId || null
   236→   236→      });
   237→   237→    }
   238→   238→
   239→   239→    updateCartUI();
   240→   240→    toast(blend.nombre + ' agregado al carrito');
   241→   241→  } else if (contexto === 'venta') {
   242→   242→    if (typeof agregarBlendAVenta === 'function') {
   243→   243→      agregarBlendAVenta(blendId, frascoId, blend.nombre, precio, disponible);
   244→   244→    }
   245→   245→    closeModal();
   246→   246→    return;
   247→   247→  }
   248→   248→
   249→   249→  closeModal();
   250→   250→}

   252→   252→function updateCartUI() {

   262→   262→function cartQty(idx, delta) {
   263→   263→  if (!cart[idx]) return;
   264→   264→  cart[idx].qty += delta;
   265→   265→  if (cart[idx].qty <= 0) {
   266→   266→    cart.splice(idx, 1);
   267→   267→  } else if (cart[idx].qty > cart[idx].maxStock) {
   268→   268→    cart[idx].qty = cart[idx].maxStock;
   269→   269→    toast('Stock maximo alcanzado', 'err');
   270→   270→  }
   271→   271→  updateCartUI();
   272→   272→  renderCartDrawer();
   273→   273→}

   365→   365→function submitOrder() {
   366→   366→  if (!cart || cart.length === 0) {
   367→   367→    toast('El carrito esta vacio', 'err');
   368→   368→    return;
   369→   369→  }
   370→   370→
   371→   371→  var db = getDB();
   372→   372→  var total = 0;
   373→   373→  var items = [];
   374→   374→
   375→   375→  cart.forEach(function(ci) {
   376→   376→    var subtotal = ci.precio * ci.qty;
   377→   377→    total += subtotal;
   378→   378→    items.push({
   379→   379→      isBlend: ci.isBlend,
   380→   380→      id: ci.id,
   381→   381→      nombre: ci.nombre,
   382→   382→      precio: ci.precio,
   383→   383→      unidad: ci.unidad,
   384→   384→      qty: ci.qty,
   385→   385→      frascoId: ci.frascoId || null,
   386→   386→      frascoNombre: ci.frascoNombre || '',
   387→   387→      etiquetaId: ci.etiquetaId || null,
   388→   388→      subtotal: subtotal
   389→   389→    });
   390→   390→  });
   391→   391→
   392→   392→  var venta = {
   393→   393→    id: nextId(),
   394→   394→    estado: 'pendiente',
   395→   395→    tipo: 'tienda',
   396→   396→    creadoPor: null,
   397→   397→    fecha: new Date().toISOString(),
   398→   398→    cliente: 'Tienda Online',
   399→   399→    items: items,
   400→   400→    total: total,
   401→   401→    notas: ''
   402→   402→  };
   403→   403→  db.ventas.push(venta);
   404→   404→
   405→   405→  // Process stock deductions
   406→   406→  cart.forEach(function(ci) {
   407→   407→    if (!ci.isBlend) {
   408→   408→      // Especia: deduct stock from product
   409→   409→      db.productos.forEach(function(p) {
   410→   410→        if (p.id === ci.id) {
   411→   411→          p.stock -= ci.qty;
   412→   412→          addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Venta tienda #' + venta.id);
   413→   413→        }
   414→   414→      });
   415→   415→    } else {
   416→   416→      // Blend: deduct blend stock
   417→   417→      db.blends.forEach(function(b) {
   418→   418→        if (b.id === ci.id) {
   419→   419→          b.stock -= ci.qty;
   420→   420→          addMovimiento('venta', b.id, b.nombre, -ci.qty, 'Venta tienda blend #' + venta.id);
   421→   421→        }
   422→   422→      });
   423→   423→      // Deduct frasco stock
   424→   424→      if (ci.frascoId) {
   425→   425→        db.productos.forEach(function(p) {
   426→   426→          if (p.id === ci.frascoId) {
   427→   427→            p.stock -= ci.qty;
   428→   428→            addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Frasco para venta #' + venta.id);
   429→   429→          }
   430→   430→        });
   431→   431→      }
   432→   432→      // Deduct etiqueta stock
   433→   433→      if (ci.etiquetaId) {
   434→   434→        db.productos.forEach(function(p) {
   435→   435→          if (p.id === ci.etiquetaId) {
   436→   436→            p.stock -= ci.qty;
   437→   437→            addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Etiqueta para venta #' + venta.id);
   438→   438→          }
   439→   439→        });

  1623→  1623→      '<span>' + fmt(cost) + '</span>' +
