    12→var storeCategory = 'todos';

    14→function renderStorefront() {
    15→  var db = getDB();
    16→  var categories = ['todos', 'blends'];
    17→  db.productos.forEach(function(p) {
    18→    if (p.tipo === 'especia' && p.precioVenta > 0 && p.stock > 0 && p.categoria) {
    19→      var found = false;
    20→      categories.forEach(function(c) { if (c === p.categoria) found = true; });
    21→      if (!found) categories.push(p.categoria);
    22→    }
    23→  });
    24→
    25→  var storeNav = document.getElementById('store-nav');
    26→  if (storeNav) {
    27→    var navHtml = '';
    28→    categories.forEach(function(cat) {
    29→      var label = cat === 'todos' ? 'Todos' : cat === 'blends' ? 'Blends' : cat;
    30→      var cls = (cat === storeCategory) ? ' store-nav-btn active' : ' store-nav-btn';
    31→      navHtml += '<button class="' + cls + '" data-cat="' + esc(cat) + '">' + esc(label) + '</button>';
    32→    });
    33→    storeNav.innerHTML = navHtml;
    34→    setupStoreNav();
    35→  }
    36→
    37→  renderStoreProducts();
    38→  updateCartUI();
    39→}

    41→function setupStoreNav() {
    42→  var btns = document.querySelectorAll('.store-nav-btn');
    43→  btns.forEach(function(btn) {
    44→    btn.addEventListener('click', function() {
    45→      storeCategory = btn.getAttribute('data-cat');
    46→      btns.forEach(function(b) { b.classList.toggle('active', b === btn); });
    47→      renderStoreProducts();
    48→    });
    49→  });
    50→}

    52→function renderStoreProducts() {
    53→  var container = document.getElementById('store-products');
    54→  var emptyEl = document.getElementById('store-empty');
    55→  if (!container) return;
    56→
    57→  var db = getDB();
    58→  var html = '';
    59→  var hasProducts = false;
    60→
    61→  db.productos.forEach(function(p) {
    62→    if (p.tipo !== 'especia') return;
    63→    if (p.precioVenta <= 0 || p.stock <= 0) return;
    64→    if (storeCategory !== 'todos' && storeCategory !== p.categoria) return;
    65→
    66→    hasProducts = true;
    67→    html += '<div class="product-card">' +
    68→      '<div class="product-card-type">Especia</div>' +
    69→      '<div class="product-card-name">' + esc(p.nombre) + '</div>' +
    70→      '<div class="product-card-desc">' + esc(p.categoria || '') + '</div>' +
    71→      '<div class="product-card-footer">' +
    72→        '<div>' +
    73→          '<div class="product-card-price">' + fmt(p.precioVenta) + '/gr</div>' +
    74→          '<div class="product-card-stock">' + p.stock + ' gr disponibles</div>' +
    75→        '</div>' +
    76→        '<button class="btn-add-cart" onclick="addToCart(false,' + p.id + ',\'' + jsEsc(p.nombre) + '\',' + p.precioVenta + ',\'' + esc(p.unidad) + '\',' + p.stock + ')">Agregar</button>' +
    77→      '</div>' +
    78→      '</div>';
    79→  });
    80→
    81→  if (storeCategory === 'todos' || storeCategory === 'blends') {
    82→    db.blends.forEach(function(b) {
    83→      if (b.precioVenta <= 0 || (b.stock || 0) <= 0) return;
    84→
    85→      var maxFrascoStock = 0;
    86→      db.productos.forEach(function(p) {
    87→        if (p.tipo === 'frasco' && p.stock > 0 && p.stock > maxFrascoStock) {
    88→          maxFrascoStock = p.stock;
    89→        }
    90→      });
    91→
    92→      var etiquetaStock = 999999;
    93→      if (b.etiquetaId) {
    94→        db.productos.forEach(function(p) {
    95→          if (p.id == b.etiquetaId) etiquetaStock = p.stock;
    96→        });
    97→      }
    98→
    99→      var efectivo = Math.min(b.stock || 0, maxFrascoStock, etiquetaStock);
   100→      if (efectivo <= 0) return;
   101→
   102→      hasProducts = true;
   103→      html += '<div class="product-card">' +
   104→        '<div class="product-card-type" style="color:#e55">Blend</div>' +
   105→        '<div class="product-card-name">' + esc(b.nombre) + '</div>' +
   106→        '<div class="product-card-desc">' + esc(b.formato || 'polvo') + ' - ' + (b.gramosPorUnidad || 100) + 'gr</div>' +
   107→        '<div class="product-card-footer">' +
   108→          '<div>' +
   109→            '<div class="product-card-price">' + fmt(b.precioVenta) + '/und</div>' +
   110→            '<div class="product-card-stock">' + efectivo + ' unidades</div>' +
   111→          '</div>' +
   112→          '<button class="btn-add-cart" onclick="mostrarSelectorFrasco(' + b.id + ',\'carrito\',\'' + jsEsc(b.nombre) + '\',' + b.precioVenta + ',' + efectivo + ')">Agregar</button>' +
   113→        '</div>' +
   114→        '</div>';
   115→    });
   116→  }
   117→
   118→  container.innerHTML = html;
   119→  if (emptyEl) emptyEl.style.display = hasProducts ? 'none' : 'block';
   120→}

   122→function addToCart(isBlend, id, nombre, precio, unidad, stock) {
   123→  var existingIdx = -1;
   124→  cart.forEach(function(item, idx) {
   125→    if (!item.isBlend && !isBlend && item.id === id) {
   126→      existingIdx = idx;
   127→    }
   128→  });
   129→
   130→  if (existingIdx >= 0) {
   131→    if (cart[existingIdx].qty < cart[existingIdx].maxStock) {
   132→      cart[existingIdx].qty++;
   133→    } else {
   134→      toast('Stock maximo alcanzado', 'err');
   135→      return;
   136→    }
   137→  } else {
   138→    cart.push({
   139→      isBlend: false,
   140→      id: id,
   141→      nombre: nombre,
   142→      precio: precio,
   143→      unidad: unidad,
   144→      qty: 1,
   145→      maxStock: stock
   146→    });
   147→  }
   148→
   149→  updateCartUI();
   150→  toast(nombre + ' agregado al carrito');
   151→}

   153→function mostrarSelectorFrasco(blendId, contexto, nombre, precio, stock) {
   154→  var db = getDB();
   155→  var frascos = [];
   156→  db.productos.forEach(function(p) {
   157→    if (p.tipo === 'frasco' && p.stock > 0) frascos.push(p);
   158→  });
   159→
   160→  if (frascos.length === 0) {
   161→    toast('No hay frascos disponibles', 'err');
   162→    return;
   163→  }
   164→
   165→  var html = '<p class="mb-12">Seleccione un frasco para <strong>' + esc(nombre) + '</strong></p>' +
   166→    '<p class="mb-12">Precio: ' + fmt(precio) + ' / unidad</p><div class="frasco-list">';
   167→
   168→  frascos.forEach(function(f) {
   169→    var disponible = Math.min(stock, f.stock);
   170→    if (disponible <= 0) return;
   171→    html += '<div class="frasco-option">' +
   172→      '<div><strong>' + esc(f.nombre) + '</strong><br>Disponible: ' + disponible + ' unidades</div>' +
   173→      '<button class="btn btn-sm" onclick="confirmarFrasco(' + blendId + ',' + f.id + ',\'' + jsEsc(contexto) + '\',' + precio + ',' + stock + ')">Seleccionar</button>' +
   174→      '</div>';
   175→  });
   176→
   177→  html += '</div>';
   178→  openModal('Seleccionar Frasco', html);
   179→}

   181→function confirmarFrasco(blendId, frascoId, contexto, precio, stock) {
   182→  var db = getDB();
   183→  var blend = null;
   184→  db.blends.forEach(function(b) { if (b.id === blendId) blend = b; });
   185→  if (!blend) return;
   186→
   187→  var frasco = null;
   188→  db.productos.forEach(function(p) { if (p.id === frascoId) frasco = p; });
   189→  if (!frasco) return;
   190→
   191→  var etiquetaStock = 999999;
   192→  if (blend.etiquetaId) {
   193→    db.productos.forEach(function(p) {
   194→      if (p.id == blend.etiquetaId) etiquetaStock = p.stock;
   195→    });
   196→  }
   197→
   198→  var disponible = Math.min(blend.stock, frasco.stock, etiquetaStock);
   199→  if (disponible <= 0) {
   200→    toast('No hay stock suficiente', 'err');
   201→    closeModal();
   202→    return;
   203→  }
   204→
   205→  if (contexto === 'carrito') {
   206→    var existingIdx = -1;
   207→    cart.forEach(function(item, idx) {
   208→      if (item.isBlend && item.id === blendId && item.frascoId === frascoId) {
   209→        existingIdx = idx;
   210→      }
   211→    });
   212→
   213→    if (existingIdx >= 0) {
   214→      if (cart[existingIdx].qty < cart[existingIdx].maxStock) {
   215→        cart[existingIdx].qty++;
   216→      } else {
   217→        toast('Stock maximo alcanzado', 'err');
   218→        closeModal();
   219→        return;
   220→      }
   221→    } else {
   222→      cart.push({
   223→        isBlend: true,
   224→        id: blendId,
   225→        nombre: blend.nombre,
   226→        precio: precio,
   227→        unidad: 'unidad',
   228→        qty: 1,
   229→        maxStock: disponible,
   230→        frascoId: frascoId,
   231→        frascoNombre: frasco.nombre,
   232→        etiquetaId: blend.etiquetaId || null
   233→      });
   234→    }
   235→
   236→    updateCartUI();
   237→    toast(blend.nombre + ' agregado al carrito');
   238→  } else if (contexto === 'venta') {
   239→    if (typeof agregarBlendAVenta === 'function') {
   240→      agregarBlendAVenta(blendId, frascoId, blend.nombre, precio, disponible);
   241→    }
   242→    closeModal();
   243→    return;
   244→  }
   245→
   246→  closeModal();
   247→}

   249→function updateCartUI() {
   250→  var countEl = document.getElementById('cart-count');
   251→  var total = 0;
   252→  cart.forEach(function(item) { total += item.qty; });
   253→  if (countEl) {
   254→    countEl.textContent = total;
   255→    countEl.style.display = total > 0 ? 'inline' : 'none';
   256→  }

   259→function cartQty(idx, delta) {
   260→  if (!cart[idx]) return;
   261→  cart[idx].qty += delta;
   262→  if (cart[idx].qty <= 0) {
   263→    cart.splice(idx, 1);
   264→  } else if (cart[idx].qty > cart[idx].maxStock) {
   265→    cart[idx].qty = cart[idx].maxStock;
   266→    toast('Stock maximo alcanzado', 'err');
   267→  }
   268→  updateCartUI();
   269→  renderCartDrawer();
   270→}

   272→function toggleCart() {
   273→  var overlay = document.getElementById('cart-overlay');
   274→  var drawer = document.getElementById('cart-drawer');
   275→  if (!overlay || !drawer) return;
   276→
   277→  var isOpen = drawer.classList.contains('open');
   278→  if (isOpen) {
   279→    drawer.classList.remove('open');
   280→    overlay.classList.remove('open');
   281→  } else {
   282→    renderCartDrawer();
   283→    drawer.classList.add('open');
   284→    overlay.classList.add('open');
   285→  }
   286→}

   288→function renderCartDrawer() {
   289→  var itemsEl = document.getElementById('cart-items');
   290→  var footerEl = document.getElementById('cart-footer');
   291→  if (!itemsEl) return;
   292→
   293→  if (cart.length === 0) {

   299→  var html = '';
   300→  var total = 0;

   302→    var subtotal = item.precio * item.qty;
   303→    total += subtotal;
   304→    html += '<div style="display:flex;align-items:center;gap:10px;padding:10px 0;border-bottom:1px solid var(--border)">' +
   305→      '<div style="flex:1">' +
   306→        '<div style="font-weight:700;font-size:.88rem">' + esc(item.nombre) + '</div>' +
   307→        '<div style="font-size:.75rem;color:var(--muted)">' + fmt(item.precio) + ' x ' + item.qty + ' = ' + fmt(subtotal) + '</div>' +
   308→      '</div>' +
   309→      '<div style="display:flex;align-items:center;gap:6px">' +
   310→        '<button class="btn-icon" onclick="cartQty(' + idx + ',-1)" style="width:28px;height:28px;font-size:1rem">-</button>' +
   311→        '<span style="min-width:20px;text-align:center;font-weight:700">' + item.qty + '</span>' +
   312→        '<button class="btn-icon" onclick="cartQty(' + idx + ',1)" style="width:28px;height:28px;font-size:1rem">+</button>' +
   313→        '<button class="btn-icon" onclick="cart.splice(' + idx + ',1);updateCartUI();toggleCart()" style="width:28px;height:28px;font-size:1rem;color:var(--red)">&times;</button>' +
   314→      '</div></div>';
   315→  });
   316→
   317→  itemsEl.innerHTML = html;
   318→  if (footerEl) {
   319→    footerEl.style.display = 'block';
   320→    var totalEl = document.getElementById('cart-total-val');
   321→    if (totalEl) totalEl.textContent = fmt(total);
   322→  }
   323→}

   325→function submitOrder() {
   326→  if (!cart || cart.length === 0) {
   327→    toast('El carrito esta vacio', 'err');
   328→    return;
   329→  }
   330→
   331→  var db = getDB();
   332→  var total = 0;
   333→  var items = [];
   334→
   335→  cart.forEach(function(ci) {
   336→    var subtotal = ci.precio * ci.qty;
   337→    total += subtotal;
   338→    items.push({
   339→      isBlend: ci.isBlend,
   340→      id: ci.id,
   341→      nombre: ci.nombre,
   342→      precio: ci.precio,
   343→      unidad: ci.unidad,
   344→      qty: ci.qty,
   345→      cantidad: ci.qty,
   346→      frascoId: ci.frascoId || null,
   347→      frascoNombre: ci.frascoNombre || '',
   348→      etiquetaId: ci.etiquetaId || null,
   349→      tipo: ci.isBlend ? 'blend' : 'especia',
   350→      blendId: ci.isBlend ? ci.id : null,
   351→      productoId: ci.isBlend ? null : ci.id,
   352→      subtotal: subtotal
   353→    });
   354→  });
   355→
   356→  var venta = {
   357→    id: nextId(),
   358→    estado: 'completada',
   359→    tipo: 'tienda',
   360→    creadoPor: null,
   361→    creadoPorNombre: 'Tienda Online',
   362→    fecha: new Date().toISOString(),
   363→    cliente: 'Tienda Online',
   364→    items: items,
   365→    total: total,
   366→    notas: ''
   367→  };
   368→  db.ventas.push(venta);
   369→
   370→  cart.forEach(function(ci) {
   371→    if (!ci.isBlend) {
   372→      db.productos.forEach(function(p) {
   373→        if (p.id === ci.id) {
   374→          p.stock = Math.round((p.stock - ci.qty) * 100) / 100;
   375→          addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Venta tienda #' + venta.id, db);
   376→        }
   377→      });
   378→    } else {
   379→      db.blends.forEach(function(b) {
   380→        if (b.id === ci.id) {
   381→          b.stock = (b.stock || 0) - ci.qty;
   382→          addMovimiento('venta', b.id, b.nombre, -ci.qty, 'Venta tienda blend #' + venta.id, db);
   383→        }
   384→      });
   385→      if (ci.frascoId) {
   386→        db.productos.forEach(function(p) {
   387→          if (p.id === ci.frascoId) {
   388→            p.stock -= ci.qty;
   389→            addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Frasco para venta #' + venta.id, db);
   390→          }
   391→        });
   392→      }
   393→      if (ci.etiquetaId) {
   394→        db.productos.forEach(function(p) {
   395→          if (p.id === ci.etiquetaId) {
   396→            p.stock -= ci.qty;
   397→            addMovimiento('venta', p.id, p.nombre, -ci.qty, 'Etiqueta para venta #' + venta.id, db);
   398→          }
   399→        });
   400→      }
   401→    }
   402→  });
   403→
   404→  cart = [];
   405→  saveDB(db);
   406→  var drawer = document.getElementById('cart-drawer');
   407→  var overlay = document.getElementById('cart-overlay');
   408→  if (drawer) drawer.classList.remove('open');
   409→  if (overlay) overlay.classList.remove('open');
   410→  toast('Pedido #' + venta.id + ' creado exitosamente');
   411→  renderStorefront();
   412→}

   424→function renderDashboard() {
   425→  var el = document.getElementById('page-dashboard');
   426→  if (!el) return;
   427→
   428→  var db = getDB();
   429→
   430→  var ventasTotal = 0;
   431→  db.ventas.forEach(function(v) {
   432→    if (v.estado !== 'cancelada') ventasTotal += (v.total || 0);
   433→  });
   434→
   435→  var comprasTotal = 0;
   436→  db.compras.forEach(function(c) {
   437→    if (c.estado === 'recibida') comprasTotal += (c.total || 0);
   438→  });
   439→
   440→  var especiaCount = 0;
   441→  var insumoCount = 0;
   442→  db.productos.forEach(function(p) {
   443→    if (p.tipo === 'especia') especiaCount++;
   444→    else insumoCount++;
   445→  });
   446→
   447→  var blendCount = db.blends.length;
   448→  var blendStock = 0;
   449→  db.blends.forEach(function(b) { blendStock += (b.stock || 0); });
   450→
   451→  var pendientes = 0;
   452→  db.ventas.forEach(function(v) { if (v.estado === 'pendiente') pendientes++; });
   453→
   454→  var lowStockEspecias = [];
   455→  db.productos.forEach(function(p) {
   456→    if (p.tipo === 'especia' && p.stock <= (p.stockMin || 0)) {
   457→      lowStockEspecias.push(p);
   458→    }
   459→  });
   460→
   461→  var lowStockBlends = [];
   462→  db.blends.forEach(function(b) {
   463→    if ((b.stock || 0) <= 0) {
   464→      lowStockBlends.push(b);
   465→    }
   466→  });
   467→
   468→  var html = '<div class="g4">' +
   469→    statCard('Ventas Totales', fmt(ventasTotal), 'Ingreso total') +
   470→    statCard('Compras', fmt(comprasTotal), 'Recibidas') +
   471→    statCard('Especias', String(especiaCount), insumoCount + ' insumos') +
   472→    statCard('Blends', String(blendCount), blendStock + ' uds stock') +
   473→    '</div>';
   474→
   475→  html += '<div class="alerts-section">';
   476→  if (pendientes > 0) {
   477→    html += '<div class="alert-card"><div><strong>' + pendientes + ' ventas pendientes</strong></div>' +
   478→      '<button class="btn btn-sm" onclick="navigateTo(\'ventas\')">Ver Ventas</button></div>';
   479→  }
   480→  if (lowStockEspecias.length > 0) {
   481→    html += '<div class="alert-card alert-warn"><div><strong>' + lowStockEspecias.length + ' especias con stock bajo</strong></div>' +
   482→      '<button class="btn btn-sm" onclick="navigateTo(\'especias\')">Ver Especias</button></div>';
   483→  }
   484→  if (lowStockBlends.length > 0) {
   485→    html += '<div class="alert-card alert-warn"><div><strong>' + lowStockBlends.length + ' blends sin stock</strong></div>' +
   486→      '<button class="btn btn-sm" onclick="navigateTo(\'blends\')">Ver Blends</button></div>';
   487→  }
   488→  if (pendientes === 0 && lowStockEspecias.length === 0 && lowStockBlends.length === 0) {
   489→    html += '<div class="alert-card"><div>Todo al dia</div></div>';
   490→  }
   491→  html += '</div>';
   492→
   493→  var sortedVentas = db.ventas.slice().sort(function(a, b) {
   494→    return new Date(b.fecha) - new Date(a.fecha);
   495→  });
   496→  var recentVentas = sortedVentas.slice(0, 5);
   497→
   498→  html += '<h3 class="section-title">Ventas Recientes</h3>';
   499→  if (recentVentas.length === 0) {
   500→    html += '<p class="empty-msg">No hay ventas registradas</p>';
   501→  } else {
   502→    html += '<div class="table-wrap"><table><thead><tr>' +
   503→      '<th>Fecha</th><th>Cliente</th><th>Total</th><th>Estado</th>' +
   504→      '</tr></thead><tbody>';
   505→    recentVentas.forEach(function(v) {
   506→      var badgeCls = v.estado === 'completada' ? 'badge bg' : v.estado === 'cancelada' ? 'badge br' : 'badge by';
   507→      html += '<tr>' +
   508→        '<td>' + fmtDateTime(v.fecha) + '</td>' +
   509→        '<td>' + esc(v.cliente || 'Mostrador') + '</td>' +
   510→        '<td>' + fmt(v.total) + '</td>' +
   511→        '<td><span class="' + badgeCls + '">' + esc(v.estado) + '</span></td>' +
   512→        '</tr>';
   513→    });
   514→    html += '</tbody></table></div>';
   515→  }

   754→function guardarCompra() {
   755→  var fechaRaw = document.getElementById('cmp-fecha').value;
   756→  var proveedor = (document.getElementById('cmp-proveedor').value || '').trim();
   757→  var notas = (document.getElementById('cmp-notas').value || '').trim();
   758→  var esRecibida = document.getElementById('cmp-recibida').checked;
   759→
   760→  var fecha = fechaRaw ? new Date(fechaRaw + 'T12:00:00').toISOString() : new Date().toISOString();
   761→
   762→  var container = document.getElementById('compra-items');
   763→  var rows = container.querySelectorAll('.compra-item-row');
   764→  var items = [];
   765→  var total = 0;
   766→
   767→  var db = getDB();
   768→
   769→  rows.forEach(function(row) {
   770→    var tipo = row.getAttribute('data-tipo') || '';
   771→    var numInputs = row.querySelectorAll('input[type="number"]');
   772→
   773→    if (tipo === 'especia') {
   774→      var sel = row.querySelector('select');
   775→      var existingId = sel ? Number(sel.value) || 0 : 0;
   776→      var nombreInput = row.querySelector('.cmp-nuevo-nombre');
   777→      var nombre = '';
   778→      var cantidad = numInputs[0] ? Number(numInputs[0].value) || 0 : 0;
   779→      var precioTotal = numInputs[1] ? Number(numInputs[1].value) || 0 : 0;
   780→
   781→      if (cantidad <= 0 || precioTotal <= 0) return;
   782→
   783→      var productId;
   784→      var productoNombre;
   785→
   786→      if (existingId) {
   787→        productId = existingId;
   788→        var found = null;
   789→        db.productos.forEach(function(p) { if (p.id == existingId) found = p; });
   790→        if (!found) return;
   791→        productoNombre = found.nombre;
   792→      } else {
   793→        nombre = nombreInput ? (nombreInput.value || '').trim() : '';
   794→        if (!nombre) return;
   795→
   796→        var existingProduct = null;
   797→        db.productos.forEach(function(p) {
   798→          if (p.nombre.toLowerCase() === nombre.toLowerCase() && p.tipo === 'especia') existingProduct = p;
   799→        });
   800→
   801→        if (existingProduct) {
   802→          productId = existingProduct.id;
   803→          productoNombre = existingProduct.nombre;
   804→        } else {
   805→          productId = nextId();
   806→          productoNombre = nombre;
   807→          var precioPorGr = Math.round(precioTotal / cantidad * 100) / 100;
   808→          db.productos.push({
   809→            id: productId,
   810→            nombre: nombre,
   811→            tipo: 'especia',
   812→            unidad: 'gr',
   813→            stock: 0,
   814→            stockMin: 0,
   815→            costoPor1000gr: Math.round(precioTotal / cantidad * 1000) / 100,
   816→            precioCosto: precioPorGr,
   817→            precioVenta: 0,
   818→            categoria: '',
   819→            proveedor: proveedor,
   820→            creadoEn: new Date().toISOString()
   821→          });
   822→        }
   823→      }
   824→
   825→      var subtotal = cantidad * (precioTotal / cantidad);
   826→      total += precioTotal;

   828→        productoId: productId,
   829→        productoNombre: productoNombre,
   830→        tipo: 'especia',
