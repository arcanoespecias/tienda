   706→   706→  _calcVentaTotal() {
   707→   707→    let total = 0;
   708→   708→    document.querySelectorAll('#f-venta-items .venta-item-row').forEach(row => {
   709→   709→      var tipo = row.querySelector('[data-field="tipo"]').value;
   710→   710→      var prodId = row.querySelector('[data-field="productoId"]').value;
   711→   711→      var tamano = row.querySelector('[data-field="tamano"]').value;
   712→   712→      var cant = Number(row.querySelector('[data-field="cantidad"]').value) || 0;
   713→   713→      var frascos = ArcanoDB.getFrascosParaVender();
   714→   714→      var prod = frascos.find(f => f.id === prodId);
   715→   715→      var precio = prod ? (tamano === 'grande' ? prod.precioGrande : prod.precioChico) : 0;
   716→   716→      total += precio * cant;
   717→   717→    });

   721→   721→  saveVenta() {
   722→   722→    const fecha = document.getElementById('f-venta-fecha').value;
   723→   723→    const vendedorId = document.getElementById('f-venta-vendedor').value;
   724→   724→    const rows = document.querySelectorAll('#f-venta-items .venta-item-row');
   725→   725→    const items = [];
   726→   726→    let total = 0;
   727→   727→    rows.forEach(row => {
   728→   728→      const tipo = row.querySelector('[data-field="tipo"]').value;
   729→   729→      const prodId = row.querySelector('[data-field="productoId"]').value;
   730→   730→      const tamano = row.querySelector('[data-field="tamano"]').value;
   731→   731→      const cant = Number(row.querySelector('[data-field="cantidad"]').value) || 0;
   732→   732→      if (prodId && cant > 0) {
   733→   733→        const frascos = ArcanoDB.getFrascosParaVender();
   734→   734→        const prod = frascos.find(f => f.id === prodId);
   735→   735→        const precio = prod ? (tamano === 'grande' ? prod.precioGrande : prod.precioChico) : 0;
   736→   736→        items.push({ tipo, productoId: prodId, tamano, cantidad: cant, precioUnitario: precio });
   737→   737→        total += precio * cant;
   738→   738→      }
   739→   739→    });
   740→   740→    if (items.length === 0) { alert('Agrega al menos un item'); return; }
   741→   741→    try {
   742→   742→      ArcanoDB.saveVenta({ fecha, vendedorId, items, total });
   743→   743→      document.querySelector('.modal-overlay').remove();
   744→   744→      App.renderPage('ventas');
   745→   745→    } catch (e) { alert(e.message); }
   746→   746→  },
