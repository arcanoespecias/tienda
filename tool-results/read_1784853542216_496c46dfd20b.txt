   165→function initDB() {
   166→  return new Promise(function(resolve) {
   167→    _initFirebase();
   168→
   169→    // Always try localStorage cache for instant UI
   170→    var cached = null;
   171→    try { cached = JSON.parse(localStorage.getItem(DB_KEY)); } catch (e) {}
   172→
   173→    if (cached && cached.meta && cached.meta.version === DB_VERSION && _ensureStructureOn(cached)) {
   174→      _db = cached;
   175→      _ensureStructure();  // ensure new fields exist on cached data
   176→      _ready = true;
   177→      _startFirebaseListener();
   178→      _startPedidosListener();
   179→      resolve();
   180→      return;
   181→    }
   182→
   183→    // Load from Firebase
   184→    if (_firebaseRef) {
   185→      _firebaseRef.once('value').then(function(snap) {
   186→        var fbData = snap.val();
   187→        if (fbData && fbData.meta && fbData.meta.version === DB_VERSION && _ensureStructureOn(fbData)) {
   188→          delete fbData.pedidos;  // pedidos managed separately, not part of admin _db
   189→          _db = fbData;
   190→          _ensureStructure();  // ensure new fields exist on Firebase data
   191→        } else {
   192→          _db = _emptyDB();
   193→          _ensureStructure();
   194→          _saveToFirebase();
   195→        }
   196→        _ready = true;
   197→        _cacheLocal();
   198→        _startFirebaseListener();
   199→        _startPedidosListener();
   200→        resolve();
   201→      }).catch(function() {
   202→        _db = _emptyDB();
   203→        _ensureStructure();
   204→        _ready = true;
   205→        resolve();
   206→      });
   207→    } else {
   208→      _db = _emptyDB();
   209→      _ensureStructure();
   210→      _ready = true;
   211→      resolve();
   212→    }
   213→  });
   214→}

   229→function _startFirebaseListener() {
   230→  if (!_firebaseRef) return;
   231→  _firebaseRef.on('value', function(snap) {
   232→    var data = snap.val();
   233→    if (!data || !data.meta || data.meta.version !== DB_VERSION) return;
   234→    // CRITICAL: skip if local save is pending to prevent overwriting unsaved changes
   235→    if (_localDirty) return;
   236→    delete data.pedidos;  // pedidos managed separately, not part of admin _db
   237→    var prevJson = JSON.stringify(_db);
   238→    _db = data;
   239→    _ensureStructure();
   240→    _cacheLocal();
   241→    var newJson = JSON.stringify(_db);
   242→    if (prevJson !== newJson) {
   243→      _notify('remote_change', '', '');
   244→    }
   245→  });
   246→}

   248→function _startPedidosListener() {
   249→  if (!_pedidosRef) return;
   250→  _pedidosRef.on('value', function(snap) {
   251→    var data = snap.val();
   252→    var prevLen = _pedidos.length;
   253→    _pedidos = [];
   254→    if (data) {
   255→      var keys = Object.keys(data);
   256→      for (var i = 0; i < keys.length; i++) {
   257→        var p = data[keys[i]];
   258→        if (p && typeof p === 'object') { p._key = keys[i]; _pedidos.push(p); }
   259→      }
   260→    }
   261→    _pedidos.sort(function(a, b) { return (b.creado || '').localeCompare(a.creado || ''); });
   262→    if (_pedidos.length !== prevLen) {
   263→      _notifyPedidos();
   264→    }
   265→  });
   266→}

   449→function saveEntrada(data) {
   450→  _ensureStructure();
   451→  var isNew = !data.id;
   452→  if (isNew) {
   453→    data.id = nextId('entradas');
   454→    data.creado = new Date().toISOString();
   455→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   456→    data.items = data.items || [];
   457→    data.total = Number(data.total) || 0;
   458→  }
   459→  if (isNew) {
   460→    for (var i = 0; i < data.items.length; i++) {
   461→      var item = data.items[i];
   462→      var tipo = item.tipo;
   463→      if (tipo === 'especia_grs') {
   464→        // Add grams to especia stockBolsa
   465→        if (item.especiaId && _db.especias[item.especiaId]) {
   466→          _db.especias[item.especiaId].stockBolsa = (_db.especias[item.especiaId].stockBolsa || 0) + (Number(item.cantidad) || 0);
   467→        }
   468→      } else if (tipo === 'envase') {
   469→        var talla = item.talla || 'chico';
   470→        if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   471→        _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) + (Number(item.cantidad) || 0);
   472→      } else if (tipo === 'sticker') {
   473→        var stk = _getOrCreateSticker(item.stickerNombre);
   474→        var t = item.talla || 'chico';
   475→        if (t === 'grande') {
   476→          stk.stockGrande = (stk.stockGrande || 0) + (Number(item.cantidad) || 0);
   477→        } else {
   478→          stk.stockChico = (stk.stockChico || 0) + (Number(item.cantidad) || 0);
   479→        }
   480→      } else if (tipo === 'bolsa') {
   481→        var tallaB = item.talla || 'chico';
   482→        if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   483→        _db.stockBolsas[tallaB] = (_db.stockBolsas[tallaB] || 0) + (Number(item.cantidad) || 0);
   484→      }
   485→    }
   486→  }
   487→  _db.entradas[data.id] = data;
   488→  _saveToFirebase(); _cacheLocal();
   489→  _notify(isNew ? 'create' : 'update', 'entradas', data.id);
   490→  return data;
   491→}

   507→function saveAjuste(data) {

   516→  var cantidad = Number(data.cantidad) || 0;
   517→  if (cantidad === 0) throw new Error('La cantidad no puede ser 0');
   518→  var cat = data.categoria; // 'especia', 'blend', 'envase', 'bolsa', 'sticker'
   519→  var sub = data.subtipo;   // 'pala', 'chico', 'grande'

   521→  if (cat === 'especia') {
   522→    var esp = _db.especias[data.productoId];

   525→      var nv = (Number(esp.stockBolsa) || 0) + cantidad;
   526→      if (nv < 0) throw new Error('Stock de pala resultante negativo (' + nv + 'g) para ' + esp.nombre);
   527→      esp.stockBolsa = nv;
   528→    } else {
   529→      var field = (sub === 'grande') ? 'stockGrande' : 'stockChico';
   530→      var nv = (Number(esp[field]) || 0) + cantidad;
   531→      if (nv < 0) throw new Error('Stock de frascos resultante negativo (' + nv + ') para ' + esp.nombre);
   532→      esp[field] = nv;

   535→  } else if (cat === 'blend') {
   536→    var bl = _db.blends[data.productoId];
   537→    if (!bl) throw new Error('Blend no encontrado');
   538→    var field = (sub === 'grande') ? 'stockGrande' : 'stockChico';
   539→    var nv = (Number(bl[field]) || 0) + cantidad;
   540→    if (nv < 0) throw new Error('Stock resultante negativo (' + nv + ') para ' + bl.nombre);
   541→    bl[field] = nv;
   542→    data.productoNombre = bl.nombre;
   543→  } else if (cat === 'envase') {
   544→    if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   545→    var t = (sub === 'grande') ? 'grande' : 'chico';
   546→    var nv = (_db.stockEnvases[t] || 0) + cantidad;
   547→    if (nv < 0) throw new Error('Stock de frascos ' + t + ' resultante negativo (' + nv + ')');
   548→    _db.stockEnvases[t] = nv;

   550→  } else if (cat === 'bolsa') {

   553→    var nv = (_db.stockBolsas[t] || 0) + cantidad;
   554→    if (nv < 0) throw new Error('Stock de bolsas ' + t + ' resultante negativo (' + nv + ')');

   583→function producirEspecia(especiaId, talla, cantidad) {
   584→  _ensureStructure();
   585→  var esp = _db.especias[especiaId];
   586→  if (!esp) throw new Error('Especia no encontrada');
   587→  talla = (talla === 'grande') ? 'grande' : 'chico';
   588→  cantidad = Number(cantidad) || 0;
   589→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   590→
   591→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   592→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   593→
   594→  var grsTotal = gramosPorFrasco * cantidad;
   595→
   596→  // Check & consume pala (raw material)
   597→  if ((esp.stockBolsa || 0) < grsTotal) {
   598→    throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   599→  }
   600→
   601→  // Check & consume envases
   602→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   603→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   604→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   605→  }
   606→
   607→  // Check & consume stickers
   608→  var stk = _findStickerByNombre(esp.nombre);
   609→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   610→  if (stkStock < cantidad) {
   611→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   612→  }
   613→
   614→  // Check & consume bolsas (packaging)
   615→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   616→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   617→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   618→  }
   619→
   620→  // All checks passed — consume
   621→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   622→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   623→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   624→  if (stk) {
   625→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   626→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   627→  }
   628→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   629→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   630→
   631→  // Record
   632→  var prodId = nextId('producciones');
   633→  var prod = {
   634→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   635→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   636→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   637→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   638→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   639→  };
   640→  _db.producciones[prodId] = prod;
   641→  _saveToFirebase(); _cacheLocal();
   642→  _notify('create', 'producciones', prodId);
   643→  _notify('update', 'especias', especiaId);
   644→  return { producto: esp, produccion: prod };
   645→}

   647→function producirBlend(blendId, talla, cantidad) {
   648→  _ensureStructure();
   649→  var blend = _db.blends[blendId];
   650→  if (!blend) throw new Error('Blend no encontrado');
   651→  talla = (talla === 'grande') ? 'grande' : 'chico';
   652→  cantidad = Number(cantidad) || 0;
   653→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   654→
   655→  var ingredientes = blend.ingredientes || [];
   656→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   657→
   658→  // Check ingredient stock
   659→  var detalleIngredientes = [];
   660→  for (var i = 0; i < ingredientes.length; i++) {
   661→    var ing = ingredientes[i];
   662→    var esp = _db.especias[ing.especiaId];
   663→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   664→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   665→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   666→    var grsNeeded = grsPorFrasco * cantidad;
   667→    if ((esp.stockBolsa || 0) < grsNeeded) {
   668→      throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   669→    }
   670→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   671→  }
   672→
   673→  // Check envases
   674→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   675→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   676→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   677→  }
   678→
   679→  // Check stickers
   680→  var stk = _findStickerByNombre(blend.nombre);
   681→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   682→  if (stkStock < cantidad) {
   683→    throw new Error('Stickers ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   684→  }
   685→
   686→  // Check & consume bolsas (packaging)
   687→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   688→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   689→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   690→  }
   691→
   692→  // All checks passed — consume
   693→  var grsTotalGeneral = 0;
   694→  for (var i = 0; i < detalleIngredientes.length; i++) {
   695→    var d = detalleIngredientes[i];
   696→    var esp = _db.especias[d.especiaId];
   697→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   698→    grsTotalGeneral += d.gramosTotal;
   699→  }
   700→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   701→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   702→  if (stk) {
   703→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   704→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   705→  }
   706→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   707→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   708→
   709→  var prodId = nextId('producciones');
   710→  var prod = {
   711→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   712→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   713→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   714→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   715→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   716→  };
   717→  _db.producciones[prodId] = prod;
   718→  _saveToFirebase(); _cacheLocal();
   719→  _notify('create', 'producciones', prodId);
   720→  _notify('update', 'blends', blendId);
   721→  return { producto: blend, produccion: prod };
   722→}

   734→function saveVenta(data) {
   735→  _ensureStructure();
   736→  var isNew = !data.id;
   737→  if (isNew) {
   738→    data.id = nextId('ventas');
   739→    data.creado = new Date().toISOString();
   740→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   741→    data.items = data.items || [];
   742→    data.total = Number(data.total) || 0;
   743→  }
   744→  if (isNew) {
   745→    for (var i = 0; i < data.items.length; i++) {
   746→      var item = data.items[i];
   747→      var producto;
   748→      if (item.tipo === 'especia') {
   749→        producto = _db.especias[item.productoId];
   750→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   751→      } else {
   752→        producto = _db.blends[item.productoId];
   753→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   754→      }
   755→      item.productoNombre = producto.nombre;
   756→      var cant = Number(item.cantidad) || 0;
   757→      var talla = item.talla || 'chico';
   758→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   759→      if ((producto[stockKey] || 0) < cant) {
   760→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   761→      }
   762→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   763→      item.precioUnitario = Number(item.precioUnitario) || 0;
   764→      item.subtotal = item.precioUnitario * cant;
   765→    }
   766→    // Recalculate total
   767→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   768→  }
   769→  _db.ventas[data.id] = data;
   770→  _saveToFirebase(); _cacheLocal();
   771→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   772→  return data;
   773→}

   907→      id: b.id, nombre: b.nombre, tipo: 'blend', categoria: b.categoria || 'Comidas',
   908→      precioChico: Number(b.precioTiendaChico) || Number(b.precioChico) || 0,
   909→      precioGrande: Number(b.precioTiendaGrande) || Number(b.precioGrande) || 0,
   910→      stockChico: b.stockChico || 0, stockGrande: b.stockGrande || 0,
   911→      region: b.region || '', uso: b.uso || ''
   912→    });
   913→  }
   914→  return products.sort(function(a, b) { return a.nombre.localeCompare(b.nombre); });
   915→}

   930→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   931→function findEspeciaByName(nombre) {
   932→  if (!nombre) return null;
   933→  var target = nombre.trim().toLowerCase();
   934→  var keys = Object.keys(_db.especias || {});
   935→  // 1. Exact match
   936→  for (var i = 0; i < keys.length; i++) {
   937→    var e = _db.especias[keys[i]];
   938→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   939→  }
   940→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   941→  for (var i = 0; i < keys.length; i++) {
   942→    var e = _db.especias[keys[i]];
   943→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   944→  }
   945→  // 3. Target is contained in especia name
   946→  for (var i = 0; i < keys.length; i++) {
   947→    var e = _db.especias[keys[i]];
   948→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   949→  }
   950→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   951→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   952→  for (var w = 0; w < targetWords.length; w++) {
   953→    for (var i = 0; i < keys.length; i++) {
   954→      var e = _db.especias[keys[i]];
   955→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   956→    }
   957→  }
   958→  return null;
   959→}

   961→/** Auto-detect categoria from blend USO field */
   962→function _categoriaFromUso(uso) {
   963→  if (!uso) return 'Comidas';
   964→  var u = uso.toLowerCase();
   965→  // Cocteleria keywords
   966→  if (/\b(gin|ron|vodka|whisky|mojito|mule|vermouth|aperitif|coctel)\b/.test(u)) return 'Cocteleria';
   967→  // Infusiones keywords
   968→  if (/\b(relajant|sueño|digestiv|energiz|té|calidez|respiratorio|meditaci|antioxidant|bienestar|infusi)\b/.test(u)) return 'Infusiones';
   969→  return 'Comidas';
   970→}

   972→/**
   973→ * Import especias and blends from parsed Excel data.
   974→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   975→ *
   976→ * especiasList: [{ nombre, categoria? }]
   977→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   978→ * gramosChico:  default grams for frasco chico (e.g. 30)
   979→ * gramosGrande: default grams for frasco grande (e.g. 80)
   980→ */
   981→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   982→  _ensureStructure();
   983→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   984→
   985→  // 1. Create especias (skip if name already exists)
   986→  for (var i = 0; i < especiasList.length; i++) {
   987→    var esp = especiasList[i];
   988→    var nombre = (esp.nombre || '').trim();
   989→    if (!nombre) continue;
   990→    var existing = findEspeciaByName(nombre);
   991→    if (existing) {
   992→      resultado.especiasExistentes++;
   993→      continue;
   994→    }
   995→    saveEspecia({
   996→      nombre: nombre,
   997→      categoria: esp.categoria || 'Comidas',
   998→      precioChico: 0,
   999→      precioGrande: 0,
  1000→      gramosChico: Number(gramosChico) || 30,
  1001→      gramosGrande: Number(gramosGrande) || 80,
  1002→      stockBolsa: 0,
  1003→      stockChico: 0,
  1004→      stockGrande: 0
  1005→    });
  1006→    resultado.especiasCreadas++;
  1007→  }
  1008→
  1009→  // 2. Create blends (skip if name already exists)
  1010→  for (var j = 0; j < blendsList.length; j++) {
  1011→    var bl = blendsList[j];
  1012→    var nombre = (bl.nombre || '').trim();
  1013→    if (!nombre) continue;
  1014→    var existingBlend = null;
  1015→    var blKeys = Object.keys(_db.blends || {});
  1016→    for (var k = 0; k < blKeys.length; k++) {
  1017→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
  1018→        existingBlend = _db.blends[blKeys[k]];
  1019→        break;
  1020→      }
  1021→    }
  1022→    if (existingBlend) {
  1023→      resultado.blendsExistentes++;
  1024→      continue;
  1025→    }
  1026→
  1027→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
  1028→    var ings = bl.ingredientes || [];
  1029→    var recipeTotal = 0;
  1030→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
  1031→    if (recipeTotal <= 0) recipeTotal = 500;
  1032→
  1033→    var resolvedIngs = [];
  1034→    for (var ii = 0; ii < ings.length; ii++) {
  1035→      var ing = ings[ii];
  1036→      var espObj = findEspeciaByName(ing.especia);
  1037→      if (!espObj) {
  1038→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
  1039→        continue;
  1040→      }
  1041→      var ingG = Number(ing.g) || 0;
  1042→      resolvedIngs.push({
  1043→        especiaId: espObj.id,
  1044→        especiaNombre: espObj.nombre,
  1045→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
  1046→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
  1047→        gramosReceta: ingG
  1048→      });
  1049→    }
  1050→
  1051→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
  1052→    saveBlend({
  1053→      nombre: nombre,
  1054→      categoria: cat,
  1055→      region: bl.region || '',
  1056→      uso: bl.uso || '',
  1057→      precioChico: 0,
  1058→      precioGrande: 0,
  1059→      ingredientes: resolvedIngs,
  1060→      stockChico: 0,
  1061→      stockGrande: 0
  1062→    });
  1063→    resultado.blendsCreados++;
  1064→  }
  1065→
  1066→  return resultado;
  1067→}

  1081→function addProductTag(cat, tagName) {
  1082→  _ensureStructure();
  1083→  tagName = (tagName || '').trim();
  1084→  if (!tagName) return false;
  1085→  if (!_db.productTags) _db.productTags = {};
  1086→  if (!_db.productTags[cat]) _db.productTags[cat] = [];
  1087→  // Check duplicate (case-insensitive)
  1088→  for (var i = 0; i < _db.productTags[cat].length; i++) {
  1089→    if (_db.productTags[cat][i].toLowerCase() === tagName.toLowerCase()) return false;
  1090→  }
  1091→  _db.productTags[cat].push(tagName);
  1092→  _saveToFirebase(); _cacheLocal();
  1093→  return true;
  1094→}

  1096→function removeProductTag(cat, tagName) {
  1097→  _ensureStructure();
  1098→  if (!_db.productTags || !_db.productTags[cat]) return false;
  1099→  var idx = -1;
  1100→  for (var i = 0; i < _db.productTags[cat].length; i++) {
  1101→    if (_db.productTags[cat][i] === tagName) { idx = i; break; }
  1102→  }
  1103→  if (idx < 0) return false;
  1104→  _db.productTags[cat].splice(idx, 1);
  1105→  // Also remove from all products that have this tag
  1106→  var allEsp = getEspecias();
  1107→  for (var e = 0; e < allEsp.length; e++) {
  1108→    if (allEsp[e].tags) {
  1109→      var ti = allEsp[e].tags.indexOf(tagName);
  1110→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
  1111→    }
  1112→  }
  1113→  var allBl = getBlends();
  1114→  for (var b = 0; b < allBl.length; b++) {
  1115→    if (allBl[b].tags) {
  1116→      var bi = allBl[b].tags.indexOf(tagName);
  1117→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
  1118→    }
  1119→  }
  1120→  _saveToFirebase(); _cacheLocal();
  1121→  return true;
  1122→}

  1126→function compressImage(file, maxW, quality, cb) {
  1127→  var reader = new FileReader();
  1128→  reader.onload = function(ev) {
  1129→    var img = new Image();
  1130→    img.onload = function() {
  1131→      var w = img.width, h = img.height;
  1132→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
  1133→      var canvas = document.createElement('canvas');
  1134→      canvas.width = w; canvas.height = h;
  1135→      var ctx = canvas.getContext('2d');
  1136→      ctx.drawImage(img, 0, 0, w, h);
  1137→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
  1138→      cb(null, dataUrl);
  1139→    };
  1140→    img.onerror = function() { cb('Error al cargar imagen'); };
  1141→    img.src = ev.target.result;
  1142→  };
  1143→  reader.onerror = function() { cb('Error al leer archivo'); };
  1144→  reader.readAsDataURL(file);
  1145→}
