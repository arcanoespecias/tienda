   165→function initDB() {
   166→  return new Promise(function(resolve) {
   167→    _initFirebase();
   168→
   169→    // Always try localStorage cache for instant UI
   170→    var cached = null;
   171→    try { cached = JSON.parse(localStorage.getItem(DB_KEY)); } catch (e) {}
   172→
   173→    if (cached && cached.meta && cached.meta.version === DB_VERSION && _ensureStructureOn(cached)) {
   174→      _db = cached;
   175→      _ensureStructure();  // ensure new fields exist on cached data
   176→      _ready = true;
   177→      _startFirebaseListener();
   178→      _startPedidosListener();
   179→      resolve();
   180→      return;
   181→    }
   182→
   183→    // Load from Firebase
   184→    if (_firebaseRef) {
   185→      _firebaseRef.once('value').then(function(snap) {
   186→        var fbData = snap.val();
   187→        if (fbData && fbData.meta && fbData.meta.version === DB_VERSION && _ensureStructureOn(fbData)) {
   188→          _db = fbData;
   189→          _ensureStructure();  // ensure new fields exist on Firebase data
   190→        } else {
   191→          _db = _emptyDB();
   192→          _ensureStructure();
   193→          _saveToFirebase();
   194→        }
   195→        _ready = true;
   196→        _cacheLocal();
   197→        _startFirebaseListener();
   198→        _startPedidosListener();
   199→        resolve();
   200→      }).catch(function() {
   201→        _db = _emptyDB();
   202→        _ensureStructure();
   203→        _ready = true;
   204→        resolve();
   205→      });
   206→    } else {
   207→      _db = _emptyDB();
   208→      _ensureStructure();
   209→      _ready = true;
   210→      resolve();
   211→    }
   212→  });
   213→}

   228→function _startFirebaseListener() {
   229→  if (!_firebaseRef) return;
   230→  _firebaseRef.on('value', function(snap) {
   231→    var data = snap.val();
   232→    if (!data || !data.meta || data.meta.version !== DB_VERSION) return;
   233→    // CRITICAL: skip if local save is pending to prevent overwriting unsaved changes
   234→    if (_localDirty) return;
   235→    var prevJson = JSON.stringify(_db);
   236→    _db = data;
   237→    _ensureStructure();
   238→    _cacheLocal();
   239→    var newJson = JSON.stringify(_db);
   240→    if (prevJson !== newJson) {
   241→      _notify('remote_change', '', '');
   242→    }
   243→  });
   244→}

   246→function _startPedidosListener() {
   247→  if (!_pedidosRef) return;
   248→  _pedidosRef.on('value', function(snap) {
   249→    var data = snap.val();

   251→    _pedidos = [];

   261→      _notifyPedidos();
   262→    }
   263→  });

   447→function saveEntrada(data) {
   448→  _ensureStructure();
   449→  var isNew = !data.id;
   450→  if (isNew) {
   451→    data.id = nextId('entradas');
   452→    data.creado = new Date().toISOString();
   453→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   454→    data.items = data.items || [];
   455→    data.total = Number(data.total) || 0;
   456→  }
   457→  if (isNew) {
   458→    for (var i = 0; i < data.items.length; i++) {
   459→      var item = data.items[i];
   460→      var tipo = item.tipo;
   461→      if (tipo === 'especia_grs') {
   462→        // Add grams to especia stockBolsa
   463→        if (item.especiaId && _db.especias[item.especiaId]) {
   464→          _db.especias[item.especiaId].stockBolsa = (_db.especias[item.especiaId].stockBolsa || 0) + (Number(item.cantidad) || 0);
   465→        }
   466→      } else if (tipo === 'envase') {
   467→        var talla = item.talla || 'chico';
   468→        if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   469→        _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) + (Number(item.cantidad) || 0);
   470→      } else if (tipo === 'sticker') {
   471→        var stk = _getOrCreateSticker(item.stickerNombre);
   472→        var t = item.talla || 'chico';
   473→        if (t === 'grande') {
   474→          stk.stockGrande = (stk.stockGrande || 0) + (Number(item.cantidad) || 0);
   475→        } else {
   476→          stk.stockChico = (stk.stockChico || 0) + (Number(item.cantidad) || 0);
   477→        }
   478→      } else if (tipo === 'bolsa') {
   479→        var tallaB = item.talla || 'chico';
   480→        if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   481→        _db.stockBolsas[tallaB] = (_db.stockBolsas[tallaB] || 0) + (Number(item.cantidad) || 0);
   482→      }
   483→    }
   484→  }
   485→  _db.entradas[data.id] = data;
   486→  _saveToFirebase(); _cacheLocal();
   487→  _notify(isNew ? 'create' : 'update', 'entradas', data.id);
   488→  return data;
   489→}

   506→  _ensureStructure();
   507→  if (!_db.ajustes) _db.ajustes = {};
   508→  var isNew = !data.id;
   509→  if (isNew) {
   510→    data.id = nextId('ajustes');
   511→    data.creado = new Date().toISOString();
   512→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   513→  }
   514→  var cantidad = Number(data.cantidad) || 0;
   515→  if (cantidad === 0) throw new Error('La cantidad no puede ser 0');
   516→  var cat = data.categoria; // 'especia', 'blend', 'envase', 'bolsa', 'sticker'
   517→  var sub = data.subtipo;   // 'pala', 'chico', 'grande'
   518→
   519→  if (cat === 'especia') {
   520→    var esp = _db.especias[data.productoId];
   521→    if (!esp) throw new Error('Especia no encontrada');
   522→    if (sub === 'pala') {
   523→      var nv = (Number(esp.stockBolsa) || 0) + cantidad;
   524→      if (nv < 0) throw new Error('Stock de pala resultante negativo (' + nv + 'g) para ' + esp.nombre);
   525→      esp.stockBolsa = nv;
   526→    } else {
   527→      var field = (sub === 'grande') ? 'stockGrande' : 'stockChico';
   528→      var nv = (Number(esp[field]) || 0) + cantidad;
   529→      if (nv < 0) throw new Error('Stock de frascos resultante negativo (' + nv + ') para ' + esp.nombre);
   530→      esp[field] = nv;
   531→    }
   532→    data.productoNombre = esp.nombre;
   533→  } else if (cat === 'blend') {
   534→    var bl = _db.blends[data.productoId];
   535→    if (!bl) throw new Error('Blend no encontrado');
   536→    var field = (sub === 'grande') ? 'stockGrande' : 'stockChico';
   537→    var nv = (Number(bl[field]) || 0) + cantidad;
   538→    if (nv < 0) throw new Error('Stock resultante negativo (' + nv + ') para ' + bl.nombre);
   539→    bl[field] = nv;
   540→    data.productoNombre = bl.nombre;
   541→  } else if (cat === 'envase') {
   542→    if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   543→    var t = (sub === 'grande') ? 'grande' : 'chico';
   544→    var nv = (_db.stockEnvases[t] || 0) + cantidad;
   545→    if (nv < 0) throw new Error('Stock de frascos ' + t + ' resultante negativo (' + nv + ')');
   546→    _db.stockEnvases[t] = nv;
   547→    data.productoNombre = 'Frascos ' + t;
   548→  } else if (cat === 'bolsa') {
   549→    if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   550→    var t = (sub === 'grande') ? 'grande' : 'chico';
   551→    var nv = (_db.stockBolsas[t] || 0) + cantidad;
   552→    if (nv < 0) throw new Error('Stock de bolsas ' + t + ' resultante negativo (' + nv + ')');
   553→    _db.stockBolsas[t] = nv;
   554→    data.productoNombre = 'Bolsas ' + t;
   555→  } else if (cat === 'sticker') {
   556→    var stk = _findStickerByNombre(data.productoNombre);
   557→    if (!stk) throw new Error('Sticker "' + (data.productoNombre||'') + '" no encontrado');
   558→    var field = (sub === 'grande') ? 'stockGrande' : 'stockChico';
   559→    var nv = (Number(stk[field]) || 0) + cantidad;
   560→    if (nv < 0) throw new Error('Stock resultante negativo (' + nv + ') para sticker ' + stk.nombre);
   561→    stk[field] = nv;
   562→  }
   563→
   564→  data.cantidad = cantidad;
   565→  _db.ajustes[data.id] = data;
   566→  _saveToFirebase(); _cacheLocal();
   567→  _notify(isNew ? 'create' : 'update', 'ajustes', data.id);
   568→  return data;
   569→}

   581→function producirEspecia(especiaId, talla, cantidad) {
   582→  _ensureStructure();
   583→  var esp = _db.especias[especiaId];
   584→  if (!esp) throw new Error('Especia no encontrada');
   585→  talla = (talla === 'grande') ? 'grande' : 'chico';
   586→  cantidad = Number(cantidad) || 0;
   587→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   588→
   589→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   590→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   591→
   592→  var grsTotal = gramosPorFrasco * cantidad;
   593→
   594→  // Check & consume pala (raw material)
   595→  if ((esp.stockBolsa || 0) < grsTotal) {
   596→    throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   597→  }
   598→
   599→  // Check & consume envases
   600→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   601→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   602→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   603→  }
   604→
   605→  // Check & consume stickers
   606→  var stk = _findStickerByNombre(esp.nombre);
   607→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   608→  if (stkStock < cantidad) {
   609→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   610→  }
   611→
   612→  // Check & consume bolsas (packaging)
   613→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   614→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   615→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   616→  }
   617→
   618→  // All checks passed — consume
   619→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   620→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   621→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   622→  if (stk) {
   623→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   624→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   625→  }
   626→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   627→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   628→
   629→  // Record
   630→  var prodId = nextId('producciones');
   631→  var prod = {
   632→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   633→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   634→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   635→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   636→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   637→  };
   638→  _db.producciones[prodId] = prod;
   639→  _saveToFirebase(); _cacheLocal();
   640→  _notify('create', 'producciones', prodId);
   641→  _notify('update', 'especias', especiaId);
   642→  return { producto: esp, produccion: prod };
   643→}

   645→function producirBlend(blendId, talla, cantidad) {
   646→  _ensureStructure();
   647→  var blend = _db.blends[blendId];
   648→  if (!blend) throw new Error('Blend no encontrado');
   649→  talla = (talla === 'grande') ? 'grande' : 'chico';
   650→  cantidad = Number(cantidad) || 0;
   651→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   652→
   653→  var ingredientes = blend.ingredientes || [];
   654→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   655→
   656→  // Check ingredient stock
   657→  var detalleIngredientes = [];
   658→  for (var i = 0; i < ingredientes.length; i++) {
   659→    var ing = ingredientes[i];
   660→    var esp = _db.especias[ing.especiaId];
   661→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   662→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   663→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   664→    var grsNeeded = grsPorFrasco * cantidad;
   665→    if ((esp.stockBolsa || 0) < grsNeeded) {
   666→      throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   667→    }
   668→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   669→  }
   670→
   671→  // Check envases
   672→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   673→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   674→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   675→  }
   676→
   677→  // Check stickers
   678→  var stk = _findStickerByNombre(blend.nombre);
   679→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   680→  if (stkStock < cantidad) {
   681→    throw new Error('Stickers ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   682→  }
   683→
   684→  // Check & consume bolsas (packaging)
   685→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   686→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   687→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   688→  }
   689→
   690→  // All checks passed — consume
   691→  var grsTotalGeneral = 0;
   692→  for (var i = 0; i < detalleIngredientes.length; i++) {
   693→    var d = detalleIngredientes[i];
   694→    var esp = _db.especias[d.especiaId];
   695→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   696→    grsTotalGeneral += d.gramosTotal;
   697→  }
   698→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   699→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   700→  if (stk) {
   701→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   702→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   703→  }
   704→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   705→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   706→
   707→  var prodId = nextId('producciones');
   708→  var prod = {
   709→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   710→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   711→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   712→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   713→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   714→  };
   715→  _db.producciones[prodId] = prod;
   716→  _saveToFirebase(); _cacheLocal();
   717→  _notify('create', 'producciones', prodId);
   718→  _notify('update', 'blends', blendId);
   719→  return { producto: blend, produccion: prod };
   720→}

   732→function saveVenta(data) {
   733→  _ensureStructure();
   734→  var isNew = !data.id;
   735→  if (isNew) {
   736→    data.id = nextId('ventas');
   737→    data.creado = new Date().toISOString();
   738→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   739→    data.items = data.items || [];
   740→    data.total = Number(data.total) || 0;
   741→  }
   742→  if (isNew) {
   743→    for (var i = 0; i < data.items.length; i++) {
   744→      var item = data.items[i];
   745→      var producto;
   746→      if (item.tipo === 'especia') {
   747→        producto = _db.especias[item.productoId];
   748→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   749→      } else {
   750→        producto = _db.blends[item.productoId];
   751→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   752→      }
   753→      item.productoNombre = producto.nombre;
   754→      var cant = Number(item.cantidad) || 0;
   755→      var talla = item.talla || 'chico';
   756→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   757→      if ((producto[stockKey] || 0) < cant) {
   758→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   759→      }
   760→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   761→      item.precioUnitario = Number(item.precioUnitario) || 0;
   762→      item.subtotal = item.precioUnitario * cant;
   763→    }
   764→    // Recalculate total
   765→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   766→  }
   767→  _db.ventas[data.id] = data;
   768→  _saveToFirebase(); _cacheLocal();
   769→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   770→  return data;
   771→}

   827→
   828→  var today = new Date().toISOString().slice(0, 10);
   829→  var mes = new Date().toISOString().slice(0, 7);
   830→  var ventasHoy = ventas.filter(function(v) { return v.fecha === today; });
   831→  var ventasMes = ventas.filter(function(v) { return v.fecha && v.fecha.startsWith(mes); });
   832→
   833→  var frascosChico = especias.reduce(function(s, e) { return s + (e.stockChico || 0); }, 0) +
   834→                     blends.reduce(function(s, b) { return s + (b.stockChico || 0); }, 0);

   836→                      blends.reduce(function(s, b) { return s + (b.stockGrande || 0); }, 0);

   838→  var stkBajo = stickers.filter(function(e) { return (e.stockChico + e.stockGrande) <= 5; });
   839→  var espBolsaBaja = especias.filter(function(e) { return (e.stockBolsa || 0) <= 50; });
   840→
   841→  return {
   842→    totalEspecias: especias.length,
   843→    totalBlends: blends.length,
   844→    totalProductos: especias.length + blends.length,
   845→    frascosChico: frascosChico,
   846→    frascosGrande: frascosGrande,
   847→    totalFrascos: frascosChico + frascosGrande,
   848→    envasesChico: envases.chico || 0,
   849→    envasesGrande: envases.grande || 0,
   850→    bolsasChico: bolsas.chico || 0,
   851→    bolsasGrande: bolsas.grande || 0,
   852→    ventasHoy: ventasHoy.length,
   853→    totalVentasHoy: ventasHoy.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   854→    ventasMes: ventasMes.length,

   856→    especiasBolsaBaja: espBolsaBaja,
   857→    stickersBajos: stkBajo
   858→  };
   859→}

   910→    });
   911→  }
   912→  return products.sort(function(a, b) { return a.nombre.localeCompare(b.nombre); });
   913→}

   928→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   929→function findEspeciaByName(nombre) {
   930→  if (!nombre) return null;
   931→  var target = nombre.trim().toLowerCase();
   932→  var keys = Object.keys(_db.especias || {});
   933→  // 1. Exact match
   934→  for (var i = 0; i < keys.length; i++) {
   935→    var e = _db.especias[keys[i]];
   936→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   937→  }
   938→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   939→  for (var i = 0; i < keys.length; i++) {
   940→    var e = _db.especias[keys[i]];
   941→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   942→  }
   943→  // 3. Target is contained in especia name
   944→  for (var i = 0; i < keys.length; i++) {
   945→    var e = _db.especias[keys[i]];
   946→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   947→  }
   948→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   949→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   950→  for (var w = 0; w < targetWords.length; w++) {
   951→    for (var i = 0; i < keys.length; i++) {
   952→      var e = _db.especias[keys[i]];
   953→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   954→    }
   955→  }
   956→  return null;
   957→}

   959→/** Auto-detect categoria from blend USO field */
   960→function _categoriaFromUso(uso) {
   961→  if (!uso) return 'Comidas';
   962→  var u = uso.toLowerCase();
   963→  // Cocteleria keywords
   964→  if (/\b(gin|ron|vodka|whisky|mojito|mule|vermouth|aperitif|coctel)\b/.test(u)) return 'Cocteleria';
   965→  // Infusiones keywords
   966→  if (/\b(relajant|sueño|digestiv|energiz|té|calidez|respiratorio|meditaci|antioxidant|bienestar|infusi)\b/.test(u)) return 'Infusiones';
   967→  return 'Comidas';
   968→}

   970→/**
   971→ * Import especias and blends from parsed Excel data.
   972→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   973→ *
   974→ * especiasList: [{ nombre, categoria? }]
   975→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   976→ * gramosChico:  default grams for frasco chico (e.g. 30)
   977→ * gramosGrande: default grams for frasco grande (e.g. 80)
   978→ */
   979→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   980→  _ensureStructure();
   981→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   982→
   983→  // 1. Create especias (skip if name already exists)
   984→  for (var i = 0; i < especiasList.length; i++) {
   985→    var esp = especiasList[i];
   986→    var nombre = (esp.nombre || '').trim();
   987→    if (!nombre) continue;
   988→    var existing = findEspeciaByName(nombre);
   989→    if (existing) {
   990→      resultado.especiasExistentes++;
   991→      continue;
   992→    }
   993→    saveEspecia({
   994→      nombre: nombre,
   995→      categoria: esp.categoria || 'Comidas',
   996→      precioChico: 0,
   997→      precioGrande: 0,
   998→      gramosChico: Number(gramosChico) || 30,
   999→      gramosGrande: Number(gramosGrande) || 80,
  1000→      stockBolsa: 0,
  1001→      stockChico: 0,
  1002→      stockGrande: 0
  1003→    });
  1004→    resultado.especiasCreadas++;
  1005→  }
  1006→
  1007→  // 2. Create blends (skip if name already exists)
  1008→  for (var j = 0; j < blendsList.length; j++) {
  1009→    var bl = blendsList[j];
  1010→    var nombre = (bl.nombre || '').trim();
  1011→    if (!nombre) continue;
  1012→    var existingBlend = null;
  1013→    var blKeys = Object.keys(_db.blends || {});
  1014→    for (var k = 0; k < blKeys.length; k++) {
  1015→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
  1016→        existingBlend = _db.blends[blKeys[k]];
  1017→        break;
  1018→      }
  1019→    }
  1020→    if (existingBlend) {
  1021→      resultado.blendsExistentes++;
  1022→      continue;
  1023→    }
  1024→
  1025→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
  1026→    var ings = bl.ingredientes || [];
  1027→    var recipeTotal = 0;
  1028→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
  1029→    if (recipeTotal <= 0) recipeTotal = 500;
  1030→
  1031→    var resolvedIngs = [];
  1032→    for (var ii = 0; ii < ings.length; ii++) {
  1033→      var ing = ings[ii];
  1034→      var espObj = findEspeciaByName(ing.especia);
  1035→      if (!espObj) {
  1036→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
  1037→        continue;
  1038→      }
  1039→      var ingG = Number(ing.g) || 0;
  1040→      resolvedIngs.push({
  1041→        especiaId: espObj.id,
  1042→        especiaNombre: espObj.nombre,
  1043→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
  1044→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
  1045→        gramosReceta: ingG
  1046→      });
  1047→    }
  1048→
  1049→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
  1050→    saveBlend({
  1051→      nombre: nombre,
  1052→      categoria: cat,
  1053→      region: bl.region || '',
  1054→      uso: bl.uso || '',
  1055→      precioChico: 0,
  1056→      precioGrande: 0,
  1057→      ingredientes: resolvedIngs,
  1058→      stockChico: 0,
  1059→      stockGrande: 0
  1060→    });
  1061→    resultado.blendsCreados++;
  1062→  }
  1063→
  1064→  return resultado;
  1065→}

  1079→function addProductTag(cat, tagName) {
  1080→  _ensureStructure();
  1081→  tagName = (tagName || '').trim();
  1082→  if (!tagName) return false;
  1083→  if (!_db.productTags) _db.productTags = {};
  1084→  if (!_db.productTags[cat]) _db.productTags[cat] = [];
  1085→  // Check duplicate (case-insensitive)
  1086→  for (var i = 0; i < _db.productTags[cat].length; i++) {
  1087→    if (_db.productTags[cat][i].toLowerCase() === tagName.toLowerCase()) return false;
  1088→  }
  1089→  _db.productTags[cat].push(tagName);
  1090→  _saveToFirebase(); _cacheLocal();
  1091→  return true;
  1092→}
  1093→
  1094→function removeProductTag(cat, tagName) {
  1095→  _ensureStructure();
  1096→  if (!_db.productTags || !_db.productTags[cat]) return false;
  1097→  var idx = -1;
  1098→  for (var i = 0; i < _db.productTags[cat].length; i++) {
  1099→    if (_db.productTags[cat][i] === tagName) { idx = i; break; }
  1100→  }
  1101→  if (idx < 0) return false;
  1102→  _db.productTags[cat].splice(idx, 1);
  1103→  // Also remove from all products that have this tag
  1104→  var allEsp = getEspecias();
  1105→  for (var e = 0; e < allEsp.length; e++) {
  1106→    if (allEsp[e].tags) {
  1107→      var ti = allEsp[e].tags.indexOf(tagName);
  1108→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
  1109→    }
  1110→  }
  1111→  var allBl = getBlends();
  1112→  for (var b = 0; b < allBl.length; b++) {
  1113→    if (allBl[b].tags) {
  1114→      var bi = allBl[b].tags.indexOf(tagName);
  1115→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
  1116→    }
  1117→  }
  1118→  _saveToFirebase(); _cacheLocal();
  1119→  return true;
  1120→}

  1124→function compressImage(file, maxW, quality, cb) {
  1125→  var reader = new FileReader();
  1126→  reader.onload = function(ev) {
  1127→    var img = new Image();
  1128→    img.onload = function() {
  1129→      var w = img.width, h = img.height;
  1130→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
  1131→      var canvas = document.createElement('canvas');
  1132→      canvas.width = w; canvas.height = h;
  1133→      var ctx = canvas.getContext('2d');
  1134→      ctx.drawImage(img, 0, 0, w, h);
  1135→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
  1136→      cb(null, dataUrl);
  1137→    };
  1138→    img.onerror = function() { cb('Error al cargar imagen'); };
  1139→    img.src = ev.target.result;
  1140→  };
  1141→  reader.onerror = function() { cb('Error al leer archivo'); };
  1142→  reader.readAsDataURL(file);
  1143→}
