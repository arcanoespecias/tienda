   600→function producirEspecia(especiaId, talla, cantidad) {
   601→  _ensureStructure();
   602→  var esp = _db.especias[especiaId];
   603→  if (!esp) throw new Error('Especia no encontrada');
   604→  talla = (talla === 'grande') ? 'grande' : 'chico';
   605→  cantidad = Number(cantidad) || 0;
   606→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   607→
   608→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   609→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   610→
   611→  var grsTotal = gramosPorFrasco * cantidad;
   612→
   613→  // Check & consume pala (raw material)
   614→  if ((esp.stockBolsa || 0) < grsTotal) {
   615→    throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   616→  }
   617→
   618→  // Check & consume envases
   619→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   620→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   621→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   622→  }
   623→
   624→  // Check & consume stickers
   625→  var stk = _findStickerByNombre(esp.nombre);
   626→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   627→  if (stkStock < cantidad) {
   628→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   629→  }
   630→
   631→  // Check & consume bolsas (packaging)
   632→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   633→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   634→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   635→  }
   636→
   637→  // All checks passed — consume
   638→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   639→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   640→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   641→  if (stk) {
   642→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   643→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   644→  }
   645→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   646→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   647→
   648→  // Record
   649→  var prodId = nextId('producciones');
   650→  var prod = {
   651→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   652→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   653→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   654→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   655→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   656→  };
   657→  _db.producciones[prodId] = prod;
   658→  _saveToFirebase(); _cacheLocal();
   659→  _notify('create', 'producciones', prodId);
   660→  _notify('update', 'especias', especiaId);
   661→  return { producto: esp, produccion: prod };
   662→}

   664→function producirBlend(blendId, talla, cantidad) {
   665→  _ensureStructure();
   666→  var blend = _db.blends[blendId];
   667→  if (!blend) throw new Error('Blend no encontrado');
   668→  talla = (talla === 'grande') ? 'grande' : 'chico';
   669→  cantidad = Number(cantidad) || 0;
   670→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   671→
   672→  var ingredientes = blend.ingredientes || [];
   673→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   674→
   675→  // Check ingredient stock
   676→  var detalleIngredientes = [];
   677→  for (var i = 0; i < ingredientes.length; i++) {
   678→    var ing = ingredientes[i];
   679→    var esp = _db.especias[ing.especiaId];
   680→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   681→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   682→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   683→    var grsNeeded = grsPorFrasco * cantidad;
   684→    if ((esp.stockBolsa || 0) < grsNeeded) {
   685→      throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   686→    }
   687→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   688→  }
   689→
   690→  // Check envases
   691→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   692→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   693→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   694→  }
   695→
   696→  // Check stickers
   697→  var stk = _findStickerByNombre(blend.nombre);
   698→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   699→  if (stkStock < cantidad) {
   700→    throw new Error('Stickers ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   701→  }
   702→
   703→  // Check & consume bolsas (packaging)
   704→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   705→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   706→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   707→  }
   708→
   709→  // All checks passed — consume
   710→  var grsTotalGeneral = 0;
   711→  for (var i = 0; i < detalleIngredientes.length; i++) {
   712→    var d = detalleIngredientes[i];
   713→    var esp = _db.especias[d.especiaId];
   714→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   715→    grsTotalGeneral += d.gramosTotal;
   716→  }
   717→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   718→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   719→  if (stk) {
   720→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   721→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   722→  }
   723→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   724→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   725→
   726→  var prodId = nextId('producciones');
   727→  var prod = {
   728→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   729→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   730→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   731→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   732→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   733→  };
   734→  _db.producciones[prodId] = prod;
   735→  _saveToFirebase(); _cacheLocal();
   736→  _notify('create', 'producciones', prodId);
   737→  _notify('update', 'blends', blendId);
   738→  return { producto: blend, produccion: prod };
   739→}

   751→function saveVenta(data) {
   752→  _ensureStructure();
   753→  var isNew = !data.id;
   754→  if (isNew) {
   755→    data.id = nextId('ventas');
   756→    data.creado = new Date().toISOString();
   757→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   758→    data.items = data.items || [];
   759→    data.total = Number(data.total) || 0;
   760→  }
   761→  if (isNew) {
   762→    for (var i = 0; i < data.items.length; i++) {
   763→      var item = data.items[i];
   764→      var producto;
   765→      if (item.tipo === 'especia') {
   766→        producto = _db.especias[item.productoId];
   767→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   768→      } else {
   769→        producto = _db.blends[item.productoId];
   770→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   771→      }
   772→      item.productoNombre = producto.nombre;
   773→      var cant = Number(item.cantidad) || 0;
   774→      var talla = item.talla || 'chico';
   775→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   776→      if ((producto[stockKey] || 0) < cant) {
   777→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   778→      }
   779→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   780→      item.precioUnitario = Number(item.precioUnitario) || 0;
   781→      item.subtotal = item.precioUnitario * cant;
   782→    }
   783→    // Recalculate total
   784→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   785→  }
   786→  _db.ventas[data.id] = data;
   787→  _saveToFirebase(); _cacheLocal();
   788→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   789→  return data;
   790→}

   947→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   948→function findEspeciaByName(nombre) {
   949→  if (!nombre) return null;
   950→  var target = nombre.trim().toLowerCase();
   951→  var keys = Object.keys(_db.especias || {});
   952→  // 1. Exact match
   953→  for (var i = 0; i < keys.length; i++) {
   954→    var e = _db.especias[keys[i]];
   955→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   956→  }
   957→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   958→  for (var i = 0; i < keys.length; i++) {
   959→    var e = _db.especias[keys[i]];
   960→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   961→  }
   962→  // 3. Target is contained in especia name
   963→  for (var i = 0; i < keys.length; i++) {
   964→    var e = _db.especias[keys[i]];
   965→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   966→  }
   967→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   968→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   969→  for (var w = 0; w < targetWords.length; w++) {
   970→    for (var i = 0; i < keys.length; i++) {
   971→      var e = _db.especias[keys[i]];
   972→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   973→    }
   974→  }
   975→  return null;
   976→}

   980→  if (!uso) return 'Comidas';

   982→  // Cocteleria keywords

   984→  // Infusiones keywords

   986→  return 'Comidas';
   987→}

   989→/**
   990→ * Import especias and blends from parsed Excel data.
   991→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   992→ *
   993→ * especiasList: [{ nombre, categoria? }]
   994→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   995→ * gramosChico:  default grams for frasco chico (e.g. 30)
   996→ * gramosGrande: default grams for frasco grande (e.g. 80)
   997→ */
   998→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   999→  _ensureStructure();
  1000→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
  1001→
  1002→  // 1. Create especias (skip if name already exists)
  1003→  for (var i = 0; i < especiasList.length; i++) {
  1004→    var esp = especiasList[i];
  1005→    var nombre = (esp.nombre || '').trim();
  1006→    if (!nombre) continue;
  1007→    var existing = findEspeciaByName(nombre);
  1008→    if (existing) {
  1009→      resultado.especiasExistentes++;
  1010→      continue;
  1011→    }
  1012→    saveEspecia({
  1013→      nombre: nombre,
  1014→      categoria: esp.categoria || 'Comidas',
  1015→      precioChico: 0,
  1016→      precioGrande: 0,
  1017→      gramosChico: Number(gramosChico) || 30,
  1018→      gramosGrande: Number(gramosGrande) || 80,
  1019→      stockBolsa: 0,
  1020→      stockChico: 0,
  1021→      stockGrande: 0
  1022→    });
  1023→    resultado.especiasCreadas++;
  1024→  }
  1025→
  1026→  // 2. Create blends (skip if name already exists)
  1027→  for (var j = 0; j < blendsList.length; j++) {
  1028→    var bl = blendsList[j];
  1029→    var nombre = (bl.nombre || '').trim();
  1030→    if (!nombre) continue;
  1031→    var existingBlend = null;
  1032→    var blKeys = Object.keys(_db.blends || {});
  1033→    for (var k = 0; k < blKeys.length; k++) {
  1034→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
  1035→        existingBlend = _db.blends[blKeys[k]];
  1036→        break;
  1037→      }
  1038→    }
  1039→    if (existingBlend) {
  1040→      resultado.blendsExistentes++;
  1041→      continue;
  1042→    }
  1043→
  1044→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
  1045→    var ings = bl.ingredientes || [];
  1046→    var recipeTotal = 0;
  1047→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
  1048→    if (recipeTotal <= 0) recipeTotal = 500;
  1049→
  1050→    var resolvedIngs = [];
  1051→    for (var ii = 0; ii < ings.length; ii++) {
  1052→      var ing = ings[ii];
  1053→      var espObj = findEspeciaByName(ing.especia);
  1054→      if (!espObj) {
  1055→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
  1056→        continue;
  1057→      }
  1058→      var ingG = Number(ing.g) || 0;
  1059→      resolvedIngs.push({
  1060→        especiaId: espObj.id,
  1061→        especiaNombre: espObj.nombre,
  1062→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
  1063→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
  1064→        gramosReceta: ingG
  1065→      });
  1066→    }
  1067→
  1068→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
  1069→    saveBlend({
  1070→      nombre: nombre,
  1071→      categoria: cat,
  1072→      region: bl.region || '',
  1073→      uso: bl.uso || '',
  1074→      precioChico: 0,
  1075→      precioGrande: 0,
  1076→      ingredientes: resolvedIngs,
  1077→      stockChico: 0,
  1078→      stockGrande: 0
  1079→    });
  1080→    resultado.blendsCreados++;
  1081→  }
  1082→
  1083→  return resultado;
  1084→}

  1143→function compressImage(file, maxW, quality, cb) {
  1144→  var reader = new FileReader();
  1145→  reader.onload = function(ev) {
  1146→    var img = new Image();
  1147→    img.onload = function() {
  1148→      var w = img.width, h = img.height;
  1149→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
  1150→      var canvas = document.createElement('canvas');
  1151→      canvas.width = w; canvas.height = h;
  1152→      var ctx = canvas.getContext('2d');
  1153→      ctx.drawImage(img, 0, 0, w, h);
  1154→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
  1155→      cb(null, dataUrl);
  1156→    };
  1157→    img.onerror = function() { cb('Error al cargar imagen'); };
