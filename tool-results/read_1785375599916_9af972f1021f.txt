   237→   237→    var prevJson = JSON.stringify(_db);

   248→   248→function _startPedidosListener() {
   249→   249→  if (!_pedidosRef) return;
   250→   250→  var _prevNuevoKeys = {};
   251→   251→  _pedidosRef.on('value', function(snap) {
   252→   252→    var data = snap.val();
   253→   253→    var prevLen = _pedidos.length;
   254→   254→    var prevNuevoKeys = Object.assign({}, _prevNuevoKeys);
   255→   255→    _pedidos = [];
   256→   256→    var nuevoKeys = {};
   257→   257→    if (data) {
   258→   258→      var keys = Object.keys(data);
   259→   259→      for (var i = 0; i < keys.length; i++) {
   260→   260→        var p = data[keys[i]];
   261→   261→        if (p && typeof p === 'object') {
   262→   262→          p._key = keys[i];
   263→   263→          _pedidos.push(p);
   264→   264→          if (p.estado === 'nuevo') nuevoKeys[keys[i]] = true;
   265→   265→        }
   266→   266→      }
   267→   267→    }
   268→   268→    _pedidos.sort(function(a, b) { return (b.creado || '').localeCompare(a.creado || ''); });
   269→   269→    _prevNuevoKeys = nuevoKeys;
   270→   270→    // Detect new pedido: a key in nuevoKeys that was NOT in prevNuevoKeys
   271→   271→    var hasNew = false;
   272→   272→    var nk = Object.keys(nuevoKeys);
   273→   273→    for (var n = 0; n < nk.length; n++) {
   274→   274→      if (!prevNuevoKeys[nk[n]]) { hasNew = true; break; }
   275→   275→    }
   276→   276→    _notifyPedidos(hasNew, _pedidos.length !== prevLen);
   277→   277→  });
   278→   278→}

   603→   603→  if (!esp) throw new Error('Especia no encontrada');
   604→   604→  talla = (talla === 'grande') ? 'grande' : 'chico';

   606→   606→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   607→   607→

   610→   610→
   611→   611→  var grsTotal = gramosPorFrasco * cantidad;
   612→   612→
   613→   613→  // Check & consume pala (raw material)
   614→   614→  if ((esp.stockBolsa || 0) < grsTotal) {
   615→   615→    throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   616→   616→  }
   617→   617→
   618→   618→  // Check & consume envases
   619→   619→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   620→   620→  if ((_db.stockEnvases[talla] || 0) < cantidad) {

   622→   622→  }

   625→   625→  var stk = _findStickerByNombre(esp.nombre);

   627→   627→  if (stkStock < cantidad) {

   635→   635→  }
   636→   636→
   637→   637→  // All checks passed — consume
   638→   638→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   639→   639→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;

   751→   751→function saveVenta(data) {

   753→   753→  var isNew = !data.id;

   758→   758→    data.items = data.items || [];
   759→   759→    data.total = Number(data.total) || 0;
   760→   760→  }
   761→   761→  if (isNew) {
   762→   762→    for (var i = 0; i < data.items.length; i++) {
   763→   763→      var item = data.items[i];
   764→   764→      var producto;

   770→   770→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);

   780→   780→      item.precioUnitario = Number(item.precioUnitario) || 0;

   782→   782→    }

   947→   947→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   948→   948→function findEspeciaByName(nombre) {
   949→   949→  if (!nombre) return null;
   950→   950→  var target = nombre.trim().toLowerCase();
   951→   951→  var keys = Object.keys(_db.especias || {});
   952→   952→  // 1. Exact match
   953→   953→  for (var i = 0; i < keys.length; i++) {
   954→   954→    var e = _db.especias[keys[i]];
   955→   955→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   956→   956→  }
   957→   957→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   958→   958→  for (var i = 0; i < keys.length; i++) {
   959→   959→    var e = _db.especias[keys[i]];
   960→   960→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   961→   961→  }
   962→   962→  // 3. Target is contained in especia name
   963→   963→  for (var i = 0; i < keys.length; i++) {
   964→   964→    var e = _db.especias[keys[i]];
   965→   965→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   966→   966→  }
   967→   967→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   968→   968→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   969→   969→  for (var w = 0; w < targetWords.length; w++) {
   970→   970→    for (var i = 0; i < keys.length; i++) {
   971→   971→      var e = _db.especias[keys[i]];
   972→   972→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   973→   973→    }
   974→   974→  }
   975→   975→  return null;
   976→   976→}

   990→   990→ * Import especias and blends from parsed Excel data.
   991→   991→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   992→   992→ *
   993→   993→ * especiasList: [{ nombre, categoria? }]

   995→   995→ * gramosChico:  default grams for frasco chico (e.g. 30)

   998→   998→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   999→   999→  _ensureStructure();
  1000→  1000→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
  1001→  1001→
  1002→  1002→  // 1. Create especias (skip if name already exists)
  1003→  1003→  for (var i = 0; i < especiasList.length; i++) {
  1004→  1004→    var esp = especiasList[i];
  1005→  1005→    var nombre = (esp.nombre || '').trim();
  1006→  1006→    if (!nombre) continue;
  1007→  1007→    var existing = findEspeciaByName(nombre);
  1008→  1008→    if (existing) {
  1009→  1009→      resultado.especiasExistentes++;
  1010→  1010→      continue;
  1011→  1011→    }
  1012→  1012→    saveEspecia({
  1013→  1013→      nombre: nombre,
  1014→  1014→      categoria: esp.categoria || 'Comidas',
  1015→  1015→      precioChico: 0,
  1016→  1016→      precioGrande: 0,
  1017→  1017→      gramosChico: Number(gramosChico) || 30,
  1018→  1018→      gramosGrande: Number(gramosGrande) || 80,
  1019→  1019→      stockBolsa: 0,
  1020→  1020→      stockChico: 0,
  1021→  1021→      stockGrande: 0
  1022→  1022→    });
  1023→  1023→    resultado.especiasCreadas++;
  1024→  1024→  }
  1025→  1025→
  1026→  1026→  // 2. Create blends (skip if name already exists)
  1027→  1027→  for (var j = 0; j < blendsList.length; j++) {
  1028→  1028→    var bl = blendsList[j];
  1029→  1029→    var nombre = (bl.nombre || '').trim();
  1030→  1030→    if (!nombre) continue;
  1031→  1031→    var existingBlend = null;
  1032→  1032→    var blKeys = Object.keys(_db.blends || {});
  1033→  1033→    for (var k = 0; k < blKeys.length; k++) {
  1034→  1034→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
  1035→  1035→        existingBlend = _db.blends[blKeys[k]];
  1036→  1036→        break;
  1037→  1037→      }
  1038→  1038→    }
  1039→  1039→    if (existingBlend) {
  1040→  1040→      resultado.blendsExistentes++;
  1041→  1041→      continue;
  1042→  1042→    }
  1043→  1043→
  1044→  1044→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
  1045→  1045→    var ings = bl.ingredientes || [];
  1046→  1046→    var recipeTotal = 0;
  1047→  1047→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
  1048→  1048→    if (recipeTotal <= 0) recipeTotal = 500;
  1049→  1049→
  1050→  1050→    var resolvedIngs = [];
  1051→  1051→    for (var ii = 0; ii < ings.length; ii++) {
  1052→  1052→      var ing = ings[ii];
  1053→  1053→      var espObj = findEspeciaByName(ing.especia);
  1054→  1054→      if (!espObj) {
  1055→  1055→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
  1056→  1056→        continue;
  1057→  1057→      }
  1058→  1058→      var ingG = Number(ing.g) || 0;
  1059→  1059→      resolvedIngs.push({
  1060→  1060→        especiaId: espObj.id,
  1061→  1061→        especiaNombre: espObj.nombre,
  1062→  1062→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
  1063→  1063→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
  1064→  1064→        gramosReceta: ingG
  1065→  1065→      });
  1066→  1066→    }
  1067→  1067→
  1068→  1068→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
  1069→  1069→    saveBlend({
  1070→  1070→      nombre: nombre,
  1071→  1071→      categoria: cat,
  1072→  1072→      region: bl.region || '',
  1073→  1073→      uso: bl.uso || '',
  1074→  1074→      precioChico: 0,
  1075→  1075→      precioGrande: 0,
  1076→  1076→      ingredientes: resolvedIngs,
  1077→  1077→      stockChico: 0,
  1078→  1078→      stockGrande: 0
  1079→  1079→    });
  1080→  1080→    resultado.blendsCreados++;
  1081→  1081→  }
  1082→  1082→
  1083→  1083→  return resultado;
  1084→  1084→}

  1143→  1143→function compressImage(file, maxW, quality, cb) {
  1144→  1144→  var reader = new FileReader();
  1145→  1145→  reader.onload = function(ev) {
  1146→  1146→    var img = new Image();
  1147→  1147→    img.onload = function() {
  1148→  1148→      var w = img.width, h = img.height;
  1149→  1149→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
  1150→  1150→      var canvas = document.createElement('canvas');
  1151→  1151→      canvas.width = w; canvas.height = h;
  1152→  1152→      var ctx = canvas.getContext('2d');
  1153→  1153→      ctx.drawImage(img, 0, 0, w, h);
  1154→  1154→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
  1155→  1155→      cb(null, dataUrl);
  1156→  1156→    };
