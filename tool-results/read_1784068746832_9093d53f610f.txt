    26→function renderStoreProducts() {
    27→  var db = getDB();
    28→  var products = [];
    29→  var container = document.getElementById('store-products');
    30→  var empty = document.getElementById('store-empty');
    31→
    32→  // Get sellable especias (with precioVenta > 0 and stock > 0)
    33→  var especias = (db.productos || []).filter(function(p) {
    34→    return p.tipo === 'especia' && p.precioVenta > 0 && p.stock > 0;
    35→  });
    36→  especias.forEach(function(p) {
    37→    // Filter by storeCategory
    38→    if (storeCategory !== 'todos' && storeCategory !== 'blends') {
    39→      if ((p.categoria || '') !== storeCategory) return;
    40→    }
    41→    products.push({
    42→      id: p.id, nombre: p.nombre, tipo: p.categoria || 'Especia',
    43→      desc: (p.usoPrincipal || p.notas || ''), precio: p.precioVenta,
    44→      stock: p.stock, unidad: p.unidad, blend: false,
    45→      origen: p.origen || '', perfil: p.perfil || ''
    46→    });
    47→  });
    48→
    49→  // Get blends with stock > 0 (only when "todos" or "blends" selected)
    50→  if (storeCategory === 'todos' || storeCategory === 'blends') {
    51→    var blends = (db.blends || []).filter(function(b) {
    52→      return b.stock > 0 && b.precioVenta > 0;
    53→    });
    54→    blends.forEach(function(b) {
    55→      // Calculate effective stock (min of blend, frascos, etiqueta)
    56→      var etiq = b.etiquetaId ? (db.productos || []).find(function(p) { return p.id === b.etiquetaId; }) : null;
    57→      var etiqStk = etiq ? etiq.stock : 0;
    58→      var frascosStk = (db.productos || []).filter(function(p) { return p.tipo === 'frasco'; });
    59→      var maxFrasco = frascosStk.length > 0 ? Math.max.apply(null, frascosStk.map(function(f) { return f.stock; })) : 0;
    60→      var stockEfectivo = Math.min(b.stock || 0, maxFrasco, etiqStk);
    61→      if (stockEfectivo <= 0) return; // Don't show if can't fulfill
    62→
    63→      products.push({
    64→        id: b.id, nombre: b.nombre + ' (' + (b.formato || 'polvo') + ')', tipo: 'Blend',
    65→        desc: b.descripcion || '', precio: b.precioVenta,
    66→        stock: stockEfectivo, unidad: 'unidad', blend: true,
    67→        origen: '', perfil: ''
    68→      });
    69→    });
    70→  }
    71→
    72→  if (products.length === 0) {
    73→    container.innerHTML = '';
    74→    empty.style.display = 'block';
    75→    return;
    76→  }
    77→  empty.style.display = 'none';
    78→
    79→  var html = '';
    80→  products.forEach(function(p) {
    81→    html += '<div class="product-card">' +
    82→      '<div class="product-card-type">' + esc(p.tipo) + (p.origen ? ' &mdash; ' + esc(p.origen) : '') + '</div>' +
    83→      '<div class="product-card-name">' + esc(p.nombre) + '</div>' +
    84→      (p.perfil ? '<div class="product-card-meta">' + esc(p.perfil) + '</div>' : '') +
    85→      (p.desc ? '<div class="product-card-desc">' + esc(p.desc) + '</div>' : '<div class="product-card-desc" style="opacity:.3">Sin descripción</div>') +
    86→      '<div class="product-card-footer">' +
    87→        '<div><div class="product-card-price">' + fmt(p.precio) + '</div>' +
    88→        '<div class="product-card-stock">' + p.stock + ' ' + esc(p.unidad) + ' disponibles</div></div>' +
    89→        '<button class="btn-add-cart" onclick="addToCart(' + (p.blend ? 'true' : 'false') + ',' + p.id + ',\'' + esc(p.nombre).replace(/'/g,"\\'") + '\',' + p.precio + ',\'' + esc(p.unidad) + '\',' + p.stock + ')">+ Agregar</button>' +
    90→      '</div></div>';
    91→  });
    92→  container.innerHTML = html;
    93→}

    95→function addToCart(isBlend, id, nombre, precio, unidad, stock) {
    96→  if (isBlend) {
    97→    mostrarSelectorFrasco(id, 'carrito', nombre, precio, stock);
    98→    return;
    99→  }
   100→  var existing = cart.find(function(c) { return c.isBlend === isBlend && c.id === id; });
   101→  if (existing) {
   102→    if (existing.qty >= stock) { toast('No hay más stock disponible', 'err'); return; }
   103→    existing.qty++;
   104→  } else {
   105→    cart.push({ isBlend: isBlend, id: id, nombre: nombre, precio: precio, unidad: unidad, qty: 1, maxStock: stock });
   106→  }
   107→  updateCartUI();
   108→  toast(nombre + ' agregado al carrito');
   109→}

   111→// Reusable frasco selector for blends
   112→function mostrarSelectorFrasco(blendId, contexto, nombre, precio, stock) {
   113→  var db = getDB();
   114→  var blend = db.blends.find(function(b) { return b.id === blendId; });
   115→  if (!blend) return;
   116→  nombre = nombre || blend.nombre;
   117→  precio = precio || blend.precioVenta;
   118→  stock = stock || blend.stock;
   119→
   120→  var frascos = (db.productos || []).filter(function(p) { return p.tipo === 'frasco' && p.stock > 0; });
   121→  if (frascos.length === 0) {
   122→    toast('No hay frascos disponibles en stock', 'err');
   123→    return;
   124→  }
   125→
   126→  var etiqueta = blend.etiquetaId ? db.productos.find(function(p) { return p.id === blend.etiquetaId; }) : null;
   127→  var etiquetaStock = etiqueta ? etiqueta.stock : 9999;
   128→
   129→  var html = '<p class="text-sm text-muted mb-12">Elegí el frasco para <strong class="text-gold">' + esc(nombre) + '</strong> (' + esc(blend.formato || 'polvo') + ')</p>';
   130→  html += '<div class="card mb-16">';
   131→  frascos.forEach(function(f) {
   132→    var disponible = Math.min(stock, f.stock, etiquetaStock);
   133→    var disabled = disponible <= 0 ? ' style="opacity:.4;pointer-events:none"' : '';
   134→    html += '<div class="mov-row" style="cursor:pointer"' + disabled + ' onclick="confirmarFrasco(' + blendId + ',' + f.id + ',\'' + contexto + '\',' + precio + ',' + stock + ')">' +
   135→      '<div><span class="fw7">' + esc(f.nombre) + '</span>' +
   136→      '<div class="text-xs text-muted">Frasco: ' + f.stock + ' | Etiqueta: ' + etiquetaStock + '</div></div>' +
   137→      '<span class="text-gold fw7">' + fmt(precio) + '</span></div>';
   138→  });
   139→  html += '</div>';
   140→  html += '<button class="btn btn-ghost btn-full" onclick="closeModal()">Cancelar</button>';
   141→  openModal('Elegir Frasco', html);
   142→}

   144→function confirmarFrasco(blendId, frascoId, contexto, precio, stock) {
   145→  var db = getDB();
   146→  var blend = db.blends.find(function(b) { return b.id === blendId; });
   147→  var frasco = db.productos.find(function(p) { return p.id === frascoId; });
   148→  var etiqueta = blend && blend.etiquetaId ? db.productos.find(function(p) { return p.id === blend.etiquetaId; }) : null;
   149→  if (!blend || !frasco) return;
   150→
   151→  var etiquetaStock = etiqueta ? etiqueta.stock : 9999;
   152→  var disponible = Math.min(blend.stock, frasco.stock, etiquetaStock);
   153→  if (disponible <= 0) { toast('Sin stock disponible', 'err'); return; }
   154→
   155→  var nombre = blend.nombre + ' (' + frasco.nombre + ')';
   156→
   157→  if (contexto === 'carrito') {
   158→    // Check if same blend+frasco already in cart
   159→    var existing = cart.find(function(c) { return c.isBlend && c.id === blendId && c.frascoId === frascoId; });
   160→    if (existing) {
   161→      if (existing.qty >= disponible) { toast('Stock máximo alcanzado', 'err'); return; }
   162→      existing.qty++;
   163→    } else {
   164→      cart.push({ isBlend: true, id: blendId, nombre: nombre, precio: precio, unidad: 'unidad', qty: 1, maxStock: disponible, frascoId: frascoId, frascoNombre: frasco.nombre, etiquetaId: blend.etiquetaId || 0 });
   165→    }
   166→    closeModal();
   167→    updateCartUI();
   168→    toast(nombre + ' agregado al carrito');
   169→  } else if (contexto === 'venta') {
   170→    agregarBlendAVenta(blendId, frascoId, nombre, precio, disponible);
   171→    closeModal();
   172→  }
   173→}

   175→function updateCartUI() {
   176→  var countEl = document.getElementById('cart-count');
   177→  var total = 0;
   178→  var count = 0;
   179→  cart.forEach(function(c) { total += c.precio * c.qty; count += c.qty; });
   180→
   181→  if (count > 0) {
   182→    countEl.style.display = 'flex';
   183→    countEl.textContent = count;
   184→  } else {
   185→    countEl.style.display = 'none';
   186→  }
   187→
   188→  var itemsEl = document.getElementById('cart-items');
   189→  var footerEl = document.getElementById('cart-footer');
   190→
   191→  if (cart.length === 0) {
   192→    itemsEl.innerHTML = '<div class="cart-empty">Tu carrito está vacío</div>';
   193→    footerEl.style.display = 'none';
   194→    return;
   195→  }
   196→
   197→  footerEl.style.display = 'block';
   198→  document.getElementById('cart-total-val').textContent = fmt(total);
   199→
   200→  var html = '';
   201→  cart.forEach(function(c, i) {
   202→    html += '<div class="cart-item">' +
   203→      '<div class="cart-item-info"><div class="cart-item-name">' + esc(c.nombre) + '</div>' +
   204→      '<div class="cart-item-price">' + fmt(c.precio) + ' / ' + esc(c.unidad) + '</div></div>' +
   205→      '<div class="cart-item-qty">' +
   206→        '<button onclick="cartQty(' + i + ',-1)">-</button>' +
   207→        '<span>' + c.qty + '</span>' +
   208→        '<button onclick="cartQty(' + i + ',1)">+</button>' +
   209→      '</div></div>';
   210→  });
   211→  itemsEl.innerHTML = html;
   212→}
   213→
   214→function cartQty(idx, delta) {
   215→  if (!cart[idx]) return;
   216→  cart[idx].qty += delta;
   217→  if (cart[idx].qty <= 0) cart.splice(idx, 1);
   218→  else if (cart[idx].qty > cart[idx].maxStock) { cart[idx].qty = cart[idx].maxStock; toast('Stock máximo alcanzado', 'err'); }
   219→  updateCartUI();
   220→}

   257→function submitOrder() {
   258→  var nombre = document.getElementById('chk-nombre').value.trim();
   259→  var tel = document.getElementById('chk-tel').value.trim();
   260→  var dir = document.getElementById('chk-dir').value.trim();
   261→  var notas = document.getElementById('chk-notas').value.trim();
   262→  var alertEl = document.getElementById('checkout-alert');
   263→
   264→  if (!nombre) {
   265→    alertEl.textContent = 'Ingresa tu nombre';
   266→    alertEl.className = 'alert alert-err show';
   267→    return;
   268→  }
   269→
   270→  var db = getDB();
   271→  var items = [];
   272→  var total = 0;
   273→
   274→  cart.forEach(function(c) {
   275→    var sub = c.precio * c.qty;
   276→    total += sub;
   277→    var item = {
   278→      tipo: c.isBlend ? 'blend' : 'especia',
   279→      id: c.id,
   280→      nombre: c.nombre,
   281→      cantidad: c.qty,
   282→      unidad: c.unidad,
   283→      precioUnitario: c.precio,
   284→      subtotal: sub
   285→    };
   286→    if (c.isBlend && c.frascoId) {
   287→      item.frascoId = c.frascoId;
   288→      item.frascoNombre = c.frascoNombre || '';
   289→      item.etiquetaId = c.etiquetaId || 0;
   290→    }
   291→    items.push(item);
   292→  });
   293→
   294→  var venta = {
   295→    id: nextId(),
   296→    fecha: new Date().toISOString(),
   297→    clienteNombre: nombre,
   298→    clienteTelefono: tel,
   299→    clienteDireccion: dir,
   300→    items: items,
   301→    total: total,
   302→    estado: 'pendiente',
   303→    tipo: 'tienda', // marks as storefront order
   304→    creadoPor: null,
   305→    creadoEn: new Date().toISOString()
   306→  };
   307→
   308→  db.ventas.push(venta);
   309→
   310→  // Update stock
   311→  items.forEach(function(it) {
   312→    if (it.tipo === 'especia') {
   313→      var prod = db.productos.find(function(p) { return p.id === it.id; });
   314→      if (prod) {
   315→        prod.stock = Math.max(0, prod.stock - it.cantidad);
   316→        prod.actualizadoEn = new Date().toISOString();
   317→        addMovimiento('venta', 'ventas', venta.id, it.nombre, -it.cantidad, 'Pedido de tienda');
   318→      }
   319→    } else if (it.tipo === 'blend') {
   320→      var blend = db.blends.find(function(b) { return b.id === it.id; });
   321→      if (blend) {
   322→        blend.stock = Math.max(0, blend.stock - it.cantidad);
   323→        blend.actualizadoEn = new Date().toISOString();
   324→        addMovimiento('venta', 'ventas', venta.id, it.nombre, -it.cantidad, 'Pedido de tienda');
   325→      }
   326→      // Deduct frasco stock
   327→      if (it.frascoId) {
   328→        var frasco = db.productos.find(function(p) { return p.id === it.frascoId; });
   329→        if (frasco) {
   330→          frasco.stock = Math.max(0, frasco.stock - it.cantidad);
   331→          frasco.actualizadoEn = new Date().toISOString();
   332→          addMovimiento('venta', 'ventas', venta.id, it.frascoNombre || 'Frasco', -it.cantidad, 'Frasco para ' + it.nombre);
   333→        }
   334→      }
   335→      // Deduct etiqueta stock
   336→      if (it.etiquetaId) {
   337→        var etiqueta = db.productos.find(function(p) { return p.id === it.etiquetaId; });
   338→        if (etiqueta) {
   339→          etiqueta.stock = Math.max(0, etiqueta.stock - it.cantidad);
   340→          etiqueta.actualizadoEn = new Date().toISOString();
   341→          addMovimiento('venta', 'ventas', venta.id, etiqueta.nombre || 'Etiqueta', -it.cantidad, 'Etiqueta para ' + it.nombre);
   342→        }
   343→      }
   344→    }
   345→  });
   346→
   347→  saveDB();
   348→  cart = [];
   349→  closeModal();
   350→  renderStoreProducts();
   351→  updateCartUI();
   352→  toast('Pedido realizado con éxito');
   353→}

   358→function renderDashboard() {
   359→  if (!currentUser || currentUser.rol !== 'admin') return;

   363→  var compras = db.compras || [];
   364→  var blends = db.blends || [];
   365→  var ventas = db.ventas || [];
   366→  var movimientos = db.movimientos || [];
   367→
   368→  // Stats
   369→  var especias = productos.filter(function(p) { return p.tipo === 'especia'; });
   370→  var totalVentas = ventas.filter(function(v) { return v.estado !== 'cancelada'; }).reduce(function(s, v) { return s + (v.total || 0); }, 0);

   374→  var etiquetas = productos.filter(function(p) { return p.tipo === 'etiqueta'; });

   376→  var lowStockEspecias = especias.filter(function(p) { return p.stock <= p.stockMin; });

   394→        '<div class="flex items-center gap-8"><span style="font-size:1.2rem">\u26A0</span>' +
   395→        '<div><div class="fw7 text-gold">' + ventasPendientes + ' ventas pendientes</div>' +

   401→        '<div class="flex items-center gap-8"><span style="font-size:1.2rem">\u26A0</span>' +
   402→        '<div><div class="fw7 text-red">' + lowStock.length + ' items con stock bajo</div>' +
   403→        '<div class="text-xs text-muted">' + lowStock.slice(0, 5).map(function(p) { return p.nombre + ' (' + p.tipo + ')'; }).join(', ') + (lowStock.length > 5 ? '...' : '') + '</div></div>' +

   554→function guardarCompra() {
   555→  var fecha = document.getElementById('cmp-fecha').value;
   556→  var proveedor = document.getElementById('cmp-proveedor').value.trim();
   557→  var notas = document.getElementById('cmp-notas').value.trim();
   558→  var alertEl = document.getElementById('cmp-alert');
   559→
   560→  if (!fecha || !proveedor) {
   561→    alertEl.textContent = 'Fecha y proveedor son obligatorios';
   562→    alertEl.className = 'alert alert-err show';
   563→    return;
   564→  }
   565→
   566→  var rows = document.querySelectorAll('#cmp-items-container .ing-row');
   567→  var items = [];
   568→  var total = 0;
   569→
   570→  rows.forEach(function(row) {
   571→    var select = row.querySelector('select');
   572→    var productId = parseInt(select.value);
   573→    var qty = parseFloat(row.querySelectorAll('input')[0].value) || 0;
   574→    var price = parseFloat(row.querySelectorAll('input')[1].value) || 0;
   575→    if (productId && qty > 0) {
   576→      var prod = (getDB().productos || []).find(function(p) { return p.id === productId; });
   577→      var sub = qty * price;
   578→      total += sub;
   579→      items.push({ productoId: productId, nombre: prod ? prod.nombre : 'Desconocido', cantidad: qty, unidad: prod ? prod.unidad : 'unidad', precioUnitario: price, subtotal: sub });
   580→    }
   581→  });
   582→
   583→  if (items.length === 0) {
   584→    alertEl.textContent = 'Agrega al menos un item';
   585→    alertEl.className = 'alert alert-err show';
   586→    return;
   587→  }
   588→
   589→  var db = getDB();
   590→  var compra = {
   591→    id: nextId(),
   592→    fecha: fecha,
   593→    proveedor: proveedor,
   594→    items: items,
   595→    total: total,
   596→    notas: notas,
   597→    estado: 'pendiente',
   598→    creadoPor: currentUser ? currentUser.id : null,
   599→    creadoEn: new Date().toISOString()
   600→  };
   601→  db.compras.push(compra);
   602→  saveDB();
   603→  closeModal();
   604→  renderCompras();
   605→  toast('Compra registrada');
   606→}

   787→function guardarProducto(existingId) {
   788→  var nombre = document.getElementById('prod-nombre').value.trim();
   789→  var tipo = document.getElementById('prod-tipo').value;
   790→  var unidad = document.getElementById('prod-unidad').value;
   791→  var costo = parseFloat(document.getElementById('prod-costo').value) || 0;
   792→  var venta = parseFloat(document.getElementById('prod-venta').value) || 0;
   793→  var stock = parseFloat(document.getElementById('prod-stock').value) || 0;
   794→  var stockMin = parseFloat(document.getElementById('prod-stockmin').value) || 0;
   795→  var proveedor = document.getElementById('prod-proveedor').value.trim();
   796→  var notas = document.getElementById('prod-notas').value.trim();
   797→  var alertEl = document.getElementById('prod-alert');
   798→
   799→  if (!nombre) {
   800→    alertEl.textContent = 'El nombre es obligatorio';
   801→    alertEl.className = 'alert alert-err show';
   802→    return;
   803→  }
   804→
   805→  var db = getDB();
   806→  var now = new Date().toISOString();
   807→
   808→  if (existingId) {
   809→    var prod = db.productos.find(function(p) { return p.id === existingId; });
   810→    if (prod) {
   811→      var oldStock = prod.stock;
   812→      prod.nombre = nombre; prod.tipo = tipo; prod.unidad = unidad;
   813→      prod.precioCosto = costo; prod.precioVenta = venta;
   814→      prod.stock = stock; prod.stockMin = stockMin;
   815→      prod.proveedor = proveedor; prod.notas = notas;
   816→      prod.actualizadoEn = now;
   817→      if (stock !== oldStock) {
   818→        addMovimiento('ajuste', 'productos', prod.id, nombre, stock - oldStock, 'Ajuste manual de stock');
   819→      }
   820→    }
   821→  } else {
   822→    db.productos.push({
   823→      id: nextId(), nombre: nombre, tipo: tipo, unidad: unidad,
   824→      precioCosto: costo, precioVenta: venta, stock: stock, stockMin: stockMin,
   825→      proveedor: proveedor, notas: notas,
   826→      creadoEn: now, actualizadoEn: now
   827→    });

   927→// Calculate blend cost per unit based on % and costo/1000gr of especias
   928→function calcBlendCost(blend) {
   929→  if (!blend || !blend.receta) return 0;
   930→  var db = getDB();
   931→  var gramosPorUnidad = blend.gramosPorUnidad || 100;
   932→  var costo = 0;
   933→  blend.receta.forEach(function(r) {
   934→    var prod = (db.productos || []).find(function(p) { return p.id === r.productoId; });
   935→    if (prod && r.porcentaje) {
   936→      // gramos de esta especia por unidad = (porcentaje / 100) * gramosPorUnidad
   937→      // costo = gramos * (precioCosto / 1000)
   938→      var gramos = (r.porcentaje / 100) * gramosPorUnidad;
   939→      costo += gramos * (prod.precioCosto / 1000);
   940→    }
   941→  });
   942→  return Math.round(costo);
   943→}

  1054→function guardarBlend(existingId) {
  1055→  var nombre = document.getElementById('bl-nombre').value.trim();
  1056→  var desc = document.getElementById('bl-desc').value.trim();
  1057→  var formato = document.getElementById('bl-formato').value;
  1058→  var etiquetaId = parseInt(document.getElementById('bl-etiqueta').value) || 0;
  1059→  var gramosPorUnidad = parseFloat(document.getElementById('bl-gramos').value) || 100;
  1060→  var precio = parseFloat(document.getElementById('bl-precio').value) || 0;
  1061→  var alertEl = document.getElementById('bl-alert');
  1062→
  1063→  if (!nombre) {
  1064→    alertEl.textContent = 'El nombre es obligatorio';
  1065→    alertEl.className = 'alert alert-err show';
  1066→    return;
  1067→  }
  1068→
  1069→  var db = getDB();
  1070→  var rows = document.querySelectorAll('#bl-receta-container .ing-row');
  1071→  var receta = [];
  1072→  var totalPct = 0;
  1073→  rows.forEach(function(row) {
  1074→    var select = row.querySelector('select');
  1075→    var pctInput = row.querySelectorAll('input')[0];
  1076→    var productId = parseInt(select.value);
  1077→    var pct = parseFloat(pctInput.value) || 0;
  1078→    if (productId && pct > 0) {
  1079→      var prod = db.productos.find(function(p) { return p.id === productId; });
  1080→      receta.push({ productoId: productId, nombre: prod ? prod.nombre : '?', porcentaje: pct });
  1081→      totalPct += pct;
  1082→    }
  1083→  });
  1084→
  1085→  if (receta.length === 0) {
  1086→    alertEl.textContent = 'Agrega al menos un ingrediente con %';
  1087→    alertEl.className = 'alert alert-err show';
  1088→    return;
  1089→  }
  1090→
  1091→  if (Math.abs(totalPct - 100) >= 1) {
  1092→    alertEl.textContent = 'Los porcentajes deben sumar 100% (actual: ' + totalPct.toFixed(1) + '%)';
  1093→    alertEl.className = 'alert alert-err show';
  1094→    return;
  1095→  }
  1096→
  1097→  // Normalize to exactly 100% if close
  1098→  if (totalPct > 0 && receta.length > 0) {
  1099→    var diff = 100 - totalPct;
  1100→    receta[receta.length - 1].porcentaje = Math.round((receta[receta.length - 1].porcentaje + diff) * 100) / 100;
  1101→  }
  1102→
  1103→  var now = new Date().toISOString();
  1104→  var blendData = { receta: receta, gramosPorUnidad: gramosPorUnidad };
  1105→  var costoCalculado = calcBlendCost(blendData);
  1106→
  1107→  if (existingId) {
  1108→    var blend = db.blends.find(function(b) { return b.id === existingId; });
  1109→    if (blend) {
  1110→      blend.nombre = nombre; blend.descripcion = desc; blend.receta = receta;
  1111→      blend.formato = formato; blend.etiquetaId = etiquetaId;
  1112→      blend.gramosPorUnidad = gramosPorUnidad;
  1113→      blend.precioVenta = precio; blend.costoUnitario = costoCalculado;
  1114→      blend.actualizadoEn = now;
  1115→    }
  1116→  } else {
  1117→    db.blends.push({
  1118→      id: nextId(), nombre: nombre, descripcion: desc, formato: formato, etiquetaId: etiquetaId,
  1119→      gramosPorUnidad: gramosPorUnidad,
  1120→      receta: receta, costoUnitario: costoCalculado, precioVenta: precio,
  1121→      stock: 0, creadoEn: now, actualizadoEn: now
  1122→    });
  1123→  }
  1124→
  1125→  saveDB();
  1126→  closeModal();
  1127→  renderBlends();
  1128→  toast(existingId ? 'Blend actualizado' : 'Blend creado');
  1129→}

  1175→function recalcProduccion(blendId) {

  1178→  if (!b) return;

  1185→    var prod = db.productos.find(function(p) { return p.id === r.productoId; });

  1201→function ejecutarProduccion(blendId) {

  1749→function handleExcelImport(input) {
  1750→  var file = input.files[0];
  1751→  if (!file) return;
  1752→
  1753→  var reader = new FileReader();
  1754→  reader.onload = function(e) {
  1755→    try {
  1756→      var data = new Uint8Array(e.target.result);
  1757→      var workbook = XLSX.read(data, { type: 'array' });
  1758→      var firstSheet = workbook.Sheets[workbook.SheetNames[0]];
  1759→      var rows = XLSX.utils.sheet_to_json(firstSheet, { defval: '' });
  1760→
  1761→      if (rows.length === 0) {
  1762→        toast('El archivo está vacío', 'err');
  1763→        return;
  1764→      }
  1765→
  1766→      // Detect column mapping
  1767→      var headers = Object.keys(rows[0]);
  1768→      var colMap = detectColumnMap(headers);
  1769→
  1770→      if (!colMap.nombre) {
  1771→        toast('No se encontró columna de nombre (Blend/Nombre/Producto)', 'err');
  1772→        return;
  1773→      }
  1774→
  1775→      // Show preview before importing
  1776→      showImportPreview(rows, colMap);
  1777→
  1778→    } catch(err) {
  1779→      toast('Error al leer el archivo: ' + err.message, 'err');
  1780→      console.error('[Import]', err);
  1781→    }
  1782→  };
  1783→  reader.readAsArrayBuffer(file);
  1784→  input.value = ''; // Reset so same file can be re-selected
  1785→}

  1787→function detectColumnMap(headers) {
  1788→  var map = { nombre: null, categoria: null, origen: null, uso: null, perfil: null, precioCosto: null, precioVenta: null, stock: null, unidad: null };
  1789→
  1790→  headers.forEach(function(h) {
  1791→    var hl = h.toLowerCase().trim();
  1792→    if (['blend', 'nombre', 'producto', 'name', 'especia', 'item'].indexOf(hl) !== -1) map.nombre = h;
  1793→    else if (['categoría', 'categoria', 'category', 'tipo', 'type'].indexOf(hl) !== -1) map.categoria = h;
  1794→    else if (['origen', 'origin', 'país', 'pais', 'procedencia'].indexOf(hl) !== -1) map.origen = h;
  1795→    else if (['uso principal', 'uso', 'use', 'aplicación', 'aplicacion'].indexOf(hl) !== -1) map.uso = h;
  1796→    else if (['perfil', 'profile', 'sabor', 'flavor'].indexOf(hl) !== -1) map.perfil = h;
  1797→    else if (['precio costo', 'costo', 'precio_compra', 'cost', 'precio de costo'].indexOf(hl) !== -1) map.precioCosto = h;
  1798→    else if (['precio venta', 'venta', 'precio', 'price', 'precio de venta'].indexOf(hl) !== -1) map.precioVenta = h;
  1799→    else if (['stock', 'cantidad', 'inventario', 'inventory', 'existencia'].indexOf(hl) !== -1) map.stock = h;
  1800→    else if (['unidad', 'unit', 'medida', 'um'].indexOf(hl) !== -1) map.unidad = h;
  1801→  });
  1802→

  1849→function executeImport() {
  1850→  var rows = window._importRows;
  1851→  var colMap = window._importColMap;
  1852→  var tipo = document.getElementById('import-tipo').value;
  1853→  var dupeAction = document.getElementById('import-dupes').value;
  1854→  var alertEl = document.getElementById('import-alert');
  1855→
  1856→  if (!rows || rows.length === 0) return;
  1857→
  1858→  var db = getDB();
  1859→  var now = new Date().toISOString();
  1860→  var imported = 0, updated = 0, skipped = 0;
  1861→
  1862→  // If overwrite, clear existing products of this type
  1863→  if (dupeAction === 'overwrite') {
  1864→    var before = db.productos.length;
  1865→    db.productos = db.productos.filter(function(p) { return p.tipo !== tipo; });
  1866→    toast('Eliminados ' + (before - db.productos.length) + ' productos de tipo ' + tipo);
  1867→  }
  1868→
  1869→  rows.forEach(function(r) {
  1870→    var nombre = String(r[colMap.nombre] || '').trim();
  1871→    if (!nombre) return;
  1872→
  1873→    var existing = db.productos.find(function(p) {
  1874→      return p.nombre.toLowerCase() === nombre.toLowerCase() && p.tipo === tipo;
  1875→    });
  1876→
  1877→    if (existing) {
  1878→      if (dupeAction === 'skip') { skipped++; return; }
  1879→      // Update existing
  1880→      if (colMap.categoria) existing.categoria = String(r[colMap.categoria] || '').trim();
  1881→      if (colMap.origen) existing.origen = String(r[colMap.origen] || '').trim();
  1882→      if (colMap.uso) existing.usoPrincipal = String(r[colMap.uso] || '').trim();
  1883→      if (colMap.perfil) existing.perfil = String(r[colMap.perfil] || '').trim();
  1884→      if (colMap.precioCosto) existing.precioCosto = parseFloat(r[colMap.precioCosto]) || existing.precioCosto;
  1885→      if (colMap.precioVenta) existing.precioVenta = parseFloat(r[colMap.precioVenta]) || existing.precioVenta;
  1886→      if (colMap.stock) existing.stock = parseFloat(r[colMap.stock]) || existing.stock;
  1887→      if (colMap.unidad) existing.unidad = String(r[colMap.unidad] || '').trim();
  1888→      existing.actualizadoEn = now;
  1889→      updated++;
  1890→    } else {
  1891→      var notas = [];
  1892→      if (colMap.uso) notas.push('Uso: ' + String(r[colMap.uso] || '').trim());
  1893→      if (colMap.perfil) notas.push('Perfil: ' + String(r[colMap.perfil] || '').trim());
  1894→
  1895→      var prod = {
  1896→        id: nextId(),
  1897→        nombre: nombre,
  1898→        tipo: tipo,
  1899→        unidad: colMap.unidad ? String(r[colMap.unidad] || 'gr').trim() : 'gr',
  1900→        precioCosto: colMap.precioCosto ? (parseFloat(r[colMap.precioCosto]) || 0) : 0,
  1901→        precioVenta: colMap.precioVenta ? (parseFloat(r[colMap.precioVenta]) || 0) : 0,
  1902→        stock: colMap.stock ? (parseFloat(r[colMap.stock]) || 0) : 0,
  1903→        stockMin: 0,
  1904→        proveedor: '',
  1905→        notas: notas.join(' | '),
  1906→        categoria: colMap.categoria ? String(r[colMap.categoria] || '').trim() : '',
  1907→        origen: colMap.origen ? String(r[colMap.origen] || '').trim() : '',
  1908→        usoPrincipal: colMap.uso ? String(r[colMap.uso] || '').trim() : '',
  1909→        perfil: colMap.perfil ? String(r[colMap.perfil] || '').trim() : '',
  1910→        creadoEn: now,
  1911→        actualizadoEn: now
  1912→      };
  1913→      db.productos.push(prod);
  1914→      imported++;
  1915→    }
  1916→  });
  1917→
  1918→  saveDB();
  1919→  closeModal();
  1920→  renderAjustes();
  1921→  toast('Importados: ' + imported + ' | Actualizados: ' + updated + ' | Ignorados: ' + skipped);
  1922→}
