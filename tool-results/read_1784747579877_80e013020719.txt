   143→function initDB() {
   144→  return new Promise(function(resolve) {
   145→    _initFirebase();
   146→
   147→    // Try localStorage cache
   148→    var cached = null;
   149→    try { cached = JSON.parse(localStorage.getItem(DB_KEY)); } catch (e) {}
   150→
   151→    if (cached && cached.meta && cached.meta.version === DB_VERSION && _ensureStructureOn(cached)) {
   152→      _db = cached;
   153→      _ready = true;
   154→      _startFirebaseListener();
   155→      resolve();
   156→      return;
   157→    }
   158→
   159→    // Load from Firebase
   160→    if (_firebaseRef) {
   161→      _firebaseRef.once('value').then(function(snap) {
   162→        var fbData = snap.val();
   163→        if (fbData && fbData.meta && fbData.meta.version === DB_VERSION && _ensureStructureOn(fbData)) {
   164→          _db = fbData;
   165→        } else {
   166→          // Fresh DB (old version or empty)
   167→          _db = _emptyDB();
   168→          _ensureStructure();
   169→          _saveToFirebase();
   170→        }
   171→        _ready = true;
   172→        _cacheLocal();
   173→        _startFirebaseListener();
   174→        resolve();
   175→      }).catch(function() {
   176→        _db = _emptyDB();
   177→        _ensureStructure();
   178→        _ready = true;
   179→        resolve();
   180→      });
   181→    } else {
   182→      _db = _emptyDB();
   183→      _ensureStructure();
   184→      _ready = true;
   185→      resolve();
   186→    }
   187→  });
   188→}

   426→function producirEspecia(especiaId, talla, cantidad) {
   427→  _ensureStructure();
   428→  var esp = _db.especias[especiaId];
   429→  if (!esp) throw new Error('Especia no encontrada');
   430→  talla = (talla === 'grande') ? 'grande' : 'chico';
   431→  cantidad = Number(cantidad) || 0;
   432→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   433→
   434→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   435→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   436→
   437→  var grsTotal = gramosPorFrasco * cantidad;
   438→
   439→  // Check & consume bolsa
   440→  if ((esp.stockBolsa || 0) < grsTotal) {
   441→    throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   442→  }
   443→
   444→  // Check & consume envases
   445→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   446→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   447→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   448→  }
   449→
   450→  // Check & consume etiquetas
   451→  var etq = _findEtiquetaByNombre(esp.nombre);
   452→  var etqStock = etq ? (Number(etq[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   453→  if (etqStock < cantidad) {
   454→    throw new Error('Etiquetas ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + etqStock);
   455→  }
   456→
   457→  // All checks passed — consume
   458→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   459→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   460→  if (etq) {
   461→    var etqKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   462→    etq[etqKey] = (etq[etqKey] || 0) - cantidad;
   463→  }
   464→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   465→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   466→
   467→  // Record
   468→  var prodId = nextId('producciones');
   469→  var prod = {
   470→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   471→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   472→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   473→    envasesConsumidos: cantidad, etiquetasConsumidas: cantidad,
   474→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   475→  };
   476→  _db.producciones[prodId] = prod;
   477→  _saveToFirebase(); _cacheLocal();
   478→  _notify('create', 'producciones', prodId);
   479→  _notify('update', 'especias', especiaId);
   480→  return { producto: esp, produccion: prod };
   481→}

   489→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   490→
   491→  var ingredientes = blend.ingredientes || [];
   492→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   493→
   494→  // Check ingredient stock
   495→  var detalleIngredientes = [];
   496→  for (var i = 0; i < ingredientes.length; i++) {
   497→    var ing = ingredientes[i];
   498→    var esp = _db.especias[ing.especiaId];
   499→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   500→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   501→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   502→    var grsNeeded = grsPorFrasco * cantidad;
   503→    if ((esp.stockBolsa || 0) < grsNeeded) {
   504→      throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   505→    }
   506→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   507→  }
   508→
   509→  // Check envases
   510→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   511→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   512→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   513→  }
   514→
   515→  // Check etiquetas
   516→  var etq = _findEtiquetaByNombre(blend.nombre);
   517→  var etqStock = etq ? (Number(etq[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   518→  if (etqStock < cantidad) {
   519→    throw new Error('Etiquetas ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + etqStock);
   520→  }
   521→
   522→  // All checks passed — consume
   523→  var grsTotalGeneral = 0;
   524→  for (var i = 0; i < detalleIngredientes.length; i++) {
   525→    var d = detalleIngredientes[i];
   526→    var esp = _db.especias[d.especiaId];
   527→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   528→    grsTotalGeneral += d.gramosTotal;
   529→  }
   530→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   531→  if (etq) {
   532→    var etqKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   533→    etq[etqKey] = (etq[etqKey] || 0) - cantidad;
   534→  }
   535→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   536→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   537→
   538→  var prodId = nextId('producciones');
   539→  var prod = {
   540→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   541→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   542→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   543→    envasesConsumidos: cantidad, etiquetasConsumidas: cantidad,
   544→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   545→  };
   546→  _db.producciones[prodId] = prod;
   547→  _saveToFirebase(); _cacheLocal();
   548→  _notify('create', 'producciones', prodId);
   549→  _notify('update', 'blends', blendId);
   550→  return { producto: blend, produccion: prod };
   551→}

   563→function saveVenta(data) {
   564→  _ensureStructure();
   565→  var isNew = !data.id;
   566→  if (isNew) {
   567→    data.id = nextId('ventas');
   568→    data.creado = new Date().toISOString();
   569→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   570→    data.items = data.items || [];
   571→    data.total = Number(data.total) || 0;
   572→  }
   573→  if (isNew) {
   574→    for (var i = 0; i < data.items.length; i++) {
   575→      var item = data.items[i];
   576→      var producto;
   577→      if (item.tipo === 'especia') {
   578→        producto = _db.especias[item.productoId];
   579→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   580→      } else {
   581→        producto = _db.blends[item.productoId];
   582→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   583→      }
   584→      item.productoNombre = producto.nombre;
   585→      var cant = Number(item.cantidad) || 0;
   586→      var talla = item.talla || 'chico';
   587→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   588→      if ((producto[stockKey] || 0) < cant) {
   589→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   590→      }
   591→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   592→      item.precioUnitario = Number(item.precioUnitario) || 0;
   593→      item.subtotal = item.precioUnitario * cant;
   594→    }
   595→    // Recalculate total
   596→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   597→  }
   598→  _db.ventas[data.id] = data;
   599→  _saveToFirebase(); _cacheLocal();
   600→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   601→  return data;
   602→}

   756→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   757→function findEspeciaByName(nombre) {
   758→  if (!nombre) return null;
   759→  var target = nombre.trim().toLowerCase();
   760→  var keys = Object.keys(_db.especias || {});
   761→  // 1. Exact match
   762→  for (var i = 0; i < keys.length; i++) {
   763→    var e = _db.especias[keys[i]];
   764→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   765→  }
   766→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   767→  for (var i = 0; i < keys.length; i++) {
   768→    var e = _db.especias[keys[i]];
   769→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   770→  }
   771→  // 3. Target is contained in especia name
   772→  for (var i = 0; i < keys.length; i++) {
   773→    var e = _db.especias[keys[i]];
   774→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   775→  }
   776→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   777→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   778→  for (var w = 0; w < targetWords.length; w++) {
   779→    for (var i = 0; i < keys.length; i++) {
   780→      var e = _db.especias[keys[i]];
   781→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   782→    }
   783→  }
   784→  return null;
   785→}

   787→/** Auto-detect categoria from blend USO field */
   788→function _categoriaFromUso(uso) {
   789→  if (!uso) return 'Comidas';
   790→  var u = uso.toLowerCase();
   791→  // Cocteleria keywords
   792→  if (/\b(gin|ron|vodka|whisky|mojito|mule|vermouth|aperitif|coctel)\b/.test(u)) return 'Cocteleria';
   793→  // Infusiones keywords
   794→  if (/\b(relajant|sueño|digestiv|energiz|té|calidez|respiratorio|meditaci|antioxidant|bienestar|infusi)\b/.test(u)) return 'Infusiones';
   795→  return 'Comidas';
   796→}

   798→/**
   799→ * Import especias and blends from parsed Excel data.
   800→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   801→ *
   802→ * especiasList: [{ nombre, categoria? }]
   803→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   804→ * gramosChico:  default grams for frasco chico (e.g. 30)
   805→ * gramosGrande: default grams for frasco grande (e.g. 80)
   806→ */
   807→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   808→  _ensureStructure();
   809→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   810→
   811→  // 1. Create especias (skip if name already exists)
   812→  for (var i = 0; i < especiasList.length; i++) {
   813→    var esp = especiasList[i];
   814→    var nombre = (esp.nombre || '').trim();
   815→    if (!nombre) continue;
   816→    var existing = findEspeciaByName(nombre);
   817→    if (existing) {
   818→      resultado.especiasExistentes++;
   819→      continue;
   820→    }
   821→    saveEspecia({
   822→      nombre: nombre,
   823→      categoria: esp.categoria || 'Comidas',
   824→      precioChico: 0,
   825→      precioGrande: 0,
   826→      gramosChico: Number(gramosChico) || 30,
   827→      gramosGrande: Number(gramosGrande) || 80,
   828→      stockBolsa: 0,
   829→      stockChico: 0,
   830→      stockGrande: 0
   831→    });
   832→    resultado.especiasCreadas++;
   833→  }
   834→
   835→  // 2. Create blends (skip if name already exists)
   836→  for (var j = 0; j < blendsList.length; j++) {
   837→    var bl = blendsList[j];
   838→    var nombre = (bl.nombre || '').trim();
   839→    if (!nombre) continue;
   840→    var existingBlend = null;
   841→    var blKeys = Object.keys(_db.blends || {});
   842→    for (var k = 0; k < blKeys.length; k++) {
   843→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
   844→        existingBlend = _db.blends[blKeys[k]];
   845→        break;
   846→      }
   847→    }
   848→    if (existingBlend) {
   849→      resultado.blendsExistentes++;
   850→      continue;
   851→    }
   852→
   853→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
   854→    var ings = bl.ingredientes || [];
   855→    var recipeTotal = 0;
   856→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
   857→    if (recipeTotal <= 0) recipeTotal = 500;
   858→
   859→    var resolvedIngs = [];
   860→    for (var ii = 0; ii < ings.length; ii++) {
   861→      var ing = ings[ii];
   862→      var espObj = findEspeciaByName(ing.especia);
   863→      if (!espObj) {
   864→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
   865→        continue;
   866→      }
   867→      var ingG = Number(ing.g) || 0;
   868→      resolvedIngs.push({
   869→        especiaId: espObj.id,
   870→        especiaNombre: espObj.nombre,
   871→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
   872→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
   873→        gramosReceta: ingG
   874→      });
   875→    }
   876→
   877→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
   878→    saveBlend({
   879→      nombre: nombre,
   880→      categoria: cat,
   881→      region: bl.region || '',
   882→      uso: bl.uso || '',
   883→      precioChico: 0,
   884→      precioGrande: 0,
   885→      ingredientes: resolvedIngs,
   886→      stockChico: 0,
   887→      stockGrande: 0
   888→    });
   889→    resultado.blendsCreados++;
   890→  }
   891→
   892→  return resultado;
   893→}

   931→  // Also remove from all products that have this tag
   932→  var allEsp = getEspecias();
   933→  for (var e = 0; e < allEsp.length; e++) {
   934→    if (allEsp[e].tags) {
   935→      var ti = allEsp[e].tags.indexOf(tagName);
   936→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
   937→    }
   938→  }
   939→  var allBl = getBlends();
   940→  for (var b = 0; b < allBl.length; b++) {
   941→    if (allBl[b].tags) {
   942→      var bi = allBl[b].tags.indexOf(tagName);
   943→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
   944→    }
   945→  }
   946→  _saveToFirebase(); _cacheLocal();
   947→  return true;
   948→}

   952→function compressImage(file, maxW, quality, cb) {
   953→  var reader = new FileReader();
   954→  reader.onload = function(ev) {
   955→    var img = new Image();
   956→    img.onload = function() {
   957→      var w = img.width, h = img.height;
   958→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
   959→      var canvas = document.createElement('canvas');
   960→      canvas.width = w; canvas.height = h;
   961→      var ctx = canvas.getContext('2d');
   962→      ctx.drawImage(img, 0, 0, w, h);
   963→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
   964→      cb(null, dataUrl);
   965→    };
   966→    img.onerror = function() { cb('Error al cargar imagen'); };
   967→    img.src = ev.target.result;
   968→  };
   969→  reader.onerror = function() { cb('Error al leer archivo'); };
   970→  reader.readAsDataURL(file);
   971→}
