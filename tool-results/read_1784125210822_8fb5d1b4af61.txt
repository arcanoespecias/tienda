   797→  2396→  if (!input) return;
   798→  2397→
   799→  2398→  input.onchange = function(e) {

   801→  2400→    if (!file) return;
   802→  2401→
   803→  2402→    var reader = new FileReader();

   823→  2422→            if (!nombre) return;
   824→  2423→            var existing = null;
   825→  2424→            db.productos.forEach(function(p) { if (p.nombre === nombre) existing = p; });
   826→  2425→            if (existing) {

   833→  2432→              if (row.PrecioVenta !== undefined) existing.precioVenta = Number(row.PrecioVenta) || 0;

   835→  2434→              if (row.Proveedor) existing.proveedor = row.Proveedor;
   836→  2435→              updated++;
   837→  2436→            } else {
   838→  2437→              db.productos.push({
   839→  2438→                id: nextId(), nombre: nombre, tipo: row.Tipo || 'especia',
   840→  2439→                unidad: row.Unidad || 'gr', stock: Number(row.Stock) || 0,
   841→  2440→                stockMin: Number(row.StockMin) || 0,

   843→  2442→                precioCosto: Number(row.PrecioCosto) || 0,
   844→  2443→                precioVenta: Number(row.PrecioVenta) || 0,
   845→  2444→                categoria: row.Categoria || '', proveedor: row.Proveedor || ''
   846→  2445→              });
   847→  2446→              added++;
   848→  2447→            }
   849→  2448→          });
   850→  2449→        }
   851→  2450→
   852→  2451→        saveDB();
   853→  2452→        refreshPage();
   854→  2453→        toast('Importados: ' + added + ' nuevos, ' + updated + ' actualizados');
   855→  2454→      } catch(err) {
   856→  2455→        toast('Error al importar: ' + (err.message || ''), 'err');
   857→  2456→      }
   858→  2457→    };
   859→  2458→    reader.readAsArrayBuffer(file);
   860→  2459→    input.value = '';
   861→  2460→  };
