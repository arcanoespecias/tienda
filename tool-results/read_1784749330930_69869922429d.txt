   388→   388→    data.items = data.items || [];
   389→   389→    data.total = Number(data.total) || 0;
   390→   390→  }
   391→   391→  if (isNew) {
   392→   392→    for (var i = 0; i < data.items.length; i++) {
   393→   393→      var item = data.items[i];
   394→   394→      var tipo = item.tipo;
   395→   395→      if (tipo === 'especia_grs') {
   396→   396→        // Add grams to especia stockBolsa
   397→   397→        if (item.especiaId && _db.especias[item.especiaId]) {
   398→   398→          _db.especias[item.especiaId].stockBolsa = (_db.especias[item.especiaId].stockBolsa || 0) + (Number(item.cantidad) || 0);
   399→   399→        }
   400→   400→      } else if (tipo === 'envase') {
   401→   401→        var talla = item.talla || 'chico';
   402→   402→        if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   403→   403→        _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) + (Number(item.cantidad) || 0);
   404→   404→      } else if (tipo === 'sticker') {
   405→   405→        var stk = _getOrCreateSticker(item.stickerNombre);
   406→   406→        var t = item.talla || 'chico';
   407→   407→        if (t === 'grande') {
   408→   408→          stk.stockGrande = (stk.stockGrande || 0) + (Number(item.cantidad) || 0);
   409→   409→        } else {
   410→   410→          stk.stockChico = (stk.stockChico || 0) + (Number(item.cantidad) || 0);
   411→   411→        }
   412→   412→      }
   413→   413→    }
   414→   414→  }
   415→   415→  _db.entradas[data.id] = data;
   416→   416→  _saveToFirebase(); _cacheLocal();
   417→   417→  _notify(isNew ? 'create' : 'update', 'entradas', data.id);

   431→   431→function producirEspecia(especiaId, talla, cantidad) {
   432→   432→  _ensureStructure();
   433→   433→  var esp = _db.especias[especiaId];
   434→   434→  if (!esp) throw new Error('Especia no encontrada');
   435→   435→  talla = (talla === 'grande') ? 'grande' : 'chico';
   436→   436→  cantidad = Number(cantidad) || 0;
   437→   437→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   438→   438→
   439→   439→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   440→   440→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   441→   441→
   442→   442→  var grsTotal = gramosPorFrasco * cantidad;
   443→   443→
   444→   444→  // Check & consume bolsa
   445→   445→  if ((esp.stockBolsa || 0) < grsTotal) {
   446→   446→    throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   447→   447→  }
   448→   448→
   449→   449→  // Check & consume envases
   450→   450→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   451→   451→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   452→   452→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   453→   453→  }
   454→   454→
   455→   455→  // Check & consume stickers
   456→   456→  var stk = _findStickerByNombre(esp.nombre);
   457→   457→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   458→   458→  if (stkStock < cantidad) {
   459→   459→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   460→   460→  }
   461→   461→
   462→   462→  // All checks passed — consume
   463→   463→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   464→   464→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   465→   465→  if (stk) {
   466→   466→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   467→   467→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   468→   468→  }
   469→   469→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   470→   470→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   471→   471→
   472→   472→  // Record
   473→   473→  var prodId = nextId('producciones');
   474→   474→  var prod = {
   475→   475→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   476→   476→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   477→   477→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   478→   478→    envasesConsumidos: cantidad, stickersConsumidos: cantidad,
   479→   479→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   480→   480→  };
   481→   481→  _db.producciones[prodId] = prod;
   482→   482→  _saveToFirebase(); _cacheLocal();
   483→   483→  _notify('create', 'producciones', prodId);
   484→   484→  _notify('update', 'especias', especiaId);
   485→   485→  return { producto: esp, produccion: prod };
   486→   486→}

   491→   491→  if (!blend) throw new Error('Blend no encontrado');

   495→   495→

   497→   497→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   498→   498→
   499→   499→  // Check ingredient stock

   501→   501→  for (var i = 0; i < ingredientes.length; i++) {

   503→   503→    var esp = _db.especias[ing.especiaId];
   504→   504→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');

   506→   506→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   507→   507→    var grsNeeded = grsPorFrasco * cantidad;
   508→   508→    if ((esp.stockBolsa || 0) < grsNeeded) {
   509→   509→      throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   510→   510→    }

   512→   512→  }
   513→   513→
   514→   514→  // Check envases
   515→   515→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   516→   516→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   517→   517→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   518→   518→  }
   519→   519→
   520→   520→  // Check stickers
   521→   521→  var stk = _findStickerByNombre(blend.nombre);
   522→   522→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;

   524→   524→    throw new Error('Stickers ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   525→   525→  }
   526→   526→
   527→   527→  // All checks passed — consume
   528→   528→  var grsTotalGeneral = 0;
   529→   529→  for (var i = 0; i < detalleIngredientes.length; i++) {

   541→   541→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   542→   542→
   543→   543→  var prodId = nextId('producciones');
   544→   544→  var prod = {
   545→   545→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   546→   546→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   547→   547→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   548→   548→    envasesConsumidos: cantidad, stickersConsumidos: cantidad,
   549→   549→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   550→   550→  };

   552→   552→  _saveToFirebase(); _cacheLocal();
   553→   553→  _notify('create', 'producciones', prodId);
   554→   554→  _notify('update', 'blends', blendId);

   568→   568→function saveVenta(data) {
   569→   569→  _ensureStructure();

   580→   580→      var item = data.items[i];
   581→   581→      var producto;
   582→   582→      if (item.tipo === 'especia') {
   583→   583→        producto = _db.especias[item.productoId];
   584→   584→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   585→   585→      } else {
   586→   586→        producto = _db.blends[item.productoId];
   587→   587→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   588→   588→      }
   589→   589→      item.productoNombre = producto.nombre;
   590→   590→      var cant = Number(item.cantidad) || 0;
   591→   591→      var talla = item.talla || 'chico';
   592→   592→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   593→   593→      if ((producto[stockKey] || 0) < cant) {
   594→   594→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   595→   595→      }
   596→   596→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   597→   597→      item.precioUnitario = Number(item.precioUnitario) || 0;
   598→   598→      item.subtotal = item.precioUnitario * cant;
   599→   599→    }
   600→   600→    // Recalculate total
   601→   601→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   602→   602→  }
   603→   603→  _db.ventas[data.id] = data;
   604→   604→  _saveToFirebase(); _cacheLocal();
   605→   605→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   606→   606→  return data;
   607→   607→}

   761→   761→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   762→   762→function findEspeciaByName(nombre) {
   763→   763→  if (!nombre) return null;
   764→   764→  var target = nombre.trim().toLowerCase();
   765→   765→  var keys = Object.keys(_db.especias || {});
   766→   766→  // 1. Exact match
   767→   767→  for (var i = 0; i < keys.length; i++) {
   768→   768→    var e = _db.especias[keys[i]];
   769→   769→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   770→   770→  }
   771→   771→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   772→   772→  for (var i = 0; i < keys.length; i++) {
   773→   773→    var e = _db.especias[keys[i]];
   774→   774→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   775→   775→  }
   776→   776→  // 3. Target is contained in especia name
   777→   777→  for (var i = 0; i < keys.length; i++) {
   778→   778→    var e = _db.especias[keys[i]];
   779→   779→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   780→   780→  }
   781→   781→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   782→   782→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   783→   783→  for (var w = 0; w < targetWords.length; w++) {
   784→   784→    for (var i = 0; i < keys.length; i++) {
   785→   785→      var e = _db.especias[keys[i]];
   786→   786→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   787→   787→    }
   788→   788→  }
   789→   789→  return null;
   790→   790→}

   792→   792→/** Auto-detect categoria from blend USO field */
   793→   793→function _categoriaFromUso(uso) {
   794→   794→  if (!uso) return 'Comidas';
   795→   795→  var u = uso.toLowerCase();
   796→   796→  // Cocteleria keywords
   797→   797→  if (/\b(gin|ron|vodka|whisky|mojito|mule|vermouth|aperitif|coctel)\b/.test(u)) return 'Cocteleria';
   798→   798→  // Infusiones keywords
   799→   799→  if (/\b(relajant|sueño|digestiv|energiz|té|calidez|respiratorio|meditaci|antioxidant|bienestar|infusi)\b/.test(u)) return 'Infusiones';
   800→   800→  return 'Comidas';
   801→   801→}

   803→   803→/**
   804→   804→ * Import especias and blends from parsed Excel data.
   805→   805→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   806→   806→ *
   807→   807→ * especiasList: [{ nombre, categoria? }]
   808→   808→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   809→   809→ * gramosChico:  default grams for frasco chico (e.g. 30)
   810→   810→ * gramosGrande: default grams for frasco grande (e.g. 80)
   811→   811→ */
   812→   812→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   813→   813→  _ensureStructure();
   814→   814→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   815→   815→
   816→   816→  // 1. Create especias (skip if name already exists)
   817→   817→  for (var i = 0; i < especiasList.length; i++) {
   818→   818→    var esp = especiasList[i];
   819→   819→    var nombre = (esp.nombre || '').trim();
   820→   820→    if (!nombre) continue;
   821→   821→    var existing = findEspeciaByName(nombre);
   822→   822→    if (existing) {
   823→   823→      resultado.especiasExistentes++;
   824→   824→      continue;
   825→   825→    }
   826→   826→    saveEspecia({
   827→   827→      nombre: nombre,
   828→   828→      categoria: esp.categoria || 'Comidas',
   829→   829→      precioChico: 0,
   830→   830→      precioGrande: 0,
   831→   831→      gramosChico: Number(gramosChico) || 30,
   832→   832→      gramosGrande: Number(gramosGrande) || 80,
   833→   833→      stockBolsa: 0,
   834→   834→      stockChico: 0,
   835→   835→      stockGrande: 0
   836→   836→    });
   837→   837→    resultado.especiasCreadas++;
   838→   838→  }
   839→   839→
   840→   840→  // 2. Create blends (skip if name already exists)
   841→   841→  for (var j = 0; j < blendsList.length; j++) {
   842→   842→    var bl = blendsList[j];
   843→   843→    var nombre = (bl.nombre || '').trim();
   844→   844→    if (!nombre) continue;
   845→   845→    var existingBlend = null;
   846→   846→    var blKeys = Object.keys(_db.blends || {});
   847→   847→    for (var k = 0; k < blKeys.length; k++) {
   848→   848→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
   849→   849→        existingBlend = _db.blends[blKeys[k]];
   850→   850→        break;
   851→   851→      }
   852→   852→    }
   853→   853→    if (existingBlend) {
   854→   854→      resultado.blendsExistentes++;
   855→   855→      continue;
   856→   856→    }
   857→   857→
   858→   858→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
   859→   859→    var ings = bl.ingredientes || [];
   860→   860→    var recipeTotal = 0;
   861→   861→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
   862→   862→    if (recipeTotal <= 0) recipeTotal = 500;
   863→   863→
   864→   864→    var resolvedIngs = [];
   865→   865→    for (var ii = 0; ii < ings.length; ii++) {
   866→   866→      var ing = ings[ii];
   867→   867→      var espObj = findEspeciaByName(ing.especia);
   868→   868→      if (!espObj) {
   869→   869→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
   870→   870→        continue;
   871→   871→      }
   872→   872→      var ingG = Number(ing.g) || 0;
   873→   873→      resolvedIngs.push({
   874→   874→        especiaId: espObj.id,
   875→   875→        especiaNombre: espObj.nombre,
   876→   876→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
   877→   877→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
   878→   878→        gramosReceta: ingG
   879→   879→      });
   880→   880→    }
   881→   881→
   882→   882→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
   883→   883→    saveBlend({
   884→   884→      nombre: nombre,
   885→   885→      categoria: cat,
   886→   886→      region: bl.region || '',
   887→   887→      uso: bl.uso || '',
   888→   888→      precioChico: 0,
   889→   889→      precioGrande: 0,
   890→   890→      ingredientes: resolvedIngs,
   891→   891→      stockChico: 0,
   892→   892→      stockGrande: 0
   893→   893→    });
   894→   894→    resultado.blendsCreados++;
   895→   895→  }
   896→   896→
   897→   897→  return resultado;
   898→   898→}

   912→   912→function addProductTag(cat, tagName) {
   913→   913→  _ensureStructure();
   914→   914→  tagName = (tagName || '').trim();
   915→   915→  if (!tagName) return false;
   916→   916→  if (!_db.productTags) _db.productTags = {};
   917→   917→  if (!_db.productTags[cat]) _db.productTags[cat] = [];
   918→   918→  // Check duplicate (case-insensitive)
   919→   919→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   920→   920→    if (_db.productTags[cat][i].toLowerCase() === tagName.toLowerCase()) return false;
   921→   921→  }
   922→   922→  _db.productTags[cat].push(tagName);
   923→   923→  _saveToFirebase(); _cacheLocal();
   924→   924→  return true;
   925→   925→}
   926→   926→
   927→   927→function removeProductTag(cat, tagName) {
   928→   928→  _ensureStructure();
   929→   929→  if (!_db.productTags || !_db.productTags[cat]) return false;
   930→   930→  var idx = -1;
   931→   931→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   932→   932→    if (_db.productTags[cat][i] === tagName) { idx = i; break; }
   933→   933→  }
   934→   934→  if (idx < 0) return false;
   935→   935→  _db.productTags[cat].splice(idx, 1);
   936→   936→  // Also remove from all products that have this tag
   937→   937→  var allEsp = getEspecias();
   938→   938→  for (var e = 0; e < allEsp.length; e++) {
   939→   939→    if (allEsp[e].tags) {
   940→   940→      var ti = allEsp[e].tags.indexOf(tagName);
   941→   941→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
   942→   942→    }
   943→   943→  }
   944→   944→  var allBl = getBlends();
   945→   945→  for (var b = 0; b < allBl.length; b++) {
   946→   946→    if (allBl[b].tags) {
   947→   947→      var bi = allBl[b].tags.indexOf(tagName);
   948→   948→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
   949→   949→    }
   950→   950→  }
   951→   951→  _saveToFirebase(); _cacheLocal();
   952→   952→  return true;
   953→   953→}

   957→   957→function compressImage(file, maxW, quality, cb) {
   958→   958→  var reader = new FileReader();
   959→   959→  reader.onload = function(ev) {
   960→   960→    var img = new Image();
   961→   961→    img.onload = function() {
   962→   962→      var w = img.width, h = img.height;
   963→   963→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
   964→   964→      var canvas = document.createElement('canvas');
   965→   965→      canvas.width = w; canvas.height = h;
   966→   966→      var ctx = canvas.getContext('2d');
   967→   967→      ctx.drawImage(img, 0, 0, w, h);
   968→   968→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
   969→   969→      cb(null, dataUrl);
   970→   970→    };
   971→   971→    img.onerror = function() { cb('Error al cargar imagen'); };
   972→   972→    img.src = ev.target.result;
   973→   973→  };
   974→   974→  reader.onerror = function() { cb('Error al leer archivo'); };
   975→   975→  reader.readAsDataURL(file);
   976→   976→}
