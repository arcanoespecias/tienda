   158→function _saveToLocal() {

   162→function _loadFromFirebase() {
   163→  return new Promise(function(resolve) {
   164→    var ref = _fbRef();
   165→    if (!ref) { resolve(_loadFromLocal()); return; }
   166→    ref.once('value').then(function(snap) {
   167→      if (snap.exists() && snap.val()) {
   168→        var data = snap.val();
   169→        if (typeof data !== 'object' || Array.isArray(data)) {
   170→          _db = _emptyDB();
   171→        } else {
   172→          _db = data;
   173→        }
   174→      } else {
   175→        _db = _emptyDB();
   176→      }
   177→      _ensureStructure();
   178→      _saveToLocal();
   179→      resolve(_db);
   180→    }).catch(function(err) {
   181→      console.error('[DB] Firebase load error:', err);
   182→      resolve(_loadFromLocal());
   183→    });
   184→  });
   185→}

   187→function _loadFromLocal() {

   191→  } catch (e) { _db = _emptyDB(); }
   192→  _ensureStructure();
   193→  return _db;

   331→/** Producir frascos de ESPECIA pura: consume grs de bolsa + etiquetas (por talla), agrega frascos */
   332→function producirEspeciaFrascos(especiaId, grsPorFrasco, cantidad, talla) {
   333→  _ensureStructure();
   334→  var esp = _db.especias[especiaId];
   335→  if (!esp) throw new Error('Especia no encontrada');
   336→  grsPorFrasco = Number(grsPorFrasco) || 0;
   337→  cantidad = Number(cantidad) || 0;
   338→  talla = (talla === 'grande') ? 'grande' : 'chico';
   339→  if (grsPorFrasco <= 0) throw new Error('Los gramos por frasco deben ser mayor a 0');
   340→  if (cantidad <= 0) throw new Error('La cantidad de frascos debe ser mayor a 0');
   341→
   342→  var grsTotal = grsPorFrasco * cantidad;
   343→  if ((esp.stockBolsa || 0) < grsTotal) {
   344→    throw new Error('Stock insuficiente en BOLSA de "' + esp.nombre + '". Necesario: ' + grsTotal + 'grs, Disponible: ' + (esp.stockBolsa || 0) + 'grs');
   345→  }
   346→
   347→  // Check etiqueta por talla
   348→  var etiqueta = _findEtiquetaByNombre(esp.nombre);
   349→  var detalleEtiqueta = null;
   350→  var stockKey = (talla === 'grande') ? 'stockGrande' : 'stockChico';
   351→  if (etiqueta) {
   352→    var etqDisp = Number(etiqueta[stockKey]) || 0;
   353→    if (etqDisp < cantidad) {
   354→      throw new Error('Stock insuficiente de ETIQUETAS ' + talla.toUpperCase() + ' de "' + esp.nombre + '". Necesario: ' + cantidad + ', Disponible: ' + etqDisp);
   355→    }
   356→    etiqueta[stockKey] = etqDisp - cantidad;
   357→    detalleEtiqueta = { etiquetaNombre: esp.nombre, talla: talla, cantidadConsumida: cantidad, stockRestante: etiqueta[stockKey] };
   358→  } else {
   359→    throw new Error('No hay etiquetas fisicas registradas para "' + esp.nombre + '". Compralas en Compras primero.');
   360→  }
   361→
   362→  // Consume bolsa, add frascos
   363→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   364→  esp.stockFrascos = (esp.stockFrascos || 0) + cantidad;
   365→
   366→  // Record
   367→  var prodId = nextId('producciones');
   368→  var produccion = {
   369→    id: prodId, tipo: 'especia', especiaId: especiaId, especiaNombre: esp.nombre,
   370→    categoria: esp.categoria || '', talla: talla, grsPorFrasco: grsPorFrasco, cantidad: cantidad,
   371→    grsConsumidos: grsTotal, etiquetaUsada: detalleEtiqueta,
   372→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   373→  };
   374→  _db.producciones[prodId] = produccion;
   375→  _saveToFirebase();
   376→  _notify('create', 'producciones', prodId);
   377→  _notify('update', 'especias', especiaId);
   378→  return { especia: esp, produccion: produccion };
   379→}

   381→/** Producir frascos de BLEND: consume grs de especias (bolsa) + etiquetas (por talla), agrega frascos al blend */
   382→function producirBlend(blendId, cantidad, talla) {
   383→  _ensureStructure();
   384→  var blend = _db.blends[blendId];
   385→  if (!blend) throw new Error('Blend no encontrado');
   386→  cantidad = Number(cantidad) || 0;
   387→  talla = (talla === 'grande') ? 'grande' : 'chico';
   388→  if (cantidad <= 0) throw new Error('Cantidad debe ser mayor a 0');
   389→
   390→  var ingredientes = blend.ingredientes || [];
   391→  var detalleIngredientes = [];
   392→  var detalleEtiqueta = null;
   393→
   394→  // 1. Check especias stockBolsa (grams)
   395→  for (var i = 0; i < ingredientes.length; i++) {
   396→    var ing = ingredientes[i];
   397→    var esp = _db.especias[ing.especiaId];
   398→    if (!esp) throw new Error('Especia no encontrada: ' + ing.especiaId);
   399→    var grsNeeded = (ing.cantidad || 0) * cantidad;
   400→    if ((esp.stockBolsa || 0) < grsNeeded) {
   401→      throw new Error('Stock insuficiente en BOLSA de "' + esp.nombre + '". Necesario: ' + grsNeeded + 'grs, Disponible: ' + (esp.stockBolsa || 0) + 'grs');
   402→    }
   403→  }
   404→  // 2. Consume especias from bolsa
   405→  for (var i = 0; i < ingredientes.length; i++) {
   406→    var ing = ingredientes[i];
   407→    var esp = _db.especias[ing.especiaId];
   408→    var grsNeeded = (ing.cantidad || 0) * cantidad;
   409→    esp.stockBolsa = (esp.stockBolsa || 0) - grsNeeded;
   410→    detalleIngredientes.push({
   411→      especiaId: ing.especiaId, especiaNombre: esp.nombre,
   412→      grsPorFrasco: ing.cantidad || 0, grsTotal: grsNeeded
   413→    });
   414→  }
   415→
   416→  // 3. Check & consume etiqueta por talla
   417→  var stockKey = (talla === 'grande') ? 'stockGrande' : 'stockChico';
   418→  var etiqueta = _findEtiquetaByNombre(blend.nombre);
   419→  if (etiqueta) {
   420→    var etqDisp = Number(etiqueta[stockKey]) || 0;
   421→    if (etqDisp < cantidad) {
   422→      throw new Error('Stock insuficiente de ETIQUETAS ' + talla.toUpperCase() + ' de "' + blend.nombre + '". Necesario: ' + cantidad + ', Disponible: ' + etqDisp);
   423→    }
   424→    etiqueta[stockKey] = etqDisp - cantidad;
   425→    detalleEtiqueta = { etiquetaNombre: etiqueta.nombre, talla: talla, cantidadConsumida: cantidad, stockRestante: etiqueta[stockKey] };
   426→  } else {
   427→    throw new Error('No hay etiquetas fisicas registradas para "' + blend.nombre + '". Compralas en Compras primero.');
   428→  }
   429→
   430→  // 4. Add blend frascos
   431→  blend.stockFrascos = (blend.stockFrascos || 0) + cantidad;
   432→
   433→  // 5. Record
   434→  var prodId = nextId('producciones');
   435→  var produccion = {
   436→    id: prodId, tipo: 'blend', blendId: blendId, blendNombre: blend.nombre,
   437→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   438→    ingredientesUsados: detalleIngredientes, etiquetaUsada: detalleEtiqueta,
   439→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   440→  };
   441→  _db.producciones[prodId] = produccion;
   442→  _saveToFirebase();
   443→  _notify('create', 'producciones', prodId);
   444→  _notify('update', 'blends', blendId);
   445→  return { blend: blend, produccion: produccion };
   446→}

   520→function saveCompra(data) {
   521→  _ensureStructure();
   522→  var isNew = !data.id;
   523→  if (isNew) {
   524→    data.id = nextId('compras');
   525→    data.creado = new Date().toISOString();
   526→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   527→    data.items = data.items || [];
   528→    data.total = Number(data.total) || 0;
   529→  }
   530→  if (isNew) {
   531→    for (var i = 0; i < data.items.length; i++) {
   532→      var item = data.items[i];
   533→      if (item.tipo === 'etiqueta') {
   534→        var etq = _getOrCreateEtiqueta(item.etiquetaNombre);
   535→        var talla = item.talla || 'chico';
   536→        if (talla === 'grande') {
   537→          etq.stockGrande = (etq.stockGrande || 0) + (Number(item.cantidad) || 0);
   538→        } else {
   539→          etq.stockChico = (etq.stockChico || 0) + (Number(item.cantidad) || 0);
   540→        }
   541→      } else if (item.tipo === 'especia' && item.especiaId && _db.especias[item.especiaId]) {
   542→        // Add to stockBolsa (grams of raw spice)
   543→        _db.especias[item.especiaId].stockBolsa = (_db.especias[item.especiaId].stockBolsa || 0) + (Number(item.cantidad) || 0);
   544→      }
   545→    }
   546→  }
   547→  _db.compras[data.id] = data;
   548→  _saveToFirebase();
   549→  _notify(isNew ? 'create' : 'update', 'compras', data.id);
   550→  return data;
   551→}

   567→function saveVenta(data) {
   568→  _ensureStructure();
   569→  var isNew = !data.id;
   570→  if (isNew) {
   571→    data.id = nextId('ventas');
   572→    data.creado = new Date().toISOString();
   573→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   574→    data.items = data.items || [];
   575→    data.total = Number(data.total) || 0;
   576→  }
   577→  if (isNew) {
   578→    for (var i = 0; i < data.items.length; i++) {
   579→      var item = data.items[i];
   580→      var producto;
   581→      if (item.tipo === 'especia') {
   582→        producto = _db.especias[item.productoId];
   583→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   584→      } else {
   585→        producto = _db.blends[item.productoId];
   586→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   587→      }
   588→      var cant = Number(item.cantidad) || 0;
   589→      var stockDisponible = (producto.stockFrascos || 0);
   590→      if (stockDisponible < cant) {
   591→        throw new Error('Stock insuficiente de FRASCOS de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + stockDisponible);
   592→      }
   593→      item.productoNombre = producto.nombre;
   594→      producto.stockFrascos = (producto.stockFrascos || 0) - cant;
   595→    }
   596→  }
   597→  _db.ventas[data.id] = data;
   598→  _saveToFirebase();
   599→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   600→  return data;
   601→}

   655→/* ==================== DASHBOARD STATS ==================== */
   656→
   657→function getStats() {
   658→  var db = getDB();
   659→  var especias = _filterValid(Object.values(db.especias || {}));
   660→  var blends = _filterValid(Object.values(db.blends || {}));
   661→  var ventas = _filterValid(Object.values(db.ventas || {}));
   662→  var compras = _filterValid(Object.values(db.compras || {}));
   663→  var etiquetas = _filterValid(Object.values(db.etiquetas || {}));
   664→
   665→  var today = new Date().toISOString().slice(0, 10);
   666→  var mes = new Date().toISOString().slice(0, 7);
   667→  var ventasHoy = ventas.filter(function(v) { return v.fecha === today; });
   668→  var ventasMes = ventas.filter(function(v) { return v.fecha && v.fecha.startsWith(mes); });
   669→  var comprasMes = compras.filter(function(c) { return c.fecha && c.fecha.startsWith(mes); });
   670→  var etqBajoStock = etiquetas.filter(function(e) {
   671→    var chico = Number(e.stockChico) || 0;
   672→    var grande = Number(e.stockGrande) || 0;
   673→    return chico + grande <= 5;
   674→  });
   675→  var totalFrascos = especias.reduce(function(s, e) { return s + (e.stockFrascos || 0); }, 0) +
   676→                      blends.reduce(function(s, b) { return s + (b.stockFrascos || 0); }, 0);
   677→
   678→  return {
   679→    totalEspecias: especias.length,
   680→    totalBlends: blends.length,
   681→    totalEtiquetas: etiquetas.length,
   682→    totalFrascos: totalFrascos,
   683→    totalVentas: ventas.length,
   684→    totalCompras: compras.length,
   685→    ventasHoy: ventasHoy.length,
   686→    totalVentasHoy: ventasHoy.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   687→    ventasMes: ventasMes.length,
   688→    totalVentasMes: ventasMes.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   689→    totalComprasMes: comprasMes.reduce(function(s, c) { return s + (Number(c.total) || 0); }, 0),
   690→    especiasBajoStockBolsa: especias.filter(function(e) { return (e.stockBolsa || 0) <= 50; }),
   691→    especiasBajoStockFrascos: especias.filter(function(e) { return (e.stockFrascos || 0) <= 3; }),
   692→    blendsBajoStockFrascos: blends.filter(function(b) { return (b.stockFrascos || 0) <= 3; }),
   693→    etiquetasBajoStock: etqBajoStock
   694→  };
   695→}
