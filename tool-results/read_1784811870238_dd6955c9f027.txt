   158→function initDB() {
   159→  return new Promise(function(resolve) {
   160→    _initFirebase();
   161→
   162→    // Always try localStorage cache for instant UI
   163→    var cached = null;
   164→    try { cached = JSON.parse(localStorage.getItem(DB_KEY)); } catch (e) {}
   165→
   166→    if (cached && cached.meta && cached.meta.version === DB_VERSION && _ensureStructureOn(cached)) {
   167→      _db = cached;
   168→      _ensureStructure();  // ensure new fields exist on cached data
   169→      _ready = true;
   170→      _startFirebaseListener();
   171→      resolve();
   172→      return;
   173→    }
   174→
   175→    // Load from Firebase
   176→    if (_firebaseRef) {
   177→      _firebaseRef.once('value').then(function(snap) {
   178→        var fbData = snap.val();
   179→        if (fbData && fbData.meta && fbData.meta.version === DB_VERSION && _ensureStructureOn(fbData)) {
   180→          _db = fbData;
   181→          _ensureStructure();  // ensure new fields exist on Firebase data
   182→        } else {
   183→          _db = _emptyDB();
   184→          _ensureStructure();
   185→          _saveToFirebase();
   186→        }
   187→        _ready = true;
   188→        _cacheLocal();
   189→        _startFirebaseListener();
   190→        resolve();
   191→      }).catch(function() {
   192→        _db = _emptyDB();
   193→        _ensureStructure();
   194→        _ready = true;
   195→        resolve();
   196→      });
   197→    } else {
   198→      _db = _emptyDB();
   199→      _ensureStructure();
   200→      _ready = true;
   201→      resolve();
   202→    }
   203→  });
   204→}

   206→function _ensureStructureOn(data) {
   207→  if (!data || typeof data !== 'object') return false;
   208→  if (!data.meta || !data.meta.version) return false;
   209→  if (data.meta.version !== DB_VERSION) return false;
   210→  // Basic check
   211→  if (!data.especias || !data.blends) return false;
   212→  return true;
   213→}

   219→function _startFirebaseListener() {
   220→  if (!_firebaseRef) return;
   221→  _firebaseRef.on('value', function(snap) {
   222→    var data = snap.val();
   223→    if (!data || !data.meta || data.meta.version !== DB_VERSION) return;
   224→    // CRITICAL: skip if local save is pending to prevent overwriting unsaved changes
   225→    if (_localDirty) return;
   226→    var prevJson = JSON.stringify(_db);
   227→    _db = data;
   228→    _ensureStructure();
   229→    _cacheLocal();
   230→    var newJson = JSON.stringify(_db);
   231→    if (prevJson !== newJson) {
   232→      _notify('remote_change', '', '');
   233→    }
   234→  });
   235→}

   401→    data.items = data.items || [];
   402→    data.total = Number(data.total) || 0;
   403→  }
   404→  if (isNew) {
   405→    for (var i = 0; i < data.items.length; i++) {
   406→      var item = data.items[i];
   407→      var tipo = item.tipo;
   408→      if (tipo === 'especia_grs') {
   409→        // Add grams to especia stockBolsa
   410→        if (item.especiaId && _db.especias[item.especiaId]) {
   411→          _db.especias[item.especiaId].stockBolsa = (_db.especias[item.especiaId].stockBolsa || 0) + (Number(item.cantidad) || 0);
   412→        }
   413→      } else if (tipo === 'envase') {
   414→        var talla = item.talla || 'chico';
   415→        if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   416→        _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) + (Number(item.cantidad) || 0);
   417→      } else if (tipo === 'sticker') {
   418→        var stk = _getOrCreateSticker(item.stickerNombre);
   419→        var t = item.talla || 'chico';
   420→        if (t === 'grande') {
   421→          stk.stockGrande = (stk.stockGrande || 0) + (Number(item.cantidad) || 0);
   422→        } else {
   423→          stk.stockChico = (stk.stockChico || 0) + (Number(item.cantidad) || 0);
   424→        }
   425→      } else if (tipo === 'bolsa') {
   426→        var tallaB = item.talla || 'chico';
   427→        if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };

   433→  _saveToFirebase(); _cacheLocal();

   446→/* ==================== PRODUCCION ==================== */
   447→
   448→function producirEspecia(especiaId, talla, cantidad) {
   449→  _ensureStructure();
   450→  var esp = _db.especias[especiaId];
   451→  if (!esp) throw new Error('Especia no encontrada');
   452→  talla = (talla === 'grande') ? 'grande' : 'chico';
   453→  cantidad = Number(cantidad) || 0;
   454→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   455→
   456→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   457→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   458→
   459→  var grsTotal = gramosPorFrasco * cantidad;
   460→
   461→  // Check & consume pala (raw material)
   462→  if ((esp.stockBolsa || 0) < grsTotal) {
   463→    throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   464→  }
   465→
   466→  // Check & consume envases
   467→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   468→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   469→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   470→  }
   471→
   472→  // Check & consume stickers
   473→  var stk = _findStickerByNombre(esp.nombre);
   474→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   475→  if (stkStock < cantidad) {
   476→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   477→  }
   478→
   479→  // Check & consume bolsas (packaging)
   480→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   481→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   482→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   483→  }
   484→
   485→  // All checks passed — consume
   486→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   487→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;

   512→function producirBlend(blendId, talla, cantidad) {
   513→  _ensureStructure();
   514→  var blend = _db.blends[blendId];
   515→  if (!blend) throw new Error('Blend no encontrado');
   516→  talla = (talla === 'grande') ? 'grande' : 'chico';
   517→  cantidad = Number(cantidad) || 0;
   518→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   519→
   520→  var ingredientes = blend.ingredientes || [];
   521→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   522→
   523→  // Check ingredient stock
   524→  var detalleIngredientes = [];
   525→  for (var i = 0; i < ingredientes.length; i++) {
   526→    var ing = ingredientes[i];
   527→    var esp = _db.especias[ing.especiaId];
   528→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   529→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   530→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   531→    var grsNeeded = grsPorFrasco * cantidad;
   532→    if ((esp.stockBolsa || 0) < grsNeeded) {
   533→      throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   534→    }
   535→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   536→  }
   537→
   538→  // Check envases
   539→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   540→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   541→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   542→  }
   543→
   544→  // Check stickers
   545→  var stk = _findStickerByNombre(blend.nombre);
   546→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   547→  if (stkStock < cantidad) {
   548→    throw new Error('Stickers ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   549→  }
   550→
   551→  // Check & consume bolsas (packaging)
   552→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   553→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   554→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   555→  }
   556→
   557→  // All checks passed — consume
   558→  var grsTotalGeneral = 0;
   559→  for (var i = 0; i < detalleIngredientes.length; i++) {
   560→    var d = detalleIngredientes[i];
   561→    var esp = _db.especias[d.especiaId];
   562→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   563→    grsTotalGeneral += d.gramosTotal;
   564→  }
   565→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   566→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   567→  if (stk) {
   568→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   569→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   570→  }
   571→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   572→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   573→
   574→  var prodId = nextId('producciones');
   575→  var prod = {
   576→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   577→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   578→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   579→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   580→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   581→  };
   582→  _db.producciones[prodId] = prod;
   583→  _saveToFirebase(); _cacheLocal();
   584→  _notify('create', 'producciones', prodId);
   585→  _notify('update', 'blends', blendId);
   586→  return { producto: blend, produccion: prod };
   587→}

   599→function saveVenta(data) {
   600→  _ensureStructure();
   601→  var isNew = !data.id;
   602→  if (isNew) {
   603→    data.id = nextId('ventas');
   604→    data.creado = new Date().toISOString();
   605→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   606→    data.items = data.items || [];
   607→    data.total = Number(data.total) || 0;
   608→  }
   609→  if (isNew) {
   610→    for (var i = 0; i < data.items.length; i++) {
   611→      var item = data.items[i];
   612→      var producto;
   613→      if (item.tipo === 'especia') {
   614→        producto = _db.especias[item.productoId];
   615→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   616→      } else {
   617→        producto = _db.blends[item.productoId];
   618→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   619→      }
   620→      item.productoNombre = producto.nombre;
   621→      var cant = Number(item.cantidad) || 0;
   622→      var talla = item.talla || 'chico';
   623→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   624→      if ((producto[stockKey] || 0) < cant) {
   625→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   626→      }
   627→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   628→      item.precioUnitario = Number(item.precioUnitario) || 0;
   629→      item.subtotal = item.precioUnitario * cant;
   630→    }
   631→    // Recalculate total
   632→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   633→  }
   634→  _db.ventas[data.id] = data;
   635→  _saveToFirebase(); _cacheLocal();
   636→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   637→  return data;
   638→}

   795→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   796→function findEspeciaByName(nombre) {
   797→  if (!nombre) return null;
   798→  var target = nombre.trim().toLowerCase();
   799→  var keys = Object.keys(_db.especias || {});
   800→  // 1. Exact match
   801→  for (var i = 0; i < keys.length; i++) {
   802→    var e = _db.especias[keys[i]];
   803→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   804→  }
   805→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   806→  for (var i = 0; i < keys.length; i++) {
   807→    var e = _db.especias[keys[i]];
   808→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   809→  }
   810→  // 3. Target is contained in especia name
   811→  for (var i = 0; i < keys.length; i++) {
   812→    var e = _db.especias[keys[i]];
   813→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   814→  }
   815→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   816→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   817→  for (var w = 0; w < targetWords.length; w++) {
   818→    for (var i = 0; i < keys.length; i++) {
   819→      var e = _db.especias[keys[i]];
   820→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   821→    }
   822→  }
   823→  return null;
   824→}

   832→  // Infusiones keywords

   834→  return 'Comidas';

   837→/**
   838→ * Import especias and blends from parsed Excel data.
   839→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   840→ *
   841→ * especiasList: [{ nombre, categoria? }]
   842→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   843→ * gramosChico:  default grams for frasco chico (e.g. 30)
   844→ * gramosGrande: default grams for frasco grande (e.g. 80)
   845→ */
   846→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   847→  _ensureStructure();
   848→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   849→
   850→  // 1. Create especias (skip if name already exists)
   851→  for (var i = 0; i < especiasList.length; i++) {
   852→    var esp = especiasList[i];
   853→    var nombre = (esp.nombre || '').trim();
   854→    if (!nombre) continue;
   855→    var existing = findEspeciaByName(nombre);
   856→    if (existing) {
   857→      resultado.especiasExistentes++;
   858→      continue;
   859→    }
   860→    saveEspecia({
   861→      nombre: nombre,
   862→      categoria: esp.categoria || 'Comidas',
   863→      precioChico: 0,
   864→      precioGrande: 0,
   865→      gramosChico: Number(gramosChico) || 30,
   866→      gramosGrande: Number(gramosGrande) || 80,
   867→      stockBolsa: 0,
   868→      stockChico: 0,
   869→      stockGrande: 0
   870→    });
   871→    resultado.especiasCreadas++;
   872→  }
   873→
   874→  // 2. Create blends (skip if name already exists)
   875→  for (var j = 0; j < blendsList.length; j++) {
   876→    var bl = blendsList[j];
   877→    var nombre = (bl.nombre || '').trim();
   878→    if (!nombre) continue;
   879→    var existingBlend = null;
   880→    var blKeys = Object.keys(_db.blends || {});
   881→    for (var k = 0; k < blKeys.length; k++) {
   882→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
   883→        existingBlend = _db.blends[blKeys[k]];
   884→        break;
   885→      }
   886→    }
   887→    if (existingBlend) {
   888→      resultado.blendsExistentes++;
   889→      continue;
   890→    }
   891→
   892→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
   893→    var ings = bl.ingredientes || [];
   894→    var recipeTotal = 0;
   895→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
   896→    if (recipeTotal <= 0) recipeTotal = 500;
   897→
   898→    var resolvedIngs = [];
   899→    for (var ii = 0; ii < ings.length; ii++) {
   900→      var ing = ings[ii];
   901→      var espObj = findEspeciaByName(ing.especia);
   902→      if (!espObj) {
   903→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
   904→        continue;
   905→      }
   906→      var ingG = Number(ing.g) || 0;
   907→      resolvedIngs.push({
   908→        especiaId: espObj.id,
   909→        especiaNombre: espObj.nombre,
   910→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
   911→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
   912→        gramosReceta: ingG
   913→      });
   914→    }
   915→
   916→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
   917→    saveBlend({
   918→      nombre: nombre,
   919→      categoria: cat,
   920→      region: bl.region || '',
   921→      uso: bl.uso || '',
   922→      precioChico: 0,
   923→      precioGrande: 0,
   924→      ingredientes: resolvedIngs,
   925→      stockChico: 0,
   926→      stockGrande: 0
   927→    });
   928→    resultado.blendsCreados++;
   929→  }
   930→
   931→  return resultado;
   932→}

   946→function addProductTag(cat, tagName) {
   947→  _ensureStructure();
   948→  tagName = (tagName || '').trim();
   949→  if (!tagName) return false;
   950→  if (!_db.productTags) _db.productTags = {};
   951→  if (!_db.productTags[cat]) _db.productTags[cat] = [];
   952→  // Check duplicate (case-insensitive)
   953→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   954→    if (_db.productTags[cat][i].toLowerCase() === tagName.toLowerCase()) return false;
   955→  }
   956→  _db.productTags[cat].push(tagName);
   957→  _saveToFirebase(); _cacheLocal();
   958→  return true;
   959→}
   960→
   961→function removeProductTag(cat, tagName) {
   962→  _ensureStructure();
   963→  if (!_db.productTags || !_db.productTags[cat]) return false;
   964→  var idx = -1;
   965→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   966→    if (_db.productTags[cat][i] === tagName) { idx = i; break; }
   967→  }
   968→  if (idx < 0) return false;
   969→  _db.productTags[cat].splice(idx, 1);
   970→  // Also remove from all products that have this tag
   971→  var allEsp = getEspecias();
   972→  for (var e = 0; e < allEsp.length; e++) {
   973→    if (allEsp[e].tags) {
   974→      var ti = allEsp[e].tags.indexOf(tagName);
   975→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
   976→    }
   977→  }
   978→  var allBl = getBlends();
   979→  for (var b = 0; b < allBl.length; b++) {
   980→    if (allBl[b].tags) {
   981→      var bi = allBl[b].tags.indexOf(tagName);
   982→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
   983→    }
   984→  }
   985→  _saveToFirebase(); _cacheLocal();
   986→  return true;
   987→}

   991→function compressImage(file, maxW, quality, cb) {
   992→  var reader = new FileReader();
   993→  reader.onload = function(ev) {
   994→    var img = new Image();
   995→    img.onload = function() {
   996→      var w = img.width, h = img.height;
   997→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
   998→      var canvas = document.createElement('canvas');
   999→      canvas.width = w; canvas.height = h;
  1000→      var ctx = canvas.getContext('2d');
  1001→      ctx.drawImage(img, 0, 0, w, h);
  1002→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
  1003→      cb(null, dataUrl);
  1004→    };
  1005→    img.onerror = function() { cb('Error al cargar imagen'); };
  1006→    img.src = ev.target.result;
  1007→  };
  1008→  reader.onerror = function() { cb('Error al leer archivo'); };
  1009→  reader.readAsDataURL(file);
  1010→}
