   141→    if (cached && cached.meta && cached.meta.version === DB_VERSION && _ensureStructureOn(cached)) {

   150→    if (_firebaseRef) {

   152→        var fbData = snap.val();
   153→        if (fbData && fbData.meta && fbData.meta.version === DB_VERSION && _ensureStructureOn(fbData)) {
   154→          _db = fbData;
   155→        } else {
   156→          // Fresh DB (old version or empty)
   157→          _db = _emptyDB();
   158→          _ensureStructure();
   159→          _saveToFirebase();
   160→        }
   161→        _ready = true;
   162→        _cacheLocal();
   163→        _startFirebaseListener();
   164→        resolve();
   165→      }).catch(function() {
   166→        _db = _emptyDB();
   167→        _ensureStructure();
   168→        _ready = true;
   169→        resolve();
   170→      });
   171→    } else {
   172→      _db = _emptyDB();
   173→      _ensureStructure();
   174→      _ready = true;
   175→      resolve();
   176→    }
   177→  });
   178→}

   416→function producirEspecia(especiaId, talla, cantidad) {
   417→  _ensureStructure();
   418→  var esp = _db.especias[especiaId];
   419→  if (!esp) throw new Error('Especia no encontrada');
   420→  talla = (talla === 'grande') ? 'grande' : 'chico';
   421→  cantidad = Number(cantidad) || 0;
   422→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   423→
   424→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   425→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   426→
   427→  var grsTotal = gramosPorFrasco * cantidad;
   428→
   429→  // Check & consume bolsa
   430→  if ((esp.stockBolsa || 0) < grsTotal) {
   431→    throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   432→  }
   433→
   434→  // Check & consume envases
   435→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   436→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   437→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   438→  }
   439→
   440→  // Check & consume etiquetas
   441→  var etq = _findEtiquetaByNombre(esp.nombre);
   442→  var etqStock = etq ? (Number(etq[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   443→  if (etqStock < cantidad) {
   444→    throw new Error('Etiquetas ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + etqStock);
   445→  }
   446→
   447→  // All checks passed — consume
   448→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   449→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   450→  if (etq) {
   451→    var etqKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   452→    etq[etqKey] = (etq[etqKey] || 0) - cantidad;
   453→  }
   454→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   455→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   456→
   457→  // Record
   458→  var prodId = nextId('producciones');
   459→  var prod = {
   460→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   461→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   462→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   463→    envasesConsumidos: cantidad, etiquetasConsumidas: cantidad,
   464→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   465→  };
   466→  _db.producciones[prodId] = prod;
   467→  _saveToFirebase(); _cacheLocal();
   468→  _notify('create', 'producciones', prodId);
   469→  _notify('update', 'especias', especiaId);
   470→  return { producto: esp, produccion: prod };
   471→}

   473→function producirBlend(blendId, talla, cantidad) {
   474→  _ensureStructure();
   475→  var blend = _db.blends[blendId];
   476→  if (!blend) throw new Error('Blend no encontrado');
   477→  talla = (talla === 'grande') ? 'grande' : 'chico';
   478→  cantidad = Number(cantidad) || 0;
   479→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   480→
   481→  var ingredientes = blend.ingredientes || [];
   482→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   483→
   484→  // Check ingredient stock
   485→  var detalleIngredientes = [];
   486→  for (var i = 0; i < ingredientes.length; i++) {
   487→    var ing = ingredientes[i];
   488→    var esp = _db.especias[ing.especiaId];
   489→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   490→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   491→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   492→    var grsNeeded = grsPorFrasco * cantidad;
   493→    if ((esp.stockBolsa || 0) < grsNeeded) {
   494→      throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   495→    }
   496→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   497→  }
   498→
   499→  // Check envases
   500→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   501→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   502→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   503→  }
   504→
   505→  // Check etiquetas
   506→  var etq = _findEtiquetaByNombre(blend.nombre);
   507→  var etqStock = etq ? (Number(etq[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   508→  if (etqStock < cantidad) {
   509→    throw new Error('Etiquetas ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + etqStock);
   510→  }
   511→
   512→  // All checks passed — consume
   513→  var grsTotalGeneral = 0;
   514→  for (var i = 0; i < detalleIngredientes.length; i++) {
   515→    var d = detalleIngredientes[i];
   516→    var esp = _db.especias[d.especiaId];
   517→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   518→    grsTotalGeneral += d.gramosTotal;
   519→  }
   520→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   521→  if (etq) {
   522→    var etqKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   523→    etq[etqKey] = (etq[etqKey] || 0) - cantidad;
   524→  }
   525→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   526→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   527→
   528→  var prodId = nextId('producciones');
   529→  var prod = {
   530→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   531→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   532→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   533→    envasesConsumidos: cantidad, etiquetasConsumidas: cantidad,
   534→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   535→  };
   536→  _db.producciones[prodId] = prod;
   537→  _saveToFirebase(); _cacheLocal();
   538→  _notify('create', 'producciones', prodId);
   539→  _notify('update', 'blends', blendId);
   540→  return { producto: blend, produccion: prod };
   541→}

   553→function saveVenta(data) {
   554→  _ensureStructure();
   555→  var isNew = !data.id;
   556→  if (isNew) {
   557→    data.id = nextId('ventas');
   558→    data.creado = new Date().toISOString();
   559→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   560→    data.items = data.items || [];
   561→    data.total = Number(data.total) || 0;
   562→  }
   563→  if (isNew) {
   564→    for (var i = 0; i < data.items.length; i++) {
   565→      var item = data.items[i];
   566→      var producto;
   567→      if (item.tipo === 'especia') {
   568→        producto = _db.especias[item.productoId];
   569→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   570→      } else {
   571→        producto = _db.blends[item.productoId];
   572→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   573→      }
   574→      item.productoNombre = producto.nombre;
   575→      var cant = Number(item.cantidad) || 0;
   576→      var talla = item.talla || 'chico';
   577→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   578→      if ((producto[stockKey] || 0) < cant) {
   579→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   580→      }
   581→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   582→      item.precioUnitario = Number(item.precioUnitario) || 0;
   583→      item.subtotal = item.precioUnitario * cant;
   584→    }
   585→    // Recalculate total
   586→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   587→  }
   588→  _db.ventas[data.id] = data;
   589→  _saveToFirebase(); _cacheLocal();
   590→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   591→  return data;
   592→}

   641→function getStats() {
   642→  var especias = _filterValid(Object.values(_db.especias || {}));
   643→  var blends = _filterValid(Object.values(_db.blends || {}));
   644→  var ventas = _filterValid(Object.values(_db.ventas || {}));
   645→  var etiquetas = _filterValid(Object.values(_db.etiquetas || {}));
   646→  var envases = _db.stockEnvases || { chico: 0, grande: 0 };
   647→
   648→  var today = new Date().toISOString().slice(0, 10);
   649→  var mes = new Date().toISOString().slice(0, 7);
   650→  var ventasHoy = ventas.filter(function(v) { return v.fecha === today; });
   651→  var ventasMes = ventas.filter(function(v) { return v.fecha && v.fecha.startsWith(mes); });
   652→
   653→  var frascosChico = especias.reduce(function(s, e) { return s + (e.stockChico || 0); }, 0) +
   654→                     blends.reduce(function(s, b) { return s + (b.stockChico || 0); }, 0);
   655→  var frascosGrande = especias.reduce(function(s, e) { return s + (e.stockGrande || 0); }, 0) +
   656→                      blends.reduce(function(s, b) { return s + (b.stockGrande || 0); }, 0);
   657→
   658→  var etqBajo = etiquetas.filter(function(e) { return (e.stockChico + e.stockGrande) <= 5; });
   659→  var espBolsaBaja = especias.filter(function(e) { return (e.stockBolsa || 0) <= 50; });
   660→
   661→  return {
   662→    totalEspecias: especias.length,
   663→    totalBlends: blends.length,
   664→    totalProductos: especias.length + blends.length,
   665→    frascosChico: frascosChico,
   666→    frascosGrande: frascosGrande,
   667→    totalFrascos: frascosChico + frascosGrande,
   668→    envasesChico: envases.chico || 0,
   669→    envasesGrande: envases.grande || 0,
   670→    ventasHoy: ventasHoy.length,
   671→    totalVentasHoy: ventasHoy.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   672→    ventasMes: ventasMes.length,
   673→    totalVentasMes: ventasMes.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   674→    especiasBolsaBaja: espBolsaBaja,
   675→    etiquetasBajas: etqBajo
   676→  };
   677→}

   680→function getFrascosParaVender() {
   681→  var items = [];
   682→  var espKeys = Object.keys(_db.especias || {});
   683→  for (var i = 0; i < espKeys.length; i++) {
   684→    var e = _db.especias[espKeys[i]];
   685→    if (!e || typeof e !== 'object') continue;
   686→    if ((e.stockChico || 0) > 0) items.push({ tipo: 'especia', id: e.id, nombre: e.nombre, talla: 'chico', stock: e.stockChico, precio: e.precioChico || 0 });
   687→    if ((e.stockGrande || 0) > 0) items.push({ tipo: 'especia', id: e.id, nombre: e.nombre, talla: 'grande', stock: e.stockGrande, precio: e.precioGrande || 0 });
   688→  }
   689→  var blKeys = Object.keys(_db.blends || {});
   690→  for (var i = 0; i < blKeys.length; i++) {
   691→    var b = _db.blends[blKeys[i]];
   692→    if (!b || typeof b !== 'object') continue;
   693→    if ((b.stockChico || 0) > 0) items.push({ tipo: 'blend', id: b.id, nombre: b.nombre, talla: 'chico', stock: b.stockChico, precio: b.precioChico || 0 });
   694→    if ((b.stockGrande || 0) > 0) items.push({ tipo: 'blend', id: b.id, nombre: b.nombre, talla: 'grande', stock: b.stockGrande, precio: b.precioGrande || 0 });
   695→  }
   696→  return items.sort(function(a, b) { return a.nombre.localeCompare(b.nombre); });
   697→}
