   397→   397→      }

   416→   416→function producirEspecia(especiaId, talla, cantidad) {
   417→   417→  _ensureStructure();
   418→   418→  var esp = _db.especias[especiaId];
   419→   419→  if (!esp) throw new Error('Especia no encontrada');
   420→   420→  talla = (talla === 'grande') ? 'grande' : 'chico';
   421→   421→  cantidad = Number(cantidad) || 0;
   422→   422→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   423→   423→
   424→   424→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   425→   425→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   426→   426→
   427→   427→  var grsTotal = gramosPorFrasco * cantidad;
   428→   428→
   429→   429→  // Check & consume bolsa
   430→   430→  if ((esp.stockBolsa || 0) < grsTotal) {
   431→   431→    throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   432→   432→  }
   433→   433→
   434→   434→  // Check & consume envases
   435→   435→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   436→   436→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   437→   437→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   438→   438→  }
   439→   439→
   440→   440→  // Check & consume etiquetas
   441→   441→  var etq = _findEtiquetaByNombre(esp.nombre);
   442→   442→  var etqStock = etq ? (Number(etq[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   443→   443→  if (etqStock < cantidad) {
   444→   444→    throw new Error('Etiquetas ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + etqStock);
   445→   445→  }
   446→   446→
   447→   447→  // All checks passed — consume
   448→   448→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   449→   449→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   450→   450→  if (etq) {
   451→   451→    var etqKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   452→   452→    etq[etqKey] = (etq[etqKey] || 0) - cantidad;
   453→   453→  }
   454→   454→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   455→   455→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   456→   456→
   457→   457→  // Record
   458→   458→  var prodId = nextId('producciones');
   459→   459→  var prod = {
   460→   460→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   461→   461→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   462→   462→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   463→   463→    envasesConsumidos: cantidad, etiquetasConsumidas: cantidad,
   464→   464→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   465→   465→  };
   466→   466→  _db.producciones[prodId] = prod;
   467→   467→  _saveToFirebase(); _cacheLocal();
   468→   468→  _notify('create', 'producciones', prodId);
   469→   469→  _notify('update', 'especias', especiaId);
   470→   470→  return { producto: esp, produccion: prod };

   473→   473→function producirBlend(blendId, talla, cantidad) {
   474→   474→  _ensureStructure();
   475→   475→  var blend = _db.blends[blendId];
   476→   476→  if (!blend) throw new Error('Blend no encontrado');
   477→   477→  talla = (talla === 'grande') ? 'grande' : 'chico';
   478→   478→  cantidad = Number(cantidad) || 0;
   479→   479→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   480→   480→
   481→   481→  var ingredientes = blend.ingredientes || [];
   482→   482→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   483→   483→
   484→   484→  // Check ingredient stock
   485→   485→  var detalleIngredientes = [];
   486→   486→  for (var i = 0; i < ingredientes.length; i++) {
   487→   487→    var ing = ingredientes[i];
   488→   488→    var esp = _db.especias[ing.especiaId];
   489→   489→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   490→   490→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   491→   491→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   492→   492→    var grsNeeded = grsPorFrasco * cantidad;
   493→   493→    if ((esp.stockBolsa || 0) < grsNeeded) {
   494→   494→      throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   495→   495→    }
   496→   496→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   497→   497→  }
   498→   498→
   499→   499→  // Check envases
   500→   500→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   501→   501→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   502→   502→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   503→   503→  }
   504→   504→
   505→   505→  // Check etiquetas
   506→   506→  var etq = _findEtiquetaByNombre(blend.nombre);
   507→   507→  var etqStock = etq ? (Number(etq[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   508→   508→  if (etqStock < cantidad) {
   509→   509→    throw new Error('Etiquetas ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + etqStock);
   510→   510→  }
   511→   511→
   512→   512→  // All checks passed — consume
   513→   513→  var grsTotalGeneral = 0;
   514→   514→  for (var i = 0; i < detalleIngredientes.length; i++) {
   515→   515→    var d = detalleIngredientes[i];
   516→   516→    var esp = _db.especias[d.especiaId];
   517→   517→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   518→   518→    grsTotalGeneral += d.gramosTotal;
   519→   519→  }
   520→   520→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   521→   521→  if (etq) {
   522→   522→    var etqKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   523→   523→    etq[etqKey] = (etq[etqKey] || 0) - cantidad;
   524→   524→  }
   525→   525→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   526→   526→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   527→   527→
   528→   528→  var prodId = nextId('producciones');
   529→   529→  var prod = {
   530→   530→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   531→   531→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   532→   532→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   533→   533→    envasesConsumidos: cantidad, etiquetasConsumidas: cantidad,
   534→   534→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   535→   535→  };
   536→   536→  _db.producciones[prodId] = prod;
   537→   537→  _saveToFirebase(); _cacheLocal();
   538→   538→  _notify('create', 'producciones', prodId);
   539→   539→  _notify('update', 'blends', blendId);
   540→   540→  return { producto: blend, produccion: prod };
   541→   541→}

   553→   553→function saveVenta(data) {
   554→   554→  _ensureStructure();
   555→   555→  var isNew = !data.id;
   556→   556→  if (isNew) {
   557→   557→    data.id = nextId('ventas');
   558→   558→    data.creado = new Date().toISOString();
   559→   559→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   560→   560→    data.items = data.items || [];
   561→   561→    data.total = Number(data.total) || 0;
   562→   562→  }
   563→   563→  if (isNew) {
   564→   564→    for (var i = 0; i < data.items.length; i++) {
   565→   565→      var item = data.items[i];
   566→   566→      var producto;
   567→   567→      if (item.tipo === 'especia') {
   568→   568→        producto = _db.especias[item.productoId];
   569→   569→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   570→   570→      } else {
   571→   571→        producto = _db.blends[item.productoId];
   572→   572→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   573→   573→      }
   574→   574→      item.productoNombre = producto.nombre;
   575→   575→      var cant = Number(item.cantidad) || 0;
   576→   576→      var talla = item.talla || 'chico';
   577→   577→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   578→   578→      if ((producto[stockKey] || 0) < cant) {
   579→   579→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   580→   580→      }
   581→   581→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   582→   582→      item.precioUnitario = Number(item.precioUnitario) || 0;
   583→   583→      item.subtotal = item.precioUnitario * cant;
   584→   584→    }
   585→   585→    // Recalculate total
   586→   586→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   587→   587→  }
   588→   588→  _db.ventas[data.id] = data;
   589→   589→  _saveToFirebase(); _cacheLocal();
   590→   590→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   591→   591→  return data;
   592→   592→}

   641→   641→function getStats() {
   642→   642→  var especias = _filterValid(Object.values(_db.especias || {}));
   643→   643→  var blends = _filterValid(Object.values(_db.blends || {}));
   644→   644→  var ventas = _filterValid(Object.values(_db.ventas || {}));
   645→   645→  var etiquetas = _filterValid(Object.values(_db.etiquetas || {}));
   646→   646→  var envases = _db.stockEnvases || { chico: 0, grande: 0 };
   647→   647→
   648→   648→  var today = new Date().toISOString().slice(0, 10);
   649→   649→  var mes = new Date().toISOString().slice(0, 7);
   650→   650→  var ventasHoy = ventas.filter(function(v) { return v.fecha === today; });
   651→   651→  var ventasMes = ventas.filter(function(v) { return v.fecha && v.fecha.startsWith(mes); });
   652→   652→
   653→   653→  var frascosChico = especias.reduce(function(s, e) { return s + (e.stockChico || 0); }, 0) +
   654→   654→                     blends.reduce(function(s, b) { return s + (b.stockChico || 0); }, 0);
   655→   655→  var frascosGrande = especias.reduce(function(s, e) { return s + (e.stockGrande || 0); }, 0) +
   656→   656→                      blends.reduce(function(s, b) { return s + (b.stockGrande || 0); }, 0);
   657→   657→
   658→   658→  var etqBajo = etiquetas.filter(function(e) { return (e.stockChico + e.stockGrande) <= 5; });
   659→   659→  var espBolsaBaja = especias.filter(function(e) { return (e.stockBolsa || 0) <= 50; });
   660→   660→
   661→   661→  return {
   662→   662→    totalEspecias: especias.length,
   663→   663→    totalBlends: blends.length,
   664→   664→    totalProductos: especias.length + blends.length,
   665→   665→    frascosChico: frascosChico,
   666→   666→    frascosGrande: frascosGrande,
   667→   667→    totalFrascos: frascosChico + frascosGrande,
   668→   668→    envasesChico: envases.chico || 0,
   669→   669→    envasesGrande: envases.grande || 0,
   670→   670→    ventasHoy: ventasHoy.length,
   671→   671→    totalVentasHoy: ventasHoy.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   672→   672→    ventasMes: ventasMes.length,
   673→   673→    totalVentasMes: ventasMes.reduce(function(s, v) { return s + (Number(v.total) || 0); }, 0),
   674→   674→    especiasBolsaBaja: espBolsaBaja,
   675→   675→    etiquetasBajas: etqBajo
   676→   676→  };
   677→   677→}

   680→   680→function getFrascosParaVender() {

   687→   687→    if ((e.stockGrande || 0) > 0) items.push({ tipo: 'especia', id: e.id, nombre: e.nombre, talla: 'grande', stock: e.stockGrande, precio: e.precioGrande || 0 });
   688→   688→  }
   689→   689→  var blKeys = Object.keys(_db.blends || {});
   690→   690→  for (var i = 0; i < blKeys.length; i++) {
   691→   691→    var b = _db.blends[blKeys[i]];
   692→   692→    if (!b || typeof b !== 'object') continue;
   693→   693→    if ((b.stockChico || 0) > 0) items.push({ tipo: 'blend', id: b.id, nombre: b.nombre, talla: 'chico', stock: b.stockChico, precio: b.precioChico || 0 });
   694→   694→    if ((b.stockGrande || 0) > 0) items.push({ tipo: 'blend', id: b.id, nombre: b.nombre, talla: 'grande', stock: b.stockGrande, precio: b.precioGrande || 0 });
