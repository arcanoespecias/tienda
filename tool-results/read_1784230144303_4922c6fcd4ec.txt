   240→   240→  _calcEspeciaProduccion(especiaId) {
   241→   241→    const esp = ArcanoDB.getEspecia(especiaId);
   242→   242→    if (!esp) return;
   243→   243→    const talla = (document.getElementById('f-pe-talla') || {}).value || 'chico';
   244→   244→    const grs = Number(document.getElementById('f-pe-grs').value) || 0;
   245→   245→    const cant = Number(document.getElementById('f-pe-cant').value) || 0;
   246→   246→    const total = grs * cant;
   247→   247→    const disponible = esp.stockBolsa || 0;
   248→   248→    const okBolsa = total > 0 && total <= disponible;
   249→   249→    const etqList = ArcanoDB.getEtiquetasStock();
   250→   250→    const etq = etqList.find(function(e) { return e.nombre === esp.nombre; });
   251→   251→    var etqDisp = 0, okEtq = false;
   252→   252→    if (etq) { etqDisp = Number(talla === 'grande' ? etq.stockGrande : etq.stockChico) || 0; okEtq = etqDisp >= cant; }
   253→   253→    const ok = okBolsa && okEtq && cant > 0;
   254→   254→    var html = '<p class="text-sm fw7 ' + (okBolsa ? 'text-green' : 'text-red') + '">' +
   255→   255→      (total > disponible ? 'FALTAN ' + (total - disponible) + 'grs en bolsa' : total > 0 ? 'OK: ' + total + 'grs de bolsa' : 'Ingresa cantidades') + '</p>';
   256→   256→    html += '<p class="text-sm fw7 ' + (okEtq ? 'text-green' : 'text-red') + '">' +
   257→   257→      (etq ? (okEtq ? 'OK: ' + cant + ' etq. ' + talla + ' disponible' : 'FALTAN ' + (cant - etqDisp) + ' etq. ' + talla) : 'SIN ETIQUETAS "' + esp.nombre + '"') + '</p>';
   258→   258→    if (ok) html += '<p class="text-green mt-8 fw7">Todo disponible. Se produciran ' + cant + ' frascos ' + talla + '.</p>';
   259→   259→    document.getElementById('f-pe-info').innerHTML = html;
   260→   260→  },

   451→   451→    const talla = (document.getElementById('f-prod-talla') || {}).value || 'chico';

   470→   470→    const etqList = ArcanoDB.getEtiquetasStock();

   472→   472→    if (etq) {
   473→   473→      var etqDisp = Number(talla === 'grande' ? etq.stockGrande : etq.stockChico) || 0;
   474→   474→      if (etqDisp < cant) { ok = false; html += '<p class="text-red mt-8 fw7">Faltan ' + (cant - etqDisp) + ' etq. ' + talla + ' de "' + blend.nombre + '"</p>'; }
