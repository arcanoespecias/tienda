   240→  _calcEspeciaProduccion(especiaId) {
   241→    const esp = ArcanoDB.getEspecia(especiaId);
   242→    if (!esp) return;
   243→    const talla = (document.getElementById('f-pe-talla') || {}).value || 'chico';
   244→    const grs = Number(document.getElementById('f-pe-grs').value) || 0;
   245→    const cant = Number(document.getElementById('f-pe-cant').value) || 0;
   246→    const total = grs * cant;
   247→    const disponible = esp.stockBolsa || 0;
   248→    const okBolsa = total > 0 && total <= disponible;
   249→    const etqList = ArcanoDB.getEtiquetasStock();
   250→    const etq = etqList.find(function(e) { return e.nombre === esp.nombre; });
   251→    var etqDisp = 0, okEtq = false;
   252→    if (etq) { etqDisp = Number(talla === 'grande' ? etq.stockGrande : etq.stockChico) || 0; okEtq = etqDisp >= cant; }
   253→    const ok = okBolsa && okEtq && cant > 0;
   254→    var html = '<p class="text-sm fw7 ' + (okBolsa ? 'text-green' : 'text-red') + '">' +
   255→      (total > disponible ? 'FALTAN ' + (total - disponible) + 'grs en bolsa' : total > 0 ? 'OK: ' + total + 'grs de bolsa' : 'Ingresa cantidades') + '</p>';
   256→    html += '<p class="text-sm fw7 ' + (okEtq ? 'text-green' : 'text-red') + '">' +
   257→      (etq ? (okEtq ? 'OK: ' + cant + ' etq. ' + talla + ' disponible' : 'FALTAN ' + (cant - etqDisp) + ' etq. ' + talla) : 'SIN ETIQUETAS "' + esp.nombre + '"') + '</p>';
   258→    if (ok) html += '<p class="text-green mt-8 fw7">Todo disponible. Se produciran ' + cant + ' frascos ' + talla + '.</p>';
   259→    document.getElementById('f-pe-info').innerHTML = html;
   260→  },

   448→  _calcBlendProduccion(blendId) {
   449→    const blend = ArcanoDB.getBlend(blendId);
   450→    if (!blend) return;
   451→    const talla = (document.getElementById('f-prod-talla') || {}).value || 'chico';
   452→    const cant = Number(document.getElementById('f-prod-cant').value) || 0;
   453→    const calcDiv = document.getElementById('f-prod-calc');
   454→    if (cant <= 0) { calcDiv.innerHTML = ''; return; }
   455→    const ings = blend.ingredientes || [];
   456→    let html = '<div class="card"><div class="card-body"><table class="table"><thead><tr><th>Especia</th><th>Grs/frasco</th><th>Total grs</th><th>Bolsa</th><th>Estado</th></tr></thead><tbody>';
   457→    let ok = true;
   458→    for (const ing of ings) {
   459→      const esp = ArcanoDB.getEspecia(ing.especiaId);
   460→      if (!esp) continue;
   461→      const total = (ing.cantidad || 0) * cant;
   462→      const disponible = esp.stockBolsa || 0;
   463→      const itemOk = disponible >= total;
   464→      if (!itemOk) ok = false;
   465→      html += '<tr><td class="fw7">' + esp.nombre + '</td><td>' + (ing.cantidad || 0) + 'grs</td><td class="fw7">' + total + 'grs</td><td>' + disponible + 'grs</td><td>' +
   466→        (itemOk ? '<span class="badge badge-green">OK</span>' : '<span class="badge badge-red">Faltan ' + (total - disponible) + 'grs</span>') + '</td></tr>';
   467→    }
   468→    html += '</tbody></table></div></div>';
   469→    // Etiqueta check por talla
   470→    const etqList = ArcanoDB.getEtiquetasStock();
   471→    const etq = etqList.find(function(e) { return e.nombre === blend.nombre; });
   472→    if (etq) {
   473→      var etqDisp = Number(talla === 'grande' ? etq.stockGrande : etq.stockChico) || 0;
   474→      if (etqDisp < cant) { ok = false; html += '<p class="text-red mt-8 fw7">Faltan ' + (cant - etqDisp) + ' etq. ' + talla + ' de "' + blend.nombre + '"</p>'; }
   475→      else { html += '<p class="text-green mt-4">OK: ' + cant + ' etq. ' + talla + ' disponible</p>'; }
   476→    } else { ok = false; html += '<p class="text-red mt-8 fw7">No hay etiquetas "' + blend.nombre + '". Compralas en Compras.</p>'; }
   477→    if (ok) html += '<p class="text-green mt-8 fw7">Todo disponible. Se produciran ' + cant + ' frascos ' + talla + '.</p>';
   478→    calcDiv.innerHTML = html;
   479→  },

   762→  _calcVentaTotal() {
   763→    let total = 0;
   764→    document.querySelectorAll('#f-venta-items .venta-item-row').forEach(row => {

   769→      var frascos = ArcanoDB.getFrascosParaVender();

   777→  saveVenta() {
   778→    const fecha = document.getElementById('f-venta-fecha').value;
   779→    const vendedorId = document.getElementById('f-venta-vendedor').value;
   780→    const rows = document.querySelectorAll('#f-venta-items .venta-item-row');
   781→    const items = [];
   782→    let total = 0;
   783→    rows.forEach(row => {
   784→      const tipo = row.querySelector('[data-field="tipo"]').value;
   785→      const prodId = row.querySelector('[data-field="productoId"]').value;
   786→      const tamano = row.querySelector('[data-field="tamano"]').value;
   787→      const cant = Number(row.querySelector('[data-field="cantidad"]').value) || 0;
   788→      if (prodId && cant > 0) {
   789→        const frascos = ArcanoDB.getFrascosParaVender();
   790→        const prod = frascos.find(f => f.id === prodId);
   791→        const precio = prod ? (tamano === 'grande' ? prod.precioGrande : prod.precioChico) : 0;
   792→        items.push({ tipo, productoId: prodId, tamano, cantidad: cant, precioUnitario: precio });
   793→        total += precio * cant;
   794→      }
   795→    });
   796→    if (items.length === 0) { alert('Agrega al menos un item'); return; }
   797→    try {
   798→      ArcanoDB.saveVenta({ fecha, vendedorId, items, total });
   799→      document.querySelector('.modal-overlay').remove();
   800→      App.renderPage('ventas');
   801→    } catch (e) { alert(e.message); }
