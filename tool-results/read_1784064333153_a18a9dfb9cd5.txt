   313→  1711→  1711→function executeImport() {
   314→  1712→  1712→  var rows = window._importRows;
   315→  1713→  1713→  var colMap = window._importColMap;
   316→  1714→  1714→  var tipo = document.getElementById('import-tipo').value;
   317→  1715→  1715→  var dupeAction = document.getElementById('import-dupes').value;
   318→  1716→  1716→  var alertEl = document.getElementById('import-alert');
   319→  1717→  1717→
   320→  1718→  1718→  if (!rows || rows.length === 0) return;
   321→  1719→  1719→
   322→  1720→  1720→  var db = getDB();
   323→  1721→  1721→  var now = new Date().toISOString();
   324→  1722→  1722→  var imported = 0, updated = 0, skipped = 0;
   325→  1723→  1723→
   326→  1724→  1724→  // If overwrite, clear existing products of this type
   327→  1725→  1725→  if (dupeAction === 'overwrite') {
   328→  1726→  1726→    var before = db.productos.length;
   329→  1727→  1727→    db.productos = db.productos.filter(function(p) { return p.tipo !== tipo; });
   330→  1728→  1728→    toast('Eliminados ' + (before - db.productos.length) + ' productos de tipo ' + tipo);
   331→  1729→  1729→  }
   332→  1730→  1730→
   333→  1731→  1731→  rows.forEach(function(r) {
   334→  1732→  1732→    var nombre = String(r[colMap.nombre] || '').trim();
   335→  1733→  1733→    if (!nombre) return;
   336→  1734→  1734→
   337→  1735→  1735→    var existing = db.productos.find(function(p) {
   338→  1736→  1736→      return p.nombre.toLowerCase() === nombre.toLowerCase() && p.tipo === tipo;
   339→  1737→  1737→    });
   340→  1738→  1738→
   341→  1739→  1739→    if (existing) {
   342→  1740→  1740→      if (dupeAction === 'skip') { skipped++; return; }
   343→  1741→  1741→      // Update existing
   344→  1742→  1742→      if (colMap.categoria) existing.categoria = String(r[colMap.categoria] || '').trim();
   345→  1743→  1743→      if (colMap.origen) existing.origen = String(r[colMap.origen] || '').trim();
   346→  1744→  1744→      if (colMap.uso) existing.usoPrincipal = String(r[colMap.uso] || '').trim();
   347→  1745→  1745→      if (colMap.perfil) existing.perfil = String(r[colMap.perfil] || '').trim();
   348→  1746→  1746→      if (colMap.precioCosto) existing.precioCosto = parseFloat(r[colMap.precioCosto]) || existing.precioCosto;
   349→  1747→  1747→      if (colMap.precioVenta) existing.precioVenta = parseFloat(r[colMap.precioVenta]) || existing.precioVenta;
   350→  1748→  1748→      if (colMap.stock) existing.stock = parseFloat(r[colMap.stock]) || existing.stock;
   351→  1749→  1749→      if (colMap.unidad) existing.unidad = String(r[colMap.unidad] || '').trim();
   352→  1750→  1750→      existing.actualizadoEn = now;
   353→  1751→  1751→      updated++;
   354→  1752→  1752→    } else {
   355→  1753→  1753→      var notas = [];
   356→  1754→  1754→      if (colMap.uso) notas.push('Uso: ' + String(r[colMap.uso] || '').trim());
   357→  1755→  1755→      if (colMap.perfil) notas.push('Perfil: ' + String(r[colMap.perfil] || '').trim());
   358→  1756→  1756→
   359→  1757→  1757→      var prod = {
   360→  1758→  1758→        id: nextId(),
   361→  1759→  1759→        nombre: nombre,
   362→  1760→  1760→        tipo: tipo,
   363→  1761→  1761→        unidad: colMap.unidad ? String(r[colMap.unidad] || 'gr').trim() : 'gr',
   364→  1762→  1762→        precioCosto: colMap.precioCosto ? (parseFloat(r[colMap.precioCosto]) || 0) : 0,
   365→  1763→  1763→        precioVenta: colMap.precioVenta ? (parseFloat(r[colMap.precioVenta]) || 0) : 0,
   366→  1764→  1764→        stock: colMap.stock ? (parseFloat(r[colMap.stock]) || 0) : 0,
   367→  1765→  1765→        stockMin: 0,
   368→  1766→  1766→        proveedor: '',
   369→  1767→  1767→        notas: notas.join(' | '),
   370→  1768→  1768→        categoria: colMap.categoria ? String(r[colMap.categoria] || '').trim() : '',
   371→  1769→  1769→        origen: colMap.origen ? String(r[colMap.origen] || '').trim() : '',
   372→  1770→  1770→        usoPrincipal: colMap.uso ? String(r[colMap.uso] || '').trim() : '',
   373→  1771→  1771→        perfil: colMap.perfil ? String(r[colMap.perfil] || '').trim() : '',
   374→  1772→  1772→        creadoEn: now,
   375→  1773→  1773→        actualizadoEn: now
   376→  1774→  1774→      };
   377→  1775→  1775→      db.productos.push(prod);
   378→  1776→  1776→      imported++;
   379→  1777→  1777→    }
   380→  1778→  1778→  });
   381→  1779→  1779→
   382→  1780→  1780→  saveDB();
   383→  1781→  1781→  closeModal();
   384→  1782→  1782→  renderAjustes();
   385→  1783→  1783→  toast('Importados: ' + imported + ' | Actualizados: ' + updated + ' | Ignorados: ' + skipped);
   386→  1784→  1784→}
