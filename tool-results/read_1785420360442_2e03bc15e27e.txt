  1328→  handleOCRResult(text) {
  1329→    var statusEl = document.getElementById('cam-status');
  1330→    var confirmArea = document.getElementById('cam-confirm-area');
  1331→    var detectedTextEl = document.getElementById('cam-detected-text');
  1332→    var prodSelect = document.getElementById('cam-prod-select');
  1333→    if (!text || text.length < 2) {
  1334→      if (statusEl) statusEl.innerHTML = '<span style="color:var(--red)">No se detecto texto. Intenta de nuevo.</span>';
  1335→      setTimeout(function() { if (statusEl) statusEl.textContent = 'Apunta la camara a la etiqueta del producto'; }, 2000);
  1336→      return;
  1337→    }
  1338→    var especias = ArcanoDB.getEspecias();
  1339→    var blends = ArcanoDB.getBlends();
  1340→    var allProducts = [];
  1341→    for (var i = 0; i < especias.length; i++) { allProducts.push({ tipo: 'especia', producto: especias[i] }); }
  1342→    for (var i = 0; i < blends.length; i++) { allProducts.push({ tipo: 'blend', producto: blends[i] }); }
  1343→    var ocrLower = text.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  1344→    var scored = [];
  1345→    for (var i = 0; i < allProducts.length; i++) {
  1346→      var p = allProducts[i];
  1347→      var name = (p.producto.nombre || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  1348→      var nameWords = name.split(/\s+/);
  1349→      var matchCount = 0;
  1350→      for (var w = 0; w < nameWords.length; w++) {
  1351→        if (nameWords[w].length < 2) continue;
  1352→        if (ocrLower.indexOf(nameWords[w]) !== -1) matchCount++;
  1353→      }
  1354→      var score = nameWords.length > 0 ? matchCount / nameWords.length : 0;
  1355→      if (ocrLower.indexOf(name) !== -1) score = Math.max(score, 1.0);
  1356→      if (name.length >= 3 && ocrLower.indexOf(name.substring(0, Math.min(name.length, 6))) !== -1) score = Math.max(score, 0.7);
  1357→      if (score >= 0.5) scored.push({ tipo: p.tipo, producto: p.producto, score: score });
  1358→    }
  1359→    scored.sort(function(a, b) { return b.score - a.score; });
  1360→    if (confirmArea) confirmArea.style.display = 'block';
  1361→    if (detectedTextEl) detectedTextEl.textContent = 'Texto leido: "' + text.substring(0, 80) + (text.length > 80 ? '...' : '') + '"';

  1366→          var s = scored[i];
  1367→          var pct = Math.round(s.score * 100);
  1368→          prodSelect.innerHTML += '<option value="' + s.tipo + '|' + s.producto.id + '">' + s.producto.nombre + ' (' + pct + '%)</option>';
  1369→        }
  1370→        if (scored[0].score >= 0.7) {
  1371→          prodSelect.value = scored[0].tipo + '|' + scored[0].producto.id;
  1372→        }
  1373→        if (statusEl) statusEl.innerHTML = '<span style="color:var(--green)">Producto detectado - confirma abajo</span>';
  1374→      } else {
  1375→        if (statusEl) statusEl.innerHTML = '<span style="color:var(--red)">No se encontro producto. Selecciona manualmente.</span>';
