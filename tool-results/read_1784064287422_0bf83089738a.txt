    26→function renderStoreProducts() {
    27→  var db = getDB();
    28→  var products = [];
    29→  var container = document.getElementById('store-products');
    30→  var empty = document.getElementById('store-empty');
    31→
    32→  // Get sellable especias (with precioVenta > 0 and stock > 0)
    33→  var especias = (db.productos || []).filter(function(p) {
    34→    return p.tipo === 'especia' && p.precioVenta > 0 && p.stock > 0;
    35→  });
    36→  especias.forEach(function(p) {
    37→    // Filter by storeCategory
    38→    if (storeCategory !== 'todos' && storeCategory !== 'blends') {
    39→      if ((p.categoria || '') !== storeCategory) return;
    40→    }
    41→    products.push({
    42→      id: p.id, nombre: p.nombre, tipo: p.categoria || 'Especia',
    43→      desc: (p.usoPrincipal || p.notas || ''), precio: p.precioVenta,
    44→      stock: p.stock, unidad: p.unidad, blend: false,
    45→      origen: p.origen || '', perfil: p.perfil || ''
    46→    });
    47→  });
    48→
    49→  // Get blends with stock > 0 (only when "todos" or "blends" selected)
    50→  if (storeCategory === 'todos' || storeCategory === 'blends') {
    51→    var blends = (db.blends || []).filter(function(b) {
    52→      return b.stock > 0 && b.precioVenta > 0;
    53→    });
    54→    blends.forEach(function(b) {
    55→      // Calculate effective stock (min of blend, frascos, etiqueta)
    56→      var etiq = b.etiquetaId ? (db.productos || []).find(function(p) { return p.id === b.etiquetaId; }) : null;
    57→      var etiqStk = etiq ? etiq.stock : 0;
    58→      var frascosStk = (db.productos || []).filter(function(p) { return p.tipo === 'frasco'; });
    59→      var maxFrasco = frascosStk.length > 0 ? Math.max.apply(null, frascosStk.map(function(f) { return f.stock; })) : 0;
    60→      var stockEfectivo = Math.min(b.stock || 0, maxFrasco, etiqStk);
    61→      if (stockEfectivo <= 0) return; // Don't show if can't fulfill
    62→
    63→      products.push({
    64→        id: b.id, nombre: b.nombre + ' (' + (b.formato || 'polvo') + ')', tipo: 'Blend',
    65→        desc: b.descripcion || '', precio: b.precioVenta,
    66→        stock: stockEfectivo, unidad: 'unidad', blend: true,
    67→        origen: '', perfil: ''
    68→      });
    69→    });
    70→  }
    71→
    72→  if (products.length === 0) {
    73→    container.innerHTML = '';
    74→    empty.style.display = 'block';
    75→    return;
    76→  }
    77→  empty.style.display = 'none';
    78→
    79→  var html = '';
    80→  products.forEach(function(p) {
    81→    html += '<div class="product-card">' +
    82→      '<div class="product-card-type">' + esc(p.tipo) + (p.origen ? ' &mdash; ' + esc(p.origen) : '') + '</div>' +
    83→      '<div class="product-card-name">' + esc(p.nombre) + '</div>' +
    84→      (p.perfil ? '<div class="product-card-meta">' + esc(p.perfil) + '</div>' : '') +
    85→      (p.desc ? '<div class="product-card-desc">' + esc(p.desc) + '</div>' : '<div class="product-card-desc" style="opacity:.3">Sin descripción</div>') +
    86→      '<div class="product-card-footer">' +
    87→        '<div><div class="product-card-price">' + fmt(p.precio) + '</div>' +
    88→        '<div class="product-card-stock">' + p.stock + ' ' + esc(p.unidad) + ' disponibles</div></div>' +
    89→        '<button class="btn-add-cart" onclick="addToCart(' + (p.blend ? 'true' : 'false') + ',' + p.id + ',\'' + esc(p.nombre).replace(/'/g,"\\'") + '\',' + p.precio + ',\'' + esc(p.unidad) + '\',' + p.stock + ')">+ Agregar</button>' +
    90→      '</div></div>';
    91→  });
    92→  container.innerHTML = html;
    93→}

    95→function addToCart(isBlend, id, nombre, precio, unidad, stock) {
    96→  if (isBlend) {
    97→    mostrarSelectorFrasco(id, 'carrito', nombre, precio, stock);
    98→    return;
    99→  }
   100→  var existing = cart.find(function(c) { return c.isBlend === isBlend && c.id === id; });
   101→  if (existing) {
   102→    if (existing.qty >= stock) { toast('No hay más stock disponible', 'err'); return; }
   103→    existing.qty++;
   104→  } else {
   105→    cart.push({ isBlend: isBlend, id: id, nombre: nombre, precio: precio, unidad: unidad, qty: 1, maxStock: stock });
   106→  }
   107→  updateCartUI();
   108→  toast(nombre + ' agregado al carrito');
   109→}

   111→// Reusable frasco selector for blends
   112→function mostrarSelectorFrasco(blendId, contexto, nombre, precio, stock) {
   113→  var db = getDB();
   114→  var blend = db.blends.find(function(b) { return b.id === blendId; });
   115→  if (!blend) return;
   116→  nombre = nombre || blend.nombre;
   117→  precio = precio || blend.precioVenta;
   118→  stock = stock || blend.stock;
   119→
   120→  var frascos = (db.productos || []).filter(function(p) { return p.tipo === 'frasco' && p.stock > 0; });
   121→  if (frascos.length === 0) {
   122→    toast('No hay frascos disponibles en stock', 'err');
   123→    return;
   124→  }
   125→
   126→  var etiqueta = blend.etiquetaId ? db.productos.find(function(p) { return p.id === blend.etiquetaId; }) : null;
   127→  var etiquetaStock = etiqueta ? etiqueta.stock : 9999;
   128→
   129→  var html = '<p class="text-sm text-muted mb-12">Elegí el frasco para <strong class="text-gold">' + esc(nombre) + '</strong> (' + esc(blend.formato || 'polvo') + ')</p>';
   130→  html += '<div class="card mb-16">';
   131→  frascos.forEach(function(f) {
   132→    var disponible = Math.min(stock, f.stock, etiquetaStock);
   133→    var disabled = disponible <= 0 ? ' style="opacity:.4;pointer-events:none"' : '';
   134→    html += '<div class="mov-row" style="cursor:pointer"' + disabled + ' onclick="confirmarFrasco(' + blendId + ',' + f.id + ',\'' + contexto + '\',' + precio + ',' + stock + ')">' +
   135→      '<div><span class="fw7">' + esc(f.nombre) + '</span>' +
   136→      '<div class="text-xs text-muted">Frasco: ' + f.stock + ' | Etiqueta: ' + etiquetaStock + '</div></div>' +
   137→      '<span class="text-gold fw7">' + fmt(precio) + '</span></div>';
   138→  });
   139→  html += '</div>';
   140→  html += '<button class="btn btn-ghost btn-full" onclick="closeModal()">Cancelar</button>';
   141→  openModal('Elegir Frasco', html);
   142→}

   144→function confirmarFrasco(blendId, frascoId, contexto, precio, stock) {
   145→  var db = getDB();
   146→  var blend = db.blends.find(function(b) { return b.id === blendId; });
   147→  var frasco = db.productos.find(function(p) { return p.id === frascoId; });
   148→  var etiqueta = blend && blend.etiquetaId ? db.productos.find(function(p) { return p.id === blend.etiquetaId; }) : null;
   149→  if (!blend || !frasco) return;
   150→
   151→  var etiquetaStock = etiqueta ? etiqueta.stock : 9999;
   152→  var disponible = Math.min(blend.stock, frasco.stock, etiquetaStock);
   153→  if (disponible <= 0) { toast('Sin stock disponible', 'err'); return; }
   154→
   155→  var nombre = blend.nombre + ' (' + frasco.nombre + ')';
   156→
   157→  if (contexto === 'carrito') {
   158→    // Check if same blend+frasco already in cart
   159→    var existing = cart.find(function(c) { return c.isBlend && c.id === blendId && c.frascoId === frascoId; });
   160→    if (existing) {
   161→      if (existing.qty >= disponible) { toast('Stock máximo alcanzado', 'err'); return; }
   162→      existing.qty++;
   163→    } else {
   164→      cart.push({ isBlend: true, id: blendId, nombre: nombre, precio: precio, unidad: 'unidad', qty: 1, maxStock: disponible, frascoId: frascoId, frascoNombre: frasco.nombre, etiquetaId: blend.etiquetaId || 0 });
   165→    }
   166→    closeModal();
   167→    updateCartUI();
   168→    toast(nombre + ' agregado al carrito');
   169→  } else if (contexto === 'venta') {
   170→    agregarBlendAVenta(blendId, frascoId, nombre, precio, disponible);
   171→    closeModal();
   172→  }
   173→}

   175→function updateCartUI() {
   176→  var countEl = document.getElementById('cart-count');
   177→  var total = 0;
   178→  var count = 0;
   179→  cart.forEach(function(c) { total += c.precio * c.qty; count += c.qty; });
   180→
   181→  if (count > 0) {
   182→    countEl.style.display = 'flex';
   183→    countEl.textContent = count;
   184→  } else {
   185→    countEl.style.display = 'none';
   186→  }
   187→
   188→  var itemsEl = document.getElementById('cart-items');
   189→  var footerEl = document.getElementById('cart-footer');
   190→
   191→  if (cart.length === 0) {
   192→    itemsEl.innerHTML = '<div class="cart-empty">Tu carrito está vacío</div>';
   193→    footerEl.style.display = 'none';
   194→    return;
   195→  }
   196→
   197→  footerEl.style.display = 'block';
   198→  document.getElementById('cart-total-val').textContent = fmt(total);
   199→
   200→  var html = '';
   201→  cart.forEach(function(c, i) {
   202→    html += '<div class="cart-item">' +
   203→      '<div class="cart-item-info"><div class="cart-item-name">' + esc(c.nombre) + '</div>' +
   204→      '<div class="cart-item-price">' + fmt(c.precio) + ' / ' + esc(c.unidad) + '</div></div>' +
   205→      '<div class="cart-item-qty">' +
   206→        '<button onclick="cartQty(' + i + ',-1)">-</button>' +
   207→        '<span>' + c.qty + '</span>' +
   208→        '<button onclick="cartQty(' + i + ',1)">+</button>' +
   209→      '</div></div>';
   210→  });
   211→  itemsEl.innerHTML = html;
   212→}
   213→
   214→function cartQty(idx, delta) {
   215→  if (!cart[idx]) return;
   216→  cart[idx].qty += delta;
   217→  if (cart[idx].qty <= 0) cart.splice(idx, 1);
   218→  else if (cart[idx].qty > cart[idx].maxStock) { cart[idx].qty = cart[idx].maxStock; toast('Stock máximo alcanzado', 'err'); }
   219→  updateCartUI();
   220→}

   257→function submitOrder() {
   258→  var nombre = document.getElementById('chk-nombre').value.trim();
   259→  var tel = document.getElementById('chk-tel').value.trim();
   260→  var dir = document.getElementById('chk-dir').value.trim();
   261→  var notas = document.getElementById('chk-notas').value.trim();
   262→  var alertEl = document.getElementById('checkout-alert');
   263→
   264→  if (!nombre) {
   265→    alertEl.textContent = 'Ingresa tu nombre';
   266→    alertEl.className = 'alert alert-err show';
   267→    return;
   268→  }
   269→
   270→  var db = getDB();
   271→  var items = [];
   272→  var total = 0;
   273→
   274→  cart.forEach(function(c) {
   275→    var sub = c.precio * c.qty;
   276→    total += sub;
   277→    var item = {
   278→      tipo: c.isBlend ? 'blend' : 'especia',
   279→      id: c.id,
   280→      nombre: c.nombre,
   281→      cantidad: c.qty,
   282→      unidad: c.unidad,
   283→      precioUnitario: c.precio,
   284→      subtotal: sub
   285→    };
   286→    if (c.isBlend && c.frascoId) {
   287→      item.frascoId = c.frascoId;
   288→      item.frascoNombre = c.frascoNombre || '';
   289→      item.etiquetaId = c.etiquetaId || 0;
   290→    }
   291→    items.push(item);
   292→  });
   293→
   294→  var venta = {
   295→    id: nextId(),
   296→    fecha: new Date().toISOString(),
   297→    clienteNombre: nombre,
   298→    clienteTelefono: tel,
   299→    clienteDireccion: dir,
   300→    items: items,
   301→    total: total,
   302→    estado: 'pendiente',
   303→    tipo: 'tienda', // marks as storefront order
   304→    creadoPor: null,
   305→    creadoEn: new Date().toISOString()
   306→  };
   307→
   308→  db.ventas.push(venta);
   309→
   310→  // Update stock
   311→  items.forEach(function(it) {
   312→    if (it.tipo === 'especia') {
   313→      var prod = db.productos.find(function(p) { return p.id === it.id; });
   314→      if (prod) {
   315→        prod.stock = Math.max(0, prod.stock - it.cantidad);
   316→        prod.actualizadoEn = new Date().toISOString();
   317→        addMovimiento('venta', 'ventas', venta.id, it.nombre, -it.cantidad, 'Pedido de tienda');
   318→      }
   319→    } else if (it.tipo === 'blend') {
   320→      var blend = db.blends.find(function(b) { return b.id === it.id; });
   321→      if (blend) {
   322→        blend.stock = Math.max(0, blend.stock - it.cantidad);
   323→        blend.actualizadoEn = new Date().toISOString();
   324→        addMovimiento('venta', 'ventas', venta.id, it.nombre, -it.cantidad, 'Pedido de tienda');
   325→      }
   326→      // Deduct frasco stock
   327→      if (it.frascoId) {
   328→        var frasco = db.productos.find(function(p) { return p.id === it.frascoId; });
   329→        if (frasco) {
   330→          frasco.stock = Math.max(0, frasco.stock - it.cantidad);
   331→          frasco.actualizadoEn = new Date().toISOString();
   332→          addMovimiento('venta', 'ventas', venta.id, it.frascoNombre || 'Frasco', -it.cantidad, 'Frasco para ' + it.nombre);
   333→        }
   334→      }
   335→      // Deduct etiqueta stock
   336→      if (it.etiquetaId) {
   337→        var etiqueta = db.productos.find(function(p) { return p.id === it.etiquetaId; });
   338→        if (etiqueta) {
   339→          etiqueta.stock = Math.max(0, etiqueta.stock - it.cantidad);
   340→          etiqueta.actualizadoEn = new Date().toISOString();
   341→          addMovimiento('venta', 'ventas', venta.id, etiqueta.nombre || 'Etiqueta', -it.cantidad, 'Etiqueta para ' + it.nombre);
   342→        }
   343→      }
   344→    }
   345→  });
   346→
   347→  saveDB();
   348→  cart = [];
   349→  closeModal();
   350→  renderStoreProducts();
   351→  updateCartUI();
   352→  toast('Pedido realizado con éxito');
   353→}

   358→function renderDashboard() {
   359→  if (!currentUser || currentUser.rol !== 'admin') return;

   363→  var compras = db.compras || [];
   364→  var blends = db.blends || [];
   365→  var ventas = db.ventas || [];
   366→  var movimientos = db.movimientos || [];
   367→
   368→  // Stats
   369→  var especias = productos.filter(function(p) { return p.tipo === 'especia'; });
   370→  var totalVentas = ventas.filter(function(v) { return v.estado !== 'cancelada'; }).reduce(function(s, v) { return s + (v.total || 0); }, 0);

   374→  var etiquetas = productos.filter(function(p) { return p.tipo === 'etiqueta'; });

   376→  var lowStockEspecias = especias.filter(function(p) { return p.stock <= p.stockMin; });

   394→        '<div class="flex items-center gap-8"><span style="font-size:1.2rem">\u26A0</span>' +
   395→        '<div><div class="fw7 text-gold">' + ventasPendientes + ' ventas pendientes</div>' +

   401→        '<div class="flex items-center gap-8"><span style="font-size:1.2rem">\u26A0</span>' +
   402→        '<div><div class="fw7 text-red">' + lowStock.length + ' items con stock bajo</div>' +
   403→        '<div class="text-xs text-muted">' + lowStock.slice(0, 5).map(function(p) { return p.nombre + ' (' + p.tipo + ')'; }).join(', ') + (lowStock.length > 5 ? '...' : '') + '</div></div>' +

   744→function guardarProducto(existingId) {
   745→  var nombre = document.getElementById('prod-nombre').value.trim();
   746→  var tipo = document.getElementById('prod-tipo').value;
   747→  var unidad = document.getElementById('prod-unidad').value;
   748→  var costo = parseFloat(document.getElementById('prod-costo').value) || 0;
   749→  var venta = parseFloat(document.getElementById('prod-venta').value) || 0;
   750→  var stock = parseFloat(document.getElementById('prod-stock').value) || 0;
   751→  var stockMin = parseFloat(document.getElementById('prod-stockmin').value) || 0;
   752→  var proveedor = document.getElementById('prod-proveedor').value.trim();
   753→  var notas = document.getElementById('prod-notas').value.trim();
   754→  var alertEl = document.getElementById('prod-alert');
   755→
   756→  if (!nombre) {
   757→    alertEl.textContent = 'El nombre es obligatorio';
   758→    alertEl.className = 'alert alert-err show';
   759→    return;
   760→  }
   761→
   762→  var db = getDB();
   763→  var now = new Date().toISOString();
   764→
   765→  if (existingId) {
   766→    var prod = db.productos.find(function(p) { return p.id === existingId; });
   767→    if (prod) {
   768→      var oldStock = prod.stock;
   769→      prod.nombre = nombre; prod.tipo = tipo; prod.unidad = unidad;
   770→      prod.precioCosto = costo; prod.precioVenta = venta;
   771→      prod.stock = stock; prod.stockMin = stockMin;
   772→      prod.proveedor = proveedor; prod.notas = notas;
   773→      prod.actualizadoEn = now;
   774→      if (stock !== oldStock) {
   775→        addMovimiento('ajuste', 'productos', prod.id, nombre, stock - oldStock, 'Ajuste manual de stock');
   776→      }
   777→    }
   778→  } else {
   779→    db.productos.push({
   780→      id: nextId(), nombre: nombre, tipo: tipo, unidad: unidad,
   781→      precioCosto: costo, precioVenta: venta, stock: stock, stockMin: stockMin,
   782→      proveedor: proveedor, notas: notas,
   783→      creadoEn: now, actualizadoEn: now
   784→    });
   785→  }
   786→
   787→  saveDB();
   788→  closeModal();
   789→  renderProductos();
   790→  toast(existingId ? 'Producto actualizado' : 'Producto creado');
   791→}

   808→      // Calculate cost from recipe

   810→      b.costoUnitario = costo;
   811→      var profit = b.precioVenta - costo;
   812→      var profitClass = profit >= 0 ? 'profit' : 'loss';

   819→      var maxFrascoStock = frascosDisp.length > 0 ? Math.max.apply(null, frascosDisp.map(function(f) { return f.stock; })) : 0;
   820→      var stockEfectivo = Math.min(b.stock || 0, maxFrascoStock, etiquetaStk);
   821→
   822→      html += '<div class="blend-card"><div class="blend-header" onclick="this.nextElementSibling.classList.toggle(\'open\')">' +

   878→function calcBlendCost(blend) {
   879→  if (!blend || !blend.receta) return 0;
   880→  var db = getDB();
   881→  var costo = 0;
   882→  blend.receta.forEach(function(r) {
   883→    var prod = (db.productos || []).find(function(p) { return p.id === r.productoId; });
   884→    if (prod) {
   885→      // Convert to same unit for cost calc (simplified: assumes recipe unit matches product unit)
   886→      var unitCost = prod.precioCosto / 1; // cost per base unit
   887→      // If product is in kg and recipe in gr, convert
   888→      var factor = 1;
   889→      if (prod.unidad === 'kg' && r.unidad === 'gr') factor = 0.001;
   890→      else if (prod.unidad === 'L' && r.unidad === 'ml') factor = 0.001;
   891→      costo += r.cantidad * unitCost * factor;
   892→    }
   893→  });
   894→  return Math.round(costo);
   895→}

  1067→function ejecutarProduccion(blendId) {
  1068→  var cant = parseInt(document.getElementById('bl-prod-cant').value) || 1;
  1069→  if (cant < 1) return;
  1070→
  1071→  var db = getDB();
  1072→  var b = db.blends.find(function(x) { return x.id === blendId; });
  1073→  if (!b) return;
  1074→
  1075→  // Check stock for multiplied recipe
  1076→  (b.receta || []).forEach(function(r) {
  1077→    var needed = r.cantidad * cant;
  1078→    var prod = db.productos.find(function(p) { return p.id === r.productoId; });
  1079→    if (!prod || prod.stock < needed) {
  1080→      toast('Stock insuficiente de ' + r.nombre, 'err');
  1081→      return;
  1082→    }
  1083→  });
  1084→
  1085→  // Deduct stock from ingredients
  1086→  (b.receta || []).forEach(function(r) {
  1087→    var needed = r.cantidad * cant;
  1088→    var prod = db.productos.find(function(p) { return p.id === r.productoId; });
  1089→    if (prod) {
  1090→      prod.stock -= needed;
  1091→      prod.actualizadoEn = new Date().toISOString();
  1092→      addMovimiento('blend', 'blends', b.id, r.nombre, -needed, 'Producción: ' + b.nombre + ' x' + cant);
  1093→    }
  1094→  });
  1095→
  1096→  b.stock += cant;
  1097→  b.actualizadoEn = new Date().toISOString();
  1098→  saveDB();
  1099→  closeModal();
  1100→  renderBlends();
  1101→  toast(cant + ' unidad(es) de ' + b.nombre + ' producida(s)');
  1102→}

  1611→function handleExcelImport(input) {
  1612→  var file = input.files[0];
  1613→  if (!file) return;
  1614→
  1615→  var reader = new FileReader();
  1616→  reader.onload = function(e) {
  1617→    try {
  1618→      var data = new Uint8Array(e.target.result);
  1619→      var workbook = XLSX.read(data, { type: 'array' });
  1620→      var firstSheet = workbook.Sheets[workbook.SheetNames[0]];
  1621→      var rows = XLSX.utils.sheet_to_json(firstSheet, { defval: '' });
  1622→
  1623→      if (rows.length === 0) {
  1624→        toast('El archivo está vacío', 'err');
  1625→        return;
  1626→      }
  1627→
  1628→      // Detect column mapping
  1629→      var headers = Object.keys(rows[0]);
  1630→      var colMap = detectColumnMap(headers);
  1631→
  1632→      if (!colMap.nombre) {
  1633→        toast('No se encontró columna de nombre (Blend/Nombre/Producto)', 'err');
  1634→        return;
  1635→      }
  1636→
  1637→      // Show preview before importing
  1638→      showImportPreview(rows, colMap);
  1639→
  1640→    } catch(err) {
  1641→      toast('Error al leer el archivo: ' + err.message, 'err');
  1642→      console.error('[Import]', err);
  1643→    }
  1644→  };
  1645→  reader.readAsArrayBuffer(file);
  1646→  input.value = ''; // Reset so same file can be re-selected
  1647→}

  1649→function detectColumnMap(headers) {
  1650→  var map = { nombre: null, categoria: null, origen: null, uso: null, perfil: null, precioCosto: null, precioVenta: null, stock: null, unidad: null };
  1651→
  1652→  headers.forEach(function(h) {
  1653→    var hl = h.toLowerCase().trim();
  1654→    if (['blend', 'nombre', 'producto', 'name', 'especia', 'item'].indexOf(hl) !== -1) map.nombre = h;
  1655→    else if (['categoría', 'categoria', 'category', 'tipo', 'type'].indexOf(hl) !== -1) map.categoria = h;
  1656→    else if (['origen', 'origin', 'país', 'pais', 'procedencia'].indexOf(hl) !== -1) map.origen = h;
  1657→    else if (['uso principal', 'uso', 'use', 'aplicación', 'aplicacion'].indexOf(hl) !== -1) map.uso = h;
  1658→    else if (['perfil', 'profile', 'sabor', 'flavor'].indexOf(hl) !== -1) map.perfil = h;
  1659→    else if (['precio costo', 'costo', 'precio_compra', 'cost', 'precio de costo'].indexOf(hl) !== -1) map.precioCosto = h;
  1660→    else if (['precio venta', 'venta', 'precio', 'price', 'precio de venta'].indexOf(hl) !== -1) map.precioVenta = h;
  1661→    else if (['stock', 'cantidad', 'inventario', 'inventory', 'existencia'].indexOf(hl) !== -1) map.stock = h;
  1662→    else if (['unidad', 'unit', 'medida', 'um'].indexOf(hl) !== -1) map.unidad = h;
  1663→  });
  1664→
  1665→  return map;
  1666→}

  1711→function executeImport() {
  1712→  var rows = window._importRows;
  1713→  var colMap = window._importColMap;
  1714→  var tipo = document.getElementById('import-tipo').value;
  1715→  var dupeAction = document.getElementById('import-dupes').value;
  1716→  var alertEl = document.getElementById('import-alert');
  1717→
  1718→  if (!rows || rows.length === 0) return;
  1719→
  1720→  var db = getDB();
  1721→  var now = new Date().toISOString();
  1722→  var imported = 0, updated = 0, skipped = 0;
  1723→
  1724→  // If overwrite, clear existing products of this type
  1725→  if (dupeAction === 'overwrite') {
  1726→    var before = db.productos.length;
  1727→    db.productos = db.productos.filter(function(p) { return p.tipo !== tipo; });
  1728→    toast('Eliminados ' + (before - db.productos.length) + ' productos de tipo ' + tipo);
  1729→  }
  1730→
  1731→  rows.forEach(function(r) {
  1732→    var nombre = String(r[colMap.nombre] || '').trim();
  1733→    if (!nombre) return;
  1734→
  1735→    var existing = db.productos.find(function(p) {
  1736→      return p.nombre.toLowerCase() === nombre.toLowerCase() && p.tipo === tipo;
  1737→    });
  1738→
  1739→    if (existing) {
  1740→      if (dupeAction === 'skip') { skipped++; return; }
  1741→      // Update existing
  1742→      if (colMap.categoria) existing.categoria = String(r[colMap.categoria] || '').trim();
  1743→      if (colMap.origen) existing.origen = String(r[colMap.origen] || '').trim();
  1744→      if (colMap.uso) existing.usoPrincipal = String(r[colMap.uso] || '').trim();
  1745→      if (colMap.perfil) existing.perfil = String(r[colMap.perfil] || '').trim();
  1746→      if (colMap.precioCosto) existing.precioCosto = parseFloat(r[colMap.precioCosto]) || existing.precioCosto;
  1747→      if (colMap.precioVenta) existing.precioVenta = parseFloat(r[colMap.precioVenta]) || existing.precioVenta;
  1748→      if (colMap.stock) existing.stock = parseFloat(r[colMap.stock]) || existing.stock;
  1749→      if (colMap.unidad) existing.unidad = String(r[colMap.unidad] || '').trim();
  1750→      existing.actualizadoEn = now;
  1751→      updated++;
  1752→    } else {
  1753→      var notas = [];
  1754→      if (colMap.uso) notas.push('Uso: ' + String(r[colMap.uso] || '').trim());
  1755→      if (colMap.perfil) notas.push('Perfil: ' + String(r[colMap.perfil] || '').trim());
  1756→
  1757→      var prod = {
  1758→        id: nextId(),
  1759→        nombre: nombre,
  1760→        tipo: tipo,
  1761→        unidad: colMap.unidad ? String(r[colMap.unidad] || 'gr').trim() : 'gr',
  1762→        precioCosto: colMap.precioCosto ? (parseFloat(r[colMap.precioCosto]) || 0) : 0,
  1763→        precioVenta: colMap.precioVenta ? (parseFloat(r[colMap.precioVenta]) || 0) : 0,
  1764→        stock: colMap.stock ? (parseFloat(r[colMap.stock]) || 0) : 0,
  1765→        stockMin: 0,
  1766→        proveedor: '',
  1767→        notas: notas.join(' | '),
  1768→        categoria: colMap.categoria ? String(r[colMap.categoria] || '').trim() : '',
  1769→        origen: colMap.origen ? String(r[colMap.origen] || '').trim() : '',
  1770→        usoPrincipal: colMap.uso ? String(r[colMap.uso] || '').trim() : '',
  1771→        perfil: colMap.perfil ? String(r[colMap.perfil] || '').trim() : '',
  1772→        creadoEn: now,
  1773→        actualizadoEn: now
  1774→      };
  1775→      db.productos.push(prod);
  1776→      imported++;
  1777→    }
  1778→  });
  1779→
  1780→  saveDB();
  1781→  closeModal();
  1782→  renderAjustes();
  1783→  toast('Importados: ' + imported + ' | Actualizados: ' + updated + ' | Ignorados: ' + skipped);
  1784→}
