  1037→  handleOCRResult(text) {
  1038→    var statusEl = document.getElementById('cam-status');
  1039→    var confirmArea = document.getElementById('cam-confirm-area');
  1040→    var detectedTextEl = document.getElementById('cam-detected-text');
  1041→    var prodSelect = document.getElementById('cam-prod-select');
  1042→    if (!text || text.length < 2) {
  1043→      if (statusEl) statusEl.innerHTML = '<span style="color:var(--red)">No se detecto texto. Intenta de nuevo.</span>';
  1044→      setTimeout(function() { if (statusEl) statusEl.textContent = 'Apunta la camara a la etiqueta del producto'; }, 2000);
  1045→      return;
  1046→    }
  1047→    var especias = ArcanoDB.getEspecias();
  1048→    var blends = ArcanoDB.getBlends();
  1049→    var allProducts = [];
  1050→    for (var i = 0; i < especias.length; i++) { allProducts.push({ tipo: 'especia', producto: especias[i] }); }
  1051→    for (var i = 0; i < blends.length; i++) { allProducts.push({ tipo: 'blend', producto: blends[i] }); }
  1052→    var ocrLower = text.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  1053→    var scored = [];
  1054→    for (var i = 0; i < allProducts.length; i++) {
  1055→      var p = allProducts[i];
  1056→      var name = (p.producto.nombre || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  1057→      var nameWords = name.split(/\s+/);
  1058→      var matchCount = 0;
  1059→      for (var w = 0; w < nameWords.length; w++) {
  1060→        if (nameWords[w].length < 2) continue;
  1061→        if (ocrLower.indexOf(nameWords[w]) !== -1) matchCount++;
  1062→      }
  1063→      var score = nameWords.length > 0 ? matchCount / nameWords.length : 0;
  1064→      if (ocrLower.indexOf(name) !== -1) score = Math.max(score, 1.0);
  1065→      if (name.length >= 3 && ocrLower.indexOf(name.substring(0, Math.min(name.length, 6))) !== -1) score = Math.max(score, 0.7);
  1066→      if (score >= 0.5) scored.push({ tipo: p.tipo, producto: p.producto, score: score });
  1067→    }
  1068→    scored.sort(function(a, b) { return b.score - a.score; });
  1069→    if (confirmArea) confirmArea.style.display = 'block';
  1070→    if (detectedTextEl) detectedTextEl.textContent = 'Texto leido: "' + text.substring(0, 80) + (text.length > 80 ? '...' : '') + '"';
  1071→    if (prodSelect) {
  1072→      prodSelect.innerHTML = '<option value="">Seleccionar producto</option>';
  1073→      if (scored.length > 0) {
  1074→        for (var i = 0; i < Math.min(scored.length, 5); i++) {
  1075→          var s = scored[i];
  1076→          var pct = Math.round(s.score * 100);
  1077→          prodSelect.innerHTML += '<option value="' + s.tipo + '|' + s.producto.id + '">' + s.producto.nombre + ' (' + pct + '%)</option>';
  1078→        }
  1079→        if (scored[0].score >= 0.7) {
  1080→          prodSelect.value = scored[0].tipo + '|' + scored[0].producto.id;
  1081→        }
  1082→        if (statusEl) statusEl.innerHTML = '<span style="color:var(--green)">Producto detectado - confirma abajo</span>';
  1083→      } else {
  1084→        if (statusEl) statusEl.innerHTML = '<span style="color:var(--red)">No se encontro producto. Selecciona manualmente.</span>';
  1085→        for (var i = 0; i < allProducts.length; i++) {
  1086→          var ap = allProducts[i];
  1087→          prodSelect.innerHTML += '<option value="' + ap.tipo + '|' + ap.producto.id + '">' + ap.producto.nombre + '</option>';
  1088→        }
  1089→      }
  1090→    }
  1091→  },
