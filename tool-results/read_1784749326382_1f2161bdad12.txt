   148→function initDB() {
   149→  return new Promise(function(resolve) {
   150→    _initFirebase();
   151→
   152→    // Try localStorage cache
   153→    var cached = null;
   154→    try { cached = JSON.parse(localStorage.getItem(DB_KEY)); } catch (e) {}
   155→
   156→    if (cached && cached.meta && cached.meta.version === DB_VERSION && _ensureStructureOn(cached)) {
   157→      _db = cached;
   158→      _ready = true;
   159→      _startFirebaseListener();
   160→      resolve();
   161→      return;
   162→    }
   163→
   164→    // Load from Firebase
   165→    if (_firebaseRef) {
   166→      _firebaseRef.once('value').then(function(snap) {
   167→        var fbData = snap.val();
   168→        if (fbData && fbData.meta && fbData.meta.version === DB_VERSION && _ensureStructureOn(fbData)) {
   169→          _db = fbData;
   170→        } else {
   171→          // Fresh DB (old version or empty)
   172→          _db = _emptyDB();
   173→          _ensureStructure();
   174→          _saveToFirebase();
   175→        }
   176→        _ready = true;
   177→        _cacheLocal();
   178→        _startFirebaseListener();
   179→        resolve();
   180→      }).catch(function() {
   181→        _db = _emptyDB();
   182→        _ensureStructure();
   183→        _ready = true;
   184→        resolve();
   185→      });
   186→    } else {
   187→      _db = _emptyDB();
   188→      _ensureStructure();
   189→      _ready = true;
   190→      resolve();
   191→    }
   192→  });
   193→}

   431→function producirEspecia(especiaId, talla, cantidad) {
   432→  _ensureStructure();
   433→  var esp = _db.especias[especiaId];
   434→  if (!esp) throw new Error('Especia no encontrada');
   435→  talla = (talla === 'grande') ? 'grande' : 'chico';
   436→  cantidad = Number(cantidad) || 0;
   437→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   438→
   439→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   440→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   441→
   442→  var grsTotal = gramosPorFrasco * cantidad;
   443→
   444→  // Check & consume bolsa
   445→  if ((esp.stockBolsa || 0) < grsTotal) {
   446→    throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   447→  }
   448→
   449→  // Check & consume envases
   450→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   451→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   452→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   453→  }
   454→
   455→  // Check & consume stickers
   456→  var stk = _findStickerByNombre(esp.nombre);
   457→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   458→  if (stkStock < cantidad) {
   459→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   460→  }
   461→
   462→  // All checks passed — consume
   463→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   464→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   465→  if (stk) {
   466→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   467→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   468→  }
   469→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   470→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   471→
   472→  // Record
   473→  var prodId = nextId('producciones');
   474→  var prod = {
   475→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   476→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   477→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   478→    envasesConsumidos: cantidad, stickersConsumidos: cantidad,
   479→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   480→  };
   481→  _db.producciones[prodId] = prod;
   482→  _saveToFirebase(); _cacheLocal();
   483→  _notify('create', 'producciones', prodId);
   484→  _notify('update', 'especias', especiaId);
   485→  return { producto: esp, produccion: prod };
   486→}

   491→  if (!blend) throw new Error('Blend no encontrado');
   492→  talla = (talla === 'grande') ? 'grande' : 'chico';
   493→  cantidad = Number(cantidad) || 0;
   494→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   495→
   496→  var ingredientes = blend.ingredientes || [];
   497→  if (ingredientes.length === 0) throw new Error('El blend no tiene ingredientes definidos. Editalo primero.');
   498→
   499→  // Check ingredient stock
   500→  var detalleIngredientes = [];
   501→  for (var i = 0; i < ingredientes.length; i++) {
   502→    var ing = ingredientes[i];
   503→    var esp = _db.especias[ing.especiaId];
   504→    if (!esp) throw new Error('Especia "' + (ing.especiaNombre || ing.especiaId) + '" no encontrada');
   505→    var grsPorFrasco = (talla === 'grande') ? (Number(ing.gramosGrande) || 0) : (Number(ing.gramosChico) || 0);
   506→    if (grsPorFrasco <= 0) throw new Error('El ingrediente "' + esp.nombre + '" no tiene gramos para frasco ' + talla);
   507→    var grsNeeded = grsPorFrasco * cantidad;
   508→    if ((esp.stockBolsa || 0) < grsNeeded) {
   509→      throw new Error('Bolsa insuficiente de "' + esp.nombre + '". Necesitas ' + grsNeeded + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   510→    }
   511→    detalleIngredientes.push({ especiaId: ing.especiaId, especiaNombre: esp.nombre, gramosPorFrasco: grsPorFrasco, gramosTotal: grsNeeded });
   512→  }
   513→
   514→  // Check envases
   515→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   516→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   517→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   518→  }
   519→
   520→  // Check stickers
   521→  var stk = _findStickerByNombre(blend.nombre);
   522→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   523→  if (stkStock < cantidad) {
   524→    throw new Error('Stickers ' + talla + ' insuficientes para "' + blend.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   525→  }
   526→
   527→  // All checks passed — consume
   528→  var grsTotalGeneral = 0;
   529→  for (var i = 0; i < detalleIngredientes.length; i++) {
   530→    var d = detalleIngredientes[i];
   531→    var esp = _db.especias[d.especiaId];
   532→    esp.stockBolsa = (esp.stockBolsa || 0) - d.gramosTotal;
   533→    grsTotalGeneral += d.gramosTotal;
   534→  }
   535→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   536→  if (stk) {
   537→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   538→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   539→  }
   540→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   541→  blend[frascoKey] = (blend[frascoKey] || 0) + cantidad;
   542→
   543→  var prodId = nextId('producciones');
   544→  var prod = {
   545→    id: prodId, tipo: 'blend', productoId: blendId, productoNombre: blend.nombre,
   546→    categoria: blend.categoria || '', talla: talla, cantidad: cantidad,
   547→    ingredientes: detalleIngredientes, gramosTotal: grsTotalGeneral,
   548→    envasesConsumidos: cantidad, stickersConsumidos: cantidad,
   549→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   550→  };
   551→  _db.producciones[prodId] = prod;
   552→  _saveToFirebase(); _cacheLocal();
   553→  _notify('create', 'producciones', prodId);
   554→  _notify('update', 'blends', blendId);
   555→  return { producto: blend, produccion: prod };
   556→}

   568→function saveVenta(data) {
   569→  _ensureStructure();
   570→  var isNew = !data.id;
   571→  if (isNew) {
   572→    data.id = nextId('ventas');
   573→    data.creado = new Date().toISOString();
   574→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   575→    data.items = data.items || [];
   576→    data.total = Number(data.total) || 0;
   577→  }
   578→  if (isNew) {
   579→    for (var i = 0; i < data.items.length; i++) {
   580→      var item = data.items[i];
   581→      var producto;
   582→      if (item.tipo === 'especia') {
   583→        producto = _db.especias[item.productoId];
   584→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   585→      } else {
   586→        producto = _db.blends[item.productoId];
   587→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   588→      }
   589→      item.productoNombre = producto.nombre;
   590→      var cant = Number(item.cantidad) || 0;
   591→      var talla = item.talla || 'chico';
   592→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   593→      if ((producto[stockKey] || 0) < cant) {
   594→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   595→      }
   596→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   597→      item.precioUnitario = Number(item.precioUnitario) || 0;
   598→      item.subtotal = item.precioUnitario * cant;
   599→    }
   600→    // Recalculate total
   601→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   602→  }
   603→  _db.ventas[data.id] = data;
   604→  _saveToFirebase(); _cacheLocal();
   605→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   606→  return data;
   607→}

   761→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   762→function findEspeciaByName(nombre) {
   763→  if (!nombre) return null;
   764→  var target = nombre.trim().toLowerCase();
   765→  var keys = Object.keys(_db.especias || {});
   766→  // 1. Exact match
   767→  for (var i = 0; i < keys.length; i++) {
   768→    var e = _db.especias[keys[i]];
   769→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   770→  }
   771→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   772→  for (var i = 0; i < keys.length; i++) {
   773→    var e = _db.especias[keys[i]];
   774→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   775→  }
   776→  // 3. Target is contained in especia name
   777→  for (var i = 0; i < keys.length; i++) {
   778→    var e = _db.especias[keys[i]];
   779→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   780→  }
   781→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   782→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   783→  for (var w = 0; w < targetWords.length; w++) {
   784→    for (var i = 0; i < keys.length; i++) {
   785→      var e = _db.especias[keys[i]];
   786→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   787→    }
   788→  }
   789→  return null;
   790→}

   792→/** Auto-detect categoria from blend USO field */
   793→function _categoriaFromUso(uso) {
   794→  if (!uso) return 'Comidas';

   796→  // Cocteleria keywords
   797→  if (/\b(gin|ron|vodka|whisky|mojito|mule|vermouth|aperitif|coctel)\b/.test(u)) return 'Cocteleria';
   798→  // Infusiones keywords
   799→  if (/\b(relajant|sueño|digestiv|energiz|té|calidez|respiratorio|meditaci|antioxidant|bienestar|infusi)\b/.test(u)) return 'Infusiones';
   800→  return 'Comidas';

   803→/**
   804→ * Import especias and blends from parsed Excel data.
   805→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   806→ *
   807→ * especiasList: [{ nombre, categoria? }]
   808→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   809→ * gramosChico:  default grams for frasco chico (e.g. 30)
   810→ * gramosGrande: default grams for frasco grande (e.g. 80)
   811→ */
   812→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   813→  _ensureStructure();
   814→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   815→
   816→  // 1. Create especias (skip if name already exists)
   817→  for (var i = 0; i < especiasList.length; i++) {
   818→    var esp = especiasList[i];
   819→    var nombre = (esp.nombre || '').trim();
   820→    if (!nombre) continue;
   821→    var existing = findEspeciaByName(nombre);
   822→    if (existing) {
   823→      resultado.especiasExistentes++;
   824→      continue;
   825→    }
   826→    saveEspecia({
   827→      nombre: nombre,
   828→      categoria: esp.categoria || 'Comidas',
   829→      precioChico: 0,
   830→      precioGrande: 0,
   831→      gramosChico: Number(gramosChico) || 30,
   832→      gramosGrande: Number(gramosGrande) || 80,
   833→      stockBolsa: 0,
   834→      stockChico: 0,
   835→      stockGrande: 0
   836→    });
   837→    resultado.especiasCreadas++;
   838→  }
   839→
   840→  // 2. Create blends (skip if name already exists)
   841→  for (var j = 0; j < blendsList.length; j++) {
   842→    var bl = blendsList[j];
   843→    var nombre = (bl.nombre || '').trim();
   844→    if (!nombre) continue;
   845→    var existingBlend = null;
   846→    var blKeys = Object.keys(_db.blends || {});
   847→    for (var k = 0; k < blKeys.length; k++) {
   848→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
   849→        existingBlend = _db.blends[blKeys[k]];
   850→        break;
   851→      }
   852→    }
   853→    if (existingBlend) {
   854→      resultado.blendsExistentes++;
   855→      continue;
   856→    }
   857→
   858→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
   859→    var ings = bl.ingredientes || [];
   860→    var recipeTotal = 0;
   861→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
   862→    if (recipeTotal <= 0) recipeTotal = 500;
   863→
   864→    var resolvedIngs = [];
   865→    for (var ii = 0; ii < ings.length; ii++) {
   866→      var ing = ings[ii];
   867→      var espObj = findEspeciaByName(ing.especia);
   868→      if (!espObj) {
   869→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
   870→        continue;
   871→      }
   872→      var ingG = Number(ing.g) || 0;
   873→      resolvedIngs.push({
   874→        especiaId: espObj.id,
   875→        especiaNombre: espObj.nombre,
   876→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
   877→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
   878→        gramosReceta: ingG
   879→      });
   880→    }
   881→
   882→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
   883→    saveBlend({
   884→      nombre: nombre,
   885→      categoria: cat,
   886→      region: bl.region || '',
   887→      uso: bl.uso || '',
   888→      precioChico: 0,
   889→      precioGrande: 0,
   890→      ingredientes: resolvedIngs,
   891→      stockChico: 0,
   892→      stockGrande: 0
   893→    });
   894→    resultado.blendsCreados++;
   895→  }
   896→
   897→  return resultado;
   898→}

   930→  var idx = -1;

   932→    if (_db.productTags[cat][i] === tagName) { idx = i; break; }
   933→  }
   934→  if (idx < 0) return false;
   935→  _db.productTags[cat].splice(idx, 1);
   936→  // Also remove from all products that have this tag
   937→  var allEsp = getEspecias();
   938→  for (var e = 0; e < allEsp.length; e++) {
   939→    if (allEsp[e].tags) {
   940→      var ti = allEsp[e].tags.indexOf(tagName);
   941→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
   942→    }
   943→  }
   944→  var allBl = getBlends();
   945→  for (var b = 0; b < allBl.length; b++) {
   946→    if (allBl[b].tags) {
   947→      var bi = allBl[b].tags.indexOf(tagName);
   948→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
   949→    }
   950→  }
   951→  _saveToFirebase(); _cacheLocal();
   952→  return true;
   953→}

   957→function compressImage(file, maxW, quality, cb) {
   958→  var reader = new FileReader();
   959→  reader.onload = function(ev) {
   960→    var img = new Image();
   961→    img.onload = function() {
   962→      var w = img.width, h = img.height;
   963→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
   964→      var canvas = document.createElement('canvas');
   965→      canvas.width = w; canvas.height = h;
   966→      var ctx = canvas.getContext('2d');
   967→      ctx.drawImage(img, 0, 0, w, h);
   968→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
   969→      cb(null, dataUrl);
   970→    };
   971→    img.onerror = function() { cb('Error al cargar imagen'); };
   972→    img.src = ev.target.result;
   973→  };
   974→  reader.onerror = function() { cb('Error al leer archivo'); };
   975→  reader.readAsDataURL(file);
   976→}
