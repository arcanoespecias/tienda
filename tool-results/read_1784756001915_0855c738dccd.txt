   383→   383→function saveEntrada(data) {

   389→   389→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   390→   390→    data.items = data.items || [];
   391→   391→    data.total = Number(data.total) || 0;
   392→   392→  }
   393→   393→  if (isNew) {
   394→   394→    for (var i = 0; i < data.items.length; i++) {
   395→   395→      var item = data.items[i];
   396→   396→      var tipo = item.tipo;
   397→   397→      if (tipo === 'especia_grs') {
   398→   398→        // Add grams to especia stockBolsa
   399→   399→        if (item.especiaId && _db.especias[item.especiaId]) {
   400→   400→          _db.especias[item.especiaId].stockBolsa = (_db.especias[item.especiaId].stockBolsa || 0) + (Number(item.cantidad) || 0);
   401→   401→        }
   402→   402→      } else if (tipo === 'envase') {
   403→   403→        var talla = item.talla || 'chico';
   404→   404→        if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   405→   405→        _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) + (Number(item.cantidad) || 0);
   406→   406→      } else if (tipo === 'sticker') {
   407→   407→        var stk = _getOrCreateSticker(item.stickerNombre);
   408→   408→        var t = item.talla || 'chico';
   409→   409→        if (t === 'grande') {
   410→   410→          stk.stockGrande = (stk.stockGrande || 0) + (Number(item.cantidad) || 0);
   411→   411→        } else {
   412→   412→          stk.stockChico = (stk.stockChico || 0) + (Number(item.cantidad) || 0);
   413→   413→        }
   414→   414→      } else if (tipo === 'bolsa') {
   415→   415→        var tallaB = item.talla || 'chico';
   416→   416→        if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   417→   417→        _db.stockBolsas[tallaB] = (_db.stockBolsas[tallaB] || 0) + (Number(item.cantidad) || 0);
   418→   418→      }
   419→   419→    }
   420→   420→  }

   437→   437→function producirEspecia(especiaId, talla, cantidad) {
   438→   438→  _ensureStructure();
   439→   439→  var esp = _db.especias[especiaId];
   440→   440→  if (!esp) throw new Error('Especia no encontrada');
   441→   441→  talla = (talla === 'grande') ? 'grande' : 'chico';
   442→   442→  cantidad = Number(cantidad) || 0;
   443→   443→  if (cantidad <= 0) throw new Error('La cantidad debe ser mayor a 0');
   444→   444→
   445→   445→  var gramosPorFrasco = (talla === 'grande') ? (Number(esp.gramosGrande) || 0) : (Number(esp.gramosChico) || 0);
   446→   446→  if (gramosPorFrasco <= 0) throw new Error('La especia no tiene gramos definidos para frasco ' + talla + '. Editala primero.');
   447→   447→
   448→   448→  var grsTotal = gramosPorFrasco * cantidad;
   449→   449→
   450→   450→  // Check & consume pala (raw material)
   451→   451→  if ((esp.stockBolsa || 0) < grsTotal) {
   452→   452→    throw new Error('Pala insuficiente de "' + esp.nombre + '". Necesitas ' + grsTotal + 'grs, tienes ' + (esp.stockBolsa || 0) + 'grs');
   453→   453→  }
   454→   454→
   455→   455→  // Check & consume envases
   456→   456→  if (!_db.stockEnvases) _db.stockEnvases = { chico: 0, grande: 0 };
   457→   457→  if ((_db.stockEnvases[talla] || 0) < cantidad) {
   458→   458→    throw new Error('Envases ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockEnvases[talla] || 0));
   459→   459→  }
   460→   460→
   461→   461→  // Check & consume stickers
   462→   462→  var stk = _findStickerByNombre(esp.nombre);
   463→   463→  var stkStock = stk ? (Number(stk[talla === 'grande' ? 'stockGrande' : 'stockChico']) || 0) : 0;
   464→   464→  if (stkStock < cantidad) {
   465→   465→    throw new Error('Stickers ' + talla + ' insuficientes para "' + esp.nombre + '". Necesitas ' + cantidad + ', tienes ' + stkStock);
   466→   466→  }
   467→   467→
   468→   468→  // Check & consume bolsas (packaging)
   469→   469→  if (!_db.stockBolsas) _db.stockBolsas = { chico: 0, grande: 0 };
   470→   470→  if ((_db.stockBolsas[talla] || 0) < cantidad) {
   471→   471→    throw new Error('Bolsas ' + talla + ' insuficientes. Necesitas ' + cantidad + ', tienes ' + (_db.stockBolsas[talla] || 0));
   472→   472→  }
   473→   473→
   474→   474→  // All checks passed — consume
   475→   475→  esp.stockBolsa = (esp.stockBolsa || 0) - grsTotal;
   476→   476→  _db.stockEnvases[talla] = (_db.stockEnvases[talla] || 0) - cantidad;
   477→   477→  _db.stockBolsas[talla] = (_db.stockBolsas[talla] || 0) - cantidad;
   478→   478→  if (stk) {
   479→   479→    var stkKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   480→   480→    stk[stkKey] = (stk[stkKey] || 0) - cantidad;
   481→   481→  }
   482→   482→  var frascoKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   483→   483→  esp[frascoKey] = (esp[frascoKey] || 0) + cantidad;
   484→   484→
   485→   485→  // Record
   486→   486→  var prodId = nextId('producciones');
   487→   487→  var prod = {
   488→   488→    id: prodId, tipo: 'especia', productoId: especiaId, productoNombre: esp.nombre,
   489→   489→    categoria: esp.categoria || '', talla: talla, cantidad: cantidad,
   490→   490→    gramosPorFrasco: gramosPorFrasco, gramosTotal: grsTotal,
   491→   491→    envasesConsumidos: cantidad, stickersConsumidos: cantidad, bolsasConsumidas: cantidad,
   492→   492→    fecha: new Date().toISOString().slice(0, 10), creado: new Date().toISOString()
   493→   493→  };
   494→   494→  _db.producciones[prodId] = prod;
   495→   495→  _saveToFirebase(); _cacheLocal();
   496→   496→  _notify('create', 'producciones', prodId);
   497→   497→  _notify('update', 'especias', especiaId);

   588→   588→function saveVenta(data) {
   589→   589→  _ensureStructure();
   590→   590→  var isNew = !data.id;
   591→   591→  if (isNew) {
   592→   592→    data.id = nextId('ventas');
   593→   593→    data.creado = new Date().toISOString();
   594→   594→    data.fecha = data.fecha || new Date().toISOString().slice(0, 10);
   595→   595→    data.items = data.items || [];
   596→   596→    data.total = Number(data.total) || 0;
   597→   597→  }
   598→   598→  if (isNew) {
   599→   599→    for (var i = 0; i < data.items.length; i++) {
   600→   600→      var item = data.items[i];
   601→   601→      var producto;
   602→   602→      if (item.tipo === 'especia') {
   603→   603→        producto = _db.especias[item.productoId];
   604→   604→        if (!producto) throw new Error('Especia no encontrada: ' + item.productoId);
   605→   605→      } else {
   606→   606→        producto = _db.blends[item.productoId];
   607→   607→        if (!producto) throw new Error('Blend no encontrado: ' + item.productoId);
   608→   608→      }
   609→   609→      item.productoNombre = producto.nombre;
   610→   610→      var cant = Number(item.cantidad) || 0;
   611→   611→      var talla = item.talla || 'chico';
   612→   612→      var stockKey = talla === 'grande' ? 'stockGrande' : 'stockChico';
   613→   613→      if ((producto[stockKey] || 0) < cant) {
   614→   614→        throw new Error('Stock insuficiente de frascos ' + talla + ' de "' + producto.nombre + '". Solicitado: ' + cant + ', Disponible: ' + (producto[stockKey] || 0));
   615→   615→      }
   616→   616→      producto[stockKey] = (producto[stockKey] || 0) - cant;
   617→   617→      item.precioUnitario = Number(item.precioUnitario) || 0;
   618→   618→      item.subtotal = item.precioUnitario * cant;
   619→   619→    }
   620→   620→    // Recalculate total
   621→   621→    data.total = data.items.reduce(function(s, it) { return s + (it.subtotal || 0); }, 0);
   622→   622→  }
   623→   623→  _db.ventas[data.id] = data;
   624→   624→  _saveToFirebase(); _cacheLocal();
   625→   625→  _notify(isNew ? 'create' : 'update', 'ventas', data.id);
   626→   626→  return data;
   627→   627→}

   775→   775→  } else if (tipo === 'blend' && _db.blends[id]) {
   776→   776→    _db.blends[id].enTienda = !_db.blends[id].enTienda;
   777→   777→  } else return;
   778→   778→  _saveToFirebase(); _cacheLocal();
   779→   779→  _notify('update', tipo === 'especia' ? 'especias' : 'blends', id);
   780→   780→}

   784→   784→/** Find especia by name with flexible matching (exact, prefix, contains, word overlap) */
   785→   785→function findEspeciaByName(nombre) {
   786→   786→  if (!nombre) return null;
   787→   787→  var target = nombre.trim().toLowerCase();
   788→   788→  var keys = Object.keys(_db.especias || {});
   789→   789→  // 1. Exact match
   790→   790→  for (var i = 0; i < keys.length; i++) {
   791→   791→    var e = _db.especias[keys[i]];
   792→   792→    if (e && (e.nombre || '').trim().toLowerCase() === target) return e;
   793→   793→  }
   794→   794→  // 2. Especia name starts with target (e.g. "Color" matches "Color (achiote/...)")
   795→   795→  for (var i = 0; i < keys.length; i++) {
   796→   796→    var e = _db.especias[keys[i]];
   797→   797→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) === 0) return e;
   798→   798→  }
   799→   799→  // 3. Target is contained in especia name
   800→   800→  for (var i = 0; i < keys.length; i++) {
   801→   801→    var e = _db.especias[keys[i]];
   802→   802→    if (e && (e.nombre || '').trim().toLowerCase().indexOf(target) >= 0) return e;
   803→   803→  }
   804→   804→  // 4. Word overlap: any word (len>=4) from target appears in especia name
   805→   805→  var targetWords = target.split(/[\s()\/,]+/).filter(function(w) { return w.length >= 4; });
   806→   806→  for (var w = 0; w < targetWords.length; w++) {
   807→   807→    for (var i = 0; i < keys.length; i++) {
   808→   808→      var e = _db.especias[keys[i]];
   809→   809→      if (e && (e.nombre || '').trim().toLowerCase().indexOf(targetWords[w]) >= 0) return e;
   810→   810→    }
   811→   811→  }
   812→   812→  return null;
   813→   813→}

   815→   815→/** Auto-detect categoria from blend USO field */
   816→   816→function _categoriaFromUso(uso) {
   817→   817→  if (!uso) return 'Comidas';
   818→   818→  var u = uso.toLowerCase();
   819→   819→  // Cocteleria keywords
   820→   820→  if (/\b(gin|ron|vodka|whisky|mojito|mule|vermouth|aperitif|coctel)\b/.test(u)) return 'Cocteleria';
   821→   821→  // Infusiones keywords
   822→   822→  if (/\b(relajant|sueño|digestiv|energiz|té|calidez|respiratorio|meditaci|antioxidant|bienestar|infusi)\b/.test(u)) return 'Infusiones';
   823→   823→  return 'Comidas';
   824→   824→}

   826→   826→/**
   827→   827→ * Import especias and blends from parsed Excel data.
   828→   828→ * Returns { especiasCreadas: N, blendsCreados: N, blendsParciales: N, errores: [] }
   829→   829→ *
   830→   830→ * especiasList: [{ nombre, categoria? }]
   831→   831→ * blendsList:   [{ nombre, region, uso, categoria, ingredientes: [{ especia, g, pct }] }]
   832→   832→ * gramosChico:  default grams for frasco chico (e.g. 30)
   833→   833→ * gramosGrande: default grams for frasco grande (e.g. 80)
   834→   834→ */
   835→   835→function importFromExcelData(especiasList, blendsList, gramosChico, gramosGrande) {
   836→   836→  _ensureStructure();
   837→   837→  var resultado = { especiasCreadas: 0, especiasExistentes: 0, blendsCreados: 0, blendsExistentes: 0, ingredientesNoResueltos: [], errores: [] };
   838→   838→
   839→   839→  // 1. Create especias (skip if name already exists)
   840→   840→  for (var i = 0; i < especiasList.length; i++) {
   841→   841→    var esp = especiasList[i];
   842→   842→    var nombre = (esp.nombre || '').trim();
   843→   843→    if (!nombre) continue;
   844→   844→    var existing = findEspeciaByName(nombre);
   845→   845→    if (existing) {
   846→   846→      resultado.especiasExistentes++;
   847→   847→      continue;
   848→   848→    }
   849→   849→    saveEspecia({
   850→   850→      nombre: nombre,
   851→   851→      categoria: esp.categoria || 'Comidas',
   852→   852→      precioChico: 0,
   853→   853→      precioGrande: 0,
   854→   854→      gramosChico: Number(gramosChico) || 30,
   855→   855→      gramosGrande: Number(gramosGrande) || 80,
   856→   856→      stockBolsa: 0,
   857→   857→      stockChico: 0,
   858→   858→      stockGrande: 0
   859→   859→    });
   860→   860→    resultado.especiasCreadas++;
   861→   861→  }
   862→   862→
   863→   863→  // 2. Create blends (skip if name already exists)
   864→   864→  for (var j = 0; j < blendsList.length; j++) {
   865→   865→    var bl = blendsList[j];
   866→   866→    var nombre = (bl.nombre || '').trim();
   867→   867→    if (!nombre) continue;
   868→   868→    var existingBlend = null;
   869→   869→    var blKeys = Object.keys(_db.blends || {});
   870→   870→    for (var k = 0; k < blKeys.length; k++) {
   871→   871→      if ((_db.blends[blKeys[k]].nombre || '').trim().toLowerCase() === nombre.toLowerCase()) {
   872→   872→        existingBlend = _db.blends[blKeys[k]];
   873→   873→        break;
   874→   874→      }
   875→   875→    }
   876→   876→    if (existingBlend) {
   877→   877→      resultado.blendsExistentes++;
   878→   878→      continue;
   879→   879→    }
   880→   880→
   881→   881→    // Resolve ingredients: map specia names to IDs and calculate grams per frasco
   882→   882→    var ings = bl.ingredientes || [];
   883→   883→    var recipeTotal = 0;
   884→   884→    for (var ii = 0; ii < ings.length; ii++) recipeTotal += (Number(ings[ii].g) || 0);
   885→   885→    if (recipeTotal <= 0) recipeTotal = 500;
   886→   886→
   887→   887→    var resolvedIngs = [];
   888→   888→    for (var ii = 0; ii < ings.length; ii++) {
   889→   889→      var ing = ings[ii];
   890→   890→      var espObj = findEspeciaByName(ing.especia);
   891→   891→      if (!espObj) {
   892→   892→        resultado.ingredientesNoResueltos.push(nombre + ' → ' + (ing.especia || '?'));
   893→   893→        continue;
   894→   894→      }
   895→   895→      var ingG = Number(ing.g) || 0;
   896→   896→      resolvedIngs.push({
   897→   897→        especiaId: espObj.id,
   898→   898→        especiaNombre: espObj.nombre,
   899→   899→        gramosChico: Math.round((ingG / recipeTotal) * (Number(gramosChico) || 30) * 100) / 100,
   900→   900→        gramosGrande: Math.round((ingG / recipeTotal) * (Number(gramosGrande) || 80) * 100) / 100,
   901→   901→        gramosReceta: ingG
   902→   902→      });
   903→   903→    }
   904→   904→
   905→   905→    var cat = bl.categoria || _categoriaFromUso(bl.uso);
   906→   906→    saveBlend({
   907→   907→      nombre: nombre,
   908→   908→      categoria: cat,
   909→   909→      region: bl.region || '',
   910→   910→      uso: bl.uso || '',
   911→   911→      precioChico: 0,
   912→   912→      precioGrande: 0,
   913→   913→      ingredientes: resolvedIngs,
   914→   914→      stockChico: 0,
   915→   915→      stockGrande: 0
   916→   916→    });
   917→   917→    resultado.blendsCreados++;
   918→   918→  }
   919→   919→
   920→   920→  return resultado;
   921→   921→}

   935→   935→function addProductTag(cat, tagName) {
   936→   936→  _ensureStructure();
   937→   937→  tagName = (tagName || '').trim();
   938→   938→  if (!tagName) return false;
   939→   939→  if (!_db.productTags) _db.productTags = {};
   940→   940→  if (!_db.productTags[cat]) _db.productTags[cat] = [];
   941→   941→  // Check duplicate (case-insensitive)
   942→   942→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   943→   943→    if (_db.productTags[cat][i].toLowerCase() === tagName.toLowerCase()) return false;
   944→   944→  }
   945→   945→  _db.productTags[cat].push(tagName);
   946→   946→  _saveToFirebase(); _cacheLocal();
   947→   947→  return true;
   948→   948→}

   950→   950→function removeProductTag(cat, tagName) {
   951→   951→  _ensureStructure();
   952→   952→  if (!_db.productTags || !_db.productTags[cat]) return false;
   953→   953→  var idx = -1;
   954→   954→  for (var i = 0; i < _db.productTags[cat].length; i++) {
   955→   955→    if (_db.productTags[cat][i] === tagName) { idx = i; break; }
   956→   956→  }
   957→   957→  if (idx < 0) return false;
   958→   958→  _db.productTags[cat].splice(idx, 1);
   959→   959→  // Also remove from all products that have this tag
   960→   960→  var allEsp = getEspecias();
   961→   961→  for (var e = 0; e < allEsp.length; e++) {
   962→   962→    if (allEsp[e].tags) {
   963→   963→      var ti = allEsp[e].tags.indexOf(tagName);
   964→   964→      if (ti >= 0) { allEsp[e].tags.splice(ti, 1); }
   965→   965→    }
   966→   966→  }
   967→   967→  var allBl = getBlends();
   968→   968→  for (var b = 0; b < allBl.length; b++) {
   969→   969→    if (allBl[b].tags) {
   970→   970→      var bi = allBl[b].tags.indexOf(tagName);
   971→   971→      if (bi >= 0) { allBl[b].tags.splice(bi, 1); }
   972→   972→    }
   973→   973→  }
   974→   974→  _saveToFirebase(); _cacheLocal();
   975→   975→  return true;
   976→   976→}

   980→   980→function compressImage(file, maxW, quality, cb) {
   981→   981→  var reader = new FileReader();
   982→   982→  reader.onload = function(ev) {
   983→   983→    var img = new Image();
   984→   984→    img.onload = function() {
   985→   985→      var w = img.width, h = img.height;
   986→   986→      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
   987→   987→      var canvas = document.createElement('canvas');
   988→   988→      canvas.width = w; canvas.height = h;
   989→   989→      var ctx = canvas.getContext('2d');
   990→   990→      ctx.drawImage(img, 0, 0, w, h);
   991→   991→      var dataUrl = canvas.toDataURL('image/jpeg', quality);
   992→   992→      cb(null, dataUrl);
   993→   993→    };
   994→   994→    img.onerror = function() { cb('Error al cargar imagen'); };
   995→   995→    img.src = ev.target.result;
   996→   996→  };
   997→   997→  reader.onerror = function() { cb('Error al leer archivo'); };
   998→   998→  reader.readAsDataURL(file);
   999→   999→}
